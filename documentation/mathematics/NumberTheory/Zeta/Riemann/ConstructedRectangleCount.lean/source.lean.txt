import GRWTSK.NumberTheory.Zeta.Riemann.RationalPoleValue
import GRWTSK.NumberTheory.Zeta.Riemann.RationalRectangleCount

set_option autoImplicit false

/-!
# The rectangle count with source-value semantics derived from finite plans

Plans contain geometry and resolutions, NEVER a claimed source-value ball. Their
complete values are generated by RationalPoleValue. All prior geometry, strict
margins, image data, logarithmic index and canonical-divisor obligations persist.
No successful large target trace is asserted. Every inherited proof still needs
pinned elaboration and trust review. SOURCE_ONLY / NOT_RUN / NOT_KERNEL_AUDITED.
-/
namespace GRWTSK.RiemannHypothesis.ConstructedRectangleCount

open GRWTSK.Analysis.ErrorBounds
open RationalComplexEnclosure RationalDyadicArithmetic
open RationalPoleContour CriticalStripHeightDescent

structure Plan where
  center : Coordinates
  inner : ℚ
  outer : ℚ
  parameters : RationalPoleValue.Parameters
  deriving DecidableEq

/-- The fallback is deliberately outside the source's positive-real-part domain.
It is total data, not evidence of a present or acceptable record. -/
def fallback : Plan := ⟨zero, 0, 0, ⟨0,0,0,0,0,0,0⟩⟩

def planAt (plans : List Plan) (i : ℕ) : Plan := plans.getD i fallback

def record (p : Plan) : EdgeData :=
  ⟨p.center, p.inner, p.outer, RationalPoleValue.valueBall p.center p.parameters⟩

def records (plans : List Plan) : List EdgeData := plans.map record

def PlansValid (n : ℕ) (plans : List Plan) : Prop :=
  0 < n ∧ plans.length = n ∧ ∀ i ∈ Finset.range n,
    RationalPoleValue.check (planAt plans i).center (planAt plans i).parameters = true

instance plansValidDecidable (n : ℕ) (plans : List Plan) : Decidable (PlansValid n plans) := by
  unfold PlansValid; infer_instance

def checkPlans (n : ℕ) (plans : List Plan) : Bool := decide (PlansValid n plans)

theorem checkPlans_iff (n : ℕ) (plans : List Plan) :
    checkPlans n plans = true ↔ PlansValid n plans := by simp [checkPlans]

@[simp] theorem records_length (plans : List Plan) : (records plans).length = plans.length := by
  simp [records]

/-- Cast-free position identity: the same plan supplies both center and value. -/
theorem edgeAt_records {plans : List Plan} {i : ℕ} (hi : i < plans.length) :
    edgeAt (records plans) i = record (planAt plans i) := by
  unfold edgeAt records planAt
  rw [List.getD_eq_getElem _ _ (by simpa only [List.length_map] using hi),
    List.getD_eq_getElem _ _ hi]
  simp

/-- No caller-supplied Enclosures, value truth annotation or exterior oracle. -/
theorem enclosures_of_checkPlans {n : ℕ} {plans : List Plan}
    (h : checkPlans n plans = true) : Enclosures n (records plans) := by
  obtain ⟨_,hl,hv⟩ := (checkPlans_iff _ _).mp h
  intro i hi
  have hip : i < plans.length := by simpa only [hl] using Finset.mem_range.mp hi
  rw [edgeAt_records hip]
  exact RationalPoleValue.valueBall_sound (hv i hi)

/-- Old guards consume generated source balls, not independently editable values. -/
def evaluateBox (l u : Coordinates) (n : ℕ) (marks : List ℚ)
    (plans : List Plan) (image : List Coordinates) : Option ℤ :=
  if checkPlans n plans then
    RationalRectangleCount.evaluateBox l u n marks (records plans) image
  else none

theorem evaluateBox_eq_some_iff (l u : Coordinates) (n : ℕ) (a : List ℚ)
    (plans : List Plan) (y : List Coordinates) (k : ℤ) :
    evaluateBox l u n a plans y = some k ↔ checkPlans n plans = true ∧
      RationalRectangleCount.evaluateBox l u n a (records plans) y = some k := by
  by_cases h : checkPlans n plans = true <;> simp [evaluateBox, h]

def evaluate (T : ℚ) (n : ℕ) (marks : List ℚ)
    (plans : List Plan) (image : List Coordinates) : Option ℤ :=
  evaluateBox RationalRectangleCount.lower (RationalRectangleCount.upper T) n marks plans image

/-- The only substantive premise is finite success for this SAME literal data.
This implication does not assert that target-height success has been obtained. -/
theorem count_eq_of_evaluate {T : ℚ} {n : ℕ} {a : List ℚ}
    {plans : List Plan} {y : List Coordinates} {k : ℤ}
    (h : evaluate T n a plans y = some k) :
    (MirrorCountingRectangle.count (T : ℝ) : ℤ) = k := by
  obtain ⟨hp,hc⟩ := (evaluateBox_eq_some_iff _ _ _ _ _ _ _).mp h
  exact RationalRectangleCount.count_eq_of_evaluate (enclosures_of_checkPlans hp) hc

theorem result_nonnegative {T : ℚ} {n : ℕ} {a : List ℚ}
    {plans : List Plan} {y : List Coordinates} {k : ℤ}
    (h : evaluate T n a plans y = some k) : 0 ≤ k := by
  rw [← count_eq_of_evaluate h]
  exact Int.natCast_nonneg _

/-- Conditional bounded completion. All finite success and all Lean acceptance
obligations remain; there is no assertion of a completed numerical inventory. -/
theorem complete_pair_of_evaluate_one {n : ℕ} {a : List ℚ}
    {plans : List Plan} {y : List Coordinates}
    (h : evaluate 15 n a plans y = some 1) :
    ∃ z : CanonicalZeroDivisor.ZeroPoint, CanonicalZeroDivisor.multiplicity z = 1 ∧
      z.1.re = (1/2 : ℝ) ∧ CanonicalZeroDivisor.heightHead (15 : ℝ) =
        {z, LocalZeroSymmetry.conjugatePoint z} := by
  obtain ⟨hp,hc⟩ := (evaluateBox_eq_some_iff _ _ _ _ _ _ _).mp h
  exact RationalRectangleCount.complete_pair_of_evaluate_one (enclosures_of_checkPlans hp) hc

end GRWTSK.RiemannHypothesis.ConstructedRectangleCount

open GRWTSK.NumberTheory.Cyclotomic.ThirdCyclotomicIdeals

open GRWTSK.NumberTheory.Cyclotomic.ThirdCyclotomicCharacters

private instance quotientFintype (P : Ideal O) [P.IsMaximal] : Fintype (O ⧸ P) :=
  Fintype.ofFinite _

private instance quotientField (P : Ideal O) [P.IsMaximal] : Field (O ⧸ P) :=
  Ideal.Quotient.field P

/-- The actual cubic Jacobi sum has residue -1, retaining its coefficient in the
integer algebra generated by omega. -/
theorem cubicJacobi_primary (P : Ideal O) [P.IsMaximal] (hgood : lambda ∉ P) :
    ∃ z ∈ Algebra.adjoin ℤ {omega},
      jacobiSum (cubicChar P hgood) (cubicChar P hgood) = -1 + z * lambda ^ 2 := by
  classical
  simpa only [lambda] using
    (exists_jacobiSum_eq_neg_one_add (n := 3) (by decide)
      (cubicChar_pow_three P hgood) (cubicChar_pow_three P hgood)
      (by simpa only [Nat.card_eq_fintype_card] using card_sub_one_div_three P hgood)
      omega_primitive)

/-- The actual Jacobi sum has residue -1 modulo lambda squared. -/
theorem lambda_sq_dvd_cubicJacobi_add_one (P : Ideal O) [P.IsMaximal]
    (hgood : lambda ∉ P) :
    lambda ^ 2 ∣ jacobiSum (cubicChar P hgood) (cubicChar P hgood) + 1 := by
  obtain ⟨z, _, hJ⟩ := cubicJacobi_primary P hgood
  refine ⟨z, ?_⟩
  rw [hJ]
  ring

/-- Negating the actual Jacobi sum gives residue +1; this asserts no generator property. -/
theorem lambda_sq_dvd_neg_cubicJacobi_sub_one (P : Ideal O) [P.IsMaximal]
    (hgood : lambda ∉ P) :
    lambda ^ 2 ∣ -(jacobiSum (cubicChar P hgood) (cubicChar P hgood)) - 1 := by
  obtain ⟨z, _, hJ⟩ := cubicJacobi_primary P hgood
  refine ⟨-z, ?_⟩
  rw [hJ]
  ring

/-- The actual ramified ideal excludes 2, so lambda does not divide 2. -/
theorem lambda_not_dvd_two : ¬lambda ∣ (2 : O) := by
  rintro ⟨a, ha⟩
  apply two_not_mem_ramifiedIdeal
  rw [ha]
  exact ramifiedIdeal.mul_mem_right a lambda_mem_ramifiedIdeal

/-- The opposite residue +1 is impossible for the actual unnegated Jacobi sum. -/
theorem lambda_sq_not_dvd_cubicJacobi_sub_one (P : Ideal O) [P.IsMaximal]
    (hgood : lambda ∉ P) :
    ¬lambda ^ 2 ∣ jacobiSum (cubicChar P hgood) (cubicChar P hgood) - 1 := by
  intro hwrong
  have htwo : lambda ^ 2 ∣ (2 : O) := by
    convert dvd_sub (lambda_sq_dvd_cubicJacobi_add_one P hgood) hwrong using 1
    ring
  exact lambda_not_dvd_two ((dvd_pow_self lambda (by decide)).trans htwo)

end

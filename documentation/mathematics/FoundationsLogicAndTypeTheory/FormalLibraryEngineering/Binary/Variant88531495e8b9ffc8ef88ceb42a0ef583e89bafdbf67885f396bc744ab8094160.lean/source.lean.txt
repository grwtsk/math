/-- Leaves of the binary Presentation syntax tree. -/
inductive PresentationLeaf where
  | one : PresentationLeaf
  | var : Nat → PresentationLeaf
deriving Repr, DecidableEq

/-- Full ordered binary trees with labelled leaves. -/
inductive FullBinaryTree (α : Type u) where
  | leaf : α → FullBinaryTree α
  | node : FullBinaryTree α → FullBinaryTree α → FullBinaryTree α
deriving Repr, DecidableEq

/-- A local bijection record, avoiding external equivalence dependencies. -/
structure Bijection (α β : Type u) where
  toFun : α → β
  invFun : β → α
  left_inv : ∀ x : α, invFun (toFun x) = x
  right_inv : ∀ y : β, toFun (invFun y) = y

namespace Expr

/-- Convert Presentation syntax to the corresponding labelled full binary tree. -/
def toBinaryTree : Expr → FullBinaryTree PresentationLeaf
  | one => .leaf .one
  | var n => .leaf (.var n)
  | node lhs rhs => .node lhs.toBinaryTree rhs.toBinaryTree

/-- Convert a labelled full binary tree to Presentation syntax. -/
def ofBinaryTree : FullBinaryTree PresentationLeaf → Expr
  | .leaf .one => one
  | .leaf (.var n) => var n
  | .node lhs rhs => node (ofBinaryTree lhs) (ofBinaryTree rhs)

/-- Presentation-to-binary-to-Presentation is the identity. -/
theorem of_toBinaryTree :
    ∀ e : Expr, ofBinaryTree (toBinaryTree e) = e := by
  intro e
  induction e with
  | one => rfl
  | var n => rfl
  | node lhs rhs ihl ihr =>
      simp [toBinaryTree, ofBinaryTree, ihl, ihr]

/-- Binary-to-Presentation-to-binary is the identity. -/
theorem to_ofBinaryTree :
    ∀ t : FullBinaryTree PresentationLeaf, toBinaryTree (ofBinaryTree t) = t := by
  intro t
  induction t with
  | leaf leaf =>
      cases leaf <;> rfl
  | node lhs rhs ihl ihr =>
      simp [toBinaryTree, ofBinaryTree, ihl, ihr]

/--
The exact syntactic bijection between Presentation expressions and labelled full binary
trees.
-/
def equivFullBinaryTree : Bijection Expr (FullBinaryTree PresentationLeaf) where
  toFun := toBinaryTree
  invFun := ofBinaryTree
  left_inv := of_toBinaryTree
  right_inv := to_ofBinaryTree

end Expr

/-- Constructor injectivity records that Presentation syntax preserves order. -/
theorem node_injective
    {a b c d : Expr}
    (h : Expr.node a b = Expr.node c d) :
    a = c ∧ b = d := by
  cases h
  exact ⟨rfl, rfl⟩

end Presentation

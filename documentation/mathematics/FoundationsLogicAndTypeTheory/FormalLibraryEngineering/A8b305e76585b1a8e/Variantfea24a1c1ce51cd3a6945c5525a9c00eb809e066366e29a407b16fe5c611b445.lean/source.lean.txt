open MathDiscovery

namespace Registry

/-- Canonical and secondary homes follow actual edges from the declared root. -/
def DomainPathsResolve (registry : Registry) : Prop :=
  ∀ taxon ∈ registry.taxa,
    taxon.principalDomain.root = registry.rootDomain ∧
      registry.domains.ValidPath taxon.principalDomain ∧
      ∀ path ∈ taxon.secondaryDomains,
        path.root = registry.rootDomain ∧ registry.domains.ValidPath path

end Registry

end MathDiscovery.Taxonomy

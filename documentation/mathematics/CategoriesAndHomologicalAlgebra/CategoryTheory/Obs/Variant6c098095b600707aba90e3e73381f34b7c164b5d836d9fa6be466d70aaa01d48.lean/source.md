# Folder.Category.Obs

Radiation role: category-interface
BinaryMatrix spine: This module supplies categorical interfaces used by sheaf and monodromy layers around the BinaryMatrix spine.

namespace Folder
namespace Category

open CategoryTheory

/-
This is the free observation category generated by the arrows below. It is a
formal indexing category, not a semantic model of an actual filesystem.
-/
inductive Obs where
  | session
  | rootHandle
  | entryHandle
  | fileEntry
  | byteInterval
  | indexSection
  | previewSection
  | historyInterval
deriving DecidableEq, Repr

namespace Obs

inductive Generator : Obs → Obs → Type
  | sessionToRoot : Generator Obs.session Obs.rootHandle
  | rootToEntry : Generator Obs.rootHandle Obs.entryHandle
  | entryToFile : Generator Obs.entryHandle Obs.fileEntry
  | fileToBytes : Generator Obs.fileEntry Obs.byteInterval
  | fileToIndex : Generator Obs.fileEntry Obs.indexSection
  | bytesToPreview : Generator Obs.byteInterval Obs.previewSection
  | sessionToHistory : Generator Obs.session Obs.historyInterval

inductive Path : Obs → Obs → Type
  | nil {U : Obs} : Path U U
  | cons {U V W : Obs} : Generator U V → Path V W → Path U W

namespace Path

def append {U V W : Obs} : Path U V → Path V W → Path U W
  | nil, q => q
  | cons g p, q => cons g (append p q)

@[simp]
theorem nil_append {U V : Obs} (p : Path U V) :
    append nil p = p :=
  rfl

@[simp]
theorem append_nil {U V : Obs} (p : Path U V) :
    append p nil = p := by
  induction p with
  | nil => rfl
  | cons g p ih =>
      simp [append, ih]

@[simp]
theorem append_assoc {U V W X : Obs}
    (p : Path U V) (q : Path V W) (r : Path W X) :
    append (append p q) r = append p (append q r) := by
  induction p with
  | nil => rfl
  | cons g p ih =>
      simp [append, ih]

end Path

instance : Category Obs where
  Hom U V := Path U V
  id U := Path.nil
  comp f g := Path.append f g
  id_comp := by
    intro X Y f
    rfl
  comp_id := by
    intro X Y f
    exact Path.append_nil f
  assoc := by
    intro W X Y Z f g h
    exact Path.append_assoc f g h

end Obs
end Category
end Folder

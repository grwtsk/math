/-- Every facet identifier resolves in the governed facet registry. -/
def FacetsResolve (registry : Registry) : Prop :=
  ∀ taxon ∈ registry.taxa,
    ∀ facetId ∈ taxon.facets,
      facetId ∈ registry.facets.facets.map (·.id)

end Registry

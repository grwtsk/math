# Exact finite heat evolution

The matrix exponential gives the exact finite heat family generated by the
square of a Hermitian matrix.  Hermiticity and the semigroup law are proved in
the kernel.  Operator-norm contraction is exposed as a separate certificate:
this module does not confuse positivity of a quadratic form with an already
formalized norm estimate for the selected matrix norm.
-/

variable {Index : Type u} [Fintype Index] [DecidableEq Index]

/-- Exact finite heat matrix `exp (-s D²)`. -/
def heatDiffuser (dirac : FiniteOperator Index Index) (s : ℝ) :
    FiniteOperator Index Index :=
  NormedSpace.exp ((-(s : ℂ)) • (dirac * dirac))

omit [DecidableEq Index] in
theorem diracSquare_isHermitian
    {dirac : FiniteOperator Index Index}
    (hermitian : dirac.IsHermitian) :
    (dirac * dirac).IsHermitian := by
  rw [Matrix.IsHermitian, Matrix.conjTranspose_mul,
    hermitian.eq]

omit [DecidableEq Index] in
theorem heatGenerator_isHermitian
    {dirac : FiniteOperator Index Index}
    (hermitian : dirac.IsHermitian)
    (s : ℝ) :
    ((-(s : ℂ)) • (dirac * dirac)).IsHermitian := by
  apply (diracSquare_isHermitian hermitian).smul
  simp [isSelfAdjoint_iff]

/-- Every finite heat matrix is Hermitian. -/
theorem heatDiffuser_selfAdjoint
    {dirac : FiniteOperator Index Index}
    (hermitian : dirac.IsHermitian)
    (s : ℝ) :
    (heatDiffuser dirac s).IsHermitian :=
  (heatGenerator_isHermitian hermitian s).exp

/-- The finite heat family obeys the exact semigroup law. -/
theorem heatDiffuser_add
    (dirac : FiniteOperator Index Index)
    (s t : ℝ) :
    heatDiffuser dirac (s + t) =
      heatDiffuser dirac s * heatDiffuser dirac t := by
  have exponent_add :
      ((-((s + t : ℝ) : ℂ)) • (dirac * dirac)) =
        (-(s : ℂ)) • (dirac * dirac) +
          (-(t : ℂ)) • (dirac * dirac) := by
    module
  unfold heatDiffuser
  rw [exponent_add]
  exact Matrix.exp_add_of_commute _ _
    (((Commute.refl (dirac * dirac)).smul_left (-(s : ℂ))).smul_right
      (-(t : ℂ)))

/-- Norm contraction is a proof-bearing capability, not a Boolean status. -/
structure HeatContractionCertificate
    (dirac : FiniteOperator Index Index) where
  operatorNorm : FiniteOperator Index Index → ℝ
  operatorNorm_nonnegative : ∀ matrix, 0 ≤ operatorNorm matrix
  contractive : ∀ s : ℝ, 0 ≤ s → operatorNorm (heatDiffuser dirac s) ≤ 1

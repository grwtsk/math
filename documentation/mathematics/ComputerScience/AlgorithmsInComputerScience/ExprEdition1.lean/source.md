import Lean
import Mathematics.Core.Id

import Mathematics.Core.IdElements.Digest

/-!
# Stable structural expression schema

Bound-variable spelling, source positions, and display metadata are not part of
`semanticDigest`.  Binder classes, constant names, literals, projections, let
structure, and normalized universe-parameter positions remain semantic.
-/

namespace Mathematics.Extractor

open Mathematics

inductive BinderClass
  | default
  | implicit
  | strictImplicit
  | instanceImplicit
  deriving DecidableEq, Repr, Inhabited

inductive CanonicalLevel
  | zero
  | succ (level : CanonicalLevel)
  | max (left right : CanonicalLevel)
  | imax (left right : CanonicalLevel)
  | param (position : Nat)
  | unresolvedParam (name : Lean.Name)
  | metavariable (description : String)
  deriving DecidableEq, Repr, Inhabited

inductive CanonicalLiteral
  | nat (value : Nat)
  | string (value : String)
  | adapterEncoded (value : String)
  deriving DecidableEq, Repr, Inhabited

/-- Structural expression node. Child expressions are addressed by `ExprId`. -/
inductive CanonicalExpr
  | bvar (index : Nat)
  | sort (level : CanonicalLevel)
  | const (name : Lean.Name) (levels : Array CanonicalLevel)
  | app (fn arg : ExprId)
  | lam (binderInfo : BinderClass) (type body : ExprId)
  | forallE (binderInfo : BinderClass) (type body : ExprId)
  | letE (type value body : ExprId) (nondep : Bool)
  | lit (literal : CanonicalLiteral)
  | mdata (payloadDigest : Digest) (expr : ExprId)
  | proj (typeName : Lean.Name) (fieldIdx : Nat) (expr : ExprId)
  | adapterStructural (encoding : String)
  | legacyText (text : String)
  | invalidFreeVariable (description : String)
  | invalidMetavariable (description : String)
  deriving DecidableEq, Repr, Inhabited

/-- Exact structural child references of one canonical expression node. -/
def CanonicalExpr.childIds : CanonicalExpr → List ExprId
  | .app function argument => [function, argument]
  | .lam _ type body => [type, body]
  | .forallE _ type body => [type, body]
  | .letE type value body _ => [type, value, body]
  | .mdata _ expression => [expression]
  | .proj _ _ expression => [expression]
  | _ => []

structure PresentationVariant where
  digest : Digest
  encodingDigest : Digest
  deriving DecidableEq, Repr, Inhabited

structure ExpressionRecord where
  exprId : ExprId
  semanticRoot : CanonicalExpr
  semanticDigest : Digest
  rawDigests : Array Digest
  presentationVariants : Array PresentationVariant
  metadataDigests : Array Digest
  deriving DecidableEq, Repr, Inhabited

/-- Path within the structural expression tree. -/
structure ExprPath where
  steps : Array Nat
  deriving DecidableEq, Repr, Inhabited

end Mathematics.Extractor

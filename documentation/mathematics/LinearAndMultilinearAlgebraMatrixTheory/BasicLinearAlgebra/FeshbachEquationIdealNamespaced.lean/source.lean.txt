# Exact equation ideals for finite Feshbach elimination

Invertible elimination of a finite block preserves the ideal generated by the
equations, not merely their common zero set. The upper-right block is arbitrary;
no adjoint, order, or positivity hypothesis occurs in the algebraic identity.
The polynomial specialization uses constant coefficient lifts and independent
variables for both blocks.

/-- The ideal generated by both blocks of a finite system of equations. -/
def blockEquationIdeal (p : ι → R) (q : κ → R) : Ideal R :=
  Ideal.span (Set.range (Sum.elim p q))

theorem mem_blockEquationIdeal_left (p : ι → R) (q : κ → R) (i : ι) :
    p i ∈ blockEquationIdeal p q :=
  Ideal.subset_span ⟨Sum.inl i, rfl⟩

theorem mem_blockEquationIdeal_right (p : ι → R) (q : κ → R) (j : κ) :
    q j ∈ blockEquationIdeal p q :=
  Ideal.subset_span ⟨Sum.inr j, rfl⟩

variable [Fintype κ]

private theorem mulVec_mem_ideal {α : Type*} {β : Type*} [Fintype β]
    (I : Ideal R) (M : Matrix α β R) (x : β → R)
    (hx : ∀ j, x j ∈ I) (i : α) : M.mulVec x i ∈ I := by
  exact I.sum_mem fun j _ ↦ I.mul_mem_left (M i j) (hx j)

/-- An invertible block row operation preserves the exact generated ideal. -/
theorem blockEquationIdeal_elimination [DecidableEq κ]
    (U : Matrix ι κ R) (B : Matrix κ κ R) [Invertible B]
    (p : ι → R) (q : κ → R) :
    blockEquationIdeal (p + U.mulVec q) (B.mulVec q) =
      blockEquationIdeal p q := by
  apply le_antisymm
  · apply Ideal.span_le.mpr
    rintro _ ⟨i | j, rfl⟩
    · exact (blockEquationIdeal p q).add_mem
        (mem_blockEquationIdeal_left p q i)
        (mulVec_mem_ideal _ U q (mem_blockEquationIdeal_right p q) i)
    · exact mulVec_mem_ideal _ B q (mem_blockEquationIdeal_right p q) j
  · apply Ideal.span_le.mpr
    have hq : ∀ j, q j ∈ blockEquationIdeal (p + U.mulVec q) (B.mulVec q) := by
      intro j
      have h := mulVec_mem_ideal
        (blockEquationIdeal (p + U.mulVec q) (B.mulVec q)) (⅟ B) (B.mulVec q)
        (mem_blockEquationIdeal_right (p + U.mulVec q) (B.mulVec q)) j
      simpa only [Matrix.mulVec_mulVec, invOf_mul_self, Matrix.one_mulVec] using h
    rintro _ ⟨i | j, rfl⟩
    · change p i ∈ blockEquationIdeal (p + U.mulVec q) (B.mulVec q)
      have h := (blockEquationIdeal (p + U.mulVec q) (B.mulVec q)).sub_mem
        (mem_blockEquationIdeal_left (p + U.mulVec q) (B.mulVec q) i)
        (mulVec_mem_ideal _ U q hq i)
      simpa only [Pi.add_apply, add_sub_cancel_right] using h
    · exact hq j

/-- The order in which the two equation blocks are listed does not affect the
generated ideal. -/
theorem blockEquationIdeal_swap (p : ι → R) (q : κ → R) :
    blockEquationIdeal p q = blockEquationIdeal q p := by
  apply le_antisymm
  all_goals
    apply Ideal.span_le.mpr
    rintro _ ⟨i | j, rfl⟩
  · exact mem_blockEquationIdeal_right q p i
  · exact mem_blockEquationIdeal_left q p j
  · exact mem_blockEquationIdeal_right p q i
  · exact mem_blockEquationIdeal_left p q j

/-- An invertible coefficient matrix acting on one equation block does not
change its ideal. -/
theorem blockEquationIdeal_mulVec_left [Fintype ι] [DecidableEq ι]
    (D : Matrix ι ι R) [Invertible D] (p : ι → R) (q : κ → R) :
    blockEquationIdeal (D.mulVec p) q = blockEquationIdeal p q := by
  rw [blockEquationIdeal_swap (D.mulVec p) q, blockEquationIdeal_swap p q]
  simpa only [Matrix.zero_mulVec, add_zero] using
    blockEquationIdeal_elimination (0 : Matrix κ ι R) D q p

/-- A lower triangular row addition also preserves the exact equation ideal. -/
theorem blockEquationIdeal_add_mulVec_right [Fintype ι]
    (K : Matrix κ ι R) (p : ι → R) (q : κ → R) :
    blockEquationIdeal p (q + K.mulVec p) = blockEquationIdeal p q := by
  classical
  let : Invertible (1 : Matrix ι ι R) := invertibleOne
  rw [blockEquationIdeal_swap p (q + K.mulVec p), blockEquationIdeal_swap p q]
  simpa only [Matrix.one_mulVec] using
    blockEquationIdeal_elimination K (1 : Matrix ι ι R) q p

/-- Full block equations and the Schur equation with reconstruction residual
generate the same ideal over every commutative ring with an invertible block. -/
theorem feshbach_equationIdeal_eq [Fintype ι] [DecidableEq κ]
    (A : Matrix ι ι R) (U : Matrix ι κ R) (C : Matrix κ ι R)
    (B : Matrix κ κ R) [Invertible B] (x : ι → R) (y : κ → R) :
    blockEquationIdeal (A.mulVec x + U.mulVec y) (C.mulVec x + B.mulVec y) =
      blockEquationIdeal ((Feshbach.schur A U C B).mulVec x)
        (y + (⅟ B).mulVec (C.mulVec x)) := by
  dsimp only [Feshbach.schur]
  have htop :
      (A - U * ⅟ B * C).mulVec x +
          U.mulVec (y + (⅟ B).mulVec (C.mulVec x)) =
        A.mulVec x + U.mulVec y := by
    rw [Matrix.sub_mulVec, Matrix.mulVec_add]
    simp only [← Matrix.mulVec_mulVec]
    abel
  have hbottom :
      B.mulVec (y + (⅟ B).mulVec (C.mulVec x)) = C.mulVec x + B.mulVec y := by
    rw [Matrix.mulVec_add, Matrix.mulVec_mulVec, mul_invOf_self, Matrix.one_mulVec]
    exact add_comm _ _
  simpa only [htop, hbottom] using
    blockEquationIdeal_elimination U B ((A - U * ⅟ B * C).mulVec x)
      (y + (⅟ B).mulVec (C.mulVec x))

/-- The full polynomial equation ideal with constant-lifted matrix coefficients
and independent retained and eliminated variables. -/
noncomputable def feshbachPolynomialEquationIdeal [Fintype ι]
    (A : Matrix ι ι R) (U : Matrix ι κ R) (C : Matrix κ ι R)
    (B : Matrix κ κ R) : Ideal (MvPolynomial (ι ⊕ κ) R) :=
  blockEquationIdeal
    ((A.map MvPolynomial.C).mulVec (fun i ↦ MvPolynomial.X (Sum.inl i)) +
      (U.map MvPolynomial.C).mulVec (fun j ↦ MvPolynomial.X (Sum.inr j)))
    ((C.map MvPolynomial.C).mulVec (fun i ↦ MvPolynomial.X (Sum.inl i)) +
      (B.map MvPolynomial.C).mulVec (fun j ↦ MvPolynomial.X (Sum.inr j)))

/-- The reduced polynomial equation ideal includes both the Schur equations and
the exact reconstruction residual; no eliminated variable is discarded. -/
noncomputable def reducedFeshbachPolynomialEquationIdeal [Fintype ι] [DecidableEq κ]
    (A : Matrix ι ι R) (U : Matrix ι κ R) (C : Matrix κ ι R)
    (B : Matrix κ κ R) [Invertible B] : Ideal (MvPolynomial (ι ⊕ κ) R) :=
  blockEquationIdeal
    (((Feshbach.schur A U C B).map MvPolynomial.C).mulVec
      (fun i ↦ MvPolynomial.X (Sum.inl i)))
    ((fun j ↦ MvPolynomial.X (Sum.inr j)) +
      ((⅟ B).map MvPolynomial.C).mulVec
        ((C.map MvPolynomial.C).mulVec (fun i ↦ MvPolynomial.X (Sum.inl i))))

/-- The equation-ideal identity in the polynomial ring on the independent
retained and eliminated coordinates. Every matrix coefficient is lifted through
`MvPolynomial.C`, including the inverse and the Schur complement. -/
theorem feshbach_polynomial_equationIdeal_eq [Fintype ι] [DecidableEq κ]
    (A : Matrix ι ι R) (U : Matrix ι κ R) (C : Matrix κ ι R)
    (B : Matrix κ κ R) [Invertible B] :
    blockEquationIdeal
        ((A.map (MvPolynomial.C : R →+* MvPolynomial (ι ⊕ κ) R)).mulVec
            (fun i ↦ MvPolynomial.X (Sum.inl i)) +
          (U.map MvPolynomial.C).mulVec (fun j ↦ MvPolynomial.X (Sum.inr j)))
        ((C.map MvPolynomial.C).mulVec (fun i ↦ MvPolynomial.X (Sum.inl i)) +
          (B.map MvPolynomial.C).mulVec (fun j ↦ MvPolynomial.X (Sum.inr j))) =
      blockEquationIdeal
        (((Feshbach.schur A U C B).map MvPolynomial.C).mulVec
          (fun i ↦ MvPolynomial.X (Sum.inl i)))
        ((fun j ↦ MvPolynomial.X (Sum.inr j)) +
          ((⅟ B).map MvPolynomial.C).mulVec
            ((C.map MvPolynomial.C).mulVec (fun i ↦ MvPolynomial.X (Sum.inl i)))) := by
  let f : R →+* MvPolynomial (ι ⊕ κ) R := MvPolynomial.C
  let : Invertible (B.map f) := Invertible.map f.mapMatrix B
  have hinv : ⅟ (B.map f) = (⅟ B).map f := rfl
  have hschur : (Feshbach.schur A U C B).map f =
      Feshbach.schur (A.map f) (U.map f) (C.map f) (B.map f) := by
    dsimp only [Feshbach.schur]
    rw [hinv]
    ext i j
    simp only [Matrix.map_apply, Matrix.sub_apply, map_sub, Matrix.mul_apply, map_sum, map_mul]
  have h := feshbach_equationIdeal_eq (A.map f) (U.map f) (C.map f) (B.map f)
    (fun i ↦ MvPolynomial.X (Sum.inl i)) (fun j ↦ MvPolynomial.X (Sum.inr j))
  rw [← hschur, hinv] at h
  exact h

variable {K : Type*} [Field K]

/-- Every finite homogeneous block-equation ideal over a field is prime,
including singular systems. This theorem requires no invertibility of either
block; it is a specialization of the canonical linear-equation theorem. -/
theorem feshbach_polynomial_equationIdeal_isPrime [Fintype ι]
    (A : Matrix ι ι K) (U : Matrix ι κ K) (C : Matrix κ ι K)
    (B : Matrix κ κ K) :
    (blockEquationIdeal
        ((A.map (MvPolynomial.C : K →+* MvPolynomial (ι ⊕ κ) K)).mulVec
            (fun i ↦ MvPolynomial.X (Sum.inl i)) +
          (U.map MvPolynomial.C).mulVec (fun j ↦ MvPolynomial.X (Sum.inr j)))
        ((C.map MvPolynomial.C).mulVec (fun i ↦ MvPolynomial.X (Sum.inl i)) +
          (B.map MvPolynomial.C).mulVec (fun j ↦ MvPolynomial.X (Sum.inr j)))).IsPrime := by
  have h := homogeneous_mulVec_equationIdeal_isPrime (Matrix.fromBlocks A U C B)
  have hcoords : (MvPolynomial.X : (ι ⊕ κ) → MvPolynomial (ι ⊕ κ) K) =
      Sum.elim (fun i ↦ MvPolynomial.X (Sum.inl i))
        (fun j ↦ MvPolynomial.X (Sum.inr j)) := by
    funext s
    cases s <;> rfl
  rw [Matrix.fromBlocks_map, hcoords, Matrix.fromBlocks_mulVec] at h
  exact h

/-- Radicality of every finite homogeneous Feshbach polynomial system over a
field, including the singular Schur case. Radicality imposes no sign condition. -/
theorem feshbach_polynomial_equationIdeal_isRadical [Fintype ι]
    (A : Matrix ι ι K) (U : Matrix ι κ K) (C : Matrix κ ι K)
    (B : Matrix κ κ K) :
    (blockEquationIdeal
        ((A.map (MvPolynomial.C : K →+* MvPolynomial (ι ⊕ κ) K)).mulVec
            (fun i ↦ MvPolynomial.X (Sum.inl i)) +
          (U.map MvPolynomial.C).mulVec (fun j ↦ MvPolynomial.X (Sum.inr j)))
        ((C.map MvPolynomial.C).mulVec (fun i ↦ MvPolynomial.X (Sum.inl i)) +
          (B.map MvPolynomial.C).mulVec (fun j ↦ MvPolynomial.X (Sum.inr j)))).IsRadical :=
  (feshbach_polynomial_equationIdeal_isPrime A U C B).isRadical

/-- Exact elimination transports the same radicality to the Schur equations and
reconstruction residual without any Schur-rank assumption. -/
theorem reduced_feshbach_polynomial_equationIdeal_isRadical [Fintype ι] [DecidableEq κ]
    (A : Matrix ι ι K) (U : Matrix ι κ K) (C : Matrix κ ι K)
    (B : Matrix κ κ K) [Invertible B] :
    (blockEquationIdeal
        (((Feshbach.schur A U C B).map
            (MvPolynomial.C : K →+* MvPolynomial (ι ⊕ κ) K)).mulVec
          (fun i ↦ MvPolynomial.X (Sum.inl i)))
        ((fun j ↦ MvPolynomial.X (Sum.inr j)) +
          ((⅟ B).map MvPolynomial.C).mulVec
            ((C.map MvPolynomial.C).mulVec
              (fun i ↦ MvPolynomial.X (Sum.inl i))))).IsRadical := by
  have h := feshbach_polynomial_equationIdeal_isRadical A U C B
  rw [feshbach_polynomial_equationIdeal_eq] at h
  exact h

/-- The exact full polynomial ideal equals the vanishing ideal of its closed
point zero locus at the finite algebraically closed field scope. -/
theorem feshbachPolynomialEquationIdeal_eq_vanishingIdeal_zeroLocus
    [IsAlgClosed K] [Fintype ι]
    (A : Matrix ι ι K) (U : Matrix ι κ K) (C : Matrix κ ι K)
    (B : Matrix κ κ K) :
    feshbachPolynomialEquationIdeal A U C B =
      MvPolynomial.vanishingIdeal K
        (MvPolynomial.zeroLocus K (feshbachPolynomialEquationIdeal A U C B)) := by
  exact ((MvPolynomial.vanishingIdeal_zeroLocus_eq_radical _).trans
    (feshbach_polynomial_equationIdeal_isRadical A U C B).radical).symm

/-- The same closed point vanishing-ideal equality holds for the reduced ideal,
including singular Schur complements, by exact full/reduced ideal transport. -/
theorem reducedFeshbachPolynomialEquationIdeal_eq_vanishingIdeal_zeroLocus
    [IsAlgClosed K] [Fintype ι] [DecidableEq κ]
    (A : Matrix ι ι K) (U : Matrix ι κ K) (C : Matrix κ ι K)
    (B : Matrix κ κ K) [Invertible B] :
    reducedFeshbachPolynomialEquationIdeal A U C B =
      MvPolynomial.vanishingIdeal K
        (MvPolynomial.zeroLocus K (reducedFeshbachPolynomialEquationIdeal A U C B)) := by
  have heq : feshbachPolynomialEquationIdeal A U C B =
      reducedFeshbachPolynomialEquationIdeal A U C B :=
    feshbach_polynomial_equationIdeal_eq A U C B
  rw [← heq]
  exact feshbachPolynomialEquationIdeal_eq_vanishingIdeal_zeroLocus A U C B

/-- The origin solves this homogeneous full polynomial system over every field.
This is an existence statement only for the defined finite linear system. -/
theorem feshbachPolynomialEquationIdeal_origin_mem_zeroLocus [Fintype ι]
    (A : Matrix ι ι K) (U : Matrix ι κ K) (C : Matrix κ ι K)
    (B : Matrix κ κ K) :
    (0 : (ι ⊕ κ) → K) ∈
      MvPolynomial.zeroLocus K (feshbachPolynomialEquationIdeal A U C B) := by
  rw [feshbachPolynomialEquationIdeal, blockEquationIdeal, MvPolynomial.zeroLocus_span]
  rintro _ ⟨i | j, rfl⟩ <;> simp [Matrix.mulVec, dotProduct]

/-- Both exactly identified homogeneous equation systems have a closed point,
namely the origin. No nonemptiness theorem for a nonlinear parameter fiber is
supplied by this result. -/
theorem feshbachPolynomialEquationIdeal_full_reduced_zeroLocus_nonempty
    [Fintype ι] [DecidableEq κ]
    (A : Matrix ι ι K) (U : Matrix ι κ K) (C : Matrix κ ι K)
    (B : Matrix κ κ K) [Invertible B] :
    (MvPolynomial.zeroLocus K (feshbachPolynomialEquationIdeal A U C B)).Nonempty ∧
      (MvPolynomial.zeroLocus K (reducedFeshbachPolynomialEquationIdeal A U C B)).Nonempty := by
  have hfull := feshbachPolynomialEquationIdeal_origin_mem_zeroLocus A U C B
  have heq : feshbachPolynomialEquationIdeal A U C B =
      reducedFeshbachPolynomialEquationIdeal A U C B :=
    feshbach_polynomial_equationIdeal_eq A U C B
  exact ⟨⟨0, hfull⟩, ⟨0, heq ▸ hfull⟩⟩

/-- At the faithful nondegenerate finite scope (both the eliminated block and
the Schur block invertible), the full polynomial equation ideal is the complete
coordinate ideal. No claim of nondegeneracy is inferred from elimination. -/
theorem nondegenerate_feshbach_polynomial_equationIdeal_eq_coordinates
    [Fintype ι] [DecidableEq ι] [DecidableEq κ]
    (A : Matrix ι ι R) (U : Matrix ι κ R) (C : Matrix κ ι R)
    (B : Matrix κ κ R) [Invertible B] [Invertible (Feshbach.schur A U C B)] :
    blockEquationIdeal
        ((A.map (MvPolynomial.C : R →+* MvPolynomial (ι ⊕ κ) R)).mulVec
            (fun i ↦ MvPolynomial.X (Sum.inl i)) +
          (U.map MvPolynomial.C).mulVec (fun j ↦ MvPolynomial.X (Sum.inr j)))
        ((C.map MvPolynomial.C).mulVec (fun i ↦ MvPolynomial.X (Sum.inl i)) +
          (B.map MvPolynomial.C).mulVec (fun j ↦ MvPolynomial.X (Sum.inr j))) =
      blockEquationIdeal
        (fun i ↦ (MvPolynomial.X (Sum.inl i) : MvPolynomial (ι ⊕ κ) R))
        (fun j ↦ MvPolynomial.X (Sum.inr j)) := by
  let f : R →+* MvPolynomial (ι ⊕ κ) R := MvPolynomial.C
  let : Invertible ((Feshbach.schur A U C B).map f) :=
    Invertible.map f.mapMatrix (Feshbach.schur A U C B)
  rw [feshbach_polynomial_equationIdeal_eq]
  rw [blockEquationIdeal_mulVec_left]
  simp only [Matrix.mulVec_mulVec]
  exact blockEquationIdeal_add_mulVec_right _ _ _

/-- The ideal of all coordinate variables is exactly the kernel of the constant
coefficient homomorphism. This records ideal equality, not a zero-locus claim. -/
theorem coordinateEquationIdeal_eq_ker {σ : Type*} :
    Ideal.span (Set.range (MvPolynomial.X : σ → MvPolynomial σ R)) =
      RingHom.ker (MvPolynomial.constantCoeff : MvPolynomial σ R →+* R) := by
  apply le_antisymm
  · apply Ideal.span_le.mpr
    rintro _ ⟨i, rfl⟩
    exact RingHom.mem_ker.mpr (MvPolynomial.constantCoeff_X R i)
  · intro p hp
    have h : p - MvPolynomial.C (MvPolynomial.constantCoeff p) ∈
        Ideal.span (Set.range (MvPolynomial.X : σ → MvPolynomial σ R)) := by
      clear hp
      induction p using MvPolynomial.induction_on with
      | C a => simp
      | add p q hp hq =>
          simp only [map_add]
          rw [show p + q - (MvPolynomial.C (MvPolynomial.constantCoeff p) +
              MvPolynomial.C (MvPolynomial.constantCoeff q)) =
              (p - MvPolynomial.C (MvPolynomial.constantCoeff p)) +
                (q - MvPolynomial.C (MvPolynomial.constantCoeff q)) by abel]
          exact (Ideal.span (Set.range (MvPolynomial.X : σ → MvPolynomial σ R))).add_mem hp hq
      | mul_X p i _ =>
          simpa only [map_mul, MvPolynomial.constantCoeff_X, mul_zero, map_zero, sub_zero]
            using (Ideal.span (Set.range (MvPolynomial.X : σ → MvPolynomial σ R))).mul_mem_left
              p (Ideal.subset_span ⟨i, rfl⟩)
    have hz : MvPolynomial.constantCoeff p = 0 := RingHom.mem_ker.mp hp
    simpa only [hz, map_zero, sub_zero] using h

/-- Negating an equation block preserves its exact ideal. This algebraic
invariance does not preserve the sign of a real quadratic form. -/
theorem blockEquationIdeal_neg_left (p : ι → R) (q : κ → R) :
    blockEquationIdeal (-p) q = blockEquationIdeal p q := by
  apply le_antisymm
  · apply Ideal.span_le.mpr
    rintro _ ⟨i | j, rfl⟩
    · exact (blockEquationIdeal p q).neg_mem (mem_blockEquationIdeal_left p q i)
    · exact mem_blockEquationIdeal_right p q j
  · apply Ideal.span_le.mpr
    rintro _ ⟨i | j, rfl⟩
    · change p i ∈ blockEquationIdeal (-p) q
      simpa only [Pi.neg_apply, neg_neg] using
        (blockEquationIdeal (-p) q).neg_mem (mem_blockEquationIdeal_left (-p) q i)
    · exact mem_blockEquationIdeal_right (-p) q j

end FormalResearch.PositiveAtlas

namespace FormalResearch.PositiveAtlas.FeshbachEquationIdealFixture

/-- Polynomial equations of the real diagonal form `diag(a,b)`. -/
noncomputable def diagonalEquationIdeal (a b : ℝ) : Ideal (MvPolynomial (Unit ⊕ Unit) ℝ) :=
  blockEquationIdeal
    (fun _ : Unit ↦ MvPolynomial.C a * MvPolynomial.X (Sum.inl ()))
    (fun _ : Unit ↦ MvPolynomial.C b * MvPolynomial.X (Sum.inr ()))

/-- `diag(1,1)` and `diag(-1,1)` have exactly the same equation ideal. -/
theorem opposite_signs_same_equationIdeal :
    diagonalEquationIdeal (-1) 1 = diagonalEquationIdeal 1 1 := by
  simp only [diagonalEquationIdeal, map_neg, map_one, neg_mul, one_mul]
  change blockEquationIdeal
      (-(fun _ : Unit ↦ (MvPolynomial.X (Sum.inl ()) : MvPolynomial (Unit ⊕ Unit) ℝ)))
      (fun _ : Unit ↦ MvPolynomial.X (Sum.inr ())) = _
  exact blockEquationIdeal_neg_left
      (fun _ : Unit ↦ (MvPolynomial.X (Sum.inl ()) : MvPolynomial (Unit ⊕ Unit) ℝ))
      (fun _ : Unit ↦ MvPolynomial.X (Sum.inr ()))

/-- The shared ideal is the coordinate ideal, hence prime and radical. -/
theorem positive_diagonal_equationIdeal_isRadical :
    (diagonalEquationIdeal 1 1).IsRadical := by
  simpa only [diagonalEquationIdeal, map_one, one_mul] using
    (blockCoordinateEquationIdeal_isRadical (R := ℝ) (ι := Unit) (κ := Unit))

/-- Radicality of the very same equation ideal holds for the indefinite form. -/
theorem indefinite_diagonal_equationIdeal_isRadical :
    (diagonalEquationIdeal (-1) 1).IsRadical := by
  rw [opposite_signs_same_equationIdeal]
  exact positive_diagonal_equationIdeal_isRadical

/-- The quadratic form of `diag(1,1)` is nonnegative. -/
theorem positive_diagonal_nonnegative (x y : ℝ) : 0 ≤ x ^ 2 + y ^ 2 :=
  add_nonneg (sq_nonneg x) (sq_nonneg y)

/-- The quadratic form of `diag(-1,1)` has a negative direction. -/
theorem indefinite_diagonal_has_negative_direction :
    ∃ x y : ℝ, -(x ^ 2) + y ^ 2 < 0 := by
  exact ⟨1, 0, by norm_num⟩

/-- Exact shared radical equations do not certify quadratic-form positivity:
the two diagonal packets share their ideal and disagree on nonnegativity. -/
theorem radical_equationIdeal_does_not_determine_sign :
    diagonalEquationIdeal (-1) 1 = diagonalEquationIdeal 1 1 ∧
      (diagonalEquationIdeal (-1) 1).IsRadical ∧
      (∀ x y : ℝ, 0 ≤ x ^ 2 + y ^ 2) ∧
      (∃ x y : ℝ, -(x ^ 2) + y ^ 2 < 0) :=
  ⟨opposite_signs_same_equationIdeal, indefinite_diagonal_equationIdeal_isRadical,
    positive_diagonal_nonnegative, indefinite_diagonal_has_negative_direction⟩

end FormalResearch.PositiveAtlas.FeshbachEquationIdealFixture

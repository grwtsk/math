/-!
# Group presentations as data

`Group.Presentation` extends `Group.Generators` with a set of relators to form
a group presentation as data.

## Main definitions

* `Group.Presentation G ι`: a presentation `⟨ι | rel⟩` of `G`, extending `Group.Generators G ι`.

## Main results

* `Group.Presentation.lift_rel_eq_one`: a relator maps to the identity under the canonical
  surjection.
* `Group.isFinitelyPresented_iff_exists_finite_presentation`: A group is finitely presented
  if and only if it admits a presentation with finitely many generators and finitely many relators.

## Implementation notes

* Relators are a `Set` because (`Subgroup.normalClosure`, `PresentedGroup`) take a set as input.
* The kernel condition `lift_ker_eq_normalClosure` uses `toGenerators.lift` as in with
`Algebra.Presentation`.

* [D. F. Holt, S. Rees, C. E. Röver, *Groups, Languages and Automata*][HoltReesRover2017], §1

variable {G ι : Type*} [Group G]

/-- A group presentation is given by a generating family (`val : ι → G`) and a set of relators
(`rel : Set (FreeGroup ι)`) such that the normal subgroup generated by the relators is exactly the
kernel of the canonical `lift` from the free group on the generators to `G`. -/
structure Group.Presentation (G ι : Type*) [Group G] extends Group.Generators G ι where
  /-- The set of relators, as words in the free group; each `r ∈ rel` is read as `r = 1` in `G`. -/
  rel : Set (FreeGroup ι)
  /-- The kernel of the canonical surjection is equal to the normal closure of the relators. -/
  lift_ker_eq_normalClosure : toGenerators.lift.ker = Subgroup.normalClosure rel

namespace Group.Presentation

variable (P : Group.Presentation G ι)

/-- The presentation given by a surjective homomorphism `φ : FreeGroup ι →* G`,
together with a set of relators whose normal closure is the kernel of `φ`. -/
@[simps!]
def ofFreeGroupHom (φ : FreeGroup ι →* G) (hφ : Function.Surjective φ) (rel : Set (FreeGroup ι))
    (hrel : φ.ker = Subgroup.normalClosure rel) : Group.Presentation G ι where
  toGenerators := Group.Generators.ofFreeGroupHom φ hφ
  rel := rel
  lift_ker_eq_normalClosure := by rwa [Generators.ofFreeGroupHom_lift]

@[simp]
lemma ofFreeGroupHom_toGenerators (φ : FreeGroup ι →* G) (hφ : Function.Surjective φ)
    (rel : Set (FreeGroup ι)) (hrel : φ.ker = Subgroup.normalClosure rel) :
    (Group.Presentation.ofFreeGroupHom φ hφ rel hrel).toGenerators =
      Group.Generators.ofFreeGroupHom φ hφ := rfl

lemma ker_lift : P.lift.ker = Subgroup.normalClosure P.rel := P.lift_ker_eq_normalClosure

/-- A relator as a free group element always maps to the identity under the canonical surjection. -/
@[simp]
theorem lift_rel_eq_one {r : FreeGroup ι} (hr : r ∈ P.rel) : P.lift r = 1 := by
  rw [← MonoidHom.mem_ker, ker_lift]
  exact Subgroup.subset_normalClosure hr

lemma isFinitelyNormallyGenerated_ker_lift (hrel : P.rel.Finite) :
    P.lift.ker.IsFinitelyNormallyGenerated := ⟨P.rel, hrel, P.ker_lift.symm⟩

/-- `G`, with presentation `P`, is isomorphic to the `PresentedGroup` given by `P.rel`. -/
noncomputable def presentedGroupEquiv : PresentedGroup P.rel ≃* G :=
  (QuotientGroup.quotientMulEquivOfEq P.ker_lift.symm).trans
    (QuotientGroup.quotientKerEquivOfSurjective P.lift P.lift_surjective)

/-- A group with a finite presentation is finitely presented. -/
theorem isFinitelyPresented [Finite ι] (hrel : P.rel.Finite) : Group.IsFinitelyPresented G :=
  .of_surjective P.lift P.lift_surjective (P.isFinitelyNormallyGenerated_ker_lift hrel)

/-- A group is finitely presented if and only if it admits a presentation
with finitely many generators and finitely many relators. -/
theorem isFinitelyPresented_iff_exists_finite_presentation :
    Group.IsFinitelyPresented G ↔ ∃ (n : ℕ) (P : Group.Presentation G (Fin n)), P.rel.Finite := by
  constructor
  · rintro ⟨n, φ, hφ, S, hSfin, hS⟩
    exact ⟨n, .ofFreeGroupHom φ hφ S hS.symm, hSfin⟩
  · rintro ⟨n, P, hP⟩
    exact P.isFinitelyPresented hP

end Group.Presentation

import FormalResearch.FiniteLatticeArithmetic.DivisorRepresentation
import Mathlib.GroupTheory.Coset.Card

/-!
# The full subgroup lattice of a finite additive cyclic group

The subgroup at `d` consists precisely of elements annihilated by `d`.
It is proved equal to the subgroup generated by `M / d`; in particular,
this construction keeps the modulus-one case.

Originating direction: grwtsk contributors. Lean construction: Codex.
-/

namespace FormalResearch.FiniteLatticeArithmetic

variable {M : ℕ} [NeZero M]

/-- The additive subgroup annihilated by the divisor, in the actual group `ZMod M`. -/
def cyclicSubgroup (d : Divisor M) : AddSubgroup (ZMod M) where
  carrier := {x | (d : ℕ) • x = 0}
  zero_mem' := nsmul_zero _
  add_mem' := by intro x y hx hy; simp only [Set.mem_ofPred_eq] at *; simp [hx, hy]
  neg_mem' := by intro x hx; simp only [Set.mem_ofPred_eq] at *; simp [hx]

omit [NeZero M] in
@[simp] theorem mem_cyclicSubgroup (d : Divisor M) (x : ZMod M) :
    x ∈ cyclicSubgroup d ↔ addOrderOf x ∣ (d : ℕ) :=
  addOrderOf_dvd_iff_nsmul_eq_zero.symm

/-- The prescribed generator has exactly the declared subgroup order. -/
theorem generator_order (d : Divisor M) :
    addOrderOf ((M / (d : ℕ) : ℕ) : ZMod M) = (d : ℕ) := by
  rw [ZMod.addOrderOf_coe _ (NeZero.ne M),
    Nat.gcd_eq_right (Nat.div_dvd_of_dvd d.property)]
  exact Nat.div_div_self d.property (NeZero.ne M)

/-- The annihilator description equals the requested generated-subgroup description. -/
theorem cyclicSubgroup_eq_zmultiples (d : Divisor M) :
    cyclicSubgroup d = AddSubgroup.zmultiples ((M / (d : ℕ) : ℕ) : ZMod M) := by
  classical
  have hle : AddSubgroup.zmultiples ((M / (d : ℕ) : ℕ) : ZMod M) ≤ cyclicSubgroup d := by
    apply AddSubgroup.zmultiples_le_of_mem
    rw [mem_cyclicSubgroup, generator_order]
  have hcard : Nat.card (cyclicSubgroup d) ≤ (d : ℕ) := by
    change Nat.card {x : ZMod M // (d : ℕ) • x = 0} ≤ (d : ℕ)
    rw [Nat.card_eq_fintype_card, Fintype.card_subtype]
    exact IsAddCyclic.card_nsmul_eq_zero_le d.pos
  apply (AddSubgroup.eq_of_le_of_card_ge hle ?_).symm
  simpa only [Nat.card_zmultiples, generator_order] using hcard

theorem cyclicSubgroup_card (d : Divisor M) : Nat.card (cyclicSubgroup d) = (d : ℕ) := by
  rw [cyclicSubgroup_eq_zmultiples, Nat.card_zmultiples, generator_order]

/-- Divisibility is exactly subgroup inclusion. -/
theorem cyclicSubgroup_le_iff (d e : Divisor M) :
    cyclicSubgroup d ≤ cyclicSubgroup e ↔ d ≤ e := by
  constructor
  · intro h
    have hg : ((M / (d : ℕ) : ℕ) : ZMod M) ∈ cyclicSubgroup d := by
      rw [mem_cyclicSubgroup, generator_order]
    simpa only [mem_cyclicSubgroup, generator_order, Divisor.le_iff] using h hg
  · intro h x hx
    exact (mem_cyclicSubgroup e x).mpr (((mem_cyclicSubgroup d x).mp hx).trans h)

/-- Every subgroup occurs; this is a full representation, not an embedding. -/
theorem cyclicSubgroup_surjective : Function.Surjective (cyclicSubgroup (M := M)) := by
  intro H
  let d : Divisor M := ⟨Nat.card H, by
    simpa only [AddSubgroup.card_top, Nat.card_eq_fintype_card, ZMod.card] using
      (AddSubgroup.card_dvd_of_le (show H ≤ ⊤ from le_top))⟩
  have hle : H ≤ cyclicSubgroup d := by
    intro x hx
    rw [mem_cyclicSubgroup]
    have h := addOrderOf_dvd_natCard (⟨x, hx⟩ : H)
    rw [← addOrderOf_injective H.subtype Subtype.coe_injective (⟨x, hx⟩ : H)] at h
    exact h
  refine ⟨d, (AddSubgroup.eq_of_le_of_card_ge hle ?_).symm⟩
  simp only [cyclicSubgroup_card, d]
  exact le_rfl

/-- Positive divisor coordinates exhaust the subgroup lattice of the additive cyclic group. -/
noncomputable def cyclicSubgroupOrderIso : Divisor M ≃o AddSubgroup (ZMod M) :=
  { Equiv.ofBijective cyclicSubgroup
      ⟨fun d e h => le_antisymm
        ((cyclicSubgroup_le_iff d e).mp h.le)
        ((cyclicSubgroup_le_iff e d).mp h.ge), cyclicSubgroup_surjective⟩ with
    map_rel_iff' := cyclicSubgroup_le_iff _ _ }

@[simp] theorem cyclicSubgroup_gcd (d e : Divisor M) :
    cyclicSubgroup (d ⊓ e) = cyclicSubgroup d ⊓ cyclicSubgroup e :=
  cyclicSubgroupOrderIso.map_inf d e

@[simp] theorem cyclicSubgroup_lcm (d e : Divisor M) :
    cyclicSubgroup (d ⊔ e) = cyclicSubgroup d ⊔ cyclicSubgroup e :=
  cyclicSubgroupOrderIso.map_sup d e

/-- Principal color as the additive order of the difference. -/
noncomputable def cyclicColor (x y : ZMod M) : Divisor M :=
  ⟨addOrderOf (x - y), by
    simpa only [Nat.card_eq_fintype_card, ZMod.card] using addOrderOf_dvd_natCard (x - y)⟩

theorem cyclicColor_threshold (x y : ZMod M) (d : Divisor M) :
    cyclicColor x y ≤ d ↔ x - y ∈ cyclicSubgroup d :=
  (mem_cyclicSubgroup d (x - y)).symm

/-- Any natural representative gives the same exact principal color. -/
theorem cyclicColor_representative (x y : ZMod M) (r : ℕ)
    (hr : (r : ZMod M) = x - y) :
    ((cyclicColor x y : Divisor M) : ℕ) = M / Nat.gcd M r := by
  change addOrderOf (x - y) = _
  rw [← hr, ZMod.addOrderOf_coe _ (NeZero.ne M)]

/-- Two representatives of the same difference give the same numerical principal color. -/
theorem cyclicColor_representative_independent (x y : ZMod M) (r s : ℕ)
    (hr : (r : ZMod M) = x - y) (hs : (s : ZMod M) = x - y) :
    M / Nat.gcd M r = M / Nat.gcd M s :=
  (cyclicColor_representative x y r hr).symm.trans (cyclicColor_representative x y s hs)

/-- The standard representative specializes the representative-independent formula. -/
theorem cyclicColor_val (x y : ZMod M) :
    ((cyclicColor x y : Divisor M) : ℕ) = M / Nat.gcd M (x - y).val :=
  cyclicColor_representative x y _ (ZMod.natCast_zmod_val _)

end FormalResearch.FiniteLatticeArithmetic

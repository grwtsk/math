/-- Integer column populations on a specified finite vertex set. -/
abbrev Population (n : ℕ) := Fin n → ℤ

/-- Only the selected coordinate changes; loops are excluded explicitly. -/
def mutate {n : ℕ} (A : Matrix (Fin n) (Fin n) ℤ) (i : Fin n)
    (x : Population n) : Population n :=
  fun j => if j = i then -x i + ∑ k ∈ Finset.univ.erase i, A i k * x k else x j

theorem mutate_other {n : ℕ} (A : Matrix (Fin n) (Fin n) ℤ)
    (i j : Fin n) (x : Population n) (h : j ≠ i) : mutate A i x j = x j := by
  simp [mutate, h]

/-- Mutation is an involution for every integer loop-excluded adjacency. -/
theorem mutate_involutive {n : ℕ} (A : Matrix (Fin n) (Fin n) ℤ) (i : Fin n) :
    Function.Involutive (mutate A i) := by
  intro x
  funext j
  by_cases h : j = i
  · subst j
    have hs : (∑ k ∈ Finset.univ.erase i, A i k * mutate A i x k) =
        ∑ k ∈ Finset.univ.erase i, A i k * x k := by
      apply Finset.sum_congr rfl
      intro k hk
      rw [mutate_other A i k x (Finset.mem_erase.mp hk).1]
    simp only [mutate]
    change -(-x i + ∑ k ∈ Finset.univ.erase i, A i k * x k) +
      (∑ k ∈ Finset.univ.erase i, A i k * mutate A i x k) = x i
    rw [hs]
    ring
  · rw [mutate_other A i j _ h, mutate_other A i j _ h]

/-- Words are executed left to right on column populations. -/
def play {n : ℕ} (A : Matrix (Fin n) (Fin n) ℤ) (w : List (Fin n))
    (x : Population n) : Population n :=
  w.foldl (fun y i => mutate A i y) x

@[simp] theorem play_nil {n : ℕ} (A : Matrix (Fin n) (Fin n) ℤ)
    (x : Population n) : play A [] x = x := rfl

@[simp] theorem play_cons {n : ℕ} (A : Matrix (Fin n) (Fin n) ℤ)
    (i : Fin n) (w : List (Fin n)) (x : Population n) :
    play A (i :: w) x = play A w (mutate A i x) := rfl

theorem play_append {n : ℕ} (A : Matrix (Fin n) (Fin n) ℤ)
    (u v : List (Fin n)) (x : Population n) :
    play A (u ++ v) x = play A v (play A u x) := by
  simp [play, List.foldl_append]

/-- Reverse words are inverse operations, not merely observed return paths. -/
theorem play_reverse {n : ℕ} (A : Matrix (Fin n) (Fin n) ℤ)
    (w : List (Fin n)) (x : Population n) : play A w.reverse (play A w x) = x := by
  induction w generalizing x with
  | nil => rfl
  | cons i w ih =>
    rw [List.reverse_cons, play_append]
    change mutate A i (play A w.reverse (play A w (mutate A i x))) = x
    rw [ih]
    exact mutate_involutive A i x

/-- The standard basis seed at the indicated vertex. -/
def seed {n : ℕ} (i : Fin n) : Population n := fun j => if j = i then 1 else 0

/-- Reachability is generated by seeds and actual moves. -/
inductive Reachable {n : ℕ} (A : Matrix (Fin n) (Fin n) ℤ) : Population n → Prop
  | seed (i : Fin n) : Reachable A (FormalResearch.MutationGame.seed i)
  | step {x : Population n} (i : Fin n) : Reachable A x → Reachable A (mutate A i x)

theorem play_reachable {n : ℕ} (A : Matrix (Fin n) (Fin n) ℤ)
    (w : List (Fin n)) {x : Population n} (hx : Reachable A x) :
    Reachable A (play A w x) := by
  induction w generalizing x with
  | nil => exact hx
  | cons i w ih => exact ih (Reachable.step i hx)

/-- Certificate data has no completeness premise. -/
structure Certificate (n m : ℕ) where
  /-- The exact incoming-row graph data. -/
  adjacency : Matrix (Fin n) (Fin n) ℤ
  /-- The proposed unquotiented populations. -/
  states : Fin m → Population n
  /-- The listed index of each standard basis seed. -/
  seeds : Fin n → Fin m
  /-- The listed index after every generator at every state. -/
  transition : Fin m → Fin n → Fin m
  /-- A seed for the independently replayed word of each state. -/
  origin : Fin m → Fin n
  /-- Finite words executed from left to right. -/
  words : Fin m → List (Fin n)

/-- The complete decidable obligations, without assuming orbit completeness. -/
def Certificate.Valid {n m : ℕ} (c : Certificate n m) : Prop :=
  (∀ i, c.states (c.seeds i) = seed i) ∧
  (∀ s i, c.states (c.transition s i) = mutate c.adjacency i (c.states s)) ∧
  (∀ s, play c.adjacency (c.words s) (seed (c.origin s)) = c.states s) ∧
  Function.Injective c.states

instance {n m : ℕ} (c : Certificate n m) : Decidable c.Valid := by
  unfold Certificate.Valid Function.Injective
  infer_instance

/-- A finite, decidable check of seeds, full transitions, replays and distinctness. -/
def Certificate.check {n m : ℕ} (c : Certificate n m) : Bool := decide c.Valid

theorem certificate_check_sound {n m : ℕ} (c : Certificate n m)
    (hc : c.check = true) :
    ∀ x, Reachable c.adjacency x ↔ ∃ s, c.states s = x := by
  have hv : c.Valid := of_decide_eq_true hc
  rcases hv with ⟨hseed, hstep, hword, _⟩
  intro x
  constructor
  · intro hx
    induction hx with
    | seed i => exact ⟨c.seeds i, hseed i⟩
    | @step x i hx ih =>
      rcases ih with ⟨s, hs⟩
      exact ⟨c.transition s i, (hstep s i).trans (congrArg (mutate c.adjacency i) hs)⟩
  · rintro ⟨s, rfl⟩
    rw [← hword s]
    exact play_reachable _ _ (Reachable.seed _)

/-- The checked distinctness fixes the unquotiented orbit cardinality. -/
theorem certificate_cardinality {n m : ℕ} (c : Certificate n m)
    (hc : c.check = true) : (Finset.univ.image c.states).card = m := by
  have hv : c.Valid := of_decide_eq_true hc
  rw [Finset.card_image_of_injective _ hv.2.2.2]
  simp

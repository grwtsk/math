# Ontological role and structural level

Role answers what a concept is. It is independent of subject domain,
representation medium, and proof authority.
-/

namespace Mathematics.Taxonomy

inductive OntologicalRole
  | object
  | structure
  | relation
  | property
  | operation
  | construction
  | invariant
  | statement
  | proof
  | method
  | representation
  | certificate
  deriving DecidableEq, Inhabited, Fintype

/-- Compatibility with the earlier concept-assignment kind. The relation is
not identity: theorem statement, proof, and certificate remain distinct. -/
def KindRoleCompatible : Concept.Kind → OntologicalRole → Prop
  | .object, .object => True
  | .object, .structure => True
  | .structure, .structure => True
  | .property, .property => True
  | .relation, .relation => True
  | .operation, .operation => True
  | .operation, .construction => True
  | .proposition, .statement => True
  | .theorem, .statement => True
  | .invariant, .invariant => True
  | .dynamics, .operation => True
  | .representation, .representation => True
  | .policy, .certificate => True
  | .policy, .method => True
  | .family, .object => True
  | _, _ => False

instance (kind : Concept.Kind) (role : OntologicalRole) :
    Decidable (KindRoleCompatible kind role) := by
  cases kind <;> cases role <;> simp only [KindRoleCompatible] <;> infer_instance

theorem proof_ne_statement :
    OntologicalRole.proof ≠ OntologicalRole.statement := by decide

theorem representation_ne_object :
    OntologicalRole.representation ≠ OntologicalRole.object := by decide

end Mathematics.Taxonomy

instance (authority : Concept.TheoremAuthority) :
    Decidable (TheoremAuthoritySufficient authority) := by
  cases authority <;>
    simp only [TheoremAuthoritySufficient] <;>
    infer_instance

def StatusAuthorityCompatible (registry : Registry) : Prop :=
  ∀ taxon ∈ registry.taxa,
    taxon.status.RequiresTheoremAuthority →
      TheoremAuthoritySufficient taxon.authority.theoremAuthority

/-- Reviewed evidence is exact, typed, fully resolved, and source-history
subordinate. Nonempty sets alone are not accepted. -/
def EvidenceResolves (registry : Registry) : Prop :=
  ∀ taxon ∈ registry.taxa,
    ∃ evidence ∈ registry.evidence,
      evidence.id = taxon.evidenceId ∧
      evidence.conceptId = taxon.conceptId ∧
      evidence.primaryRole.role = taxon.primaryRole ∧
      evidence.primaryRole.witnesses.Nonempty ∧
      registry.WitnessesResolveAs evidence.primaryRole.witnesses .roleReview ∧
      evidence.principalDomain.domain = taxon.principalDomain ∧
      evidence.principalDomain.criteria.Nonempty ∧
      evidence.principalDomain.witnesses.Nonempty ∧
      registry.WitnessesResolveAs evidence.principalDomain.witnesses .domainReview ∧
      evidence.status.status = taxon.status ∧
      evidence.status.witnesses.Nonempty ∧
      registry.WitnessesResolveAs evidence.status.witnesses .statusReview ∧
      evidence.id.Valid ∧
      evidence.reviewer.Valid ∧
      evidence.digest.Valid ∧
      evidence.sourceHistorySubordinate = true ∧
      (∀ role ∈ taxon.secondaryRoles,
        ∃ item ∈ evidence.secondaryRoles,
          item.role = role ∧ item.witnesses.Nonempty ∧
            registry.WitnessesResolveAs item.witnesses .roleReview) ∧
      (∀ path ∈ taxon.secondaryDomains,
        ∃ item ∈ evidence.secondaryDomains,
          item.domain = path ∧ item.criteria.Nonempty ∧ item.witnesses.Nonempty ∧
            registry.WitnessesResolveAs item.witnesses .domainReview) ∧
      (∀ facetId ∈ taxon.facets,
        ∃ item ∈ evidence.facets,
          item.facetId = facetId ∧ item.witnesses.Nonempty ∧
            registry.WitnessesResolveAs item.witnesses .facetReview)

/-- Typed relation endpoints resolve in the semantic concept ledger. -/
def RelationsResolve
    (registry : Registry) (concepts : List Concept.Record) : Prop :=
  ∀ relation ∈ registry.relations,
    relation.source ∈ concepts.map (·.id) ∧
      relation.target ∈ concepts.map (·.id)

/-- A derived tree/DAG view remains explicitly navigation-only. -/
def ProjectionsResolve (registry : Registry) : Prop :=
  ∀ projection ∈ registry.projections,
    projection.rootDomain ∈ registry.domains.domains.map (·.id) ∧
      projection.authority = .navigationOnly

/-- Complete formal target for a reviewed mathematical taxonomy. -/

  registry.CoordinatesSeparated ∧
  registry.domains.UniqueIds ∧
  registry.domains.ParentsResolve ∧
  registry.domains.Acyclic ∧
  registry.RootValid ∧
  registry.facets.UniqueIds ∧
  registry.UniqueTaxonIds ∧
  registry.UniqueConceptTaxa ∧
  registry.UniqueEvidenceIds ∧
  registry.UniqueWitnessEvidenceIds ∧
  registry.UniqueRelationIds ∧
  registry.UniqueProjectionIds ∧
  registry.TaxaResolveConcepts concepts ∧
  registry.CoversConcepts concepts ∧
  registry.DomainPathsResolve ∧
  registry.FacetsResolve ∧
  registry.FacetRolesCompatible ∧
  registry.RolesCompatible concepts ∧
  registry.AuthorityAgrees concepts ∧
  registry.StatusAuthorityCompatible ∧
  registry.EvidenceResolves ∧
  registry.RelationsResolve concepts ∧
  registry.ProjectionsResolve

/-- A valid registry cannot repeat its principal domain as a secondary home. -/
theorem secondary_domain_not_duplicate_primary
    {registry : Registry} {concepts : List Concept.Record}
    (valid : registry.ValidFor concepts)
    {taxon : MathematicalTaxon} (member : taxon ∈ registry.taxa) :
    taxon.principalDomain ∉ taxon.secondaryDomains :=
  (valid.1 taxon member).2

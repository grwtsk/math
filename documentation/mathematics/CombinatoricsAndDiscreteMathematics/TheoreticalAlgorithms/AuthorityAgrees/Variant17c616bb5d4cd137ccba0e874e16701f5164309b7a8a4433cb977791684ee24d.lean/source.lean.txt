namespace Registry

/-- Taxonomic classification cannot silently replace the concept authority. -/
def AuthorityAgrees
    (registry : Registry) (concepts : List Concept.Record) : Prop :=
  ∀ taxon ∈ registry.taxa,
    ∀ concept ∈ concepts,
      taxon.conceptId = concept.id →
        taxon.authority = concept.authority

end Registry

end Mathematics.Taxonomy

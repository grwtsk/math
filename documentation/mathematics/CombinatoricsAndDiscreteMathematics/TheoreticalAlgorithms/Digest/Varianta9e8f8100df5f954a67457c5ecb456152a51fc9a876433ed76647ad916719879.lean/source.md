/-- A lower-case hexadecimal digest with its declared algorithm. -/
structure Digest where
  algorithm : String := "sha256"
  value : String
  deriving DecidableEq, Repr, Inhabited

/-- Canonical SHA-256 text uses exactly 64 lower-case hexadecimal digits. -/
def Digest.Valid (digest : Digest) : Prop :=
  digest.algorithm = "sha256" ∧
    digest.value.length = 64 ∧
    digest.value.toList.all (fun character ↦
      "0123456789abcdef".contains character) = true

/-- Digest equality is byte identity only. -/
def Digest.SameBytes (left right : Digest) : Prop := left = right

instance (left right : Digest) : Decidable (left.SameBytes right) :=
  inferInstanceAs (Decidable (left = right))

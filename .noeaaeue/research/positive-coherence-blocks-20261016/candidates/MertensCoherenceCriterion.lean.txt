/-- Square variation of the exact endpoint-defect increments before `n`. -/
def criticalSquareVariation (n : ℕ) : ℝ :=
  ∑ j ∈ Finset.Icc 1 (n - 1),
    (pickupMertensPrefix j * criticalEndpointGap j) ^ 2

/-- Positive part of the complete signed cumulative coherence. -/
def criticalCoherenceExcess (n : ℕ) : ℝ :=
  max 0 (criticalEndpointDefect n ^ 2 - criticalSquareVariation n)

/-- Logarithmically weighted positive coherence prefix. -/
def criticalCoherenceExcessSum (N : ℕ) : ℝ :=
  ∑ n ∈ Finset.Icc 1 N,
    Real.log (((n + 1 : ℕ) : ℝ) / (n : ℝ)) * criticalCoherenceExcess n

/-- Positive coherence mass on one dyadic collision block. -/
def dyadicCoherenceExcessBlock (j : ℕ) : ℝ :=
  ∑ n ∈ Finset.Ico (2 ^ j) (2 ^ (j + 1)),
    Real.log (((n + 1 : ℕ) : ℝ) / (n : ℝ)) * criticalCoherenceExcess n

def PositiveCoherenceExcessSubpower : Prop :=
  ∀ ε : ℝ, 0 < ε → ∃ C : ℝ, 0 ≤ C ∧
    ∀ N : ℕ, 1 ≤ N →
      criticalCoherenceExcessSum N ≤ C * ((N + 1 : ℕ) : ℝ) ^ ε

def DyadicPositiveCoherenceBlockSubpower : Prop :=
  ∀ ε : ℝ, 0 < ε → ∃ C : ℝ, 0 ≤ C ∧
    ∀ j : ℕ, dyadicCoherenceExcessBlock j ≤ C * (2 : ℝ) ^ (ε * j)

def PositiveCoherenceExcessPolylog : Prop :=
  ∃ C r : ℝ, 0 ≤ C ∧ 0 ≤ r ∧
    ∀ N : ℕ, 1 ≤ N →
      criticalCoherenceExcessSum N ≤
        C * (1 + Real.log ((N + 1 : ℕ) : ℝ)) ^ r

/-- Written source bridge.  The accompanying derivation proves these
relationships; this candidate keeps them explicit until native admission. -/
def PositiveCoherenceGrowthBridge : Prop :=
  (PositiveCoherenceExcessSubpower ↔ PickupEnergySubexp) ∧
  (DyadicPositiveCoherenceBlockSubpower ↔ PickupEnergySubexp) ∧
  (PositiveCoherenceExcessPolylog ↔ PickupCrossPoly)

theorem riemannHypothesis_of_positiveCoherenceExcessSubpower

  exact riemannHypothesis_of_pickupEnergySubexp (hBridge.1.mp h)

/-- General square-Mellin coefficient premise. -/
def MertensSquareSummableAt (p : ℝ) : Prop :=
  Summable (fun n : ℕ =>
    if n = 0 then 0 else
      pickupMertensPrefix n ^ 2 * (n : ℝ) ^ (-p))

/-- The direct discrete Abel series. -/
def mertensAbelSeries (s : ℂ) : ℂ :=
  ∑' n : ℕ,
    if n = 0 then 0 else
      (pickupMertensPrefix n : ℂ) *
        ((n : ℂ) ^ (-s) - ((n + 1 : ℕ) : ℂ) ^ (-s))

/-- Exact target for the locally uniform Cauchy--Schwarz argument. -/
def MertensSquareMellinBridge (p : ℝ) : Prop :=
  MertensSquareSummableAt p →
    AnalyticOnNhd mertensAbelSeries
      {s : ℂ | (p - 1) / 2 < s.re}

/-- Exact zero-free consumer intended after the regularized-zeta identity has
been transported to the Abel series. -/
def MertensSquareZeroFreeBridge (p : ℝ) : Prop :=
  MertensSquareSummableAt p →
    ∀ s : ℂ,
      (p - 1) / 2 < s.re →
      (¬ ∃ n : ℕ, s = -2 * (n + 1)) →
      s ≠ 1 →
      riemannZeta s ≠ 0

/-- One-third specialization of the exact boundary arithmetic. -/
theorem eightThirds_boundary :
    (((8 / 3 : ℝ) - 1) / 2) = 5 / 6 := by
  norm_num

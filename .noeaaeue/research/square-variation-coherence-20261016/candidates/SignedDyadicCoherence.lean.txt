/-- One-sided subpower upper growth for a sequence indexed by dyadic scale. -/
def DyadicSubpowerUpper (f : ℕ → ℝ) : Prop :=
  ∀ ε : ℝ, 0 < ε → ∃ A : ℝ, 0 ≤ A ∧
    ∀ j : ℕ, f j ≤ A * (2 : ℝ) ^ (ε * j)

/-- A fixed additive discrepancy does not change the dyadic subpower class. -/
theorem dyadicSubpowerUpper_iff_of_uniform_difference
    {f g : ℕ → ℝ} {D : ℝ} (hD : 0 ≤ D)
    (hfg : ∀ j, f j ≤ g j)
    (hgf : ∀ j, g j ≤ f j + D) :
    DyadicSubpowerUpper f ↔ DyadicSubpowerUpper g := by
  constructor
  · intro hf ε hε
    obtain ⟨A,hA,hbound⟩ := hf ε hε
    refine ⟨A + D, add_nonneg hA hD, ?_⟩
    intro j
    have hpow : 1 ≤ (2 : ℝ) ^ (ε * j) := by
      exact Real.one_le_rpow (by norm_num) (mul_nonneg hε.le (Nat.cast_nonneg j))
    calc
      g j ≤ f j + D := hgf j
      _ ≤ A * (2 : ℝ) ^ (ε * j) + D := add_le_add_right (hbound j) D
      _ ≤ (A + D) * (2 : ℝ) ^ (ε * j) := by
        nlinarith [mul_nonneg hD hpow]
  · intro hg ε hε
    obtain ⟨A,hA,hbound⟩ := hg ε hε
    exact ⟨A,hA,fun j => (hfg j).trans (hbound j)⟩

end GRWTSK.NumberTheory.Moebius.SignedDyadicCoherence

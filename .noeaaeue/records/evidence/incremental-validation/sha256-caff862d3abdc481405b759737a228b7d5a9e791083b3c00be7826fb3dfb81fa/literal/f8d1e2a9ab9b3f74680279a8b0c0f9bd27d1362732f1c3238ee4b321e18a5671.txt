"""Python-only hostile controls; these do not establish Lean proof acceptance."""
import argparse
import copy
import importlib.util
import json
from pathlib import Path
import tempfile
import types
import unittest

ROOT = Path(__file__).resolve().parents[2]


def module(name):
    spec = importlib.util.spec_from_file_location(name, ROOT / '.noeaaeue/scripts' / (name + '.py'))
    value = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(value)
    return value


class JacobiReplayGuards(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.driver = module('check_third_cyclotomic_jacobi')
        cls.guard = module('check_math_trust')

    def test_positive_only_and_negative_import_closures_are_fresh(self):
        bodies = {
            'GRWTSK/Base.lean': b'/- import GRWTSK.Fake -/\n',
            'GRWTSK/Selected.lean': b'import GRWTSK.Base\n',
            'GRWTSK/PositiveOnly.lean': b'',
            'GRWTSK/Positive.lean': b'public import GRWTSK.Selected GRWTSK.PositiveOnly\n',
            'GRWTSK/NegativeDependency.lean': b'',
            'GRWTSK/Checks/Negative/Bad.lean': b'import GRWTSK.Positive GRWTSK.NegativeDependency\n',
        }
        fake = types.SimpleNamespace(read_source=lambda _, name: bodies[name],
                                     project_imports=self.guard.project_imports)
        frozen, library, positive = self.driver.freeze_selected_sources(
            fake, Path('/unused'), ['GRWTSK/Selected.lean'], ['GRWTSK/Positive.lean'],
            [('GRWTSK/Checks/Negative/Bad.lean', ('unused',))])
        self.assertEqual(library, ['GRWTSK/Base.lean', 'GRWTSK/Selected.lean'])
        self.assertEqual(positive, ['GRWTSK/PositiveOnly.lean', 'GRWTSK/Positive.lean',
                                    'GRWTSK/NegativeDependency.lean'])
        self.assertEqual(frozen, bodies)
        self.assertNotIn('GRWTSK/Checks/Negative/Bad.lean', library + positive)

    def test_cyclic_and_negative_accepted_dependencies_reject(self):
        for bodies in (
            {'GRWTSK/A.lean': b'import GRWTSK.B\n', 'GRWTSK/B.lean': b'import GRWTSK.A\n'},
            {'GRWTSK/A.lean': b'import GRWTSK.Checks.Negative.Bad\n'},
        ):
            with self.subTest(bodies=bodies):
                fake = types.SimpleNamespace(read_source=lambda _, name: bodies[name],
                                             project_imports=self.guard.project_imports)
                with self.assertRaises(RuntimeError):
                    self.driver.freeze_selected_sources(fake, Path('/unused'),
                                                        ['GRWTSK/A.lean'], [], [])

    def test_missing_local_dependency_cannot_reuse_an_object(self):
        bodies = {'GRWTSK/A.lean': b'import GRWTSK.Missing\n'}
        fake = types.SimpleNamespace(read_source=lambda _, name: bodies[name],
                                     project_imports=self.guard.project_imports)
        with self.assertRaises(KeyError):
            self.driver.freeze_selected_sources(fake, Path('/unused'), ['GRWTSK/A.lean'], [], [])

    def test_intended_negative_and_hostile_near_misses(self):
        expected = ('producer P hgood', 'lambda ^ 2 ∣ J + 1', 'lambda ^ 2 ∣ J - 1')
        diagnostic = ('Bad.lean:12:2: error: Type mismatch\n  producer P hgood\nhas type\n'
                      '  lambda ^ 2 ∣ J + 1\nbut is expected to have type\n  lambda ^ 2 ∣ J - 1\n')
        self.driver.validate_intended_rejection(self.guard, 1, diagnostic, expected)
        self.driver.validate_intended_rejection(self.guard, 1,
            diagnostic.replace('error:', 'error(lean.typeMismatch):'), expected)
        cases = [
            (0, diagnostic), (-9, diagnostic), (2, diagnostic),
            (1, diagnostic.replace('Type mismatch', 'Unknown identifier')),
            (1, diagnostic + 'Bad.lean:99:1: error: second failure\n'),
            (1, diagnostic + 'Bad.lean:99:1: error(lean.unknownIdentifier): second failure\n'),
            (1, diagnostic.replace('lambda ^ 2 ∣ J - 1', 'lambda ^ 2 ∣ J + 2')),
            (1, diagnostic.replace('but is expected to have type', 'unrelated prose')),
            (1, diagnostic.replace('producer P hgood', 'differentProducer') +
             'Bad.lean:90:1: warning: producer P hgood\n'),
        ]
        for status, text in cases:
            with self.subTest(status=status, text=text):
                with self.assertRaises(RuntimeError):
                    self.driver.validate_intended_rejection(self.guard, status, text, expected)

    def test_generated_object_family_drift_rejects(self):
        for mutation in ('unchanged', 'replace', 'truncate', 'delete-main', 'replace-private',
                         'delete-private', 'add-server', 'link-main', 'link-private'):
            with self.subTest(mutation=mutation), tempfile.TemporaryDirectory(
                    prefix='jacobi-objects-', dir=ROOT / '.noeaaeue/.local') as raw:
                folder = Path(raw)
                main = folder / 'Selected.olean'
                private = folder / 'Selected.olean.private'
                main.write_bytes(b'original')
                private.write_bytes(b'private')
                frozen = {str(main): self.driver.object_family(main)}
                if mutation == 'replace':
                    main.write_bytes(b'changed')
                elif mutation == 'truncate':
                    main.write_bytes(b'')
                elif mutation == 'delete-main':
                    main.unlink()
                elif mutation == 'replace-private':
                    private.write_bytes(b'changed')
                elif mutation == 'delete-private':
                    private.unlink()
                elif mutation == 'add-server':
                    (folder / 'Selected.olean.server').write_bytes(b'extra')
                elif mutation.startswith('link-'):
                    target = main if mutation == 'link-main' else private
                    other = folder / 'same-bytes'
                    other.write_bytes(target.read_bytes())
                    target.unlink()
                    target.symlink_to(other)
                if mutation == 'unchanged':
                    self.driver.verify_project_objects(frozen)
                else:
                    with self.assertRaises(RuntimeError):
                        self.driver.verify_project_objects(frozen)

    def test_namespace_absence_and_symlink_receipt_guards(self):
        name = '.noeaaeue/evidence/NumberTheory/Cyclotomic/third-cyclotomic-jacobi-test.json'
        with tempfile.TemporaryDirectory(prefix='jacobi-receipt-', dir=ROOT / '.noeaaeue/.local') as raw:
            root = Path(raw)
            path = self.driver.receipt_destination(root, name)
            path.parent.mkdir(parents=True)
            path.write_bytes(b'preserved receipt')
            with self.assertRaises(RuntimeError):
                self.driver.receipt_destination(root, name)
            path.unlink()
            path.symlink_to(root / 'absent')
            with self.assertRaises(RuntimeError):
                self.driver.receipt_destination(root, name)
            path.unlink()
            for bad in ('/absolute.json', '../escape.json', name.replace('jacobi-', 'characters-'),
                        name.replace('/evidence/', '//evidence/')):
                with self.subTest(path=bad), self.assertRaises(RuntimeError):
                    self.driver.receipt_destination(root, bad)
            path.parent.rmdir()
            path.parent.symlink_to(root)
            with self.assertRaises(RuntimeError):
                self.driver.receipt_destination(root, name)

    def test_memory_budget_is_explicit_and_bounded(self):
        for value in ('1024', '12288', '65536'):
            self.assertEqual(self.driver.memory_budget(value), int(value))
        for value in ('0', '1023', '65537', '-1', '1.5', 'NaN', '12g'):
            with self.subTest(value=value), self.assertRaises(argparse.ArgumentTypeError):
                self.driver.memory_budget(value)

    def test_current_census_rejects_forged_owner_or_axiom(self):
        owner, digest = 'GRWTSK.SelectedJacobiGuardFixture', '1' * 64
        row = {'module': owner, 'name': owner + '.ok', 'kind': 'theorem', 'type': 'True',
               'level_params': [], 'unsafe': False, 'partial': False,
               'known_generated_runtime_helper': False, 'mathematical_admission': True,
               'runtime_helper_parent': '', 'runtime_helper_same_owner': False,
               'runtime_helper_type_equal': False, 'runtime_helper_level_params_equal': False,
               'runtime_helper_profile_source_sha256': '', 'runtime_helper_owner_source_sha256': digest,
               'direct_type_references_to_runtime_helpers': [],
               'direct_value_references_to_runtime_helpers': [],
               'direct_value_reference_to_runtime_helper': False, 'axioms': ['propext']}

        def validate(declaration):
            diagnostic = '\n'.join(['MATH_TRUST_DECL::' + json.dumps(declaration),
                'MATH_TRUST_MODULE::' + json.dumps({'module': owner, 'declarations': 1}),
                'MATH_TRUST_COUNT::1', 'MATH_TRUST_MODULE_COUNT::1', ''])
            return self.guard.owned_audit_protocol.validate_census(
                diagnostic, [owner], {owner: digest}, {}, self.guard.census_records)

        self.assertEqual(len(validate(row)[0]), 1)
        for changed in ({'runtime_helper_owner_source_sha256': '2' * 64},
                        {'axioms': ['sorryAx']}, {'partial': True}, {'unsafe': True},
                        {'mathematical_admission': False}, {'known_generated_runtime_helper': True}):
            with self.subTest(changed=changed):
                bad = copy.deepcopy(row)
                bad.update(changed)
                with self.assertRaises(RuntimeError):
                    validate(bad)


if __name__ == '__main__':
    unittest.main()

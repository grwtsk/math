def canonical(v):
    return json.dumps(v, sort_keys=True, separators=(',', ':'), ensure_ascii=True, allow_nan=False).encode('utf-8')

def sha(b):
    return hashlib.sha256(b).hexdigest()

def load_json(b):
    def pairs(items):
        d = {}
        for k, v in items:
            if k in d:
                raise ValueError('duplicate JSON key ' + k)
            d[k] = v
        return d
    return json.loads(b.decode('utf-8'), object_pairs_hook=pairs, parse_constant=lambda v: (_ for _ in ()).throw(ValueError(v)))

def read_literal(path):
    current = ROOT
    for part in pathlib.PurePosixPath(path).parts:
        current = current / part
        if stat.S_ISLNK(current.lstat().st_mode):
            raise ValueError('symlink literal input rejected: ' + path)
    if not stat.S_ISREG(current.lstat().st_mode):
        raise ValueError('nonregular literal input: ' + path)
    return current.read_bytes()

def write_new(path, data):
    if not (path == NEW_SCHEMA or path.startswith(OUT)):
        raise ValueError('outside assigned output boundary: ' + path)
    target = ROOT / path
    target.parent.mkdir(parents=True, exist_ok=True)
    with target.open('xb') as f:
        f.write(data)

def pointer(v, loc):
    if not loc.startswith('/'):
        raise ValueError('not a JSON pointer')
    for part in loc[1:].split('/'):
        key = part.replace('~1','/').replace('~0','~')
        v = v[int(key)] if isinstance(v, list) else v[key]
    return v

def token(key):
    return str(key).replace('~','~0').replace('/','~1')

def has_float(value):
    if isinstance(value,float): return True
    if isinstance(value,dict): return any(has_float(v) for v in value.values())
    if isinstance(value,list): return any(has_float(v) for v in value)
    return False

def find_members(value, path='', inventory=False):
    found = []
    if isinstance(value,dict):
        paths = [(k,value[k]) for k in PATH_KEYS if isinstance(value.get(k),str) and value[k]]

        if path and paths and digests:

                raise ValueError('unexpected nondeterministic float in selected source-binding member')
            found.append((path,value,paths[0][1],'explicit path field and explicit SHA-256 field in the same source record'))
        for k,v in value.items():
            found.extend(find_members(v,path+'/'+token(k)))
    elif isinstance(value,list):
        for k,v in enumerate(value):
            found.extend(find_members(v,path+'/'+str(k)))
    return found

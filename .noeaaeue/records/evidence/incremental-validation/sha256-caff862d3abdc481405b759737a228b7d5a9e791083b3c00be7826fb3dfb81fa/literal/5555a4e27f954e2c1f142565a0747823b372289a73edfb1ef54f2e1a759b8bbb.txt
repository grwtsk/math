def canon(x):
    return (json.dumps(x, sort_keys=True, ensure_ascii=False, separators=(',', ':'), allow_nan=False) + '\n').encode()

def sha(b):
    return hashlib.sha256(b).hexdigest()

def digest_file(p):
    h = hashlib.sha256()
    size = 0
    with p.open('rb') as f:
        for b in iter(lambda: f.read(1048576), b''):
            h.update(b)
            size += len(b)
    return h.hexdigest(), size

def regular(p):
    if not p.is_absolute():
        p = ROOT / p
    p.relative_to(ROOT)
    for parent in (p,) + tuple(p.parents):
        if parent == ROOT.parent:
            break
        if parent.is_symlink():
            raise ValueError('symlink source: ' + str(parent))
    if not stat.S_ISREG(p.stat().st_mode):
        raise ValueError('not regular: ' + str(p))
    return p

def load(p):
    def unique(pairs):
        d = {}
        for k, v in pairs:
            if k in d:
                raise ValueError('duplicate JSON key: ' + k)
            d[k] = v
        return d
    return json.loads(regular(p).read_bytes(), object_pairs_hook=unique)

def rel(p):
    return str(p.relative_to(ROOT))

def safe_label(s):
    if unicodedata.normalize('NFC', s) != s or '\\' in s:
        raise ValueError('unnormalized label')
    q = PurePosixPath(s)
    if q.is_absolute() or str(q) != s or any(x in ('', '.', '..') for x in q.parts):
        raise ValueError('unsafe label: ' + s)
    return s

def asset(p, role, expected=None, length=None):
    p = regular(p)
    label = safe_label('math/' + rel(p))
    b = p.read_bytes()
    h = sha(b)
    if expected is not None and h != expected:
        raise ValueError('digest mismatch: ' + str(p))
    if length is not None and len(b) != length:
        raise ValueError('length mismatch: ' + str(p))
    row = {'label': label, 'original_path': rel(p), 'sha256': h, 'size_bytes': len(b), 'role': role}
    if label in assets and assets[label]['sha256'] != h:
        raise ValueError('source changed during capture: ' + label)
    if label not in assets:
        assets[label] = row
        payloads[label] = b
    return row

def write_new(path, b):
    p = ROOT / path
    q = PurePosixPath(path)
    if not allowed_output(path) or '..' in q.parts or p.suffix == '.lean':
        raise ValueError('unauthorized output')
    if p.exists() or path in planned:
        raise ValueError('create-only collision: ' + path)
    for parent in (p,) + tuple(p.parents):
        if parent == ROOT.parent:
            break
        if parent.is_symlink():
            raise ValueError('symlink output')
    if publishing:
        p.parent.mkdir(parents=True, exist_ok=True)
        with p.open('xb') as f:
            f.write(b)
    else:
        planned[path] = b
    created.append({'path': path, 'sha256': sha(b), 'size_bytes': len(b)})

# Closed schema for every emitted envelope and each explicitly authored data shape.
def shape(v, key=''):
    if v is None:
        return {'type': 'null'}
    if isinstance(v, bool):
        return {'type': 'boolean', 'const': v}
    if isinstance(v, int):
        return {'type': 'integer', 'minimum': -9007199254740991 if key == 'exit_code' else 0, 'maximum': 9007199254740991}
    if isinstance(v, str):
        if key.endswith('sha256') or key == 'sha256':
            return {'type': 'string', 'pattern': '^[0-9a-f]{64}$'}
        if key in ('path', 'retained_path', 'manifest', 'selected_input_manifest'):
            return {'type': 'string', 'pattern': '^(?!/)(?!.*(?:^|/)\\.\\.(?:/|$))(?!.*\\\\).+$'}
        return {'type': 'string'}
    if isinstance(v, list):
        choices = []
        for x in v:
            z = shape(x, key)
            if z not in choices:
                choices.append(z)
        return {'type': 'array', 'items': choices[0] if len(choices) == 1 else {'anyOf': choices} if choices else {'type': 'string'}}
    if isinstance(v, dict):
        return {'type': 'object', 'properties': {k: shape(x, k) for k, x in v.items()}, 'required': sorted(v), 'additionalProperties': False}
    raise ValueError('noncanonical authoring type')

#!/usr/bin/env python3
"""Replay actual cubic Jacobi congruence and sign controls in good maximal quotients from verified selected objects.

Pinned dependency objects are reused; this is not a Mathlib source rebuild,
whole-repository admission, independent review, or unconditional RH proof.
"""
import argparse
import datetime
import fcntl
import hashlib
import importlib.util
import json
import os
from pathlib import Path
import re
import stat
import subprocess
import tempfile
import time

from selected_incremental import SelectedReplayCache, freeze_incremental_policy

ROOT = Path(__file__).resolve().parents[2]
LIBRARY = [
    "GRWTSK/NumberTheory/Cyclotomic/ThirdCyclotomicJacobi.lean",
]
POSITIVE = [
    "GRWTSK/NumberTheory/Cyclotomic/Checks/Positive/ThirdCyclotomicJacobi.lean",
]
NEGATIVE = [
    ("GRWTSK/NumberTheory/Cyclotomic/Checks/Negative/ThirdCyclotomicJacobiWrongSign.lean",
     ("lambda_sq_dvd_cubicJacobi_add_one P hgood",
      "jacobiSum (cubicChar P hgood) (cubicChar P hgood)", "+ 1", "- 1")),
]

AUDIT = "GRWTSK/Logic/ProofChecking/Checks/OwnedDeclarationAudit.lean"


def require(value, message):
    if not value:
        raise RuntimeError(message)


def sha(data):
    return hashlib.sha256(data).hexdigest()


def command(*args):
    return subprocess.check_output(args, cwd=ROOT, text=True).strip()


def memory_budget(value):
    if not isinstance(value, str) or not re.fullmatch(r"[0-9]+", value):
        raise argparse.ArgumentTypeError("memory budget must be an integer MiB value")
    result = int(value)
    if not 1024 <= result <= 65536:
        raise argparse.ArgumentTypeError("memory budget must be between 1024 and 65536 MiB")
    return result


def freeze_selected_sources(guard, root, libraries, positives, negatives):
    """Fresh local closure, including positive-only and negative-import dependencies.

    Negative source bodies are frozen for later rejection, never compiled as
    accepted modules. Cycles and negative-source imports fail closed.
    """
    frozen, order, visiting = {}, [], set()
    negative_paths = {path for path, _ in negatives}

    def imports(body):
        for module in guard.project_imports(body.decode("utf-8")):
            require(re.fullmatch(r"GRWTSK(?:\.[^\W\d][\w']*)*", module),
                    "unsupported local module: " + module)
            yield module.replace(".", "/") + ".lean"

    def visit(relative):
        require(relative not in negative_paths and "/Checks/Negative/" not in relative,
                "rejected control cannot be an accepted dependency: " + relative)
        require(relative not in visiting, "cyclic local source dependency: " + relative)
        if relative in frozen:
            return
        visiting.add(relative)
        body = guard.read_source(root, relative)
        for dependency in imports(body):
            visit(dependency)
        frozen[relative] = body
        order.append(relative)
        visiting.remove(relative)

    for relative in libraries:
        visit(relative)
    library_order = list(order)
    for relative in positives:
        visit(relative)
    for relative, _ in negatives:
        body = guard.read_source(root, relative)
        for dependency in imports(body):
            visit(dependency)
        frozen[relative] = body
    return frozen, library_order, order[len(library_order):]


def validate_intended_rejection(guard, returncode, diagnostic, expected):
    """A sole actual type diagnostic must contain every frozen sign fragment.

    A marker in a later warning/info message cannot satisfy the oracle. Native
    signal/timeout failures and a second coded/plain error are not acceptance.
    """
    errors = guard.error_messages(diagnostic)
    require(returncode == 1 and len(errors) == 1
            and "type mismatch" in errors[0].casefold(), "not one intended type error")
    headers = list(re.finditer(r"^.*?\berror(?:\([^()\r\n]+\))?:[^\r\n]*", diagnostic, re.M))
    require(len(headers) == 1, "ambiguous type-error diagnostic")
    tail = diagnostic[headers[0].start():]
    following = re.search(r"^.*?\b(?:warning|info)(?:\([^()\r\n]+\))?:", tail, re.M)
    if following:
        tail = tail[:following.start()]
    normalized = " ".join(tail.split())
    require("but is expected to have type" in normalized
            and expected and all(" ".join(fragment.split()) in normalized for fragment in expected),
            "type error did not match every frozen expected fragment")


def receipt_destination(root, name):
    relative = Path(name)
    require(relative.as_posix() == name and not relative.is_absolute()
            and ".." not in relative.parts
            and relative.parts[:4] == (".noeaaeue", "evidence", "NumberTheory", "Cyclotomic")
            and relative.name.startswith("third-cyclotomic-jacobi-")
            and relative.suffix == ".json", "invalid receipt destination")
    destination = root / relative
    require(not os.path.lexists(destination), "receipt already exists")
    cursor = root
    for component in relative.parts[:-1]:
        cursor /= component
        if os.path.lexists(cursor):
            require(stat.S_ISDIR(cursor.lstat().st_mode), "receipt ancestor is linked/non-directory")
    return destination


def object_family(path):
    """Exact generated Lean import-object family; additional members are drift too."""
    candidates = [path, Path(str(path) + ".private"), Path(str(path) + ".server")]
    require(path.is_file() and not path.is_symlink(), "missing or linked project object: " + str(path))
    identities = []
    for item in candidates:
        if item.exists() or item.is_symlink():
            require(item.is_file() and not item.is_symlink(), "linked/nonregular project object: " + str(item))
            raw = item.read_bytes()
            identities.append({"path": str(item), "bytes": len(raw), "sha256": sha(raw)})
    return identities


def verify_project_objects(identities):
    for main, expected in identities.items():
        require(object_family(Path(main)) == expected, "generated project object drift: " + main)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--receipt", help="absent repository-relative durable JSON destination")
    parser.add_argument("--memory-mib", type=memory_budget, default=12288,
                        help="per-Lean-process MiB limit (1024..65536; default12288)")
    parser.add_argument("--cold", action="store_true",
                        help="recompile and revalidate every selected module without cache reuse")
    args = parser.parse_args()
    destination = None
    if args.receipt:
        destination = receipt_destination(ROOT, args.receipt)
    require(all(expected and all(fragment and fragment != "STILL_TO_FREEZE" for fragment in expected)
                for _, expected in NEGATIVE), "negative diagnostic must be frozen before native replay")
    spec = importlib.util.spec_from_file_location("math_guard", ROOT / ".noeaaeue/scripts/check_math_trust.py")
    guard = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(guard)
    lock_path = ROOT / ".noeaaeue/.local/math-trust/replay.lock"
    lock_path.parent.mkdir(parents=True, exist_ok=True)
    with lock_path.open("a+") as lock:
        try:
            fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        except BlockingIOError:
            print("DEFERRED: another native replay holds the shared lock", flush=True)
            return 2
        require(not guard.competing_jobs(ROOT), "native replay busy")
        return replay(guard, destination, args.memory_mib, cold=args.cold)


def replay(guard, destination, memory_mib=12288, cold=False):
    require((ROOT / "lean-toolchain").read_text().strip() == "leanprover/lean4:v4.33.0",
            "Lean pin mismatch")
    pins = {p["name"]: p["rev"] for p in json.loads((ROOT / "lake-manifest.json").read_text())["packages"]}
    require(pins["mathlib"] == guard.MATHLIB_REV, "Mathlib pin mismatch")
    require(set(guard.ALLOWED_AXIOMS) == {"propext", "Classical.choice", "Quot.sound"},
            "selected Jacobi closure permits only the three standard axioms")
    lean = str(Path(command("elan", "which", "lean")).resolve())
    version = command(lean, "--version")
    require(guard.LEAN_VERSION in version and guard.LEAN_COMMIT in version, "compiler mismatch")
    frozen_bytes, library_order, positive_order = freeze_selected_sources(
        guard, ROOT, LIBRARY, POSITIVE, NEGATIVE)
    for rel in dict.fromkeys([*guard.EXTRAS, ".noeaaeue/scripts/check_third_cyclotomic_jacobi.py"]):
        frozen_bytes[rel] = guard.read_source(ROOT, rel)
    base = ROOT / ".noeaaeue/.local/third-cyclotomic-jacobi-replay"
    base.mkdir(parents=True, exist_ok=True)
    run = Path(tempfile.mkdtemp(prefix="replay-", dir=base))
    source, build = run / "source", run / "build"
    source.mkdir()
    build.mkdir()
    freeze_incremental_policy(guard, ROOT, frozen_bytes)
    for rel, body in frozen_bytes.items():
        path = source / rel
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_bytes(body)
        path.chmod(0o444)
    paths, removed = guard.dependency_search_path(command("lake", "env", "printenv", "LEAN_PATH"), ROOT, lean)
    require(paths, "no dependency paths")
    env = dict(os.environ, LEAN_PATH=os.pathsep.join([str(build)] + paths))
    env.pop("LEAN_SRC_PATH", None)
    receipt = {
        "schema": "third-cyclotomic-jacobi-selected-replay-v1", "outcome": "running",
        "at": datetime.datetime.now(datetime.timezone.utc).isoformat(),
        "head_observed": command("git", "rev-parse", "HEAD"),
        "source_hashes": {rel: sha(body) for rel, body in frozen_bytes.items()},
        "compiler_sha256": sha(Path(lean).read_bytes()), "lean_version": version,
        "pins": pins,
        "library_order": library_order, "accepted_control_order": positive_order,
        "negative_oracles": {path: list(expected) for path, expected in NEGATIVE},
        "trust": 0, "fatal_warnings": True,
        "lean_memory_limit_mib": memory_mib,
        "canonical_project_objects_excluded": True, "dependency_objects_reused": True,
        "full_repository_build": False, "mathlib_source_rebuild": False,
        "independent_review": False, "rh_proved": False,
        "allowed_axioms": sorted(guard.ALLOWED_AXIOMS),
        "dependency_search_paths": paths, "removed_project_search_paths": removed,
        "dependency_cache_policy": (
            "Existing objects from pinned clean dependency checkouts are reused and "
            "kernel-checked by Lean at trust zero. Every imported dependency object's "
            "resolved .olean family is fingerprinted after the owned-declaration audit "
            "and checked again before acceptance; dependency sources are not rebuilt."),
        "scope": (
            "Only the selected third-cyclotomic-jacobi module and its complete local import "
            "closure are validated from immutable source copies with explicit fresh or cached compilation provenance. Positive "
            "consumers and intended type-rejection controls are checked separately; "
            "all owned declarations and transitive axioms in that library closure and accepted controls are "
            "audited. No runtime compression, hash injectivity, filesystem refinement, whole-library "
            "acceptance, cold dependency rebuild, or independent review is claimed."),
        "results": [], "selected_checks": {}, "generated_project_objects": {},
    }

    def save():
        (run / "receipt.json").write_text(json.dumps(receipt, indent=2) + "\n")

    def stable():
        verify_project_objects(receipt["generated_project_objects"])
        for rel, body in frozen_bytes.items():
            require(guard.read_source(ROOT, rel) == body, "source drift: " + rel)
            require((source / rel).read_bytes() == body, "frozen source drift: " + rel)
        require(sha(Path(lean).read_bytes()) == receipt["compiler_sha256"], "compiler drift")
        for name, revision in pins.items():
            package = ROOT / ".lake/packages" / name
            require(command("git", "-C", str(package), "rev-parse", "HEAD") == revision,
                    "dependency revision mismatch: " + name)
            require(not command("git", "-C", str(package), "status", "--porcelain", "--untracked-files=no"),
                    "dirty dependency: " + name)
        for family in receipt.get("dependency_object_identities", {}).values():
            for identity in family:
                obj = Path(identity["path"])
                require(obj.is_file() and obj.stat().st_size == identity["bytes"]
                        and guard.digest(obj) == identity["sha256"],
                        "dependency object drift: " + str(obj))
        if "audit_wrapper_sha256" in receipt:
            require(sha((source / "SelectedAudit.lean").read_bytes())
                    == receipt["audit_wrapper_sha256"], "audit wrapper drift")

    incremental = SelectedReplayCache(
        ROOT, guard, source, build, frozen_bytes, paths, lean, receipt, cold=cold)

    def compile_source(rel, output=False, expected=None):
        stable()
        argv = [lean, "-j1", "--trust=0", "--memory=" + str(memory_mib),
                "-DwarningAsError=true", "--root=" + str(source)]
        if output:
            obj = build / Path(rel).with_suffix(".olean")
            obj.parent.mkdir(parents=True, exist_ok=True)
            argv += ["-o", str(obj)]
        argv.append(str(source / rel))
        started = time.monotonic()
        process = incremental.run(rel, argv=argv, cwd=source, env=env, output=output, expected=expected, timeout=600)
        diagnostic = process.stdout + process.stderr
        (run / (str(len(receipt["results"])) + ".log")).write_text(diagnostic)
        receipt["results"].append({"path": rel, "argv": argv, "exit_code": process.returncode,
                                  "seconds": time.monotonic() - started,
                                  "output_sha256": sha(diagnostic.encode()), "expected_rejection": expected is not None})
        incremental.annotate_result(receipt["results"][-1])
        save()
        if expected is None:
            require(process.returncode == 0, "Lean rejection: " + rel + "\n" +
                    "\n".join(diagnostic.splitlines()[-25:]))
        else:
            validate_intended_rejection(guard, process.returncode, diagnostic, expected)
        if output:
            receipt["generated_project_objects"][str(obj)] = object_family(obj)
            save()
        print(rel + (" INTENDED REJECTION" if expected else " PASSED"), flush=True)
        receipt["selected_checks"][rel] = "passed"
        return diagnostic

    derived_controls = []
    receipt["positive_control_modules_audited"] = list(positive_order)

    save()
    print("RUN " + str(run), flush=True)
    try:
        stable()
        for rel in library_order:
            compile_source(rel, output=True)
        for rel in positive_order + derived_controls:
            compile_source(rel, output=True)
        selected = library_order + positive_order + derived_controls
        modules = [Path(rel).with_suffix("").as_posix().replace("/", ".") for rel in selected]
        owner_hashes = {module: sha(frozen_bytes[rel]) for module, rel in zip(modules, selected)}
        audit_body, source_profile = guard.owned_audit_protocol.wrapper_body(
            frozen_bytes[AUDIT], "project", owner_hashes)
        profiles = guard.owned_audit_protocol.runtime_profiles(frozen_bytes[AUDIT], "project")
        receipt["owned_audit_protocol"] = "current-json-source-profiles-v1"
        receipt["owned_audit_source_profile"] = source_profile
        receipt["runtime_helper_exemptions"] = []
        imports = ["import " + Path(rel).with_suffix("").as_posix().replace("/", ".") for rel in selected]
        wrapper = "\n".join(imports + ["import Lean", "import Lean.Compiler.Old", audit_body, ""])
        (source / "SelectedAudit.lean").write_text(wrapper)
        (source / "SelectedAudit.lean").chmod(0o444)
        receipt["audit_wrapper_sha256"] = sha(wrapper.encode())
        diagnostic = compile_source("SelectedAudit.lean")
        declarations, owners = guard.owned_audit_protocol.validate_census(
            diagnostic, modules, owner_hashes, profiles, guard.census_records)
        require(all(d["mathematical_admission"] for d in declarations)
                and not any(d["known_generated_runtime_helper"] for d in declarations),
                "selected source closures require all declarations mathematical and no runtime helper exemptions")
        receipt["declarations"], receipt["modules"] = declarations, owners
        receipt["dependency_object_identities"] = guard.dependency_object_identities(
            diagnostic, paths, lean)
        save()
        for rel, expected in NEGATIVE:
            compile_source(rel, expected=expected)
        stable()
        incremental.accept()
        receipt["outcome"] = "passed"
        receipt["mathematical_outcomes"] = ["PARTIAL_RESULT_CHECKED"]
        save()
        if destination:
            destination.parent.mkdir(parents=True, exist_ok=True)
            with destination.open("x") as stream:
                stream.write(json.dumps(receipt, indent=2) + "\n")
        print(json.dumps({"outcome": "passed", "declarations": len(declarations),
                          "receipt": str(destination or run / "receipt.json")}), flush=True)
        return 0
    except Exception as error:
        if "incremental" in locals():
            incremental.close()
        receipt["outcome"], receipt["failure"] = "failed", str(error)
        save()
        raise


if __name__ == "__main__":
    raise SystemExit(main())

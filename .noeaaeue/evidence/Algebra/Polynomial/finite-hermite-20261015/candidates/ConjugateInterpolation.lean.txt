import Mathlib.LinearAlgebra.Lagrange
import Mathlib.Data.Complex.Basic
import Mathlib.Tactic

set_option autoImplicit false
noncomputable section

namespace GRWTSK.Polynomial.ConjugateInterpolation

/-- Distinct support only. Analytic multiplicities are retained separately. -/
def StarClosed (s : Finset ℂ) : Prop := ∀ z ∈ s, star z ∈ s

/-- Exact coefficient reality, not projection of evaluated imaginary parts. -/
def RealCoefficients (p : Polynomial ℂ) : Prop :=
  p.map (starRingEnd ℂ) = p

theorem eval_map_star (p : Polynomial ℂ) (z : ℂ) :
    (p.map (starRingEnd ℂ)).eval z = star (p.eval (star z)) := by
  simpa using
    (Polynomial.eval_map_apply (p := p) (f := starRingEnd ℂ) (star z))

theorem coeff_im_eq_zero {p : Polynomial ℂ} (hp : RealCoefficients p) (n : ℕ) :
    (p.coeff n).im = 0 := by
  have hc : star (p.coeff n) = p.coeff n := by
    simpa [RealCoefficients] using congrArg (fun q : Polynomial ℂ => q.coeff n) hp
  have hi := congrArg Complex.im hc
  change -(p.coeff n).im = (p.coeff n).im at hi
  linarith

theorem eval_im_eq_zero {p : Polynomial ℂ} (hp : RealCoefficients p)
    {z : ℂ} (hz : z.im = 0) : (p.eval z).im = 0 := by
  have hs : star z = z := by
    apply Complex.ext <;> simp [hz]
  have he : p.eval z = star (p.eval z) := by
    calc
      p.eval z = (p.map (starRingEnd ℂ)).eval z := by rw [hp]
      _ = star (p.eval (star z)) := eval_map_star p z
      _ = star (p.eval z) := by rw [hs]
  have hi := congrArg Complex.im he
  change (p.eval z).im = -(p.eval z).im at hi
  linarith

/-- Reuse the pinned interpolator and uniqueness theorem; no new interpolation axiom. -/
theorem exists_real_interpolant (s : Finset ℂ) (hs : StarClosed s)
    (values : ℂ → ℂ)
    (hv : ∀ z ∈ s, values (star z) = star (values z)) :
    ∃ p : Polynomial ℂ, RealCoefficients p ∧ p.degree < s.card ∧
      ∀ z ∈ s, p.eval z = values z := by
  classical
  let p := Lagrange.interpolate s id values
  have hdeg : p.degree < s.card :=
    Lagrange.degree_interpolate_lt values Function.injective_id.injOn
  have heval : ∀ z ∈ s, p.eval z = values z := by
    intro z hz
    exact Lagrange.eval_interpolate_at_node values Function.injective_id.injOn hz
  refine ⟨p, ?_, hdeg, heval⟩
  apply Polynomial.eq_of_degrees_lt_of_eval_finset_eq s
    (lt_of_le_of_lt Polynomial.degree_map_le hdeg) hdeg
  intro z hz
  calc
    (p.map (starRingEnd ℂ)).eval z = star (p.eval (star z)) := eval_map_star p z
    _ = star (values (star z)) := by rw [heval (star z) (hs z hz)]
    _ = values z := by rw [hv z hz]; simp
    _ = p.eval z := (heval z hz).symm

/-- Prescribed conjugate values; no absolute-square substitution. -/
def pairValues (z η x : ℂ) : ℂ := by
  classical
  exact if x = z then η else if x = star z then star η else 0

theorem pairValues_star (z η : ℂ) (hz : star z ≠ z) (x : ℂ) :
    pairValues z η (star x) = star (pairValues z η x) := by
  classical
  by_cases hx : x = z
  · subst x
    simp [pairValues, hz]
  by_cases hx' : x = star z
  · subst x
    simp [pairValues, hz, hz.symm]
  have h₁ : star x ≠ z := by
    intro h
    apply hx'
    simpa using congrArg star h
  have h₂ : star x ≠ star z := by
    intro h
    apply hx
    simpa using congrArg star h
  simp [pairValues, hx, hx', h₁, h₂]

theorem exists_pair_interpolant (s : Finset ℂ) (hs : StarClosed s)
    (z η : ℂ) (hz : star z ≠ z) :
    ∃ p : Polynomial ℂ, RealCoefficients p ∧ p.degree < s.card ∧
      ∀ x ∈ s, p.eval x = pairValues z η x :=
  exists_real_interpolant s hs (pairValues z η) (fun x _ => pairValues_star z η hz x)

end GRWTSK.Polynomial.ConjugateInterpolation

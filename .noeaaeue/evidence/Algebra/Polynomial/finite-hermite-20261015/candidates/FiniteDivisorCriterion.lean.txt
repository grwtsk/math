/-- Exactly -I*(s-1/2); the sign agrees with the supplied quartic coordinate. -/
def ordinate (s : ℂ) : ℂ := ⟨s.im, 1 / 2 - s.re⟩

theorem ordinate_eq (s : ℂ) : ordinate s = -Complex.I * (s - 1 / 2) := by
  apply Complex.ext <;> simp [ordinate, Complex.mul_re, Complex.mul_im] <;> ring

/-- The remaining finite positive family is an EXPLICIT premise, not a data source
manufactured by an automatically positive squared-modulus Gram. -/
def PositiveTraceFamily : Prop :=
  ∀ z : ℂ, GRWTSK.RiemannHypothesis.NontrivialZetaZero z →
    ∃ (nodes : Finset ℂ) (mass : ℂ → ℝ),
      StarClosed nodes ∧ ordinate z ∈ nodes ∧
      (∀ x ∈ nodes, 0 < mass x) ∧ NonnegativeTests nodes mass 2

/-- Root target is reused, not redefined. The formal argument remains conditional. -/
theorem riemannHypothesis_of_positiveTraceFamily (h : PositiveTraceFamily) :
    _root_.RiemannHypothesis := by
  intro z hz ht hp
  obtain ⟨nodes, mass, hclosed, hmem, hmass, hpos⟩ := h z ⟨hz, ht, hp⟩
  have hreal := (square_nonnegative_iff nodes hclosed mass hmass).mp hpos
  have hzreal := hreal (ordinate z) hmem
  change 1 / 2 - z.re = 0 at hzreal
  linarith

end GRWTSK.Zeta.FiniteDivisorCriterion

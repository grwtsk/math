namespace GRWTSK.Polynomial.ConjugatePair

/-- Extra conjugate factors enforce reality even when S is not conjugation-closed.
These are interpolation multiplicities, not analytic multiplicities of a source. -/
def pairedProduct (S : Finset ℂ) (z : ℂ) : ℂ :=
  ∏ w ∈ S, (z - w) * (z - conj w)

theorem pairedProduct_conj (S : Finset ℂ) (z : ℂ) :
    pairedProduct S (conj z) = conj (pairedProduct S z) := by
  classical
  simp only [pairedProduct, map_prod]
  apply Finset.prod_congr rfl
  intro w _
  apply Complex.ext <;> simp [Complex.mul_re, Complex.mul_im] <;> ring

theorem pairedProduct_zero (S : Finset ℂ) (w : ℂ) (hw : w ∈ S) :
    pairedProduct S w = 0 := by
  classical
  unfold pairedProduct
  exact Finset.prod_eq_zero hw (by simp)

theorem pairedProduct_ne_zero (S : Finset ℂ) (r : ℂ)
    (hr : r ∉ S) (hrc : conj r ∉ S) : pairedProduct S r ≠ 0 := by
  classical
  apply Finset.prod_ne_zero_iff.mpr
  intro w hw
  apply mul_ne_zero
  · apply sub_ne_zero.mpr
    intro heq
    apply hr
    simpa only [heq] using hw
  · apply sub_ne_zero.mpr
    intro heq
    have hc : conj r = w := by simpa using congrArg conj heq
    apply hrc
    simpa only [hc] using hw

/-- A real-coefficient affine function with prescribed value d at nonreal r. -/
def normalizedLinear (r d z : ℂ) : ℂ :=
  (d.re : ℂ) + ((d.im / r.im : ℝ) : ℂ) * (z - (r.re : ℂ))

theorem normalizedLinear_at (r d : ℂ) (hr : r.im ≠ 0) :
    normalizedLinear r d r = d := by
  apply Complex.ext <;>
    simp [normalizedLinear, Complex.mul_re, Complex.mul_im] <;> field_simp [hr]

theorem normalizedLinear_conj (r d z : ℂ) :
    normalizedLinear r d (conj z) = conj (normalizedLinear r d z) := by
  apply Complex.ext <;> simp [normalizedLinear, Complex.mul_re, Complex.mul_im] <;> ring

def isolator (S : Finset ℂ) (r z : ℂ) : ℂ :=
  pairedProduct S z * normalizedLinear r (Complex.I / pairedProduct S r) z

def nodePolynomial (S : Finset ℂ) : ℂ[X] :=
  ∏ w ∈ S, (X - C w) * (X - C (conj w))

def linearPolynomial (r d : ℂ) : ℂ[X] :=
  C (d.re : ℂ) + C ((d.im / r.im : ℝ) : ℂ) * (X - C (r.re : ℂ))

def isolatingPolynomial (S : Finset ℂ) (r : ℂ) : ℂ[X] :=
  nodePolynomial S * linearPolynomial r (Complex.I / pairedProduct S r)

theorem eval_isolatingPolynomial (S : Finset ℂ) (r z : ℂ) :
    (isolatingPolynomial S r).eval z = isolator S r z := by
  classical
  simp [isolatingPolynomial, nodePolynomial, linearPolynomial,
    isolator, normalizedLinear, pairedProduct]

theorem isolator_at (S : Finset ℂ) (r : ℂ) (hi : r.im ≠ 0)
    (hr : r ∉ S) (hrc : conj r ∉ S) : isolator S r r = Complex.I := by
  rw [isolator, normalizedLinear_at r _ hi]
  field_simp [pairedProduct_ne_zero S r hr hrc]

theorem isolator_conj (S : Finset ℂ) (r z : ℂ) :
    isolator S r (conj z) = conj (isolator S r z) := by
  simp only [isolator, pairedProduct_conj, normalizedLinear_conj, map_mul]

theorem isolator_at_conj (S : Finset ℂ) (r : ℂ) (hi : r.im ≠ 0)
    (hr : r ∉ S) (hrc : conj r ∉ S) : isolator S r (conj r) = -Complex.I := by
  rw [isolator_conj, isolator_at S r hi hr hrc]
  simp

theorem isolator_zero (S : Finset ℂ) (r w : ℂ) (hw : w ∈ S) :
    isolator S r w = 0 := by
  simp [isolator, pairedProduct_zero S w hw]

/-- Unequal real weights are retained. Positivity is needed only for the sign,
not for this exact identity. No orbit-stabilizer multiplicity is imposed. -/
theorem weighted_trace (S : Finset ℂ) (r : ℂ) (hi : r.im ≠ 0)
    (hr : r ∉ S) (hrc : conj r ∉ S) (m : ℂ → ℝ) :
    (m r : ℂ) * isolator S r r ^ 2 +
      (m (conj r) : ℂ) * isolator S r (conj r) ^ 2 +
      (∑ w ∈ S, (m w : ℂ) * isolator S r w ^ 2) =
      -((m r + m (conj r) : ℝ) : ℂ) := by
  have hz : (∑ w ∈ S, (m w : ℂ) * isolator S r w ^ 2) = 0 := by
    apply Finset.sum_eq_zero
    intro w hw
    simp [isolator_zero S r w hw]
  rw [isolator_at S r hi hr hrc, isolator_at_conj S r hi hr hrc, hz]
  simp [Complex.I_sq] <;> ring

end GRWTSK.Polynomial.ConjugatePair

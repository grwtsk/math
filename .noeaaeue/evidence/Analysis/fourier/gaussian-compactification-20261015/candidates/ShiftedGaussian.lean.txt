theorem shifted_gaussian_integral (L t x : ℝ) (hL : 0 < L) :
    (∫ u : ℝ, Complex.exp (-(L : ℂ) * ((u : ℂ) - (t : ℂ)) ^ 2 -
      Complex.I * (x : ℂ) * (u : ℂ))) =
    ((Real.pi : ℂ) / (L : ℂ)) ^ (1 / 2 : ℂ) *
      Complex.exp (-(x : ℂ) ^ 2 / (4 * (L : ℂ)) -
        Complex.I * (t : ℂ) * (x : ℂ)) := by
  have hLc : (L : ℂ) ≠ 0 := by exact_mod_cast (ne_of_gt hL)
  have hn : (-(L : ℂ)).re < 0 := by simp only [Complex.neg_re, Complex.ofReal_re]; linarith
  have he (u : ℝ) :
      -(L : ℂ) * ((u : ℂ) - (t : ℂ)) ^ 2 - Complex.I * (x : ℂ) * (u : ℂ) =
      (-(L : ℂ)) * (u : ℂ) ^ 2 +
        (2 * (L : ℂ) * (t : ℂ) - Complex.I * (x : ℂ)) * (u : ℂ) +
        (-(L : ℂ) * (t : ℂ) ^ 2) := by ring
  simp_rw [he]
  rw [integral_cexp_quadratic hn]
  simp only [neg_neg]
  congr 1
  congr 1
  field_simp [hLc]
  simp only [Complex.I_sq]
  ring

end GRWTSK.Analysis.Fourier.GaussianCompactification

/-- The signed square is retained; the norm bounds only its perturbation. -/
theorem square_difference (z w : ℂ) :
    ‖z ^ 2 - w ^ 2‖ ≤ ‖z - w‖ * (2 * ‖w‖ + ‖z - w‖) := by
  have hz : ‖z‖ ≤ ‖z - w‖ + ‖w‖ := by
    calc
      ‖z‖ = ‖(z - w) + w‖ := by congr 1; ring
      _ ≤ ‖z - w‖ + ‖w‖ := norm_add_le _ _
  have hf : z ^ 2 - w ^ 2 = (z - w) * (z + w) := by ring
  rw [hf, norm_mul]
  apply mul_le_mul_of_nonneg_left _ (norm_nonneg _)
  calc
    ‖z + w‖ ≤ ‖z‖ + ‖w‖ := norm_add_le _ _
    _ ≤ 2 * ‖w‖ + ‖z - w‖ := by linarith

/-- Denominator-free version of the weighted strip error bound. -/
theorem weighted_square_difference (z w : ℂ) (q B δ : ℝ)
    (hq : 1 ≤ q) (hB : 0 ≤ B) (hδ : 0 ≤ δ)
    (hw : ‖w‖ ≤ B) (hd : q * ‖z - w‖ ≤ δ) :
    q * ‖z ^ 2 - w ^ 2‖ ≤ δ * (2 * B + δ) := by
  have hq0 : 0 ≤ q := by linarith
  have hd0 : 0 ≤ ‖z - w‖ := norm_nonneg _
  have hsmall : ‖z - w‖ ≤ δ := by nlinarith
  calc
    q * ‖z ^ 2 - w ^ 2‖ ≤ q * (‖z - w‖ * (2 * ‖w‖ + ‖z - w‖)) :=
      mul_le_mul_of_nonneg_left (square_difference z w) hq0
    _ = (q * ‖z - w‖) * (2 * ‖w‖ + ‖z - w‖) := by ring
    _ ≤ δ * (2 * B + δ) :=
      mul_le_mul hd (by linarith) (by positivity) hδ

/-- Every weight survives. Summability of the infinite source is separate. -/
theorem finite_trace_budget {ι : Type*} (S : Finset ι)
    (m q : ι → ℝ) (f g : ι → ℂ) (B δ : ℝ)
    (hB : 0 ≤ B) (hδ : 0 ≤ δ)
    (hm : ∀ i ∈ S, 0 ≤ m i) (hq : ∀ i ∈ S, 1 ≤ q i)
    (hg : ∀ i ∈ S, ‖g i‖ ≤ B)
    (hd : ∀ i ∈ S, q i * ‖f i - g i‖ ≤ δ) :
    ‖∑ i ∈ S, (m i : ℂ) * (f i ^ 2 - g i ^ 2)‖ ≤
      δ * (2 * B + δ) * ∑ i ∈ S, m i / q i := by
  calc
    _ ≤ ∑ i ∈ S, ‖(m i : ℂ) * (f i ^ 2 - g i ^ 2)‖ := norm_sum_le _ _
    _ ≤ ∑ i ∈ S, m i * (δ * (2 * B + δ) / q i) := by
      apply Finset.sum_le_sum
      intro i hi
      have hqi : 0 < q i := by linarith [hq i hi]
      have hb := weighted_square_difference (f i) (g i) (q i) B δ
        (hq i hi) hB hδ (hg i hi) (hd i hi)
      have hb' : ‖f i ^ 2 - g i ^ 2‖ ≤ δ * (2 * B + δ) / q i := by
        apply (le_div_iff₀ hqi).2
        simpa only [mul_comm] using hb
      rw [norm_mul, Complex.norm_real, Real.norm_eq_abs, abs_of_nonneg (hm i hi)]
      exact mul_le_mul_of_nonneg_left hb' (hm i hi)
    _ = δ * (2 * B + δ) * ∑ i ∈ S, m i / q i := by
      rw [Finset.mul_sum]
      apply Finset.sum_congr rfl
      intro i _
      ring

/-- Two separately proved errors below a quarter of the retained mass suffice. -/
theorem negative_of_two_errors (W M tail cut : ℝ) (hM : 0 < M)
    (hW : W ≤ -M + tail + cut) (ht : tail ≤ M / 4) (hc : cut ≤ M / 4) :
    W ≤ -M / 2 ∧ W < 0 := by
  constructor <;> linarith

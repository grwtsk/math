import Mathlib.Algebra.BigOperators.Group.Finset.Basic
import Mathlib.Tactic.Ring

set_option autoImplicit false

namespace GRWTSK.Analysis.Sequences.ClosedWeightedAbelFlux

open scoped BigOperators

/-- Weighted endpoint energy on `0,...,N-1`. -/
def endpointEnergy (q v : ℕ → ℝ) (N : ℕ) : ℝ :=
  ∑ n ∈ Finset.range N, q n ^ 2 * (v n - v (n + 1))

/-- Signed interior work. -/
def openFlux (q d v : ℕ → ℝ) (N : ℕ) : ℝ :=
  ∑ n ∈ Finset.range N, 2 * q n * d n * v (n + 1)

/-- Positive square correction. -/
def correction (d v : ℕ → ℝ) (N : ℕ) : ℝ :=
  ∑ n ∈ Finset.range N, d n ^ 2 * v (n + 1)

/-- The incoming seed and terminal charge are both retained. -/
def closedFlux (q d v : ℕ → ℝ) (N : ℕ) : ℝ :=
  q 0 ^ 2 * v 0 + openFlux q d v N - q N ^ 2 * v N

/-- Exact closed Abel identity. -/
theorem endpointEnergy_eq_closedFlux_add_correction
    (q d v : ℕ → ℝ)
    (hrec : ∀ n, q (n + 1) = q n + d n) :
    ∀ N : ℕ,
      endpointEnergy q v N =
        closedFlux q d v N + correction d v N := by
  intro N
  induction N with
  | zero => simp [endpointEnergy, closedFlux, openFlux, correction]
  | succ N ih =>
      rw [endpointEnergy, openFlux, correction]
      simp only [Finset.sum_range_succ]
      rw [endpointEnergy, openFlux, correction] at ih
      unfold closedFlux at ih ⊢
      rw [hrec N]
      linarith [sq_nonneg (q N + d N)]

/-- If all correction weights are nonnegative, endpoint-energy boundedness
immediately bounds the closed flux from above. -/
theorem closedFlux_le_endpointEnergy
    (q d v : ℕ → ℝ)
    (hrec : ∀ n, q (n + 1) = q n + d n)
    (hv : ∀ n, 0 ≤ v (n + 1)) (N : ℕ) :
    closedFlux q d v N ≤ endpointEnergy q v N := by
  rw [endpointEnergy_eq_closedFlux_add_correction q d v hrec N]
  have hc : 0 ≤ correction d v N := by
    unfold correction
    exact Finset.sum_nonneg (fun n _ => mul_nonneg (sq_nonneg _) (hv n))
  linarith

/-- A uniform correction allowance converts a closed-flux upper bound into an
endpoint-energy upper bound, with the full correction constant retained. -/
theorem endpointEnergy_le_of_closedFlux_le
    (q d v : ℕ → ℝ)
    (hrec : ∀ n, q (n + 1) = q n + d n)
    {C R : ℝ} (hF : ∀ N, closedFlux q d v N ≤ C)
    (hR : ∀ N, correction d v N ≤ R) (N : ℕ) :
    endpointEnergy q v N ≤ C + R := by
  rw [endpointEnergy_eq_closedFlux_add_correction q d v hrec N]
  exact add_le_add (hF N) (hR N)

end GRWTSK.Analysis.Sequences.ClosedWeightedAbelFlux

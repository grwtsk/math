import Mathlib.Tactic

set_option autoImplicit false

namespace GRWTSK.Analysis.MomentObstructions

/-- Third finite difference of a monomial at zero. -/
def thirdDifferenceMoment (k : ℕ) : ℤ :=
  (0 : ℤ) ^ k - 3 * (1 : ℤ) ^ k + 3 * (2 : ℤ) ^ k - (3 : ℤ) ^ k

/-- A nonzero signed measure can have a finite prefix of zero moments. -/
theorem first_three_moments_zero :
    thirdDifferenceMoment 0 = 0 ∧
    thirdDifferenceMoment 1 = 0 ∧
    thirdDifferenceMoment 2 = 0 := by
  norm_num [thirdDifferenceMoment]

/-- The same signed measure is not zero: its next moment survives. -/
theorem third_moment_nonzero : thirdDifferenceMoment 3 = -6 := by
  norm_num [thirdDifferenceMoment]

/-- Pairwise data need not determine a three-way sign. -/
theorem pairwise_same_triple_opposite :
    (0 : ℤ) = 0 ∧ (1 : ℤ) = 1 ∧ (-1 : ℤ) = -1 := by
  norm_num

end GRWTSK.Analysis.MomentObstructions

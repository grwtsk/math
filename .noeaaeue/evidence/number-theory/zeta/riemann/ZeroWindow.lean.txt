namespace GRWTSK.Zeta.Riemann

/-- Compact source rectangle, not a finite numerical zero atlas. -/
def zeroWindow (t : ℝ) : Set ℂ :=
  (fun p : ℝ × ℝ => (p.1 : ℂ) + (p.2 : ℂ) * Complex.I) ''
    (Set.Icc (0 : ℝ) 1 ×ˢ Set.Icc (t - 1) (t + 1))

theorem isCompact_zeroWindow (t : ℝ) : IsCompact (zeroWindow t) := by
  apply (isCompact_Icc.prod isCompact_Icc).image
  fun_prop

theorem zeroWindow_zeros_finite (t : ℝ) :
    (zeroWindow t ∩ riemannZetaZeros).Finite :=
  (isCompact_zeroWindow t).inter_riemannZetaZeros_finite

def windowZeros (t : ℝ) : Finset ℂ := (zeroWindow_zeros_finite t).toFinset

theorem nontrivial_zero_mem_window (t : ℝ) (s : ℂ)
    (hs : NontrivialZetaZero s) (ht : |s.im - t| ≤ 1) : s ∈ windowZeros t := by
  apply Set.Finite.mem_toFinset.mpr
  refine ⟨?_, hs.1⟩
  refine ⟨(s.re, s.im), ?_, ?_⟩
  · have hb := abs_le.mp ht
    exact ⟨⟨hs.mem_critical_strip.1.le, hs.mem_critical_strip.2.le⟩,
      ⟨by linarith, by linarith⟩⟩
  · exact Complex.re_add_im s

/-- Frequency of the critical reflection is complex conjugation of frequency. -/
theorem frequency_criticalReflection (s : ℂ) :
    frequency (criticalReflection s) = conj (frequency s) := by
  rw [critical_reflection_eq_composite]
  apply Complex.ext <;> simp [frequency, zeroReflect, zeroConjugate] <;> ring

/-- Finite cancellation nodes exclude the selected pair by definition. -/
def otherWindowFrequencies (s : ℂ) : Finset ℂ :=
  (((windowZeros s.im).image frequency).erase (frequency s)).erase (conj (frequency s))

theorem pair_not_mem_otherWindowFrequencies (s : ℂ) :
    frequency s ∉ otherWindowFrequencies s ∧
      conj (frequency s) ∉ otherWindowFrequencies s := by
  classical
  simp [otherWindowFrequencies]

/-- Every other actual source zero in the fixed closed height window is cancelled.
No cardinality, zero location, analytic multiplicity or computable enumeration is inferred. -/
theorem nontrivial_zero_frequency_mem (s u : ℂ) (hu : NontrivialZetaZero u)
    (ht : |u.im - s.im| ≤ 1) (hne : frequency u ≠ frequency s)
    (hnec : frequency u ≠ conj (frequency s)) :
    frequency u ∈ otherWindowFrequencies s := by
  classical
  simp only [otherWindowFrequencies, Finset.mem_erase, Finset.mem_image]
  exact ⟨hnec, hne, u, nontrivial_zero_mem_window s.im u hu ht, rfl⟩

/-- The actual source strip gives the imaginary-frequency bound used by the tail estimate. -/
theorem frequency_mem_half_strip (s : ℂ) (hs : NontrivialZetaZero s) :
    |(frequency s).im| ≤ 1 / 2 := by
  have hstrip := hs.mem_critical_strip
  change |-(s.re - 1 / 2)| ≤ 1 / 2
  rw [abs_neg, abs_le]
  constructor <;> linarith

/-- The coordinate change cannot merge distinct analytic zero locations. -/
theorem frequency_injective : Function.Injective frequency := by
  intro s u h
  have hre := congrArg Complex.re h
  have him := congrArg Complex.im h
  change s.im = u.im at hre
  change -(s.re - 1 / 2) = -(u.re - 1 / 2) at him
  apply Complex.ext
  · linarith
  · exact hre

theorem frequency_im_ne_zero (s : ℂ) (hoff : s.re ≠ 1 / 2) :
    (frequency s).im ≠ 0 := by
  change -(s.re - 1 / 2) ≠ 0
  intro h
  apply hoff
  linarith

/-- An actual-source polynomial test selected from the finite zeta window.
Classical finite-set formation is not a numerical algorithm for locating zeros. -/
def zeroPairTest (s : ℂ) : ℂ → ℂ :=
  GRWTSK.Polynomial.ConjugatePair.isolator (otherWindowFrequencies s) (frequency s)

theorem zeroPairTest_at (s : ℂ) (hoff : s.re ≠ 1 / 2) :
    zeroPairTest s (frequency s) = Complex.I := by
  exact GRWTSK.Polynomial.ConjugatePair.isolator_at _ _
    (frequency_im_ne_zero s hoff)
    (pair_not_mem_otherWindowFrequencies s).1
    (pair_not_mem_otherWindowFrequencies s).2

theorem zeroPairTest_at_conj (s : ℂ) (hoff : s.re ≠ 1 / 2) :
    zeroPairTest s (conj (frequency s)) = -Complex.I := by
  exact GRWTSK.Polynomial.ConjugatePair.isolator_at_conj _ _
    (frequency_im_ne_zero s hoff)
    (pair_not_mem_otherWindowFrequencies s).1
    (pair_not_mem_otherWindowFrequencies s).2

theorem zeroPairTest_vanishes (s u : ℂ) (hu : NontrivialZetaZero u)
    (ht : |u.im - s.im| ≤ 1) (hne : frequency u ≠ frequency s)
    (hnec : frequency u ≠ conj (frequency s)) :
    zeroPairTest s (frequency u) = 0 := by
  exact GRWTSK.Polynomial.ConjugatePair.isolator_zero _ _ _
    (nontrivial_zero_frequency_mem s u hu ht hne hnec)

def zeroGaussianTest (s : ℂ) (n : ℕ) : ℂ → ℂ :=
  GRWTSK.Analysis.GaussianPair.localized ((n : ℝ) + 1) (zeroPairTest s) (frequency s)

theorem zeroGaussianTest_at (s : ℂ) (n : ℕ) (hoff : s.re ≠ 1 / 2) :
    zeroGaussianTest s n (frequency s) = Complex.I := by
  unfold zeroGaussianTest
  rw [GRWTSK.Analysis.GaussianPair.localized_at, zeroPairTest_at s hoff]

theorem zeroGaussianTest_at_conj (s : ℂ) (n : ℕ) (hoff : s.re ≠ 1 / 2) :
    zeroGaussianTest s n (conj (frequency s)) = -Complex.I := by
  unfold zeroGaussianTest
  rw [GRWTSK.Analysis.GaussianPair.localized_at_conj, zeroPairTest_at_conj s hoff]

/-- Source-specific pointwise estimate; its infinite summation still requires
actual analytic multiplicities and a summable weighted Gaussian majorant. -/
theorem zeroGaussianTest_tail (s u : ℂ) (hu : NontrivialZetaZero u)
    (hfar : 1 ≤ |u.im - s.im|) (n : ℕ) :
    ‖zeroGaussianTest s n (frequency u)‖ ^ 2 ≤
      Real.exp (-(3 / 2 : ℝ) * n) * ‖zeroGaussianTest s 0 (frequency u)‖ ^ 2 := by
  simpa only [zeroGaussianTest, Nat.cast_zero, zero_add] using
    GRWTSK.Analysis.GaussianPair.geometric_envelope (zeroPairTest s)
      (frequency s) (frequency u) n (frequency_mem_half_strip u hu) hfar

end GRWTSK.Zeta.Riemann

namespace GRWTSK.Zeta.Riemann

open GRWTSK.RiemannHypothesis (NontrivialZetaZero)

/-- Exact equality through a natural height, including the boundary. -/
def RHThrough (N : ℕ) : Prop :=
  ∀ s : ℂ, NontrivialZetaZero s → |s.im| ≤ (N : ℝ) → s.re = 1 / 2

theorem all_heights_iff : (∀ N : ℕ, RHThrough N) ↔ _root_.RiemannHypothesis := by
  constructor
  · intro h s hz htriv hpole
    obtain ⟨N, hN⟩ := exists_nat_gt |s.im|
    exact h N s ⟨hz, htriv, hpole⟩ (le_of_lt hN)
  · intro h N s hs _
    exact h s hs.1 hs.2.1 hs.2.2

/-- Data for an actual induction, not a finite run or an unproved search totality claim. -/
structure HeightScheme where
  State : ℕ → Type
  base : State 0
  step : ∀ N : ℕ, State N → State (N + 1)
  sound : ∀ N : ℕ, State N → RHThrough N

def HeightScheme.stateAt (scheme : HeightScheme) : ∀ N : ℕ, scheme.State N :=
  Nat.rec scheme.base (fun N state => scheme.step N state)

theorem HeightScheme.riemannHypothesis (scheme : HeightScheme) :
    _root_.RiemannHypothesis :=
  all_heights_iff.mp (fun N => scheme.sound N (scheme.stateAt N))

end GRWTSK.Zeta.Riemann

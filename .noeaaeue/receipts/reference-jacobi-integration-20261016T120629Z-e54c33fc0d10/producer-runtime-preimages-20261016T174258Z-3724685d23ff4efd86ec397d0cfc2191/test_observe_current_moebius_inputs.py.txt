#!/usr/bin/env python3
"""Isolated observation-producer controls: no project imports or native jobs.

Only the producer's stdlib-only import and explicitly selected pure AST helpers
execute. Repository observations, process routes, locator and publication are
injected. All written fixtures and outputs live under TemporaryDirectory.
These controls establish Python behavior, not native or mathematical admission.
"""

import builtins
from contextlib import ExitStack, contextmanager
from dataclasses import dataclass
import hashlib

import os
from pathlib import Path, PurePosixPath
import re

import sys

ROOT = Path(__file__).resolve().parents[2]
PRODUCER_PATH = ROOT / ".noeaaeue/binding-repair/observe_current_moebius_inputs.py"
SOURCE_PATHS = (
    PRODUCER_PATH,
    ROOT / ".noeaaeue/scripts/check_moebius_response.py",
    ROOT / ".noeaaeue/scripts/incremental_lean.py",
    ROOT / ".noeaaeue/scripts/repository_paths.py",
)
SOURCES = {str(path): path.read_bytes() for path in SOURCE_PATHS}
EXTERNAL_ATTEMPTS = []


def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def encoded(value):
    return (json.dumps(value, sort_keys=True) + "\n").encode()


def forbidden(*args, **kwargs):
    EXTERNAL_ATTEMPTS.append("external process/location route")
    raise AssertionError("External execution is forbidden in producer controls")


@contextmanager
def isolated_execution():
    """Block process APIs and every non-stdlib import during tested execution."""
    real_import = builtins.__import__

    def stdlib_only(name, *args, **kwargs):
        if name.split(".")[0] not in sys.stdlib_module_names:
            EXTERNAL_ATTEMPTS.append("blocked import: " + name)
            raise AssertionError("Project/helper imports forbidden: " + name)
        return real_import(name, *args, **kwargs)

    with ExitStack() as stack:
        stack.enter_context(patch("builtins.__import__", side_effect=stdlib_only))
        for name in ("Popen", "run", "call", "check_call", "check_output"):
            stack.enter_context(patch.object(subprocess, name, side_effect=forbidden))
        for name in ("system", "popen", "fork", "forkpty", "posix_spawn", "posix_spawnp",
                     "execl", "execle", "execlp", "execlpe", "execv", "execve",
                     "execvp", "execvpe", "spawnl", "spawnle", "spawnlp", "spawnlpe",
                     "spawnv", "spawnve", "spawnvp", "spawnvpe", "startfile"):
            if hasattr(os, name):
                stack.enter_context(patch.object(os, name, side_effect=forbidden))
        yield


def extract(path, function_names, constant_names=(), namespace=None):
    """Execute selected definitions only, never the project module/imports."""
    tree = ast.parse(SOURCES[str(path)], filename=str(path))
    functions = [node for node in tree.body
                 if isinstance(node, (ast.FunctionDef, ast.ClassDef))
                 and node.name in function_names]
    if {node.name for node in functions} != set(function_names):
        raise AssertionError("Pure helper API changed: " + str(path))
    result = dict(namespace or {})
    for name in constant_names:
        matches = [node.value for node in tree.body if isinstance(node, ast.Assign)
                   and any(isinstance(target, ast.Name) and target.id == name
                           for target in node.targets)]
        if len(matches) != 1:
            raise AssertionError("Constant API changed: " + name)
        result[name] = ast.literal_eval(matches[0])
    selected = ast.Module(body=functions, type_ignores=[])
    if any(isinstance(node, (ast.Import, ast.ImportFrom)) for node in ast.walk(selected)):
        raise AssertionError("Selected pure helper acquired an import")
    with isolated_execution():
        exec(compile(selected, str(path), "exec"), result)
    return result


def pure_helpers():
    paths = ROOT / ".noeaaeue/scripts/repository_paths.py"
    tree = ast.parse(SOURCES[str(paths)])
    # These are literal frozenset construction calls, not general eval.
    constants = {}
    for node in tree.body:
        if isinstance(node, ast.Assign) and len(node.targets) == 1:
            name = getattr(node.targets[0], "id", None)
            if name in {"_PROCESS_ROOTS", "_REFERENCE_FILES"}:
                value = node.value
                if (not isinstance(value, ast.Call) or not isinstance(value.func, ast.Name)
                        or value.func.id != "frozenset" or len(value.args) != 1 or value.keywords):
                    raise AssertionError("Relocation constants no longer literal")
                constants[name] = frozenset(ast.literal_eval(value.args[0]))
    relocation = extract(paths, {"relocated_name"}, namespace={
        **constants, "PurePosixPath": PurePosixPath})["relocated_name"]
    checker = extract(ROOT / ".noeaaeue/scripts/check_moebius_response.py",
                      {"incremental_asset_keys"},
                      {"INCREMENTAL_CONTRACT", "INCREMENTAL_ASSETS", "PUBLIC_COMPILATION_ASSETS", "DEFAULT_CURRENT_MANIFEST"},
                      {"re": re, "relocated_name": relocation})
    checker.update(extract(ROOT / ".noeaaeue/scripts/check_moebius_response.py",
                           {"safe_absolute", "directory_descriptor", "atomic_json", "write_exclusive"},
                           namespace={"Path": Path, "os": os, "stat": stat, "json": json, "uuid": uuid}))
    origin = extract(ROOT / ".noeaaeue/scripts/incremental_lean.py",
                     {"CacheError", "_artifact_schema", "_command_schema", "_origin_payload"},
                     {"ORIGIN_SCHEMA", "OBJECT_SUFFIXES"}, {"re": re})
    return checker, origin


def load_producer():
    namespace = {"__file__": str(PRODUCER_PATH), "__name__": "isolated_observer_under_test"}
    # An inert import may read its fixed stdlib modules; no writer is authorized.
    real_open, real_os_open = builtins.open, os.open
    def read_only_open(file, mode="r", *args, **kwargs):
        if any(flag in mode for flag in "wax+"):
            return forbidden()
        return real_open(file, mode, *args, **kwargs)
    def read_only_os_open(path, flags, *args, **kwargs):
        if flags & (os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC | os.O_APPEND):
            return forbidden()
        return real_os_open(path, flags, *args, **kwargs)
    with isolated_execution(), ExitStack() as stack:
        stack.enter_context(patch("builtins.open", side_effect=read_only_open))
        stack.enter_context(patch.object(os, "open", side_effect=read_only_os_open))
        for name in ("write_bytes", "write_text", "touch", "mkdir", "unlink", "rename", "replace"):
            stack.enter_context(patch.object(Path, name, side_effect=forbidden))
        exec(compile(SOURCES[str(PRODUCER_PATH)], str(PRODUCER_PATH), "exec"), namespace)
    return SimpleNamespace(**namespace)


PURE_CHECKER, PURE_ORIGIN = pure_helpers()
PRODUCER = load_producer()


@dataclass(frozen=True)
class Observation:
    path: Path
    data: bytes
    identity: tuple

    def checked_path(self, path):
        path = Path(path).absolute()
        path.relative_to(self.root)
        return path

    def put(self, path, raw):
        path = Path(path)
        path = self.checked_path(path if path.is_absolute() else self.root / path)
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_bytes(raw)

    def stable_read(self, path):
        path = self.checked_path(path)
        descriptor = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK)
        with os.fdopen(descriptor, "rb") as stream:
            before = os.fstat(stream.fileno())
            if not stat.S_ISREG(before.st_mode):
                raise ValueError("fixture nonregular input")
            raw = stream.read()
            after = os.fstat(stream.fileno())
        current = path.lstat()
        identity = lambda s: (s.st_dev, s.st_ino, s.st_mode, s.st_size, s.st_mtime_ns, s.st_ctime_ns)
        if identity(before) != identity(after) or identity(after) != identity(current):
            raise ValueError("fixture read collision")
        self.reads.append(path)
        return Observation(path, raw, identity(after))

    def read_regular(self, root, relative):
        if Path(root) != self.root or Path(relative).is_absolute() or ".." in Path(relative).parts:
            raise AssertionError("fixture root/path escape")
        result = self.stable_read(self.root / relative)
        if relative == self.historical and self.before_final is not None and not self.triggered:
            self.triggered = True
            self.before_final()
        return result

    def source_assets(self, directories, umbrella_root=None):
        if not directories or any(not path.is_relative_to(self.root) for path in directories):
            raise AssertionError("fixture source discovery escaped")
        return [self.root / name for name in self.extra_assets]

    @staticmethod
    def imports(text):
        return [line.split()[1] for line in text.splitlines() if line.startswith("import ")]

    def locate(self, root):
        if Path(root) != self.root:
            raise AssertionError("unexpected locator root")
        self.locator_calls.append(str(root))
        return self.compiler

    def write_exclusive(self, path, raw, readonly=False):
        path = self.checked_path(path)
        archive = self.active.parent / "moebius-current-inputs-history" / ("prior-" + digest(raw) + ".json")
        if path not in (self.output, archive):
            raise AssertionError("unexpected fixture publication")
        if readonly is not True:
            raise AssertionError("Version and archive publication must be readonly")
        PURE_CHECKER["write_exclusive"](path, raw, readonly=readonly)
        self.writes.append((path, raw))
        self.readonly_writes.append(path)
        if path == self.output and self.after_version is not None:
            self.after_version()

    def directory_descriptor(self, path, create=False):
        return PURE_CHECKER["directory_descriptor"](self.checked_path(path), create=create)

    def atomic_json(self, path, value):
        path = self.checked_path(path)
        if path != self.root / PRODUCER.ACTIVE_DEFAULT:
            raise AssertionError("Unexpected activation destination")
        self.atomic_calls.append((path, value))
        return PURE_CHECKER["atomic_json"](path, value)

    def set_origin(self, payload=None, identity_changes=None):
        self.origin_payload = payload or {
            "schema": "proof-validation-origin-v1", "attempt_id": "fixture-public-attempt",
            "outcome": "passed", "terminal": True,
            "evidence": {"commands": [], "validation": {"mode": "all-project-replay",
                "status": "ALL_PROJECT_SOURCE_AND_AXIOM_CHECKS_PASSED"}},
        }
        raw = encoded(self.origin_payload)
        self.put(self.origin_path, raw)
        self.origin_identity = {"path": str(self.origin_path), "sha256": digest(raw),
                                "bytes": len(raw), "attempt_id": self.origin_payload["attempt_id"]}
        self.origin_identity.update(identity_changes or {})
        self.put(self.descriptor_path, encoded(self.origin_identity))

    def observe(self, *, incremental=True, public=False, **kwargs):
        options = dict(checker=self.checker, census=self.census, incremental=incremental,
                       lean_locator=self.locate)
        if public:
            options.update(public_origin_descriptor=self.descriptor_path,
                           origin_payload_validator=PURE_ORIGIN["_origin_payload"])
        options.update(kwargs)
        with isolated_execution(), patch.object(PRODUCER, "_locate_lean", side_effect=forbidden):
            return PRODUCER.observe_current_inputs(self.root, self.output, **options)

    def document(self):
        return json.loads(self.output.read_bytes())

    @property
    def active(self):
        return self.root / PRODUCER.ACTIVE_DEFAULT

    @property
    def lease(self):
        return self.active.parent / ".moebius-current-inputs-current.activation-lease"

    def hook_activation(self, hook):
        fixture = self
        class Hooked:
            def __enter__(self):
                self.actual = PRODUCER.CurrentDefaultActivation(fixture.root, fixture.checker)
                self.actual.__enter__()
                return self
            def publish(self, contract, raw, recheck):
                hook()
                return self.actual.publish(contract, raw, recheck)
            def __exit__(self, *args):
                return self.actual.__exit__(*args)
        return lambda root, checker: Hooked()


class ProducerControls(unittest.TestCase):
    def setUp(self):
        self.temporary = tempfile.TemporaryDirectory(prefix="producer-controls-")
        self.addCleanup(self.temporary.cleanup)
        self.fixture = Fixture(self.temporary.name)

    def reject(self, call, pattern=None):
        context = self.assertRaisesRegex((ValueError, RuntimeError, OSError), pattern) if pattern else self.assertRaises((ValueError, RuntimeError, OSError))
        with context:
            call()
        self.assertEqual(self.fixture.writes, [])
        self.assertFalse(self.fixture.output.exists())

    def binding(self, **kwargs):
        assets = {name: "a" * 64 for name in PURE_CHECKER["INCREMENTAL_ASSETS"] + PURE_CHECKER["PUBLIC_COMPILATION_ASSETS"]}
        with isolated_execution():
            return PRODUCER.build_incremental_binding(self.fixture.checker, assets, **kwargs)

    def test_import_is_stdlib_only_and_inert(self):
        before = set(sys.modules)
        module = load_producer()
        self.assertTrue(callable(module.observe_current_inputs))
        self.assertEqual(set(sys.modules) - before, set())
        self.assertEqual(self.fixture.writes, [])
        self.assertEqual(self.fixture.locator_calls, [])

    def test_source_main_guard_only_invokes_cli_under_main(self):
        tree = ast.parse(SOURCES[str(PRODUCER_PATH)])
        guards = [n for n in tree.body if isinstance(n, ast.If)]
        self.assertEqual(len(guards), 1)
        self.assertEqual(ast.unparse(guards[0].test), "__name__ == '__main__'")
        self.assertEqual(ast.unparse(guards[0].body[0]), "main()")

    def test_legacy_binding_absent_and_no_incremental_checker_needed(self):
        with isolated_execution():
            self.assertIsNone(PRODUCER.build_incremental_binding(object(), {}, enabled=False))

    def test_incremental_exact_seven_field_schema(self):
        actual = self.binding(enabled=True)
        self.assertEqual(actual, PRODUCER.EXPECTED_INCREMENTAL_CONTRACT)
        self.assertEqual(len(actual), 7)
        self.assertIs(actual["source_module_cache"], True)

    def test_builder_default_is_incremental(self):
        actual = self.binding()
        self.assertEqual(actual, PRODUCER.EXPECTED_INCREMENTAL_CONTRACT)

    def test_observer_default_is_incremental(self):
        with isolated_execution():
            report = PRODUCER.observe_current_inputs(self.fixture.root, self.fixture.output,
                checker=self.fixture.checker, census=self.fixture.census, lean_locator=self.fixture.locate)
        self.assertEqual(report["asset_count"], 28)
        self.assertEqual(self.fixture.document()["incremental_validation"], PRODUCER.EXPECTED_INCREMENTAL_CONTRACT)

    def test_cli_incremental_default_is_literal_true(self):
        tree = ast.parse(SOURCES[str(PRODUCER_PATH)])
        main = next(node for node in tree.body if isinstance(node, ast.FunctionDef) and node.name == "main")
        calls = [node for node in ast.walk(main) if isinstance(node, ast.Call)
                 and isinstance(node.func, ast.Attribute) and node.func.attr == "add_argument"
                 and node.args and isinstance(node.args[0], ast.Constant)
                 and node.args[0].value == "--incremental"]
        self.assertEqual(len(calls), 1)
        default = next(keyword.value for keyword in calls[0].keywords if keyword.arg == "default")
        self.assertIs(ast.literal_eval(default), True)

    def test_incremental_flag_rejects_nonliteral_booleans(self):
        for value in (0, 1, "true", None, [], {}):
            with self.subTest(value=value), self.assertRaisesRegex(ValueError, "literal Boolean"):
                self.binding(enabled=value)

    def test_template_boolean_and_value_drift_rejected(self):
        for key, value in (("source_module_cache", 1), ("schema", "other"), ("complete_owner_census", "cached")):
            original = dict(self.fixture.checker.INCREMENTAL_CONTRACT)
            self.fixture.checker.INCREMENTAL_CONTRACT[key] = value
            with self.subTest(key=key), self.assertRaisesRegex(ValueError, "API has changed"):
                self.binding(enabled=True)
            self.fixture.checker.INCREMENTAL_CONTRACT = original

    def test_template_extra_and_missing_fields_rejected(self):
        original = dict(self.fixture.checker.INCREMENTAL_CONTRACT)
        for changed in ({**original, "extra": True}, {k: v for k, v in original.items() if k != "schema"}):
            self.fixture.checker.INCREMENTAL_CONTRACT = changed
            with self.assertRaisesRegex(ValueError, "API has changed"):
                self.binding(enabled=True)

    def test_assets_reject_malformed_digest_or_key(self):
        for assets in ({"asset": "A" * 64}, {"asset": "a" * 63}, {1: "a" * 64}, {"asset": True}, []):
            with self.subTest(assets=assets), self.assertRaisesRegex(ValueError, "SHA256"):
                PRODUCER.build_incremental_binding(self.fixture.checker, assets, enabled=True)

    def test_assets_require_every_runtime_member(self):
        assets = {name: "a" * 64 for name in PURE_CHECKER["INCREMENTAL_ASSETS"][1:]}
        with self.assertRaisesRegex(ValueError, "exactly once"):
            PRODUCER.build_incremental_binding(self.fixture.checker, assets, enabled=True)

    def test_assets_alias_collision_rejected_by_actual_helper(self):
        assets = {name: "a" * 64 for name in PURE_CHECKER["INCREMENTAL_ASSETS"]}
        assets["scripts/repository_paths.py"] = "b" * 64
        with self.assertRaisesRegex(ValueError, "exactly once"):
            PRODUCER.build_incremental_binding(self.fixture.checker, assets, enabled=True)

    def test_assets_single_legacy_alias_supported(self):
        assets = {name: "a" * 64 for name in PURE_CHECKER["INCREMENTAL_ASSETS"]}
        assets["scripts/repository_paths.py"] = assets.pop(".noeaaeue/scripts/repository_paths.py")
        actual = PRODUCER.build_incremental_binding(self.fixture.checker, assets, enabled=True)
        self.assertEqual(actual, PRODUCER.EXPECTED_INCREMENTAL_CONTRACT)

    def test_public_origin_requires_incremental_selection(self):
        for kwargs in ({"public_origin": self.fixture.origin_identity}, {"origin_bytes": b"x"}):
            with self.assertRaisesRegex(ValueError, "explicit incremental"):
                self.binding(enabled=False, **kwargs)

    def test_orphan_origin_bytes_rejected(self):
        with self.assertRaisesRegex(ValueError, "no selected identity"):
            self.binding(enabled=True, origin_bytes=b"x")

    def test_public_exact_origin_is_identity_only(self):
        origin = self.fixture.origin_path.read_bytes()
        actual = self.binding(enabled=True, public_origin=self.fixture.origin_identity,
                              origin_bytes=origin, origin_payload_validator=PURE_ORIGIN["_origin_payload"])
        self.assertEqual(actual["public_compilation_origin"], self.fixture.origin_identity)
        self.assertEqual(len(actual), 8)
        self.assertNotIn("kernel_checked", actual)

    def test_public_descriptor_members_paths_types_and_digest(self):
        original = self.fixture.origin_identity
        variants = [[], {**original, "extra": 1}, {k: v for k, v in original.items() if k != "path"}]
        for key, value in (("path", "relative.json"), ("path", "/tmp/../origin"),
                           ("path", ""), ("sha256", "A" * 64), ("sha256", "0"),
                           ("bytes", True), ("bytes", 0), ("bytes", "1"),
                           ("attempt_id", ""), ("attempt_id", 12)):
            variants.append({**original, key: value})
        for value in variants:
            with self.subTest(value=value), self.assertRaisesRegex(ValueError, "exact explicit"):
                PRODUCER._public_identity(value)

    def test_public_requires_raw_bytes_and_bound_parser(self):
        for raw, parser in ((bytearray(b"x"), PURE_ORIGIN["_origin_payload"]),
                            (self.fixture.origin_path.read_bytes(), None)):
            with self.assertRaisesRegex(ValueError, "exact bytes.*bound origin parser"):
                self.binding(enabled=True, public_origin=self.fixture.origin_identity,
                             origin_bytes=raw, origin_payload_validator=parser)

    def test_public_hash_and_length_mismatch_rejected(self):
        for changes in ({"sha256": "a" * 64}, {"bytes": self.fixture.origin_identity["bytes"] + 1}):

    def routed_main(self, argv, *, checker=None):
        calls = []
        checker = checker or self.fixture.checker
        def observe(*args, **kwargs):
            calls.append((args, kwargs))
            return {"fixture": "routing only, no observation or acceptance"}
        namespace = PRODUCER.main.__globals__
        output = io.StringIO()
        with isolated_execution(), ExitStack() as stack:
            previous_import = builtins.__import__
            allowed = {"check_moebius_response": checker, "moebius_owner_census": self.fixture.census,
                       "incremental_lean": SimpleNamespace(_origin_payload=PURE_ORIGIN["_origin_payload"])}
            def fixture_import(name, *args, **kwargs):
                if name in allowed:
                    return allowed[name]
                return previous_import(name, *args, **kwargs)
            stack.enter_context(patch("builtins.__import__", side_effect=fixture_import))
            stack.enter_context(patch.dict(namespace, ROOT=self.fixture.root, observe_current_inputs=observe))
            stack.enter_context(patch.object(sys, "path", list(sys.path)))
            stack.enter_context(patch.object(sys, "stdout", output))
            PRODUCER.main(argv)
        return calls, output.getvalue()

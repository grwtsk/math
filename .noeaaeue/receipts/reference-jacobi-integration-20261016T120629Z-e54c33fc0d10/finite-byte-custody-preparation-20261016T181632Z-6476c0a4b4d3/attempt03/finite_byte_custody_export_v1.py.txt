#!/usr/bin/env python3
"""Inert finite raw-byte custody exporter; no proof or native-cache authority.

Only --execute plus an externally supplied exact request SHA enables copying.
No receipt discovery, recursive walks, subprocesses, project imports or code
execution occur. The CLI is fixed to ROOT; fixtures call the pure filesystem
API with their own isolated root. A successful export means byte custody only.
The exact externally root-approved request SHA is the authorization input;
the request's approval file is opaque identity-bound provenance, not authority.
The final rereads bound cooperative drift; they are not a hostile-host atomic
snapshot. Failures retain partial copies and a best-effort failure record.
An inventory is always PREPARED. Completion requires its separate terminal
record AND a retained successful API return/source-bound CLI exit-zero result;
absence of a best-effort failure marker alone never establishes completion.

Reader/writer patterns adapt retained serial_native.py safe_read/write_raw_new
(SHA64000202b88d2d7c3fcd05aabd476164cf8b7aeea846fe714639aeea4162a15b),
adding a 100000000-byte limit, descriptor ancestry checks and finite policy.
"""

MAX_REQUEST_BYTES = 10000000
MAX_FILES = 20000
CONTEXTS = frozenset(('repaired-public-first', 'repaired-public-warm', 'application',

ROLES = frozenset(('receipt', 'origin', 'cache-publication', 'frozen-inputs',

EXCLUDED_INDEXES = (

class CustodyError(ValueError):
    pass


def need(condition, message):
    if not condition:
        raise CustodyError(message)


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def decoded(raw):
    def pairs(items):
        answer = {}
        for key, value in items:
            need(key not in answer, 'duplicate JSON key')
            answer[key] = value
        return answer
    def invalid(_):
        raise CustodyError('nonfinite JSON number')
    return json.loads(raw, object_pairs_hook=pairs, parse_constant=invalid)


def encoded(value):
    return (json.dumps(value, sort_keys=True, indent=2, allow_nan=False) + '\n').encode()


def absolute(value):
    need(type(value) is str and value.startswith('/') and '\\' not in value,
         'absolute path required')
    need(value != '/' and all(part not in ('', '.', '..') for part in value.split('/')[1:]),
         'noncanonical/traversal path')
    path = Path(value)
    need(str(path) == value, 'noncanonical path')
    return path


def relative(value):
    need(type(value) is str and not value.startswith('/') and '\\' not in value,
         'relative path required')
    need(all(part not in ('', '.', '..') for part in value.split('/')), 'unsafe relative path')
    return value


def below(path, root):
    try:
        return path.relative_to(root)
    except ValueError as error:
        raise CustodyError('path outside source root') from error


def file_identity(value):
    need(type(value) is dict and set(value) == {'path', 'sha256', 'bytes'}, 'closed identity required')
    path = absolute(value['path'])
    need(type(value['sha256']) is str and re.fullmatch('[0-9a-f]{64}', value['sha256']),
         'lowercase SHA256 required')
    need(type(value['bytes']) is int and 0 <= value['bytes'] <= MAX_BYTES, 'invalid/overlimit size')
    return path


def stat_key(item):
    return (item.st_dev, item.st_ino, item.st_mode, item.st_size,
            item.st_mtime_ns, item.st_ctime_ns, item.st_nlink)


def parent_fd(path):
    """Open every parent without following links; return its exact inode chain."""
    path = absolute(str(path))
    fd = os.open('/', os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
    chain = [(os.fstat(fd).st_dev, os.fstat(fd).st_ino)]
    try:
        for part in path.parts[1:-1]:
            child = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fd)
            os.close(fd)
            fd = child
            item = os.fstat(fd)
            chain.append((item.st_dev, item.st_ino))
        return fd, tuple(chain)
    except BaseException:
        os.close(fd)
        raise


@dataclass(frozen=True)
class Read:
    data: bytes
    identity: tuple
    parents: tuple


def safe_read(path, limit=MAX_BYTES):
    """Reject nonregular/symlink/changed paths, including a stat→FIFO race."""
    path = absolute(str(path))
    fd, chain = parent_fd(path)
    try:
        before = os.stat(path.name, dir_fd=fd, follow_symlinks=False)
        need(stat.S_ISREG(before.st_mode) and before.st_size <= limit, 'nonregular/overlimit input')
        leaf = os.open(path.name, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK, dir_fd=fd)
        with os.fdopen(leaf, 'rb') as stream:
            opened = os.fstat(stream.fileno())
            need(stat.S_ISREG(opened.st_mode) and stat_key(opened) == stat_key(before),
                 'opened input identity changed')
            raw = stream.read(limit + 1)
            after = os.fstat(stream.fileno())
        final = os.stat(path.name, dir_fd=fd, follow_symlinks=False)
        need(len(raw) <= limit and stat_key(before) == stat_key(after) == stat_key(final),
             'input changed during bounded read')
        check, live_chain = parent_fd(path)
        os.close(check)
        need(chain == live_chain, 'input parent path drift')
        return Read(raw, stat_key(before), chain)
    finally:
        os.close(fd)


def describe(path, raw):
    return {'path': str(path), 'sha256': sha(raw), 'bytes': len(raw)}


def bound_read(identity):
    path = file_identity(identity)
    observed = safe_read(path)
    need(sha(observed.data) == identity['sha256'] and len(observed.data) == identity['bytes'],
         'stale file hash/size: ' + str(path))
    return observed


def reread(path, observed):
    need(safe_read(path) == observed, 'source/path drift: ' + str(path))


@dataclass(frozen=True)
class Policy:
    root: Path
    excluded: frozenset
    indexes: tuple = ()


def production_policy():
    excluded = set(FIXED_EXCLUSIONS)
    indexes = []
    for name, digest, size in EXCLUDED_INDEXES:
        path = ROOT / name
        observed = bound_read({'path': str(path), 'sha256': digest, 'bytes': size})
        data = decoded(observed.data)
        rows = data['data']['created_files']
        need(type(rows) is list and len(rows) <= MAX_FILES, 'invalid exclusion index')
        for row in rows:
            need(type(row) is dict and 'path' in row, 'invalid excluded occurrence')
            excluded.add(relative(row['path']))
        excluded.add(name)
        indexes.append((path, observed))
    return Policy(ROOT, frozenset(excluded), tuple(indexes))


def allowed(path, policy):
    name = below(path, policy.root).as_posix()
    parts = Path(name).parts
    need(name not in policy.excluded, 'excluded APP-WSC/foreign occurrence')
    need(not any(part in ('.git', '.lake', '__pycache__') for part in parts), 'excluded build/cache path')
    need(not any(name == prefix or name.startswith(prefix + '/') for prefix in (
        '.noeaaeue/cache/objects', '.noeaaeue/records', '.noeaaeue/spec', '.noeaaeue/interfaces')),
         'excluded bulk/profile data path')
    need(path.suffix.lower() not in ('.pyc', '.pyo', '.olean', '.ilean', '.o', '.obj', '.a',
                                     '.so', '.dylib', '.dll', '.exe', '.c', '.bc'),
         'excluded native/bytecode object')
    need(not re.search(r'\.(?:olean|ilean|pyc|pyo|ir|o|obj|a|so|dylib|dll|exe|c|bc)(?:\.|$)', path.name.lower()),
         'excluded native/bytecode object sidecar or raw wrapper')
    if name.startswith('.noeaaeue/.local/incremental-lean/'):
        need(path.name in ('proof-validation-origin.json', 'cache-publication.json'),
             'excluded native compilation cache bulk')


def validate_request(value, policy, own_identity):
    need(type(value) is dict and set(value) == {
        'schema', 'scope', 'root', 'destination', 'exporter', 'approval', 'files'}, 'closed request required')
    need(value['schema'] == 'finite-raw-byte-custody-request-v1'
         and value['scope'] == 'raw-byte-custody-only', 'wrong request schema/scope')
    need(absolute(value['root']) == policy.root, 'wrong source root')
    need(value['exporter'] == own_identity, 'exporter identity mismatch')
    file_identity(value['exporter'])
    approval_path = file_identity(value['approval'])
    allowed(approval_path, policy)
    destination = absolute(value['destination'])
    target_name = below(destination, policy.root / '.noeaaeue/receipts')
    need(target_name.parts and destination != policy.root / '.noeaaeue/receipts', 'invalid custody destination')
    need(not destination.exists() and not destination.is_symlink(), 'destination already exists')
    rows = value['files']
    need(type(rows) is list and 0 < len(rows) <= MAX_FILES, 'finite nonempty file list required')
    seen, seen_original = set(), set()
    for row in rows:
        need(type(row) is dict and set(row) == {'original', 'original_absolute', 'sha256', 'bytes',
             'context', 'role', 'origin_attempt_id', 'retention'}, 'closed file row required')
        original = row['original']
        need(type(original) is str, 'original occurrence must be a string')
        resolved = absolute(original) if original.startswith('/') else policy.root / relative(original)
        path = file_identity({'path': row['original_absolute'], 'sha256': row['sha256'], 'bytes': row['bytes']})
        need(resolved == path, 'original occurrence mapping mismatch')
        allowed(path, policy)
        need(path != destination and destination not in path.parents, 'source overlaps destination')
        need(str(path) not in seen and original not in seen_original, 'duplicate original occurrence')
        seen.add(str(path)); seen_original.add(original)
        need(type(row['context']) is str and row['context'] in CONTEXTS, 'invalid custody context')
        need(type(row['role']) is str and row['role'] in ROLES, 'invalid custody role')
        attempt = row['origin_attempt_id']
        need(attempt is None or type(attempt) is str and 0 < len(attempt) <= 256, 'invalid original attempt ID')
        need(row['retention'] in ('new-exact-copy', 'already-durable-reference'), 'invalid retention')
        need(row['retention'] != 'already-durable-reference' or '.local' not in path.parts,
             'ignored scratch cannot be declared durable reference')
    return destination


def write_raw_new(path, raw):
    need(type(raw) is bytes and len(raw) <= MAX_BYTES, 'overlimit/nonbyte output')
    fd, chain = parent_fd(path)
    try:
        leaf = os.open(path.name, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o400, dir_fd=fd)
        with os.fdopen(leaf, 'wb') as stream:
            stream.write(raw); stream.flush(); os.fsync(stream.fileno())
        check, live_chain = parent_fd(path)
        os.close(check)
        need(chain == live_chain, 'output parent path drift')
        observed = safe_read(path)
        need(observed.data == raw and stat.S_IMODE(observed.identity[2]) == 0o400
             and observed.identity[6] == 1, 'immutable output readback mismatch')
        return observed
    finally:
        os.close(fd)


def mkdir_new(path):
    fd, chain = parent_fd(path)
    try:
        os.mkdir(path.name, mode=0o700, dir_fd=fd)
        child = os.open(path.name, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fd)
        identity = os.fstat(child)
        os.close(child)
        check, live_chain = parent_fd(path)
        live = os.stat(path.name, dir_fd=check, follow_symlinks=False)
        os.close(check)
        need(chain == live_chain and (identity.st_dev, identity.st_ino) == (live.st_dev, live.st_ino),
             'new destination path drift')
        return (identity.st_dev, identity.st_ino)
    finally:
        os.close(fd)


def directory_guard(path, expected):
    fd, _ = parent_fd(path)
    try:
        item = os.stat(path.name, dir_fd=fd, follow_symlinks=False)
        need(stat.S_ISDIR(item.st_mode) and (item.st_dev, item.st_ino) == expected,
             'custody destination changed')
    finally:
        os.close(fd)


def export_request(request_path, request_sha, *, policy, own_path):
    """Explicit root/fixture caller only. No real export is run by preparation."""
    request_path, own_path = absolute(str(request_path)), absolute(str(own_path))
    need(type(request_sha) is str and re.fullmatch('[0-9a-f]{64}', request_sha), 'request SHA required')
    request_observed = safe_read(request_path, MAX_REQUEST_BYTES)
    need(sha(request_observed.data) == request_sha, 'request SHA mismatch')
    own_observed = safe_read(own_path)
    own_identity = describe(own_path, own_observed.data)
    value = decoded(request_observed.data)
    destination = validate_request(value, policy, own_identity)
    approval = bound_read(value['approval'])
    observations = []
    for row in value['files']:
        path = Path(row['original_absolute'])
        observed = bound_read({'path': str(path), 'sha256': row['sha256'], 'bytes': row['bytes']})
        observations.append((path, observed.identity, observed.parents))
    # Persist only bounded identities here; file bytes are read one at a time.
    def sources_guard():
        reread(request_path, request_observed)
        reread(own_path, own_observed)
        reread(Path(value['approval']['path']), approval)
        for path, observed in policy.indexes:
            reread(path, observed)
        for row, (path, identity, parents) in zip(value['files'], observations):
            observed = bound_read({'path': str(path), 'sha256': row['sha256'], 'bytes': row['bytes']})
            need((observed.identity, observed.parents) == (identity, parents), 'source path identity drift')
    sources_guard()
    made, copies, objects = False, [], {}
    try:
        destination_identity = mkdir_new(destination); made = True
        objects_dir = destination / 'sha256'
        objects_identity = mkdir_new(objects_dir)
        request_copy = destination / 'request.json.txt'
        request_copy_observed = write_raw_new(request_copy, request_observed.data)
        for row, (path, identity, parents) in zip(value['files'], observations):
            original = bound_read({'path': str(path), 'sha256': row['sha256'], 'bytes': row['bytes']})
            need((original.identity, original.parents) == (identity, parents), 'source drift before copy')
            if row['retention'] == 'already-durable-reference':
                copy = path
            else:
                copy = objects_dir / (row['sha256'] + '.txt')
                if row['sha256'] in objects:
                    # A matching digest is never enough: compare actual bytes.
                    observed = safe_read(copy)
                    need(observed.data == original.data
                         and (observed.identity, observed.parents) == objects[row['sha256']],
                         'digest collision or dedup object drift')
                else:
                    observed = write_raw_new(copy, original.data)
                    objects[row['sha256']] = (observed.identity, observed.parents)
            need(safe_read(copy).data == original.data, 'copy raw-byte mismatch')
            reread(path, original)
            copies.append({**row, 'copy': str(copy)})
        sources_guard()
        def copies_guard():
            for row, (path, identity, parents) in zip(value['files'], observations):
                original = bound_read({'path': str(path), 'sha256': row['sha256'], 'bytes': row['bytes']})
                need((original.identity, original.parents) == (identity, parents), 'late source path drift')
                copy = path if row['retention'] == 'already-durable-reference' else objects_dir / (row['sha256'] + '.txt')
                observed = safe_read(copy)
                need(observed.data == original.data, 'late copy raw-byte mismatch')
                if row['retention'] == 'new-exact-copy':
                    need((observed.identity, observed.parents) == objects[row['sha256']], 'late copy identity drift')
        copies_guard()
        reread(request_copy, request_copy_observed)
        directory_guard(destination, destination_identity)
        directory_guard(objects_dir, objects_identity)
        manifest = {
            'schema': 'native-receipt-byte-custody-v1', 'status': 'PREPARED_RAW_BYTE_INVENTORY_NOT_COMPLETION',
            'scope': 'Exact raw bytes only; no proof, replay, portable cache or publication authority.',
            'request': describe(request_path, request_observed.data),
            'request_copy': describe(request_copy, request_copy_observed.data),
            'exporter': own_identity, 'approval': value['approval'],
            'authorization_boundary': 'Exact externally root-approved request SHA; approval file is identity-only provenance.',
            'policy_indexes': [describe(path, observed.data) for path, observed in policy.indexes],
            'copies': copies, 'unique_new_objects': len(objects), 'omissions': [],
            'mathematical_acceptance': False, 'native_jobs': 0,
            'limits': ['Finite approved allowlist only; no recursive discovery.',
                       'No build/cache objects preserved; no expanded code executed.',
                       'Cooperative stable rereads; not an atomic hostile-host snapshot.'],
        }
        manifest_path = destination / 'manifest.json'
        manifest_observed = write_raw_new(manifest_path, encoded(manifest))
        # The prepared inventory is never a terminal success attestation.
        sources_guard()
        copies_guard()
        reread(request_copy, request_copy_observed)
        reread(manifest_path, manifest_observed)
        directory_guard(destination, destination_identity)
        directory_guard(objects_dir, objects_identity)
        completion_path = destination / 'completion.json'
        completion = {
            'schema': 'finite-byte-custody-terminal-record-v1',
            'status': 'TERMINAL_CUSTODY_RECORD_CLI_SUCCESS_REQUIRED',
            'manifest': describe(manifest_path, manifest_observed.data),
            'request': describe(request_path, request_observed.data), 'exporter': own_identity,
            'required_external_attestation': 'Retain successful API result or source-bound CLI exit-zero result binding this exact terminal record. This file alone is insufficient.',
            'reject_if_failure_record_present': True, 'mathematical_acceptance': False, 'native_jobs': 0,
        }
        completion_observed = write_raw_new(completion_path, encoded(completion))
        sources_guard(); copies_guard()
        reread(manifest_path, manifest_observed)
        reread(completion_path, completion_observed)
        directory_guard(destination, destination_identity)
        directory_guard(objects_dir, objects_identity)
        return {'schema': 'finite-byte-custody-export-result-v1', 'status': 'RAW_BYTE_CUSTODY_API_RETURNED_SUCCESS',
                'manifest': describe(manifest_path, manifest_observed.data),
                'completion': describe(completion_path, completion_observed.data),
                'request': describe(request_path, request_observed.data),
                'mathematical_acceptance': False, 'native_jobs': 0}
    except BaseException as error:
        if made:
            failure = {'schema': 'finite-byte-custody-failure-v1', 'status': 'FAILED_INCOMPLETE',
                       'request': describe(request_path, request_observed.data),
                       'completed_mapping_rows': len(copies), 'error_type': type(error).__name__,
                       'error': str(error), 'mathematical_acceptance': False, 'native_jobs': 0,
                       'rule': 'Reject this directory as complete even if inventory/terminal records exist. A successful external return is required.'}
            try:
                directory_guard(destination, destination_identity)
                write_raw_new(destination / 'failure.json', encoded(failure))
            except BaseException:
                # Never delete or overwrite partial artifacts to conceal failure.
                pass
        raise


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--execute', action='store_true')
    parser.add_argument('--request', type=Path)
    parser.add_argument('--request-sha256')
    args = parser.parse_args(argv)
    need(args.execute and args.request is not None and args.request_sha256 is not None,
         'inert by default; explicit exact request and --execute required')
    policy = production_policy()
    result = export_request(args.request, args.request_sha256, policy=policy, own_path=Path(__file__).absolute())
    print(json.dumps(result, sort_keys=True))


if __name__ == '__main__':
    main()

import Lean

/- A syntax inventory only. This driver never elaborates the inventoried files,
   follows their imports, expands their macros, or inspects generated constants. -/
open Lean

namespace SourceDeclarationCensus

def declarationKind (stx : Syntax) : Option String :=
  if stx.isOfKind ``Parser.Command.definition then some "def"
  else if stx.isOfKind ``Parser.Command.theorem then some "theorem"
  else if stx.isOfKind ``Parser.Command.abbrev then some "abbrev"
  else if stx.isOfKind ``Parser.Command.opaque then some "opaque"
  else if stx.isOfKind ``Parser.Command.instance then some "instance"
  else if stx.isOfKind ``Parser.Command.axiom then some "axiom"
  else if stx.isOfKind ``Parser.Command.example then some "example"
  else if stx.isOfKind ``Parser.Command.inductive then some "inductive"
  else if stx.isOfKind ``Parser.Command.coinductive then some "coinductive"
  else if stx.isOfKind ``Parser.Command.classInductive then some "class inductive"
  else if stx.isOfKind ``Parser.Command.structure then
    some (if stx[0].isOfKind ``Parser.Command.classTk then "class" else "structure")
  else none

/-- Descend only into known command wrappers, never into a declaration body,
    field, constructor, quotation, macro/elaborator body, or arbitrary syntax. -/
partial def commandKinds (stx : Syntax) : Array String := Id.run do
  if stx.isOfKind ``Parser.Command.declaration then
    return (declarationKind stx[1]).toArray
  if stx.getKind == `lemma then return #["lemma"]
  if stx.isOfKind ``Parser.Command.in then return commandKinds stx[2]
  if stx.isOfKind ``Parser.Command.withWeakNamespace then return commandKinds stx[2]
  if stx.isOfKind ``Parser.Command.withExporting then return commandKinds stx[1]
  if stx.getKind == `Lean.guardMsgsCmd then return commandKinds stx[4]
  if stx.isOfKind ``Parser.Command.mutual then
    let mut result := #[]
    for cmd in stx[1].getArgs do result := result ++ commandKinds cmd
    return result
  return #[]

def histogram (kinds : Array String) : Json :=
  let keys := kinds.toList.eraseDups
  Json.mkObj <| keys.map fun key => (key, toJson (kinds.filter (· == key)).size)

def parseSource (env : Environment) (path contents : String) : IO Json := do
  let ctx := Parser.mkInputContext contents path
  let (_, firstState, firstMessages) ← Parser.parseHeader ctx
  let mut state := firstState
  let mut messages := firstMessages
  let mut kinds : Array String := #[]
  let mut done := false
  while !done do
    let (stx, nextState, nextMessages) :=
      Parser.parseCommand ctx { env, options := {} } state messages
    kinds := kinds ++ commandKinds stx
    state := nextState
    messages := nextMessages
    done := stx.isOfKind ``Parser.Command.eoi
  let mut errors : Array String := #[]
  for message in messages.toList do errors := errors.push (← message.toString)
  if !errors.isEmpty then
    return Json.mkObj [("path", toJson path), ("status", toJson "unparsed"),
      ("kinds", Json.mkObj []), ("errors", toJson errors)]
  let declarations := (kinds.filter (· != "example")).size
  let theorems := (kinds.filter fun kind => kind == "theorem" || kind == "lemma").size
  return Json.mkObj [("path", toJson path), ("status", toJson "parsed"),
    ("counts", Json.mkObj [("declarations", toJson declarations), ("theorems", toJson theorems)]),
    ("kinds", histogram kinds), ("errors", toJson errors)]

unsafe def run (args : List String) : IO UInt32 := do
  let [manifestPath] := args | throw <| IO.userError "expected one frozen manifest path"
  let manifestText ← IO.FS.readFile manifestPath
  let manifest ← IO.ofExcept <| Json.parse manifestText
  let sources ← IO.ofExcept <| manifest.getObjValAs? (Array Json) "sources"
  initSearchPath (← findSysroot)
  enableInitializersExecution
  let env ← importModules #[{module := `Mathlib}, {module := `Lean}] {} 0 (loadExts := true)
  for source in sources do
    let path ← IO.ofExcept <| source.getObjValAs? String "path"
    let frozenPath ← IO.ofExcept <| source.getObjValAs? String "frozen_path"
    let contents ← IO.FS.readFile frozenPath
    IO.println s!"LEAN_SOURCE_CENSUS::{(← parseSource env path contents).compress}"
  return 0

end SourceDeclarationCensus

unsafe def main (args : List String) : IO UInt32 := SourceDeclarationCensus.run args

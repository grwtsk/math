/-
Mathlib sieve/coverage candidate surface for the finite bit-matrix site.

This file translates a finite `CoveringFamily` into Mathlib's presieve and
generated-sieve vocabulary over the already-defined `Category (SiteObject M)`.
It is deliberately still a candidate layer: it does not define a Mathlib
`GrothendieckTopology`, a sheaf category, a Lawvere-Tierney topology, an
essential geometric morphism, a topos, or theorem-promotion authority.
-/

/-- A type-valued index for the pieces of a finite covering family. -/
structure CoverPieceIndex {M : BitMatrix} (cover : CoveringFamily M) where
  piece : BitInterval M
  pieceMem : piece ∈ cover.pieces

namespace CoverPieceIndex

def sourceObject {M : BitMatrix} {cover : CoveringFamily M}
    (idx : CoverPieceIndex cover) : SiteObject M :=
  SiteObject.ofInterval idx.piece

def arrow {M : BitMatrix} {cover : CoveringFamily M}
    (idx : CoverPieceIndex cover) :
    idx.sourceObject ⟶ cover.asSiteObject :=
  { morphism := cover.pieceMorphism idx.piece idx.pieceMem }

end CoverPieceIndex

/-- The Mathlib presieve generated by the finite cover-piece arrows. -/
def coveringPresieve {M : BitMatrix} (cover : CoveringFamily M) :
    Presieve cover.asSiteObject :=
  Presieve.ofArrows
    (fun idx : CoverPieceIndex cover => idx.sourceObject)
    (fun idx : CoverPieceIndex cover => idx.arrow)

/-- The Mathlib sieve generated by a finite covering presieve. -/
def coveringSieveGenerated {M : BitMatrix} (cover : CoveringFamily M) :
    Sieve cover.asSiteObject :=
  Sieve.generate (coveringPresieve cover)

theorem covering_presieve_contains_piece_arrow
    {M : BitMatrix} (cover : CoveringFamily M)
    (idx : CoverPieceIndex cover) :
    coveringPresieve cover idx.arrow := by
  exact Presieve.ofArrows.mk idx

theorem covering_sieve_generated_contains_piece_arrow
    {M : BitMatrix} (cover : CoveringFamily M)
    (idx : CoverPieceIndex cover) :
    coveringSieveGenerated cover idx.arrow := by
  exact (Sieve.le_generate (coveringPresieve cover)) _
    idx.arrow (covering_presieve_contains_piece_arrow cover idx)

Mathlib Grothendieck-topology value for the finite bit-matrix site.

This is the first actual `CategoryTheory.GrothendieckTopology` object attached
to the finite `SiteObject M` category. It is defined as the infimum of all
Mathlib Grothendieck topologies that contain the generated sieves coming from
finite `CoveringFamily` records.

This file still does not construct a sheaf category, Lawvere-Tierney topology,
essential geometric morphism, topos, or theorem-promotion bridge.
-/

/--
The Mathlib Grothendieck topology generated by the finite covering-family
sieves. It is the smallest Mathlib topology, in the complete-lattice sense,
that contains every `coveringSieveGenerated cover`.
-/
def coverGeneratedTopology (M : BitMatrix) :
    CategoryTheory.GrothendieckTopology (SiteObject M) :=
  sInf
    { J : CategoryTheory.GrothendieckTopology (SiteObject M) |
      ∀ cover : CoveringFamily M,
        coveringSieveGenerated cover ∈ J cover.asSiteObject }

theorem covering_sieve_mem_coverGeneratedTopology
    {M : BitMatrix} (cover : CoveringFamily M) :
    coveringSieveGenerated cover ∈
      (coverGeneratedTopology M) cover.asSiteObject := by
  rw [coverGeneratedTopology, CategoryTheory.GrothendieckTopology.mem_sInf]
  intro J hJ
  exact hJ cover

theorem cover_generated_topology_has_maximal_sieve
    (M : BitMatrix) (object : SiteObject M) :
    (⊤ : Sieve object) ∈ (coverGeneratedTopology M) object := by
  exact CategoryTheory.GrothendieckTopology.top_mem
    (coverGeneratedTopology M) object

theorem cover_generated_topology_pullback_stable
    {M : BitMatrix} {source target : SiteObject M}
    (sieve : Sieve target) (arrow : source ⟶ target)
    (hsieve : sieve ∈ (coverGeneratedTopology M) target) :
    sieve.pullback arrow ∈ (coverGeneratedTopology M) source := by
  exact CategoryTheory.GrothendieckTopology.pullback_stable
    (coverGeneratedTopology M) arrow hsieve

theorem cover_generated_topology_transitive
    {M : BitMatrix} {object : SiteObject M}
    {covering : Sieve object}
    (hcovering : covering ∈ (coverGeneratedTopology M) object)
    (candidate : Sieve object)
    (hlocal :
      ∀ ⦃source : SiteObject M⦄ ⦃arrow : source ⟶ object⦄,

def homInduction {P : {x y : C ⊕ D} → (x ⟶ y) → Sort*}
    (inl : ∀ x y : C, (f : x ⟶ y) → P ((inl_ C D).map f))
    (inr : ∀ x y : D, (f : x ⟶ y) → P ((inr_ C D).map f))
    {x y : C ⊕ D} (f : x ⟶ y) : P f :=
  match x, y, f with
  | .inl x, .inl y, f => inl x y f.down
  | .inr x, .inr y, f => inr x y f.down

@[simp]

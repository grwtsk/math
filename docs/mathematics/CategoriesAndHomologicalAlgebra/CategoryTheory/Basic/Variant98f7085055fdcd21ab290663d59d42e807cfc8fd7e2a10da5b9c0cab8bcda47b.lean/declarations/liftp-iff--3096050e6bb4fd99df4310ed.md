theorem liftP_iff {α : TypeVec n} (p : ∀ ⦃i⦄, α i → Prop) (x : F α) :
    LiftP p x ↔ ∃ a f, x = abs ⟨a, f⟩ ∧ ∀ i j, p (f i j) := by

    cases' h : repr y with a f

    intro i j
    apply (f i j).property

  use abs ⟨a, fun i j => ⟨f i j, h₁ i j⟩⟩

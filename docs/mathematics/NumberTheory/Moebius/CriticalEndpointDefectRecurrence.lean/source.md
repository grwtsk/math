# Recurrence for the critical endpoint defect

The new Möbius coefficient cancels exactly when the endpoint defect advances
one step.  Consequently the defect is a smoothed signed transform of the
ordinary Mertens prefix, with kernel
`n⁻¹ᐟ² - (n+1)⁻¹ᐟ²`.  This identifies the remaining critical-boundary debt
without imposing a pointwise Mertens power saving.
-/

/-- One-step recurrence for the square-root-weighted Möbius prefix. -/
theorem pickupWeightedPrefix_succ (n : ℕ) :
    pickupWeightedPrefix (n + 1) =
      pickupWeightedPrefix n +
        moebiusReal (n + 1) *
          (Real.sqrt ((n + 1 : ℕ) : ℝ))⁻¹ := by
  unfold pickupWeightedPrefix
  rw [Finset.sum_Icc_succ_top (by omega : 1 ≤ n + 1)]

/-- Exact one-step evolution of the endpoint defect.  The new value
`μ(n+1)` cancels between the two prefixes. -/
theorem criticalEndpointDefect_succ (n : ℕ) :
    criticalEndpointDefect (n + 1) =
      criticalEndpointDefect n +
        pickupMertensPrefix n * criticalEndpointGap n := by
  rw [criticalEndpointDefect, criticalEndpointDefect,
    criticalEndpointGap, pickupWeightedPrefix_succ,
    pickupMertensPrefix_succ]
  ring

/-- The endpoint defect starts at zero. -/
theorem criticalEndpointDefect_zero : criticalEndpointDefect 0 = 0 := by
  simp [criticalEndpointDefect, pickupWeightedPrefix, pickupMertensPrefix]

/-- Smoothed signed Mertens transform generated by the endpoint-gap kernel. -/
def criticalMertensTransform (n : ℕ) : ℝ :=
  ∑ j ∈ Finset.range n,
    pickupMertensPrefix j * criticalEndpointGap j

/-- Telescoping the recurrence identifies the endpoint defect with the
smoothed Mertens transform. -/
theorem criticalEndpointDefect_eq_mertensTransform (n : ℕ) :
    criticalEndpointDefect n = criticalMertensTransform n := by
  induction n with
  | zero => simp [criticalEndpointDefect_zero, criticalMertensTransform]
  | succ n ih =>
      rw [criticalEndpointDefect_succ, ih]
      unfold criticalMertensTransform
      rw [Finset.sum_range_succ]

/-- Square summability of the smoothed signed Mertens transform at the exact
critical cell mass. -/
def CriticalMertensTransformSummable : Prop :=
  Summable (fun n : ℕ =>
    criticalMertensTransform n ^ 2 * exponentialLogDrop 1 n)

theorem endpointDefectSummable_iff_mertensTransformSummable :
    CriticalEndpointDefectSummable ↔ CriticalMertensTransformSummable := by
  unfold CriticalEndpointDefectSummable CriticalMertensTransformSummable
  unfold criticalEndpointDefectEnergy
  simp_rw [criticalEndpointDefect_eq_mertensTransform]

/-- The completed critical boundary is exactly the square-summability problem
for the smoothed signed Mertens transform. -/
theorem criticalBoundary_iff_mertensTransformSummable :
    CriticalBoundaryVerticalLineUniform ↔ CriticalMertensTransformSummable :=
  criticalBoundary_iff_endpointDefectSummable.trans
    endpointDefectSummable_iff_mertensTransformSummable

/-!
# Source-bound mechanism-to-packing recurrence rule

The mechanism bridge no longer supplies a completed infinite packing sequence.
It supplies only:

1. an initializer for the first three source-valid points; and
2. a one-step rule producing one new point and one new valid source label.

The infinite positive packing is then generated canonically by recursion.  All
stage nonnegativity follows from the volume/Gram geometry.  The remaining
analytic obligation is the signed-quartic comparison with positive limiting
weight and complete residual decay.

SOURCE_DRAFT / NOT_RUN / NOT_KERNEL_AUDITED /
MECHANISM_TO_RECURRENCE_RULE_OPEN.
-/

namespace GRWTSK.RiemannHypothesis

/-- Source-bound finite-state generator driven by the local witness obtained
from the accepted complete mechanism trace. -/
structure SourceBoundPackingRecurrenceProgram
    (Source Local : Type*)
    [AddCommGroup Source] [AddCommGroup Local]
    [Module ℝ Source] [Module ℝ Local]
    [TopologicalSpace Source] [TopologicalSpace Local] where
  accepted : SourceBoundMechanismBridgeAccepted Source Local
  IsPrimePowerSource : ℕ → Prop
  initial : Local → ThreePointPackingState IsPrimePowerSource
  nextPoint :
    Local → ℕ → ThreePointPackingState IsPrimePowerSource → PackingPoint
  nextSource :
    Local → ℕ → ThreePointPackingState IsPrimePowerSource → ℕ
  nextSource_ok :
    ∀ localData n state,
      IsPrimePowerSource (nextSource localData n state)

namespace SourceBoundPackingRecurrenceProgram

/-- The unique local witness extracted from the accepted source-bound mechanism
trace and its continuous linear/form bridge. -/
def localWitness
    {Source Local : Type*}
    [AddCommGroup Source] [AddCommGroup Local]
    [Module ℝ Source] [Module ℝ Local]
    [TopologicalSpace Source] [TopologicalSpace Local]
    (program : SourceBoundPackingRecurrenceProgram Source Local) : Local :=
  program.accepted.bridge.localWitness

/-- One source-driven recurrence transition. -/
def advanceAt
    {Source Local : Type*}
    [AddCommGroup Source] [AddCommGroup Local]
    [Module ℝ Source] [Module ℝ Local]
    [TopologicalSpace Source] [TopologicalSpace Local]
    (program : SourceBoundPackingRecurrenceProgram Source Local)
    (n : ℕ)
    (state : ThreePointPackingState program.IsPrimePowerSource) :
    ThreePointPackingState program.IsPrimePowerSource :=
  state.advance
    (program.nextPoint program.localWitness n state)
    (program.nextSource program.localWitness n state)
    (program.nextSource_ok program.localWitness n state)

/-- Canonical state sequence generated from the initializer and one-new-point
rule. No independent infinite sequence is supplied. -/
def stateAt
    {Source Local : Type*}
    [AddCommGroup Source] [AddCommGroup Local]
    [Module ℝ Source] [Module ℝ Local]
    [TopologicalSpace Source] [TopologicalSpace Local]
    (program : SourceBoundPackingRecurrenceProgram Source Local) :
    ℕ → ThreePointPackingState program.IsPrimePowerSource
  | 0 => program.initial program.localWitness
  | n + 1 => program.advanceAt n (program.stateAt n)

/-- Signed-quartic comparison data at one nontrivial zero for one generated
packing program. -/
structure PackingRecurrenceComparisonAtZero
    {Source Local : Type*}
    [AddCommGroup Source] [AddCommGroup Local]
    [Module ℝ Source] [Module ℝ Local]
    [TopologicalSpace Source] [TopologicalSpace Local]
    (program : SourceBoundPackingRecurrenceProgram Source Local)
    (s : ℂ) where
  weight : ℕ → ℝ
  residual : PrimeQuarticResidualLedger
  limitWeight : ℝ
  limitWeight_pos : 0 < limitWeight
  identity :
    ∀ n,
      program.valueAt n =
        weight n * witnessTrace (s.re - 1 / 2) s.im + residual.total n
  weight_tendsto : Tendsto weight atTop (𝓝 limitWeight)

/-- Structural positivity plus the comparison/limit data force the signed
quartic witness nonnegative. -/
theorem PackingRecurrenceComparisonAtZero.witnessTrace_nonnegative
    {Source Local : Type*}
    [AddCommGroup Source] [AddCommGroup Local]
    [Module ℝ Source] [Module ℝ Local]
    [TopologicalSpace Source] [TopologicalSpace Local]
    {program : SourceBoundPackingRecurrenceProgram Source Local}
    {s : ℂ}
    (comparison : PackingRecurrenceComparisonAtZero program s) :
    0 ≤ witnessTrace (s.re - 1 / 2) s.im := by
  have hlim :
      Tendsto program.valueAt atTop
        (𝓝 (comparison.limitWeight *
          witnessTrace (s.re - 1 / 2) s.im)) := by
    simp_rw [funext comparison.identity]
    simpa using
      (comparison.weight_tendsto.mul_const
        (witnessTrace (s.re - 1 / 2) s.im)).add
          comparison.residual.total_tendsto
  have hnonneg :
      0 ≤ comparison.limitWeight *
        witnessTrace (s.re - 1 / 2) s.im :=
    ge_of_tendsto hlim
      (Eventually.of_forall program.valueAt_nonnegative)
  exact
    (mul_nonneg_iff_of_pos_left comparison.limitWeight_pos).mp hnonneg

/-- Exact all-zero comparison claim for one source-bound recurrence program. -/
def PackingRecurrenceQuarticClaim
    {Source Local : Type*}
    [AddCommGroup Source] [AddCommGroup Local]
    [Module ℝ Source] [Module ℝ Local]
    [TopologicalSpace Source] [TopologicalSpace Local]
    (program : SourceBoundPackingRecurrenceProgram Source Local) : Type :=
  ∀ s : ℂ, NontrivialZetaZero s →
    PackingRecurrenceComparisonAtZero program s

  sourceIdentity : String

  pntIndependent :

/-- Once the source program and audited comparison are supplied for every
nontrivial zero, the existing signed-quartic endpoint yields RH. -/
theorem riemannHypothesis_of_auditedPackingRecurrenceExtraction
    {Source Local : Type*}
    [AddCommGroup Source] [AddCommGroup Local]
    [Module ℝ Source] [Module ℝ Local]
    [TopologicalSpace Source] [TopologicalSpace Local]
    {program : SourceBoundPackingRecurrenceProgram Source Local}
    (extraction : AuditedPackingRecurrenceExtraction program) :
    _root_.RiemannHypothesis := by
  apply
    GRWTSK.NumberTheory.Zeta.Riemann.riemannHypothesis_of_trace_extraction
  intro s hs
  let comparison := extraction.pntIndependent.theoremTerm s hs
  exact
    ⟨program.valueAt, comparison.weight, comparison.residual.total,
      comparison.limitWeight, comparison.limitWeight_pos,
      comparison.identity, program.valueAt_nonnegative,
      comparison.weight_tendsto, comparison.residual.total_tendsto⟩

end GRWTSK.RiemannHypothesis

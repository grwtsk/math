/-- Compact source definition; reducing it still entails the full original work. -/
def packets : List CheckedPoleJet.Packet :=
  CanonicalCheckedJetContour.packets plans

def result : Option ℤ := HeightFifteenCheckedJets.result packets

/-- Exact representational equality, not a proof of the common value. -/
theorem result_eq_original : result = HeightFifteenJetCandidate.result := by
  unfold result HeightFifteenCheckedJets.result packets HeightFifteenJetCandidate.result
  exact CanonicalCheckedJetContour.evaluateRectangle_packets_eq 15 edges marks plans image

/-- The canonical checked target has precisely the old success proposition. -/
theorem success_iff : result = some 1 ↔ HeightFifteenJetCandidate.Success := by
  rw [result_eq_original]
  rfl

/-- The SAME 38 block obligations are sufficient; no new target is substituted. -/
theorem result_one_of_blocks (h : HeightFifteenBlockDecomposition.BlockObligations) :
    result = some 1 :=
  success_iff.mpr (HeightFifteenBlockDecomposition.success_of_blocks h)

theorem count_one_of_result (h : result = some 1) :

  HeightFifteenCheckedJets.count_one_of_checked (by
    simpa only [result,packets] using h)

namespace GRWTSK.RiemannHypothesis.Checks.HeightFifteenJetCandidate

open GRWTSK.Analysis.Complex GRWTSK.Analysis.ErrorBounds
open RationalComplexEnclosure
open RationalNewtonDisk (magnitude)

/-- Pure finite mesh data: (start, step, number of cells), with no modular endpoint. -/
def blocks : List (ℚ × ℚ × ℕ) :=
  [(0,1/32,4), (1/8,1/16,6), (1/2,1/8,4),
   (1,1/128,13), (141/128,1/256,78), (45/32,1/128,37), (217/128,1/256,78),
   (2,1/16,2), (17/8,1/32,17), (85/32,1/64,16), (93/32,1/128,12),
   (3,1/2048,67), (6211/2048,1/4096,324), (6373/2048,1/2048,1639),
   (2003/512,1/1024,74), (255/64,1/512,8)]

def marks : List ℚ :=
  blocks.flatMap (fun b => (List.range b.2.2).map (fun j => b.1 + (j : ℚ)*b.2.1)) ++ [4]

def edges : ℕ := 2379

/-- Less point precision than the old pilot, but all original error rules remain. -/
def parameters : RationalPoleValue.Parameters := ⟨16,7,80,16,16,7,16⟩

def domain : List Coordinates :=
  RationalRectangleMesh.vertices RationalRectangleCount.lower (RationalRectangleCount.upper 15) marks

def lengthAt (i : ℕ) : ℚ :=
  magnitude (sub (RationalPoleContour.pointAt domain (i+1)) (RationalPoleContour.pointAt domain i))

def plan (i : ℕ) : ConstructedRectangleCount.Plan :=
  let a := RationalPoleContour.pointAt domain i
  let b := RationalPoleContour.pointAt domain (i+1)
  let c := mul (1/2,0) (add a b)
  ⟨c, (3/4)*lengthAt i, c.1/2, parameters⟩

def plans : List ConstructedRectangleCount.Plan := (List.range edges).map plan

def prior (i : ℕ) : ℕ := if i = 0 then edges-1 else i-1

/-- This merely proposes a rational surrogate. BOTH incident disks still check it.
The inverse-length weights reflect adjacent midpoint locations; no accuracy
claim is inferred from this interpolation formula. -/
def imageAt (i : ℕ) : Coordinates :=
  let j := prior i
  let li := lengthAt i
  let lj := lengthAt j
  let vi := (JetPoleContour.value (plan i)).mid
  let vj := (JetPoleContour.value (plan j)).mid
  mul (1/(li+lj),0) (add (mul (li,0) vj) (mul (lj,0) vi))

def image : List Coordinates := (List.range edges).map imageAt ++ [imageAt 0]

/-- Small geometric decisions remain unexecuted Lean scripts. -/
theorem mesh_shape : marks.length = edges+1 ∧
    RationalRectangleMesh.check edges marks = true := by decide

theorem plan_length : plans.length = edges := by simp [plans]

theorem image_length : image.length = edges+1 := by simp [image]

/-- The large finite equality is deliberately NOT asserted as a theorem here. -/
def result : Option ℤ := JetRectangleCount.evaluate 15 edges marks plans image

def Success : Prop := result = some 1

/-- This conditional composition is not a report of a Lean evaluation. -/
theorem count_one_of_success (h : Success) : MirrorCountingRectangle.count (15 : ℝ) = 1 := by
  have hc := JetRectangleCount.count_eq_of_evaluate h
  exact_mod_cast hc

theorem complete_pair_of_success (h : Success) :
    ∃ z : CanonicalZeroDivisor.ZeroPoint, CanonicalZeroDivisor.multiplicity z = 1 ∧
      z.1.re = (1/2 : ℝ) ∧ CanonicalZeroDivisor.heightHead (15 : ℝ) =
        {z, LocalZeroSymmetry.conjugatePoint z} :=
  JetRectangleCount.complete_pair_of_evaluate_one h

end GRWTSK.RiemannHypothesis.Checks.HeightFifteenJetCandidate

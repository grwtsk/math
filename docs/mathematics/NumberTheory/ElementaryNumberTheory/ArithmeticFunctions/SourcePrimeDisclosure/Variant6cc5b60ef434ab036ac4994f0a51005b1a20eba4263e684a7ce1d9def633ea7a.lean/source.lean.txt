/-- Squarefree divisors generated by actual prime subsets, retaining the real short cutoff. -/
def primeDivisors (P : Finset ℕ) (U : ℕ) : Finset ℕ :=
  (P.powerset.image (fun S => ∏ p ∈ S, p)).filter (· ≤ U)

theorem primeDivisors_fresh_coprime (P : Finset ℕ) (U p : ℕ)
    (hP : ∀ r ∈ P, r.Prime) (hp : p.Prime) (hpP : p ∉ P) :
    ∀ e ∈ primeDivisors P U, Nat.Coprime p e := by
  classical
  intro e he
  obtain ⟨S, hS, rfl⟩ := Finset.mem_image.mp (Finset.mem_filter.mp he).1
  have hSP := Finset.mem_powerset.mp hS
  have hfac := Nat.primeFactors_prod (fun r hr => hP r (hSP hr))
  apply hp.coprime_iff_not_dvd.mpr
  intro hd
  have hn : (∏ r ∈ S, r) ≠ 0 := Finset.prod_ne_zero_iff.mpr
    (fun r hr => (hP r (hSP hr)).ne_zero)
  have hf := Nat.mem_primeFactors.mpr ⟨hp, hd, hn⟩
  rw [hfac] at hf
  exact hpP (hSP hf)

theorem primeDivisors_insert (P : Finset ℕ) (U p : ℕ)
    (hp : 0 < p) (hpP : p ∉ P) :
    primeDivisors (insert p P) U =
      primeDivisors P U ∪ (primeDivisors P (U / p)).image (p * ·) := by
  classical
  have hi : ((insert p P).powerset.image (fun S => ∏ r ∈ S, r)) =
      (P.powerset.image (fun S => ∏ r ∈ S, r)) ∪
        (P.powerset.image (fun S => ∏ r ∈ S, r)).image (p * ·) := by
    rw [Finset.powerset_insert, Finset.image_union, Finset.image_image, Finset.image_image]
    congr 1
    apply Finset.image_congr
    intro S hS
    exact Finset.prod_insert (fun h => hpP ((Finset.mem_powerset.mp hS) h))
  unfold primeDivisors
  rw [hi, Finset.filter_union]
  congr 1
  ext e
  simp only [Finset.mem_filter, Finset.mem_image]
  constructor
  · rintro ⟨⟨a, ha, rfl⟩, hcut⟩
    exact ⟨a, ⟨ha, (Nat.le_div_iff_mul_le hp).mpr (by simpa [Nat.mul_comm] using hcut)⟩, rfl⟩
  · rintro ⟨a, ⟨ha, hcut⟩, rfl⟩
    exact ⟨⟨a, ha, rfl⟩, by simpa [Nat.mul_comm] using (Nat.le_div_iff_mul_le hp).mp hcut⟩

theorem primeDivisors_insert_disjoint (P : Finset ℕ) (U p : ℕ)
    (hP : ∀ r ∈ P, r.Prime) (hp : p.Prime) (hpP : p ∉ P) :
    Disjoint (primeDivisors P U) ((primeDivisors P (U / p)).image (p * ·)) := by
  classical
  apply Finset.disjoint_left.mpr
  intro e he hi
  obtain ⟨a, _, rfl⟩ := Finset.mem_image.mp hi
  have hc := primeDivisors_fresh_coprime P U p hP hp hpP _ he
  exact (hp.coprime_iff_not_dvd.mp hc) (dvd_mul_right p a)

def centeredDisclosure (D : Finset ℕ) (n : ℕ) : ℝ :=
  ∑ d ∈ D, (ArithmeticFunction.moebius d : ℝ) *
    (1 / (d : ℝ) - if d ∣ n then 1 else 0)

def disclosedMean (E : Finset ℕ) : ℝ :=
  ∑ e ∈ E, (ArithmeticFunction.moebius e : ℝ) / (e : ℝ)

def disclosedDivisorSum (E : Finset ℕ) (n : ℕ) : ℝ :=
  ∑ e ∈ E, if e ∣ n then (ArithmeticFunction.moebius e : ℝ) else 0

def freshPrimeCorrection (p : ℕ) (E : Finset ℕ) (n : ℕ) : ℝ :=
  -(1 / (p : ℝ)) * disclosedMean E +
    if p ∣ n then disclosedDivisorSum E n else 0

theorem moebius_fresh_prime_mul {p e : ℕ} (hp : p.Prime) (hpe : Nat.Coprime p e) :
    (ArithmeticFunction.moebius (p * e) : ℝ) =
      -(ArithmeticFunction.moebius e : ℝ) := by
  rw [ArithmeticFunction.isMultiplicative_moebius.map_mul_of_coprime hpe,
    ArithmeticFunction.moebius_apply_prime hp]
  push_cast
  ring

theorem fresh_prime_discrepancy {p e n : ℕ} (hp : p.Prime)
    (hpe : Nat.Coprime p e) :
    (ArithmeticFunction.moebius (p * e) : ℝ) *
        (1 / ((p * e : ℕ) : ℝ) - if p * e ∣ n then 1 else 0) =
      -(1 / (p : ℝ)) * ((ArithmeticFunction.moebius e : ℝ) / (e : ℝ)) +
        (if p ∣ n then if e ∣ n then (ArithmeticFunction.moebius e : ℝ) else 0 else 0) := by
  have hd : p * e ∣ n ↔ p ∣ n ∧ e ∣ n :=
    ⟨fun h => ⟨(dvd_mul_right p e).trans h, (dvd_mul_left e p).trans h⟩,
      fun h => hpe.mul_dvd_of_dvd_of_dvd h.1 h.2⟩
  rw [moebius_fresh_prime_mul hp hpe, Nat.cast_mul]
  simp only [hd, div_eq_mul_inv, mul_inv_rev]
  split_ifs <;> simp_all <;> ring

theorem centeredDisclosure_insert_prime (D E : Finset ℕ) (p n : ℕ)
    (hp : p.Prime) (hE : ∀ e ∈ E, Nat.Coprime p e)
    (hdisj : Disjoint D (E.image (p * ·))) :
    centeredDisclosure (D ∪ E.image (p * ·)) n =
      centeredDisclosure D n + freshPrimeCorrection p E n := by
  classical
  have hinj : Function.Injective (p * · : ℕ → ℕ) :=
    fun a b h => Nat.eq_of_mul_eq_mul_left hp.pos h
  unfold centeredDisclosure
  rw [Finset.sum_union hdisj, Finset.sum_image (fun _ _ _ _ h => hinj h)]
  congr 1
  have hs : (∑ e ∈ E, (ArithmeticFunction.moebius (p * e) : ℝ) *
      (1 / ((p * e : ℕ) : ℝ) - if p * e ∣ n then 1 else 0)) =
      ∑ e ∈ E, (-(1 / (p : ℝ)) * ((ArithmeticFunction.moebius e : ℝ) / (e : ℝ)) +
        (if p ∣ n then if e ∣ n then (ArithmeticFunction.moebius e : ℝ) else 0 else 0)) := by
    apply Finset.sum_congr rfl
    intro e he
    exact fresh_prime_discrepancy hp (hE e he)
  rw [hs]
  unfold freshPrimeCorrection disclosedMean disclosedDivisorSum
  rw [Finset.sum_add_distrib, ← Finset.mul_sum]
  split_ifs <;> simp_all

/-- The arbitrary-prime update uses the genuine U/p truncation, including U<p. -/
theorem centeredDisclosure_primeDivisors_insert (P : Finset ℕ) (U p n : ℕ)
    (hP : ∀ r ∈ P, r.Prime) (hp : p.Prime) (hpP : p ∉ P) :
    centeredDisclosure (primeDivisors (insert p P) U) n =
      centeredDisclosure (primeDivisors P U) n +
        freshPrimeCorrection p (primeDivisors P (U / p)) n := by
  rw [primeDivisors_insert P U p hp.pos hpP]
  exact centeredDisclosure_insert_prime _ _ _ _ hp
    (primeDivisors_fresh_coprime P (U / p) p hP hp hpP)
    (primeDivisors_insert_disjoint P U p hP hp hpP)

/-- A growing finite prime horizon resolves the complete short row; squareful zero slots stay zero. -/
theorem centeredDisclosure_coverage (P : Finset ℕ) (U n : ℕ)
    (hP : ∀ r ∈ P, r.Prime) (hcover : ∀ r, r.Prime → r ≤ U → r ∈ P) :
    centeredDisclosure (primeDivisors P U) n = centeredDisclosure (Finset.Icc 1 U) n := by
  classical
  have hsub : primeDivisors P U ⊆ Finset.Icc 1 U := by
    intro e he
    obtain ⟨he, hcut⟩ := Finset.mem_filter.mp he
    obtain ⟨S, hS, rfl⟩ := Finset.mem_image.mp he
    exact Finset.mem_Icc.mpr ⟨Finset.prod_pos (fun r hr =>
      (hP r ((Finset.mem_powerset.mp hS) hr)).pos), hcut⟩
  unfold centeredDisclosure
  apply Finset.sum_subset hsub
  intro d hd hnot
  have hnsq : ¬ Squarefree d := by
    intro hsq
    apply hnot
    apply Finset.mem_filter.mpr
    refine ⟨Finset.mem_image.mpr ⟨d.primeFactors, ?_, Nat.prod_primeFactors_of_squarefree hsq⟩,
      (Finset.mem_Icc.mp hd).2⟩
    apply Finset.mem_powerset.mpr
    intro r hr
    obtain ⟨hrp, hrd, _⟩ := Nat.mem_primeFactors.mp hr
    exact hcover r hrp ((Nat.le_of_dvd (Finset.mem_Icc.mp hd).1 hrd).trans
      (Finset.mem_Icc.mp hd).2)
  rw [ArithmeticFunction.moebius_eq_zero_of_not_squarefree hnsq]
  simp

def disclosureSource (q : ComplexL2) (I D : Finset ℕ) : ComplexL2 :=
  finiteProfile q I false false (fun n => ((centeredDisclosure D n /
    Real.sqrt (n : ℝ) : ℝ) : ℂ))

def correctionSource (q : ComplexL2) (I E : Finset ℕ) (p : ℕ) : ComplexL2 :=
  finiteProfile q I false false (fun n => ((freshPrimeCorrection p E n /
    Real.sqrt (n : ℝ) : ℝ) : ℂ))

theorem disclosureSource_insert_prime (q : ComplexL2) (I D E : Finset ℕ) (p : ℕ)
    (hp : p.Prime) (hE : ∀ e ∈ E, Nat.Coprime p e)
    (hdisj : Disjoint D (E.image (p * ·))) :
    disclosureSource q I (D ∪ E.image (p * ·)) =
      disclosureSource q I D + correctionSource q I E p := by
  classical
  unfold disclosureSource correctionSource finiteProfile
  simp_rw [centeredDisclosure_insert_prime D E p _ hp hE hdisj, add_div,
    Complex.ofReal_add, add_smul]
  exact Finset.sum_add_distrib

theorem disclosureSource_primeDivisors_insert (q : ComplexL2)
    (I P : Finset ℕ) (U p : ℕ) (hP : ∀ r ∈ P, r.Prime)
    (hp : p.Prime) (hpP : p ∉ P) :
    disclosureSource q I (primeDivisors (insert p P) U) =
      disclosureSource q I (primeDivisors P U) +
        correctionSource q I (primeDivisors P (U / p)) p := by
  rw [primeDivisors_insert P U p hp.pos hpP]
  exact disclosureSource_insert_prime _ _ _ _ _ hp
    (primeDivisors_fresh_coprime P (U / p) p hP hp hpP)
    (primeDivisors_insert_disjoint P U p hP hp hpP)

def plainSource (q : ComplexL2) (I : Finset ℕ) : ComplexL2 :=
  finiteProfile q I false false (fun n => ((1 / Real.sqrt (n : ℝ) : ℝ) : ℂ))

def primeMultipleSource (q : ComplexL2) (I E : Finset ℕ) (p : ℕ) : ComplexL2 :=
  finiteProfile q I false false (fun n =>
    (((if p ∣ n then disclosedDivisorSum E n else 0) /
      Real.sqrt (n : ℝ) : ℝ) : ℂ))

theorem correctionSource_eq_mean_and_multiples (q : ComplexL2)
    (I E : Finset ℕ) (p : ℕ) :
    correctionSource q I E p =
      ((-(1 / (p : ℝ)) * disclosedMean E : ℝ) : ℂ) • plainSource q I +
        primeMultipleSource q I E p := by
  classical
  unfold correctionSource plainSource primeMultipleSource finiteProfile freshPrimeCorrection
  rw [Finset.smul_sum, ← Finset.sum_add_distrib]
  apply Finset.sum_congr rfl
  intro n _
  dsimp only
  rw [smul_smul, ← add_smul]
  congr 1
  push_cast
  ring

/-- This is the literal physical same-index reflection diagonal, with every n retained. -/
def physicalDiagonal (q : ComplexL2) (I : Finset ℕ) (a b : ℕ → ℂ) : ℝ :=
  (∑ n ∈ I, star (a n) * b n *
    ⟪orbitAtom q false false n, reflection (orbitAtom q false false n)⟫_ℂ).re

def compensatedReadout (q : ComplexL2) (I : Finset ℕ)
    (a b : ℕ → ℂ) (M : ℝ) : ℝ :=
  (⟪finiteProfile q I false false a,
    reflection (finiteProfile q I false false b)⟫_ℂ).re - physicalDiagonal q I a b - M

/-- The full source correction, physical deleted diagonal and independently fixed M travel together. -/
theorem compensatedReadout_row_increment (q : ComplexL2) (I : Finset ℕ)
    (a b d : ℕ → ℂ) (M : ℝ) :
    compensatedReadout q I a (b + d) M - compensatedReadout q I a b M =
      (⟪finiteProfile q I false false a,
        reflection (finiteProfile q I false false d)⟫_ℂ).re - physicalDiagonal q I a d := by
  have hs : finiteProfile q I false false (b + d) =
      finiteProfile q I false false b + finiteProfile q I false false d := by
    simp only [finiteProfile, Pi.add_apply, add_smul, Finset.sum_add_distrib]
  have hd : physicalDiagonal q I a (b + d) =
      physicalDiagonal q I a b + physicalDiagonal q I a d := by
    simp only [physicalDiagonal, Pi.add_apply, mul_add, add_mul,
      Finset.sum_add_distrib, Complex.add_re]
  simp only [compensatedReadout, hs, hd, map_add, inner_add_right, Complex.add_re]
  ring

/-- If both arithmetic rows advance, the two cross terms and new/new work remain together. -/
theorem compensatedReadout_two_row_increment (q : ComplexL2) (I : Finset ℕ)
    (a b c d : ℕ → ℂ) (M : ℝ) :
    compensatedReadout q I (a + c) (b + d) M - compensatedReadout q I a b M =
      compensatedReadout q I a d 0 + compensatedReadout q I c b 0 +
        compensatedReadout q I c d 0 := by
  have hs (f g : ℕ → ℂ) : finiteProfile q I false false (f + g) =
      finiteProfile q I false false f + finiteProfile q I false false g := by
    simp only [finiteProfile, Pi.add_apply, add_smul, Finset.sum_add_distrib]
  have hd : physicalDiagonal q I (a + c) (b + d) =
      physicalDiagonal q I a b + physicalDiagonal q I a d +
        physicalDiagonal q I c b + physicalDiagonal q I c d := by
    simp only [physicalDiagonal, Pi.add_apply, star_add, add_mul, mul_add,
      Finset.sum_add_distrib, Complex.add_re]
    ring
  simp only [compensatedReadout, hs, hd, map_add, inner_add_left, inner_add_right,
    Complex.add_re]
  ring

/-- Möbius disclosure has no repeated-prime divisor contribution. -/
theorem repeated_prime_divisor_zero (p k : ℕ) (hp : p.Prime) (hk : 2 ≤ k) :
    ArithmeticFunction.moebius (p ^ k) = 0 := by
  rw [ArithmeticFunction.moebius_apply_prime_pow hp (by omega), if_neg (by omega)]

/-- Every nonzero prime-power occurrence remains in the literal Mangoldt source row. -/
theorem prime_power_mangoldt_coefficient (p k : ℕ) (hp : p.Prime) (hk : 0 < k) :
    ArithmeticFunction.vonMangoldt (p ^ k) / Real.sqrt ((p ^ k : ℕ) : ℝ) =
      Real.log (p : ℝ) / Real.sqrt ((p ^ k : ℕ) : ℝ) := by
  rw [ArithmeticFunction.vonMangoldt_apply_pow (by omega),
    ArithmeticFunction.vonMangoldt_apply_prime hp]

/-- At the actual short horizon U=6, the prime-three correction changes a nonmultiple,
and the signs on multiples depend on the complete disclosed divisor sum. -/
theorem prime_three_centering_fixture :
    freshPrimeCorrection 3 (primeDivisors {2} (6 / 3)) 7 = -(1 / 6 : ℝ) ∧
    freshPrimeCorrection 3 (primeDivisors {2} (6 / 3)) 9 = (5 / 6 : ℝ) ∧
    freshPrimeCorrection 3 (primeDivisors {2} (6 / 3)) 12 = -(1 / 6 : ℝ) := by
  have hE : primeDivisors {2} (6 / 3) = {1, 2} := by
    change primeDivisors (insert 2 ∅) 2 = {1, 2}
    rw [primeDivisors_insert ∅ 2 2 (by decide) (by simp)]
    norm_num [primeDivisors, Finset.filter_singleton]
  rw [hE]
  norm_num [freshPrimeCorrection, disclosedMean, disclosedDivisorSum,
    ArithmeticFunction.moebius_apply_prime Nat.prime_two]

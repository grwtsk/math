private theorem powers_between (h : ℝ) (hh : 1 ≤ h) (n k : ℕ) (hnk : n ≤ k) :
    1 ≤ h ^ n ∧ h ^ n ≤ h ^ k := by
  have hp : ∀ m : ℕ, 1 ≤ h ^ m := by
    intro m
    induction m with
    | zero => simp
    | succ m hm =>
      rw [pow_succ]
      nlinarith [mul_nonneg (sub_nonneg.mpr hm) (sub_nonneg.mpr hh)]
  refine ⟨hp n, ?_⟩
  have h1 := hp (k - n)
  have h2 := hp n
  calc
    h ^ n ≤ h ^ n * h ^ (k - n) :=
      le_mul_of_one_le_right (by linarith) h1

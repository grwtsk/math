private lemma sum_k_mul_half_pow_aux (K : ℕ) :
    (∑ k ∈ Icc 1 K, (k : ℝ) * (2⁻¹) ^ k) + ((K : ℝ) + 2) * (2⁻¹) ^ K ≤ 2 := by
  induction K with
  | zero => simp
  | succ m ih =>
    rw [Finset.sum_Icc_succ_top (by omega : 1 ≤ m + 1)]
    have hp : ((2 : ℝ)⁻¹) ^ (m + 1) = ((2 : ℝ)⁻¹) ^ m * (2 : ℝ)⁻¹ := pow_succ _ _
    push_cast
    nlinarith [ih, hp, pow_nonneg (by norm_num : (0 : ℝ) ≤ 2⁻¹) m]

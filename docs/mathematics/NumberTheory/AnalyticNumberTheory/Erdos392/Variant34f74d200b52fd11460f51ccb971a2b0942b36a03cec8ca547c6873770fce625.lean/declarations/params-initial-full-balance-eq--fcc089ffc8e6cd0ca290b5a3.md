lemma Params.initial_full_balance_eq (P : Params) (p : ℕ) :
    (initial_full P).balance p =
      (P.M : ℤ) * (∑ m ∈ Finset.Ico (P.n - P.n / P.M) P.n, (m.factorization p : ℤ)) -

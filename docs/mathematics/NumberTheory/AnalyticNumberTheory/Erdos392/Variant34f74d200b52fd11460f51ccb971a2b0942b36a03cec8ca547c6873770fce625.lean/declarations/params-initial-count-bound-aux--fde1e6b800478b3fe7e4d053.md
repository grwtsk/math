lemma Params.initial.count_bound_aux (n M p : ℕ) (hp : p > 0) :
    M * ((n / M + p - 1) / p) ≤ n / p + M := by
  have h_ceil_le : (n / M + p - 1) / p ≤ n / M / p + 1 :=

  have h_mul_div : M * (n / M / p) ≤ n / p := by

/-!
# The phase-one characteristic quotient and its genuine radical fiber

The characteristic polynomial of the actual diagonal phase rotation is the
conjugate-root quadratic. At phase one its characteristic quotient is
`ℂ[X] / ((X - 1)^2)`. This quotient retains a nonzero square-zero coordinate,
whereas polynomial evaluation at the actual rotation `I` kills that coordinate.
The evaluation kernel really equals the native ring Jacobson radical.

This is a different algebra from the full phase matrix algebra and from the
represented algebra generated by `I`. In particular, it does not identify the
existing plus-line observation kernel with a matrix Jacobson radical. No action
of the original reflection factors on this characteristic quotient is supplied.

The actual matrix signed trace pulls back through the evaluation and descends
through its radical. Its value on the distinguished generator remains `-2 * w`.
The zero-frequency generator is bound to the actual logarithmic phase; the
uniform arithmetic compensation estimate remains open.

AI attribution: Codex authored this module and its proofs.
-/

noncomputable section

/-- The actual phase rotation has the conjugate-root characteristic polynomial. -/
theorem phase_rotation_charpoly (χ : ℂ) :
    (phaseRotation χ).charpoly = (X - C χ) * (X - C (conj χ)) := by
  rw [phase_rotation_eq_diagonal, Matrix.charpoly_diagonal]
  simp [Fin.prod_univ_two]

/-- The characteristic polynomial retains multiplicity two when the phase is one. -/
theorem phase_rotation_one_charpoly :
    (phaseRotation 1).charpoly = (X - C (1 : ℂ)) ^ 2 := by
  simp [phase_rotation_charpoly, pow_two]

/-- The phase-one characteristic algebra, before evaluation in the actual matrix algebra. -/
abbrev CharacteristicOne := AdjoinRoot ((X - C (1 : ℂ)) ^ 2)

/-- Scalar evaluation at the actual phase-one eigenvalue. -/
def observeOne : CharacteristicOne →ₐ[ℂ] ℂ :=
  AdjoinRoot.liftAlgHom _ (Algebra.ofId ℂ ℂ) 1 (by simp)

/-- Constants make this evaluation surjective. -/
theorem observe_one_surjective : Function.Surjective observeOne := by
  intro z
  exact ⟨AdjoinRoot.of _ z, by simp [observeOne]⟩

/-- Every invisible element in this characteristic quotient has square zero. -/
theorem observe_one_kernel_square_zero {x : CharacteristicOne}
    (hx : x ∈ RingHom.ker observeOne.toRingHom) : x ^ 2 = 0 := by
  obtain ⟨p, rfl⟩ := AdjoinRoot.mk_surjective x
  have hp : p.eval 1 = 0 := by
    simpa [observeOne] using RingHom.mem_ker.mp hx
  obtain ⟨q, hq⟩ := dvd_iff_isRoot.mpr hp
  rw [hq, map_mul, mul_pow, ← map_pow, AdjoinRoot.mk_self, zero_mul]

/-- The characteristic quotient gives a genuine kernel equals Jacobson radical model. -/
theorem observe_one_ker_eq_jacobson :
    RingHom.ker observeOne.toRingHom = Ring.jacobson CharacteristicOne := by
  apply le_antisymm
  · intro x hx
    rw [← Ideal.jacobson_bot]
    exact Ideal.radical_le_jacobson
      (mem_nilradical.mpr ⟨2, observe_one_kernel_square_zero hx⟩)
  · let : (RingHom.ker observeOne.toRingHom).IsMaximal :=
      RingHom.ker_isMaximal_of_surjective observeOne.toRingHom observe_one_surjective
    exact Ring.jacobson_le_of_isMaximal _

/-- The same equality with the scalar types needed by the generic linear fiber API. -/
theorem observe_one_linear_ker_eq_jacobson :
    LinearMap.ker observeOne.toLinearMap =
      (Ring.jacobson CharacteristicOne).restrictScalars ℂ := by
  ext x
  rw [LinearMap.mem_ker]
  change observeOne x = 0 ↔ x ∈ Ring.jacobson CharacteristicOne
  rw [← observe_one_ker_eq_jacobson, RingHom.mem_ker]
  rfl

/-- The scalar observation embedded in the actual complex matrix algebra. -/
def observePhaseOne : CharacteristicOne →ₐ[ℂ] PhaseMatrix :=
  (Algebra.ofId ℂ PhaseMatrix).comp observeOne

/-- This representation is precisely polynomial evaluation at the actual rotation `I`. -/
theorem observe_phase_one_apply_mk (p : ℂ[X]) :
    observePhaseOne (AdjoinRoot.mk ((X - C (1 : ℂ)) ^ 2) p) =
      aeval (phaseRotation 1) p := by
  have hscalar : (1 : PhaseMatrix) = algebraMap ℂ PhaseMatrix 1 := by simp
  rw [phase_rotation_one, hscalar, aeval_algebraMap_apply_eq_algebraMap_eval]
  simp [observePhaseOne, observeOne]

/-- The distinguished polynomial coordinate represents the actual phase-one rotation. -/
theorem observe_phase_one_root :
    observePhaseOne (AdjoinRoot.root ((X - C (1 : ℂ)) ^ 2)) = phaseRotation 1 := by
  simpa using observe_phase_one_apply_mk X

/-- Matrix evaluation has the same genuine radical kernel, since scalar matrices embed. -/
theorem observe_phase_one_ker_eq_jacobson :
    RingHom.ker observePhaseOne.toRingHom = Ring.jacobson CharacteristicOne := by
  change RingHom.ker ((algebraMap ℂ PhaseMatrix).comp observeOne.toRingHom) = _
  rw [RingHom.ker_comp_of_injective observeOne.toRingHom
    (FaithfulSMul.algebraMap_injective ℂ PhaseMatrix)]
  exact observe_one_ker_eq_jacobson

/-- The original reference reflection is absent from the represented scalar algebra. -/
theorem reference_reflection_not_in_characteristic_image :
    phaseReflection 1 ∉ AlgHom.range observePhaseOne := by
  rintro ⟨z, hz⟩
  have h01 := congrArg (fun M : PhaseMatrix ↦ M 0 1) hz
  simp [observePhaseOne, Algebra.ofId_apply, Matrix.algebraMap_eq_diagonal,
    phaseReflection] at h01

/-- No linear operator on this quotient intertwines the original left reflection action.
The image already fails invariance at the identity. -/
theorem reference_left_action_not_preserved :
    ¬ ∃ L : CharacteristicOne →ₗ[ℂ] CharacteristicOne,
      ∀ z, observePhaseOne (L z) = phaseReflection 1 * observePhaseOne z := by
  rintro ⟨L, hL⟩
  apply reference_reflection_not_in_characteristic_image
  exact ⟨L 1, by simpa using hL 1⟩

/-- The characteristic coordinate minus its observed phase-one value. -/
def infinitesimalOne : CharacteristicOne :=
  AdjoinRoot.mk ((X - C (1 : ℂ)) ^ 2) (X - C 1)

/-- Characteristic multiplicity supplies a genuine nonzero infinitesimal. -/
theorem infinitesimal_one_ne_zero : infinitesimalOne ≠ 0 := by
  apply AdjoinRoot.mk_ne_zero_of_natDegree_lt ((monic_X_sub_C 1).pow 2)
  · exact (monic_X_sub_C 1).ne_zero
  · rw [natDegree_pow, natDegree_X_sub_C]
    norm_num

/-- The surviving formal infinitesimal has square zero. -/
theorem infinitesimal_one_square_zero : infinitesimalOne ^ 2 = 0 := by
  rw [infinitesimalOne, ← map_pow, AdjoinRoot.mk_self]

/-- Evaluation removes the infinitesimal, rather than proving that it was zero. -/
theorem observe_one_infinitesimal : observeOne infinitesimalOne = 0 := by
  simp [observeOne, infinitesimalOne]

/-- The nonzero infinitesimal belongs to the native characteristic-algebra radical. -/
theorem infinitesimal_one_mem_jacobson : infinitesimalOne ∈ Ring.jacobson CharacteristicOne := by
  rw [← observe_one_ker_eq_jacobson, RingHom.mem_ker]
  exact observe_one_infinitesimal

/-- The generic affine fiber theorem is instantiated with a proved radical identification. -/
theorem observe_one_fiber_eq_affine_coset (a x : CharacteristicOne) :
    observeOne x = observeOne a ↔ ∃ j ∈ Ring.jacobson CharacteristicOne, x = a + j :=
  ring_radical_observation_fiber observeOne.toLinearMap observe_one_linear_ker_eq_jacobson a x

/-- Pull back the actual signed matrix trace through the characteristic representation. -/
def characteristicSignedTrace (w : ℝ) (z : CharacteristicOne) : ℝ :=
  -2 * w * (observeOne z).re

/-- The pullback retains the actual two-dimensional signed trace normalization. -/
theorem characteristic_signed_trace_eq_matrix_packet (w : ℝ) (z : CharacteristicOne) :
    characteristicSignedTrace w z = signedTracePacket w (observePhaseOne z) := by
  simp [characteristicSignedTrace, signedTracePacket, observePhaseOne,
    Algebra.ofId_apply, Matrix.algebraMap_eq_diagonal]
  ring

/-- The signed trace is constant on all radical cosets, not merely zero on the radical. -/
theorem characteristic_signed_trace_coset_invariant (w : ℝ) :
    CosetInvariant ((Ring.jacobson CharacteristicOne).restrictScalars ℂ)
      (characteristicSignedTrace w) := by
  intro z j hj
  have hj' : j ∈ Ring.jacobson CharacteristicOne := hj
  rw [← observe_one_ker_eq_jacobson, RingHom.mem_ker] at hj'
  change observeOne j = 0 at hj'
  simp [characteristicSignedTrace, map_add, hj']

/-- The actual signed trace admits a factor through this genuine radical quotient. -/
theorem characteristic_signed_trace_quotient_factor (w : ℝ) :
    ∃ Qbar : CharacteristicOne ⧸ (Ring.jacobson CharacteristicOne).restrictScalars ℂ → ℝ,
      characteristicSignedTrace w =
        Qbar ∘ ((Ring.jacobson CharacteristicOne).restrictScalars ℂ).mkQ :=
  (coset_invariant_iff_quotient_factor _ _).mp (characteristic_signed_trace_coset_invariant w)

/-- Quotient descent preserves the full negative value at the distinguished generator. -/
theorem characteristic_signed_trace_identity (w : ℝ) :
    characteristicSignedTrace w (AdjoinRoot.root ((X - C (1 : ℂ)) ^ 2)) = -2 * w := by
  simp [characteristicSignedTrace, observeOne]

/-- The represented generator is the actual logarithmic rotation at zero frequency. -/
theorem observe_phase_one_root_zero_frequency (x : ℝ) :
    observePhaseOne (AdjoinRoot.root ((X - C (1 : ℂ)) ^ 2)) = phaseRotation (chi 0 x) := by
  simp [observe_phase_one_root, chi]

end

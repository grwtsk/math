/-!
# Whole-matrix Christoffel reduction for the theta derivative tape

The verified finite mode expansion cannot be made positive term by term, nor
by merely summing column permutations.  This module therefore moves back to
the whole finite Hankel matrix.

Introduce the simpler base polynomial sequence `Q_n` generated by the same
theta derivative operator from `Q_0 = 1`.  The actual derivative
polynomials `P_n` satisfy the exact all-order identity

`P_n = -(1/2) * (Q_(n+1) + Q_n)`.

Consequently every finite-mode source Hankel matrix is a scalar multiple of
the one-step Christoffel transform of the finite `Q` tape.  This is an exact
whole-determinant reorganization and retains all cancellation.  No positivity
of the transformed determinant is asserted here.
-/

/-- The theta derivative operator is homogeneous over real scalar
multiplication. -/
theorem thetaPhiDerivativeStep_smul
    (c : ℝ) (p : ℝ[X]) :
    thetaPhiDerivativeStep (c • p) =
      c • thetaPhiDerivativeStep p := by
  unfold thetaPhiDerivativeStep
  rw [Polynomial.sum_smul_index p c _ (by
    intro i
    simp)]
  rw [Polynomial.smul_sum]
  rw [Polynomial.sum_def, Polynomial.sum_def]
  apply Finset.sum_congr rfl
  intro n hn
  simp only [smul_add, Polynomial.smul_monomial, smul_eq_mul]
  apply congrArg₂ (· + ·)
  · congr 1
    ring
  · congr 1
    ring

/-- The simpler base polynomial sequence generated by the same theta
derivative operator, starting from the constant polynomial one. -/
def thetaPhiBasePolynomial : ℕ → ℝ[X]
  | 0 => Polynomial.monomial 0 1
  | n + 1 =>
      thetaPhiDerivativeStep (thetaPhiBasePolynomial n)

/-- The first base polynomial is `5 - 4y`. -/
theorem thetaPhiBasePolynomial_one_eval (y : ℝ) :
    (thetaPhiBasePolynomial 1).eval y = 5 - 4 * y := by
  change
    (thetaPhiDerivativeStep
      (Polynomial.monomial 0 1)).eval y = _
  rw [thetaPhiDerivativeStep_monomial]
  simp [Polynomial.eval_add, Polynomial.eval_monomial]
  ring

/-- The base sequence obeys the same derivative-step recurrence. -/
@[simp] theorem thetaPhiBasePolynomial_succ (n : ℕ) :
    thetaPhiBasePolynomial (n + 1) =
      thetaPhiDerivativeStep (thetaPhiBasePolynomial n) := rfl

/-- All-order polynomial identity placing the actual theta derivative
polynomial tape as a one-step Christoffel transform of the base tape. -/
theorem thetaPhiDerivativePolynomial_christoffel
    (n : ℕ) :
    thetaPhiBasePolynomial (n + 1) +
        thetaPhiBasePolynomial n =
      (-2 : ℝ) • thetaPhiDerivativePolynomial n := by
  induction n with
  | zero =>
      change
        thetaPhiDerivativeStep
            (Polynomial.monomial 0 1) +
            Polynomial.monomial 0 1 =
          (-2 : ℝ) •
            (Polynomial.monomial 0 (-3) +
              Polynomial.monomial 1 2)
      rw [thetaPhiDerivativeStep_monomial]
      simp only [smul_add, Polynomial.smul_monomial,
        smul_eq_mul]
      simp_rw [← Polynomial.C_mul_X_pow_eq_monomial]
      norm_num
      ext n
      by_cases hn : n = 0
      · subst n
        simp
        norm_num
      · simp [Polynomial.coeff_C, Polynomial.coeff_one, hn]
  | succ n ih =>
      change
        thetaPhiDerivativeStep
            (thetaPhiBasePolynomial (n + 1)) +
            thetaPhiDerivativeStep
              (thetaPhiBasePolynomial n) =
          (-2 : ℝ) •
            thetaPhiDerivativeStep
              (thetaPhiDerivativePolynomial n)
      rw [← thetaPhiDerivativeStep_add, ih,
        thetaPhiDerivativeStep_smul]

/-- Evaluation form of the Christoffel identity. -/
theorem thetaPhiDerivativePolynomial_eval_christoffel
    (n : ℕ) (y : ℝ) :
    (thetaPhiDerivativePolynomial n).eval y =
      (-1 / 2 : ℝ) *
        ((thetaPhiBasePolynomial (n + 1)).eval y +
          (thetaPhiBasePolynomial n).eval y) := by
  have h := congrArg
    (fun p : ℝ[X] => p.eval y)
    (thetaPhiDerivativePolynomial_christoffel n)
  simp only [Polynomial.eval_add, Polynomial.eval_smul,
    smul_eq_mul] at h
  linarith

/-- One base-polynomial mode contribution with the same positive theta mode
weight as the literal derivative tape. -/
def thetaPhiBaseModeValue
    (n m : ℕ) (t : ℝ) : ℝ :=
  thetaPhiDerivativeModeWeight m t *
    (thetaPhiBasePolynomial n).eval (thetaModeY m t)

/-- The actual derivative mode contribution is the corresponding one-step
Christoffel transform of the base mode contribution. -/
theorem thetaPhiDerivativeModeValue_christoffel
    (n m : ℕ) (t : ℝ) :
    thetaPhiDerivativeModeValue n m t =
      (-1 / 2 : ℝ) *
        (thetaPhiBaseModeValue (n + 1) m t +
          thetaPhiBaseModeValue n m t) := by
  unfold thetaPhiDerivativeModeValue thetaPhiBaseModeValue
  rw [thetaPhiDerivativePolynomial_eval_christoffel]
  ring

/-- Finite-mode truncation of the base polynomial tape. -/
def thetaPhiBaseTapeTruncation
    (N n : ℕ) (t : ℝ) : ℝ :=
  ∑ m : Fin N, thetaPhiBaseModeValue n m.val t

/-- Every finite literal derivative coordinate is the one-step Christoffel
transform of the corresponding finite base-tape coordinates. -/
theorem thetaPhiDerivativeTapeTruncation_christoffel
    (N n : ℕ) (t : ℝ) :
    thetaPhiDerivativeTapeTruncation N n t =
      (-1 / 2 : ℝ) *
        (thetaPhiBaseTapeTruncation N (n + 1) t +
          thetaPhiBaseTapeTruncation N n t) := by
  unfold thetaPhiDerivativeTapeTruncation
    thetaPhiBaseTapeTruncation
  simp_rw [thetaPhiDerivativeModeValue_christoffel]
  rw [← Finset.mul_sum, Finset.sum_add_distrib]

/-- Whole Hankel matrix formed from the one-step Christoffel transform of a
scalar tape. -/
def thetaPhiChristoffelHankel
    (k : ℕ) (q : ℕ → ℝ) :
    Matrix (Fin k) (Fin k) ℝ :=
  Matrix.of fun i j =>
    q (i.val + j.val + 1) +
      q (i.val + j.val)

/-- The finite literal theta derivative Hankel matrix is exactly
`-1/2` times the Christoffel-Hankel matrix of the finite base tape. -/
theorem thetaPhiDerivativeHankelTruncation_eq_christoffel
    (k N : ℕ) (t : ℝ) :
    Matrix.of
        (fun i j : Fin k =>
          thetaPhiDerivativeTapeTruncation
            N (i.val + j.val) t) =
      (-1 / 2 : ℝ) •
        thetaPhiChristoffelHankel k
          (fun n => thetaPhiBaseTapeTruncation N n t) := by
  ext i j
  simp only [Matrix.of_apply, Matrix.smul_apply,
    smul_eq_mul, thetaPhiChristoffelHankel]
  rw [thetaPhiDerivativeTapeTruncation_christoffel]

/-- Exact whole-determinant Christoffel reduction of every finite-mode signed
theta Hankel determinant.  No sign is asserted for the base determinant. -/
theorem signedJetDet_truncation_eq_christoffel
    (k N : ℕ) (t : ℝ) :
    signedJetDet k
        (fun n => thetaPhiDerivativeTapeTruncation N n t) =
      (-1 : ℝ) ^ (k * (k - 1) / 2) *
        (-1 / 2 : ℝ) ^ k *
          Matrix.det
            (thetaPhiChristoffelHankel k
              (fun n =>
                thetaPhiBaseTapeTruncation N n t)) := by
  unfold signedJetDet
  rw [thetaPhiDerivativeHankelTruncation_eq_christoffel]
  rw [Matrix.det_smul]
  simp only [Fintype.card_fin]
  ring

end

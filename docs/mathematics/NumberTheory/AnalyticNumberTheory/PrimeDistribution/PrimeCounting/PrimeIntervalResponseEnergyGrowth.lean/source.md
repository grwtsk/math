The actual source is generated by primeBand (3/4) T. Every composite note lies

/-- Empty and singleton subsets exhaust a powerset sum when all larger terms vanish. -/
theorem powerset_sum_first_order (P : Finset ℕ) (F : Finset ℕ → ℝ)
    (hlarge : ∀ U ⊆ P, 1 < U.card → F U = 0) :
    (∑ U ∈ P.powerset, F U) = F ∅ + ∑ p ∈ P, F {p} := by
  classical
  let small : Finset (Finset ℕ) := insert ∅ (P.image (fun p => ({p} : Finset ℕ)))
  have hs : small ⊆ P.powerset := by
    intro U hU
    rcases Finset.mem_insert.mp hU with hU | hU
    · subst U
      exact Finset.mem_powerset.mpr (Finset.empty_subset _)
    · rcases Finset.mem_image.mp hU with ⟨p, hp, rfl⟩
      exact Finset.mem_powerset.mpr (Finset.singleton_subset_iff.mpr hp)
  have hout : ∀ U ∈ P.powerset, U ∉ small → F U = 0 := by
    intro U hU hnot
    apply hlarge U (Finset.mem_powerset.mp hU)
    have hne : U ≠ ∅ := by
      intro h
      apply hnot
      simp only [small, h, Finset.mem_insert]
      exact Or.inl trivial
    have hc : 0 < U.card := Finset.card_pos.mpr (Finset.nonempty_iff_ne_empty.mpr hne)
    have hcne : U.card ≠ 1 := by
      intro hc1
      rcases Finset.card_eq_one.mp hc1 with ⟨p, rfl⟩
      apply hnot
      apply Finset.mem_insert.mpr
      right
      exact Finset.mem_image.mpr ⟨p,
        (Finset.mem_powerset.mp hU) (Finset.mem_singleton_self p), rfl⟩
    omega
  rw [← Finset.sum_subset hs hout]
  have hzero : (∅ : Finset ℕ) ∉ P.image (fun p => ({p} : Finset ℕ)) := by
    simp
  rw [show small = insert ∅ (P.image (fun p => ({p} : Finset ℕ))) from rfl,
    Finset.sum_insert hzero, Finset.sum_image]
  intro p _ q _ hpq
  exact Finset.singleton_injective hpq

/-- Distinct-prime pair delays beyond a horizon force every higher product beyond it. -/
theorem product_above_horizon_of_pair (P U : Finset ℕ)
    (hP : ∀ p ∈ P, p.Prime) (hU : U ⊆ P) {T : ℝ}
    (hpair : ∀ p ∈ P, ∀ q ∈ P, p ≠ q → Real.exp T < (p : ℝ) * q)
    (hc : 1 < U.card) : Real.exp T < (primeProduct U : ℝ) := by
  classical
  rcases Finset.one_lt_card.mp hc with ⟨p, hp, q, hq, hpq⟩
  have hsub : ({p, q} : Finset ℕ) ⊆ U := by
    intro r hr
    simp only [Finset.mem_insert, Finset.mem_singleton] at hr
    rcases hr with rfl | rfl
    · exact hp
    · exact hq
  have hd : p * q ∣ primeProduct U := by
    have h := Finset.prod_dvd_prod_of_subset ({p,q} : Finset ℕ) U (fun n => n) hsub
    simpa [primeProduct, hpq] using h
  have hpos := primeProduct_positive U (fun p hp => hP p (hU hp))
  have hle : p * q ≤ primeProduct U := Nat.le_of_dvd hpos hd
  have hcast : (p : ℝ) * q ≤ (primeProduct U : ℝ) := by exact_mod_cast hle
  exact (hpair p (hU hp) q (hU hq) hpq).trans_le hcast

/-- Actual source flattening; higher composite notes vanish by the declared delay geometry. -/
theorem primeState_first_order (P : Finset ℕ) (hP : ∀ p ∈ P, p.Prime) {T u : ℝ}
    (hpair : ∀ p ∈ P, ∀ q ∈ P, p ≠ q → Real.exp T < (p : ℝ) * q)
    (hu : u ≤ T) :
    PPNCP04.primeState P u = PPNCP04.baseWave u - ∑ p ∈ P, PPNCP04.atom p u := by
  classical
  change primeState P u = _
  rw [primeState_exact P hP]
  rw [powerset_sum_first_order]
  · simp only [primeProduct, Finset.prod_empty, ArithmeticFunction.moebius_apply_one,
      Int.cast_one, one_mul, Finset.prod_singleton]
    simp only [PPNCP04.atom_source_bridge]
    rw [← PPNCP04.baseWave_eq_atom_one]
    have hs : (∑ p ∈ P, (ArithmeticFunction.moebius p : ℝ) * PPNCP04.atom p u) =
        -(∑ p ∈ P, PPNCP04.atom p u) := by
      rw [← Finset.sum_neg_distrib]
      apply Finset.sum_congr rfl
      intro p hp
      rw [ArithmeticFunction.moebius_apply_prime (hP p hp)]
      simp only [Int.cast_neg, Int.cast_one, neg_one_mul]
    rw [hs]
    rfl
  · intro U hU hc
    have hp := product_above_horizon_of_pair P U hP hU hpair hc
    have hlog : T < Real.log (primeProduct U : ℝ) := by
      simpa only [Real.log_exp] using Real.log_lt_log (Real.exp_pos T) hp
    have he : ¬ Real.log (primeProduct U : ℝ) ≤ u := not_le.mpr (hu.trans_lt hlog)
    simp [atom, he]

/-- A supplied pointwise negative amplitude gives an actual continuous-source energy bound. -/
theorem sourceEnergy_lower_on_interval (P : Finset ℕ) (hP : ∀ p ∈ P, p.Prime)
    {T a b amplitude : ℝ} (ha : 0 ≤ a) (hab : a ≤ b) (hb : b ≤ T)
    (hamp : 0 ≤ amplitude)
    (hnegative : ∀ u ∈ Icc a b, PPNCP04.primeState P u ≤ -amplitude) :
    amplitude ^ 2 * (b - a) ≤ PPNCP04.sourceEnergy P T := by
  have hc : Continuous (PPNCP04.primeState P) := primeState_continuous P hP
  have hi : IntegrableOn (fun u => PPNCP04.primeState P u ^ 2) (Icc 0 T) :=
    (hc.pow 2).integrableOn_Icc
  have hiw : IntegrableOn (fun u => PPNCP04.primeState P u ^ 2) (Icc a b) :=
    (hc.pow 2).integrableOn_Icc
  have hconst : IntegrableOn (fun _ : ℝ => amplitude ^ 2) (Icc a b) :=
    continuous_const.integrableOn_Icc
  have hs : Icc a b ⊆ Icc 0 T := fun _ hu => ⟨ha.trans hu.1, hu.2.trans hb⟩
  calc
    _ = ∫ _u in Icc a b, amplitude ^ 2 := by
      rw [setIntegral_const, Real.volume_real_Icc_of_le hab]
      simp only [smul_eq_mul]
      ring
    _ ≤ ∫ u in Icc a b, PPNCP04.primeState P u ^ 2 := by
      apply setIntegral_mono_on hconst hiw measurableSet_Icc
      intro u hu
      have hn := hnegative u hu
      nlinarith
    _ ≤ ∫ u in Icc 0 T, PPNCP04.primeState P u ^ 2 :=
      setIntegral_mono_set hi (Eventually.of_forall (fun u => sq_nonneg _))
        (Eventually.of_forall hs)
    _ = _ := rfl

/-- The terminal window specialization has no extra analytic hypotheses. -/
theorem sourceEnergy_lower_on_terminal_interval (P : Finset ℕ)
    (hP : ∀ p ∈ P, p.Prime) {T δ amplitude : ℝ} (hδ : 0 ≤ δ) (hT : δ ≤ T)
    (hamp : 0 ≤ amplitude)
    (hnegative : ∀ u ∈ Icc (T - δ) T, PPNCP04.primeState P u ≤ -amplitude) :
    amplitude ^ 2 * δ ≤ PPNCP04.sourceEnergy P T := by
  have h := sourceEnergy_lower_on_interval P hP
    (a := T - δ) (b := T) (by linarith) (by linarith) le_rfl hamp hnegative
  simpa only [show T - (T - δ) = δ by ring] using h

theorem exp_large {T : ℝ} (hT : 1024 ≤ T) : 32 * (T + 1) ≤ Real.exp T := by

theorem lower_count {T : ℝ} (hT : 1024 ≤ T) :
    Real.exp T / (8 * T) ≤ (Nat.primeCounting ⌊Real.exp T / 2⌋₊ : ℝ) := by
  have ht : 0 < T := by linarith
  have he := exp_large hT
  have heone : 1 ≤ Real.exp T := by nlinarith
  have hy : 1 < Real.exp T / 2 := by nlinarith
  have hl2 : (1/2 : ℝ) ≤ Real.log 2 := by
    have h := Real.one_sub_inv_le_log_of_pos (by norm_num : (0 : ℝ) < 2)
    norm_num at h
    linarith
  have hl4 : Real.log 4 ≤ 3 := by
    have h := Real.log_le_sub_one_of_pos (by norm_num : (0 : ℝ) < 4)
    norm_num at h
    exact h
  have hlogs : Real.log (Real.exp T / 2 + 2) ≤ T + 3 := by
    have hle : Real.exp T / 2 + 2 ≤ 4 * Real.exp T := by nlinarith
    calc
      Real.log (Real.exp T / 2 + 2) ≤ Real.log (4 * Real.exp T) :=
        Real.log_le_log (by positivity) hle
      _ = Real.log 4 + T := by rw [Real.log_mul (by norm_num) (Real.exp_ne_zero T), Real.log_exp]
      _ ≤ T + 3 := by linarith
  have hn : Real.exp T / 8 ≤
      (Real.exp T / 2 - 1) * Real.log 2 - Real.log (Real.exp T / 2 + 2) := by
    have hm := mul_le_mul_of_nonneg_left hl2 (show 0 ≤ Real.exp T / 2 - 1 by linarith)
    nlinarith
  have hden : Real.log (Real.exp T / 2) ≤ T := by
    calc
      Real.log (Real.exp T / 2) ≤ Real.log (Real.exp T) :=
        Real.log_le_log (by positivity) (by nlinarith [Real.exp_pos T])
      _ = T := Real.log_exp T
  have hlpos : 0 < Real.log (Real.exp T / 2) := Real.log_pos hy
  calc
    Real.exp T / (8 * T) = (Real.exp T / 8) / T := by ring
    _ ≤ ((Real.exp T / 2 - 1) * Real.log 2 - Real.log (Real.exp T / 2 + 2)) / T :=
      div_le_div_of_nonneg_right hn ht.le
    _ ≤ ((Real.exp T / 2 - 1) * Real.log 2 - Real.log (Real.exp T / 2 + 2)) /
        Real.log (Real.exp T / 2) :=
      div_le_div_of_nonneg_left (by nlinarith [Real.exp_pos T]) hlpos hden
    _ ≤ (Nat.primeCounting ⌊Real.exp T / 2⌋₊ : ℝ) := Chebyshev.pi_ge' hy

theorem upper_count {T : ℝ} (hT : 1024 ≤ T) :
    (Nat.primeCounting ⌊Real.exp T / 256⌋₊ : ℝ) ≤ Real.exp T / (16 * T) := by
  have ht : 0 < T := by linarith
  have he := exp_large hT
  have hx : 2 ≤ Real.exp T / 256 := by nlinarith
  have hl256 : Real.log 256 ≤ 255 := by
    have h := Real.log_le_sub_one_of_pos (by norm_num : (0 : ℝ) < 256)
    norm_num at h
    exact h
  have hl4 : Real.log 4 ≤ 3 := by
    have h := Real.log_le_sub_one_of_pos (by norm_num : (0 : ℝ) < 4)
    norm_num at h
    exact h
  have hc : PrimeHarmonicLog.countConstant ≤ 8 := by
    unfold PrimeHarmonicLog.countConstant
    linarith
  have hlog : T/2 ≤ Real.log (Real.exp T / 256) := by
    rw [Real.log_div (Real.exp_ne_zero T) (by norm_num), Real.log_exp]
    linarith
  have hlogpos : 0 < Real.log (Real.exp T / 256) := by linarith
  calc
    (Nat.primeCounting ⌊Real.exp T / 256⌋₊ : ℝ) ≤
        PrimeHarmonicLog.countConstant * (Real.exp T / 256) / Real.log (Real.exp T / 256) :=
      PrimeHarmonicLog.prime_count_bound hx
    _ ≤ 8 * (Real.exp T / 256) / Real.log (Real.exp T / 256) :=
      div_le_div_of_nonneg_right (mul_le_mul_of_nonneg_right hc (by positivity)) hlogpos.le
    _ ≤ 8 * (Real.exp T / 256) / (T/2) :=
      div_le_div_of_nonneg_left (by positivity) (by linarith) hlog
    _ = Real.exp T / (16 * T) := by ring

theorem wideBand_eq_sdiff (T : ℝ) :
    wideBand T = Nat.primesLE ⌊Real.exp T / 2⌋₊ \ Nat.primesLE ⌊Real.exp T / 256⌋₊ := by
  ext p
  simp only [wideBand, Finset.mem_filter, Finset.mem_sdiff]
  constructor
  · rintro ⟨hp, hgt⟩
    refine ⟨hp, ?_⟩
    intro hlow
    have hle : (p : ℝ) ≤ Real.exp T / 256 :=
      (Nat.le_floor_iff (by positivity)).mp (Nat.le_of_mem_primesLE hlow)
    linarith
  · rintro ⟨hp, hn⟩
    refine ⟨hp, ?_⟩
    by_contra h
    apply hn
    exact Nat.mem_primesLE.mpr
      ⟨(Nat.le_floor_iff (by positivity)).mpr (le_of_not_gt h), Nat.prime_of_mem_primesLE hp⟩

theorem wideBand_count {T : ℝ} (hT : 1024 ≤ T) :
    Real.exp T / (16*T) ≤ ((wideBand T).card : ℝ) := by
  have hs : Nat.primesLE ⌊Real.exp T / 256⌋₊ ⊆ Nat.primesLE ⌊Real.exp T / 2⌋₊ :=
    Nat.primesLE_mono (Nat.floor_mono (by nlinarith [Real.exp_pos T]))
  have heq : (wideBand T).card + (Nat.primesLE ⌊Real.exp T / 256⌋₊).card =
      (Nat.primesLE ⌊Real.exp T / 2⌋₊).card := by
    rw [wideBand_eq_sdiff, Finset.card_sdiff_of_subset hs]
    exact Nat.sub_add_cancel (Finset.card_le_card hs)
  have heqReal : ((wideBand T).card : ℝ) +
      (Nat.primeCounting ⌊Real.exp T / 256⌋₊ : ℝ) =
      (Nat.primeCounting ⌊Real.exp T / 2⌋₊ : ℝ) := by
    exact_mod_cast (by simpa only [Nat.primesLE_card_eq_primeCounting] using heq)
  have hl := lower_count hT
  have hu := upper_count hT
  have hscale : Real.exp T / (8*T) = 2 * (Real.exp T / (16*T)) := by ring
  linarith

theorem band_atom_sum_lower {T u : ℝ} (hT : 1024 ≤ T) (hu : T-tailWidth ≤ u) :
    atomGain * Real.exp (T/2) / (16*T) ≤
      ∑ p ∈ Obstruction.primeBand (3/4) T, atom p u := by
  have hgain : 0 ≤ atomGain * Real.exp (-T/2) :=
    mul_nonneg atomGain_positive.le (Real.exp_pos _).le
  have hcount := mul_le_mul_of_nonneg_right (wideBand_count hT) hgain
  have hprod : Real.exp T * Real.exp (-T/2) = Real.exp (T/2) := by
    rw [← Real.exp_add]
    congr 1
    ring
  calc
    atomGain * Real.exp (T/2) / (16*T) =
        (Real.exp T / (16*T)) * (atomGain * Real.exp (-T/2)) := by
      rw [show (Real.exp T / (16*T)) * (atomGain * Real.exp (-T/2)) =
        atomGain * (Real.exp T * Real.exp (-T/2)) / (16*T) by ring, hprod]
    _ ≤ ((wideBand T).card : ℝ) * (atomGain * Real.exp (-T/2)) := hcount
    _ = ∑ _p ∈ wideBand T, atomGain * Real.exp (-T/2) := by simp
    _ ≤ ∑ p ∈ wideBand T, atom p u := by
      apply Finset.sum_le_sum
      intro p hp
      exact atom_tail_lower (Nat.prime_of_mem_primesLE (Finset.mem_filter.mp hp).1).pos
        (wideBand_mem_upper hp) hu
    _ ≤ ∑ p ∈ Obstruction.primeBand (3/4) T, atom p u :=
      Finset.sum_le_sum_of_subset_of_nonneg (wideBand_subset_primeBand hT)
        (fun p hp _ => atom_nonneg (Obstruction.primeBand_mem_prime hp).pos u)

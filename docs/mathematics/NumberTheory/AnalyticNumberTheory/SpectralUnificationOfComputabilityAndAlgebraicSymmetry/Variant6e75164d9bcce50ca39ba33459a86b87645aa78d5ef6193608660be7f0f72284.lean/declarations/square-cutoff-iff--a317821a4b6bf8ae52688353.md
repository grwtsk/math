theorem square_cutoff_iff (A : ℝ) (n : ℕ) :
    ((n ^ 2 : ℕ) : ℝ) ≤ Real.exp (2 * A) ↔ (n : ℝ) ≤ Real.exp A := by
  rw [Nat.cast_pow, show 2 * A = A + A by ring, Real.exp_add, ← pow_two]

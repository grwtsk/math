def lemma_3_2_bound (ε c P₀ : ℝ) : Prop :=
  ∀ (P : ℝ) (f : ℝ → ℝ),
    P₀ ≤ P →
    (∀ t, P ≤ t → 0 ≤ f t) →
    AntitoneOn f (Set.Ici P) →
    ContDiffOn ℝ 1 f (Set.Ici P) →
    Filter.Tendsto (fun t ↦ t * f t) Filter.atTop (nhds 0) →
    ∑' p : ℕ, (if Nat.Prime p ∧ P ≤ (p : ℝ) then f p * log p else 0) ≤
      (1 + ε) * ∫ t in Set.Ici P, f t +
      ε * P * f P +
      c * P * f P / (log P) ^ 2

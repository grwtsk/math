theorem memℒp_indicator_const (p : ℝ≥0∞) (hs : MeasurableSet s) (c : E) (hμsc : c = 0 ∨ μ s ≠ ∞) :
    Memℒp (s.indicator fun _ => c) p μ := by

  · exact zero_memℒp

    apply memℒp_const

/-- The `ℒ^p` norm of the indicator of a set is uniformly small if the set itself has small measure,

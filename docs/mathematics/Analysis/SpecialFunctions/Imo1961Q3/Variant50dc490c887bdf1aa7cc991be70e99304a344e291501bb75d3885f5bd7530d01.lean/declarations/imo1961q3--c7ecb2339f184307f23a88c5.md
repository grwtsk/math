public theorem Imo1961Q3 {n : ℕ} {x : ℝ} (h₀ : n ≠ 0) :

      (∃ k : ℤ, k * π = x) ∧ Even n ∨ (∃ k : ℤ, k * (2 * π) = x) ∧ Odd n ∨
        (∃ k : ℤ, -(π / 2) + k * (2 * π) = x) ∧ Odd n := by
  constructor
  · intro h
    rcases eq_or_ne (sin x) 0 with hsinx | hsinx

        cos_eq_neg_one_iff] at h

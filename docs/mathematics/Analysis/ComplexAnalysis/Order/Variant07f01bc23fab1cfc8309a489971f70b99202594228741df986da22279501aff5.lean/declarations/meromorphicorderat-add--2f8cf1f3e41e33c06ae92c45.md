  · simp only [h₂f₂, le_top, inf_of_le_left]

    filter_upwards [meromorphicOrderAt_eq_top_iff.1 h₂f₂]

  lift meromorphicOrderAt f₁ x to ℤ using h₂f₁ with n₁ hn₁

  let n := min n₁ n₂

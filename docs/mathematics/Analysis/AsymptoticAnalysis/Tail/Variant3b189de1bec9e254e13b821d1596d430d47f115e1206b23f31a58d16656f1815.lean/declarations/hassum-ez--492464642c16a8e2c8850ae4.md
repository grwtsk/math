      (Z.Ez P T) := by

  have hG : HasSum (fun ρ : Z.carrier => Z.Gsummand P T k l ρ) (Z.Gz P T k l) :=
    (H.summable_Gsummand k l).hasSum
  have hE : HasSum (fun x : {ρ : Z.carrier // ρ ∉ H.sA} => Z.Gsummand P T k l (x : Z.carrier))

private lemma map_range_succ {α : Type*} (f : ℕ → α) (n : ℕ) :
    (List.range (n + 1)).map f = f 0 :: (List.range n).map (fun i => f (i + 1)) := by
  induction n with
  | zero =>
      simp
  | succ n ih =>
      -- range (n+2) = range (n+1) ++ [n+1]
      calc
        (List.range (n + 2)).map f

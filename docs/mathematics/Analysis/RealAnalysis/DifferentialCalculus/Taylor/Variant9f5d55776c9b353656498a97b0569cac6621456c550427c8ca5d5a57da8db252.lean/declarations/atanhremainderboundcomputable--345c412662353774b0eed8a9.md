def atanhRemainderBoundComputable (r : ℚ) (n : ℕ) : IntervalRat :=
  let r' := |r|  -- Use absolute value to ensure non-negativity
  if h : r' ≥ 1 then
    ⟨-1000, 1000, by norm_num⟩  -- Fallback for bad input
  else
    let R := r' ^ (n + 1) / (1 - r')
    ⟨-R, R, by
      have hr : r' < 1 := not_le.mp h

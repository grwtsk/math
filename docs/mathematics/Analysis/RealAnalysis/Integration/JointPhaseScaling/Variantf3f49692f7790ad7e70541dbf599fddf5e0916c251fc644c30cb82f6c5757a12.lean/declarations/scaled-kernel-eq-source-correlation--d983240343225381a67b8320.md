    scaledKernel A s = sourceCorrelation explicitSecondDerivative A (A * s) := by

  have hne : A ≠ 0 := ne_of_gt hpos
  have heq : (fun u : ℝ ↦ Real.cos (u * s) * rescaledDensity A u) =
      fun u ↦ (Real.cos ((u / A) * (A * s)) * spectralDensity A (u / A)) / A := by

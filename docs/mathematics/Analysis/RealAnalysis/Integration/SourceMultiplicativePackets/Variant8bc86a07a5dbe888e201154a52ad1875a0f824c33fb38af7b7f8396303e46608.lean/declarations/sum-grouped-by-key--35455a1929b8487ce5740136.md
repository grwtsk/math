theorem sum_grouped_by_key {ι κ : Type*} [DecidableEq κ] (s : Finset ι)
    (key : ι → κ) (weight : ι → ℝ) (response : κ → ℝ) :
    (∑ i ∈ s, weight i * response (key i)) =
      ∑ r ∈ s.image key, (∑ i ∈ s with key i = r, weight i) * response r := by

theorem complex_cosh_norm_le_exp_norm (z : ℂ) :
    ‖Complex.cosh z‖ ≤ Real.exp ‖z‖ := by
  rw [Complex.cosh]
  calc
    ‖(Complex.exp z + Complex.exp (-z)) / 2‖

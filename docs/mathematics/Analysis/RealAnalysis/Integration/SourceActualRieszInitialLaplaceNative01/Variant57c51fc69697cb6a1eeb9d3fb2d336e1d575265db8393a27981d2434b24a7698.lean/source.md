  field_simp
  ring

private theorem realRamp_tail_integrable {σ : ℝ} (hσ : 0 < σ) (a : ℝ) :
    IntegrableOn (fun t : ℝ => (t - a) * Real.exp (-σ * t)) (Ioi a) := by

    (realRampAntiderivative_tendsto hσ a)

private theorem realRamp_tail_integral {σ : ℝ} (hσ : 0 < σ) (a : ℝ) :

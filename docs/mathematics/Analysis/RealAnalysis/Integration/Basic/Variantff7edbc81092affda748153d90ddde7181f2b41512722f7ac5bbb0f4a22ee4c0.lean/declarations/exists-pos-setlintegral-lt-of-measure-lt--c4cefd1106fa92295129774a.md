  rcases φ.exists_forall_le with ⟨C, hC⟩

  refine fun s hs => lt_of_le_of_lt ?_ hε₂ε

  intro ψ hψ
  calc
    (map (↑) ψ).lintegral (μ.restrict s) ≤

instance : Fintype { p : α × ℕ | p.2 < m.count p.1 } :=
  Fintype.ofFinset
    (m.toFinset.biUnion fun x ↦ (Finset.range (m.count x)).map ⟨_, Prod.mk_right_injective x⟩)
    (by
      rintro ⟨x, i⟩

      simp only [← and_assoc, exists_eq_right, and_iff_right_iff_imp]

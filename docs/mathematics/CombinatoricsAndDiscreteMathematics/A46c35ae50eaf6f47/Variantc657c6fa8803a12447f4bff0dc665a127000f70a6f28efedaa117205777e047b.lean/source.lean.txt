/-- Authority carried by a kernel-derived candidate. -/
inductive CandidateAuthority
  | navigationOnly
  deriving DecidableEq, Inhabited

/-- Raw kernel evidence for one entity/concept pair. -/
structure KernelCandidate where
  entityId : EntityId
  conceptId : ConceptId
  kernelId : StableId
  kernelValue : ℂ
  authority : CandidateAuthority := .navigationOnly
  contributingChannels : Finset ChannelId
  ambiguities : Finset Nonclaim
  deriving DecidableEq

/-- A resolver consumes both kernel evidence and a separate reviewed boundary
certificate.  No overload accepts a kernel score alone. -/
structure ReviewedCandidate where
  candidate : KernelCandidate
  entityProfile : EntityProfile
  concept : Record
  boundaryAccepted : concept.AcceptsEntity entityProfile
  evidence : AssignmentEvidence
  definitionDigestCurrent : evidence.reviewedDefinitionDigest = concept.definitionDigest
  boundaryDigestCurrent : evidence.reviewedBoundaryDigest = concept.boundaryDigest
  authorityDigestCurrent : evidence.reviewedAuthorityDigest = concept.authorityDigest

/-- Promote a reviewed candidate to a primary assignment. -/
def ReviewedCandidate.toOwnedAssignment
    (reviewed : ReviewedCandidate)
    (sourceDigest semanticFingerprint : Digest) : Assignment where
  entityId := reviewed.candidate.entityId
  disposition := .ownedBy reviewed.concept.id
  references := ∅
  sourceDigest := sourceDigest
  semanticFingerprint := semanticFingerprint
  evidence := reviewed.evidence
  evidenceDigest := reviewed.evidence.digest

/-- Candidate authority has no theorem-authority constructor. -/
theorem candidate_authority_is_navigation
    (candidate : KernelCandidate) :
    candidate.authority = .navigationOnly := by
  cases candidate.authority
  rfl

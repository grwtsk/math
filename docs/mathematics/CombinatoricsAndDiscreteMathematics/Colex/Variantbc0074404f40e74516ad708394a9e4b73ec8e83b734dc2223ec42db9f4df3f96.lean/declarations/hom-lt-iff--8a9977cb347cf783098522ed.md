    (h₁ : StrictMono f) (A B : Finset α) :
    (A.image f).toColex < (B.image f).toColex ↔ A.toColex < B.toColex := by
  simp only [Colex.lt_def, not_exists, mem_image, exists_prop, not_and]

  rintro ⟨k, z, ka, _⟩

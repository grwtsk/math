lemma add_eq_left_of_lt : f (b + a) = f b := by
  by_contra! h
  have h1 : f (b + a) ≤ f b := (hna b a).trans_eq (max_eq_left_of_lt h_lt)
  apply lt_irrefl (f b)
  calc
    f b = f (b + a + -a) := by simp
    _   ≤ max (f (b + a)) (f (-a)) := hna (b + a) (-a)
    _   < max (f b) (f b) := by
      rw [max_self, f_neg]
      exact max_lt (lt_of_le_of_ne h1 h) h_lt
    _   = f b := max_self (f b)

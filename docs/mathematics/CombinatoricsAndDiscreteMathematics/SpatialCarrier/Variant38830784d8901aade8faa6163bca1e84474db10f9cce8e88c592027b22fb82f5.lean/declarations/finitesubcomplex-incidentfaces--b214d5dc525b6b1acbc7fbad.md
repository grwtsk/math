def FiniteSubcomplex.incidentFaces {complex : FiniteGradedCarrier3}
    (subcomplex : FiniteSubcomplex complex) (edge : complex.Cell) :
    Finset complex.Cell :=
  subcomplex.cells.filter fun face ↦
    (complex.dimension face).1 = 2 ∧ edge ∈ complex.faces face

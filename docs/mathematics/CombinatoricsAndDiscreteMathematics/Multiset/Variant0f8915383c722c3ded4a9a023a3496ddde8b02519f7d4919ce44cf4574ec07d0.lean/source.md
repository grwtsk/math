variable [AddCommMonoid α] [PartialOrder α] [IsOrderedCancelAddMonoid α]

lemma map_add_left_Icc (a b c : α) : (Icc a b).map (c + ·) = Icc (c + a) (c + b) := by
  classical rw [Icc, Icc, ← Finset.image_add_left_Icc, Finset.image_val,

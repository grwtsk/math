import DAF.Manifest

/-!
# Stable identities for the DAF universe

The types in this module are intentionally opaque wrappers.  Hash equality is
an identity check for generated bytes; no theorem here turns ordinary digest
equality into semantic equivalence.
-/

namespace DAF.Universe

/-- Stable textual identifier supplied by the canonical universe generator. -/
structure StableId where
  value : String
  deriving DecidableEq, Repr, Inhabited

/-- Lower-case hexadecimal digest text together with its declared algorithm. -/
structure Digest where
  algorithm : String := "sha256"
  value : String
  deriving DecidableEq, Repr, Inhabited

/-- Identity of one immutable repository snapshot. -/
structure SnapshotId where
  value : String
  deriving DecidableEq, Repr, Inhabited

/-- Exact source position of a generated terrain node. -/
structure SourceLocation where
  path : String
  line : Nat
  column : Nat
  deriving DecidableEq, Repr, Inhabited

/-- Pretty-printed elaborated type retained by the compiled-environment exporter. -/
structure NormalizedType where
  text : String
  deriving DecidableEq, Repr, Inhabited

/-- An explicit statement that an interpretation is not established. -/
structure Nonclaim where
  id : StableId
  statement : String
  deriving DecidableEq, Repr, Inhabited

abbrev TerrainNodeId := StableId
abbrev TerrainEdgeId := StableId
abbrev TerrainCellId := StableId
abbrev TheoremCellId := StableId
abbrev FoldId := StableId
abbrev RuntimeBindingId := StableId
abbrev RuntimeTraceId := StableId
abbrev OperatorTraceId := StableId
abbrev ExternalAuthorityId := StableId
abbrev ConceptId := StableId
abbrev ClaimId := StableId
abbrev QueryId := StableId
abbrev HarmonicTubeId := StableId
abbrev DirichletCharacterId := StableId
abbrev CandidatePatchId := StableId

/-- Digest equality is only a byte-identity proposition. -/
def Digest.SameBytes (left right : Digest) : Prop := left = right

instance (left right : Digest) : Decidable (left.SameBytes right) :=
  inferInstanceAs (Decidable (left = right))

end DAF.Universe

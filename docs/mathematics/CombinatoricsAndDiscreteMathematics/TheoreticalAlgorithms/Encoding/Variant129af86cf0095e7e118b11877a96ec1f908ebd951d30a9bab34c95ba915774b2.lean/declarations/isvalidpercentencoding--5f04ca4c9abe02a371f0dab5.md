def isValidPercentEncoding (ba : ByteArray) : Bool :=
  let rec loop (i : Nat) : Bool :=
    if h : i < ba.size then
      let c := ba[i]'h
      if c = '%'.toUInt8 then
          if h₂ : i + 2 < ba.size then
            let d1 := ba[i + 1]'(by omega)
            let d2 := ba[i + 2]'h₂
            if isHexDigitByte d1 && isHexDigitByte d2 then
              loop (i + 3)
          else false
        else false
      else loop (i + 1)
    else true
  termination_by ba.size - i
  loop 0

/--

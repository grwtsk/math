def isValidScope (scope : String) : Bool :=
  !scope.isEmpty && scope.length ≤ 64 &&
    scope.all fun c => c.isDigit || ('a' ≤ c && c ≤ 'f') || ('A' ≤ c && c ≤ 'F')

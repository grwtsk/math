def rightInvSeq (ω : List B) : List W :=
  match ω with
  | [] => []
  | i :: ω => (π ω)⁻¹ * (s i) * (π ω) :: rightInvSeq ω

/-- The left inversion sequence of `ω`. The left inversion sequence of a word

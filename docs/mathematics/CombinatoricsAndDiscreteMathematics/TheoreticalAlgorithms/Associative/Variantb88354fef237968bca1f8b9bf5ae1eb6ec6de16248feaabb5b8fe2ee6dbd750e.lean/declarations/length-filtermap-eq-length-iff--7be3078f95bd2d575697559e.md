theorem length_filterMap_eq_length_iff [BEq α] [LawfulBEq α] {f : (a : α) → β a → Option (γ a)}
    {l : List ((a : α) × β a)} (distinct : DistinctKeys l) :

      ∀ (a : α) (h : containsKey a l), (f a (getValueCast a l h)).isSome := by

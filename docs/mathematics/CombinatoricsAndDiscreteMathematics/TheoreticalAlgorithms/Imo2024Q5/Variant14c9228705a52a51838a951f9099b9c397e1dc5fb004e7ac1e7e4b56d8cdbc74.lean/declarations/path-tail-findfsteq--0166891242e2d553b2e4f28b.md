lemma Path.tail_findFstEq (p : Path N) {r : Fin (N + 2)} (hr : r ≠ 0) :

  by_cases h : (p.cells[1]'p.one_lt_length_cells).1 = 0
  · simp_rw [Path.tail, if_pos h]

    {State : Type u} {Alphabet : Type v}
    (G : LabeledGraph State Alphabet) («prefix» suffix : List Alphabet) :
    ∀ {start finish : State},
      WordPath G start («prefix» ++ suffix) finish →
        ∃ middle, WordPath G start «prefix» middle ∧
          WordPath G middle suffix finish := by

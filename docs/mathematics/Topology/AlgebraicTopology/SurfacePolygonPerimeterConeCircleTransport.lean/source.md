/-!
# Transporting the authentic perimeter cone to the standard circle cone

The parent source constructs the genuine cone of the exact unpaired
polygon perimeter by quotienting `UnpairedPerimeter k × unitInterval`
only at radial coordinate zero. An earlier source supplies the actual
homeomorphism from that perimeter to Mathlib's `UnitAddCircle`.

This file defines independently the standard topological cone

  (UnitAddCircle × unitInterval) / ((z, 0) ~ (w, 0)),

computes its exact equivalence fibers, and descends
`perimeterUnitCircleHomeomorph k × id` and its inverse through the two
quotients. The descended maps are continuous inverses, hence give a
genuine homeomorphism

  PerimeterCone k ≃ₜ StandardCircleCone.

The homeomorphism preserves the real radial coordinate, the apex, and
the exact outer boundary parameterization. This removes all
polygon-specific topology from the remaining standard theorem.

No homeomorphism from `StandardCircleCone` to a complex closed disk is
constructed here. No pushout, CW surface, manifold, cellular/singular
comparison, van Kampen, Hurewicz, hyperbolic, trace-formula, GUE,
sofic, Chowla, zeta-divisor, or Riemann-hypothesis statement follows.
-/

/-- The genuine source cylinder for the standard cone on Mathlib's
unit additive circle. -/
abbrev StandardCircleConeCylinder : Type :=
  UnitAddCircle × unitInterval

/-- The only primitive relation for the standard circle cone:
all radius-zero points are identified. -/
inductive StandardCircleConeStep :
    StandardCircleConeCylinder →
      StandardCircleConeCylinder → Prop where
  | apex (x y : UnitAddCircle) :
      StandardCircleConeStep (x, 0) (y, 0)

/-- The exact equivalence closure of the standard cone's apex
identifications. -/
def StandardCircleConeConnected :
    StandardCircleConeCylinder →
      StandardCircleConeCylinder → Prop :=
  Relation.EqvGen StandardCircleConeStep

/-- The generated relation has no hidden positive-radius
identifications. -/
theorem standardCircleConeConnected_iff
    (p q : StandardCircleConeCylinder) :
    StandardCircleConeConnected p q ↔
      p = q ∨ (p.2 = 0 ∧ q.2 = 0) := by
  constructor
  · intro h
    induction h with
    | rel a b hab =>
        cases hab
        exact Or.inr ⟨rfl, rfl⟩
    | refl a =>
        exact Or.inl rfl
    | symm a b hab ih =>
        rcases ih with heq | ⟨ha, hb⟩
        · exact Or.inl heq.symm
        · exact Or.inr ⟨hb, ha⟩
    | trans a b c hab hbc ih₁ ih₂ =>
        rcases ih₁ with heq | ⟨ha, hb⟩
        · subst b
          exact ih₂
        · rcases ih₂ with heq | ⟨hb', hc⟩
          · subst c
            exact Or.inr ⟨ha, hb⟩
          · exact Or.inr ⟨ha, hc⟩
  · rintro (heq | ⟨hp, hq⟩)
    · subst q
      exact Relation.EqvGen.refl _
    · rcases p with ⟨x, r⟩
      rcases q with ⟨y, s⟩
      dsimp at hp hq
      subst r
      subst s
      exact Relation.EqvGen.rel _ _
        (StandardCircleConeStep.apex x y)

/-- The Setoid generated by precisely the standard cone apex
relation. -/
def standardCircleConeSetoid :
    Setoid StandardCircleConeCylinder where
  r := StandardCircleConeConnected
  iseqv := Relation.EqvGen.is_equivalence _

/-- The actual topological cone on `UnitAddCircle`. -/
def StandardCircleCone : Type :=
  Quotient standardCircleConeSetoid

local instance standardConeSourceSetoid : Setoid StandardCircleConeCylinder :=
  standardCircleConeSetoid

instance standardCircleConeTopology : TopologicalSpace StandardCircleCone :=
  inferInstanceAs (TopologicalSpace (Quotient standardCircleConeSetoid))

/-- Its topology is the true coinduced quotient topology. -/
theorem standardCircleConeTopology_coinduced :

/-- The genuine quotient projection to the standard circle cone. -/
def standardCircleConeProjection :
    StandardCircleConeCylinder → StandardCircleCone :=
  Quotient.mk standardCircleConeSetoid

theorem standardCircleConeProjection_continuous :
    Continuous standardCircleConeProjection := by
  change Continuous
    (Quotient.mk' :
      StandardCircleConeCylinder → StandardCircleCone)
  exact continuous_quotient_mk'

theorem standardCircleConeProjection_isQuotientMap :

  change IsQuotientMap
    (Quotient.mk' :
      StandardCircleConeCylinder → StandardCircleCone)
  exact isQuotientMap_quotient_mk'

/-- Away from radius zero, equality in the standard cone is exactly
equality of the original cylinder points. -/
theorem standardCircleConeProjection_eq_iff_of_ne_zero
    (p q : StandardCircleConeCylinder) (hp : p.2 ≠ 0) :
    standardCircleConeProjection p =
        standardCircleConeProjection q ↔
      p = q := by
  constructor
  · intro heq
    rcases (standardCircleConeConnected_iff p q).mp
        (Quotient.exact heq) with
      hpq | ⟨hz, _⟩
    · exact hpq
    · exact False.elim (hp hz)
  · intro hpq
    rw [hpq]

/-- The standard cone relation preserves the actual radial
coordinate. -/
theorem standardCircleConeConnected_preserves_radius
    {p q : StandardCircleConeCylinder}
    (h : StandardCircleConeConnected p q) :
    p.2 = q.2 := by
  rcases (standardCircleConeConnected_iff p q).mp h with
    hpq | ⟨hp, hq⟩
  · rw [hpq]
  · exact hp.trans hq.symm

/-- The real radial coordinate on the standard circle cone. -/
def standardCircleConeRadius :
    StandardCircleCone → unitInterval :=
  Quotient.lift
    (fun p : StandardCircleConeCylinder => p.2)
    (by
      intro p q h
      exact standardCircleConeConnected_preserves_radius h)

/-- A selected apex of the standard cone. -/
def standardCircleConeApex : StandardCircleCone :=
  standardCircleConeProjection (0, 0)

/-- Every radius-zero point is the selected standard apex. -/
theorem standardCircleCone_zero_eq_apex
    (z : UnitAddCircle) :
    standardCircleConeProjection (z, 0) =
      standardCircleConeApex := by
  apply Quotient.sound
  exact (standardCircleConeConnected_iff _ _).mpr
    (Or.inr ⟨rfl, rfl⟩)

/-- The standard cone's outer boundary inclusion. -/
def standardCircleConeBoundary :
    UnitAddCircle → StandardCircleCone :=
  fun z => standardCircleConeProjection (z, 1)

theorem standardCircleConeBoundary_continuous :
    Continuous standardCircleConeBoundary := by
  unfold standardCircleConeBoundary
  exact standardCircleConeProjection_continuous.comp
    (continuous_id.prodMk continuous_const)

/-- No two standard-circle points are identified on the outer
boundary. -/
theorem standardCircleConeBoundary_injective :
    Function.Injective standardCircleConeBoundary := by
  intro z w hzw
  have hsource :
      ((z, (1 : unitInterval)) :
          StandardCircleConeCylinder) = (w, 1) :=
    (standardCircleConeProjection_eq_iff_of_ne_zero
      _ _ (by norm_num)).mp hzw
  exact congrArg Prod.fst hsource

/-- Compactness of the standard cone, inherited from the genuine
compact circle-cylinder quotient. -/
theorem standardCircleConeCompactSpace :
    CompactSpace StandardCircleCone := by
  let : Fact (0 < (1 : ℝ)) := ⟨by norm_num⟩
  have : CompactSpace StandardCircleConeCylinder := by
    infer_instance
  exact Quotient.compactSpace

/-- Source-level product of the authentic perimeter-circle
homeomorphism with the identity radial coordinate. -/
def perimeterToStandardCircleConeSource (k : ℕ) :
    ConeCylinder k → StandardCircleConeCylinder :=
  fun p => (perimeterUnitCircleHomeomorph k p.1, p.2)

theorem perimeterToStandardCircleConeSource_continuous
    (k : ℕ) :
    Continuous (perimeterToStandardCircleConeSource k) := by
  exact
    ((perimeterUnitCircleHomeomorph k).continuous.comp
      continuous_fst).prodMk continuous_snd

/-- The forward cylinder transport respects the two genuine cone
relations. -/
theorem coneConnected_perimeterToStandardCircleConeSource
    (k : ℕ) {p q : ConeCylinder k}
    (h : ConeConnected k p q) :
    StandardCircleConeConnected
      (perimeterToStandardCircleConeSource k p)
      (perimeterToStandardCircleConeSource k q) := by
  rcases (coneConnected_iff k p q).mp h with
    hpq | hzero
  · subst q
    exact Relation.EqvGen.refl _
  · apply (standardCircleConeConnected_iff _ _).mpr
    right
    simpa [perimeterToStandardCircleConeSource] using hzero

/-- The descended forward map from the authentic perimeter cone to
the standard circle cone. -/
def perimeterConeToStandardCircleCone (k : ℕ) :
    PerimeterCone k → StandardCircleCone :=
  Quotient.lift
    (fun p =>
      standardCircleConeProjection
        (perimeterToStandardCircleConeSource k p))
    (by
      intro p q h
      exact Quotient.sound
        (coneConnected_perimeterToStandardCircleConeSource
          k h))

@[simp] theorem perimeterConeToStandardCircleCone_projection
    (k : ℕ) (p : ConeCylinder k) :
    perimeterConeToStandardCircleCone k
        (coneProjection k p) =
      standardCircleConeProjection
        (perimeterToStandardCircleConeSource k p) := rfl

theorem perimeterConeToStandardCircleCone_continuous
    (k : ℕ) :
    Continuous (perimeterConeToStandardCircleCone k) := by
  exact Continuous.quotient_lift
    (standardCircleConeProjection_continuous.comp
      (perimeterToStandardCircleConeSource_continuous k))
    (fun _ _ h => Quotient.sound (coneConnected_perimeterToStandardCircleConeSource k h))

/-- Every authentic perimeter point at radius zero maps to the selected
standard cone apex. -/
theorem perimeterConeToStandardCircleCone_zero
    (k : ℕ) (x : UnpairedPerimeter k) :
    perimeterConeToStandardCircleCone k
        (coneProjection k (x, 0)) =
      standardCircleConeApex := by
  rw [perimeterConeToStandardCircleCone_projection]
  simpa [perimeterToStandardCircleConeSource] using
    standardCircleCone_zero_eq_apex
      (perimeterUnitCircleHomeomorph k x)

/-- Source-level inverse product transport. -/
def standardCircleToPerimeterConeSource (k : ℕ) :

  fun p =>
    ((perimeterUnitCircleHomeomorph k).symm p.1, p.2)

theorem standardCircleToPerimeterConeSource_continuous

    Continuous (standardCircleToPerimeterConeSource k) := by
  exact
    ((perimeterUnitCircleHomeomorph k).symm.continuous.comp
      continuous_fst).prodMk continuous_snd

/-- The inverse cylinder transport respects the standard and authentic

theorem standardCircleConeConnected_standardCircleToPerimeterConeSource
    (k : ℕ) {p q : StandardCircleConeCylinder}

    ConeConnected k

  rcases (standardCircleConeConnected_iff p q).mp h with
    hpq | hzero
  · subst q
    exact Relation.EqvGen.refl _
  · apply (coneConnected_iff k _ _).mpr
    right

def standardCircleConeToPerimeterCone (k : ℕ) :

/-- The two descended cone maps are left inverses. -/
theorem standardCircleConeToPerimeterCone_leftInverse
    (k : ℕ) :
    Function.LeftInverse
      (standardCircleConeToPerimeterCone k)
      (perimeterConeToStandardCircleCone k) := by
  intro z
  refine Quotient.inductionOn z ?_
  rintro ⟨x, r⟩
  change
    coneProjection k
        (standardCircleToPerimeterConeSource k
          (perimeterToStandardCircleConeSource k (x, r))) =
      coneProjection k (x, r)
  change coneProjection k
    ((perimeterUnitCircleHomeomorph k).symm (perimeterUnitCircleHomeomorph k x), r) = _
  rw [(perimeterUnitCircleHomeomorph k).symm_apply_apply]

/-- The two descended cone maps are right inverses. -/
theorem standardCircleConeToPerimeterCone_rightInverse
    (k : ℕ) :
    Function.RightInverse
      (standardCircleConeToPerimeterCone k)
      (perimeterConeToStandardCircleCone k) := by
  intro z
  refine Quotient.inductionOn z ?_
  rintro ⟨x, r⟩
  change
    standardCircleConeProjection
        (perimeterToStandardCircleConeSource k
          (standardCircleToPerimeterConeSource k (x, r))) =
      standardCircleConeProjection (x, r)
  change standardCircleConeProjection
    ((perimeterUnitCircleHomeomorph k) ((perimeterUnitCircleHomeomorph k).symm x), r) = _
  rw [(perimeterUnitCircleHomeomorph k).apply_symm_apply]

/-- The source-exact equivalence of the two cone quotient carriers. -/
def perimeterConeEquivStandardCircleCone (k : ℕ) :
    PerimeterCone k ≃ StandardCircleCone where
  toFun := perimeterConeToStandardCircleCone k
  invFun := standardCircleConeToPerimeterCone k
  left_inv := standardCircleConeToPerimeterCone_leftInverse k
  right_inv := standardCircleConeToPerimeterCone_rightInverse k

/-- The authentic perimeter cone is genuinely homeomorphic to the
independently defined standard unit-circle cone. -/
noncomputable def perimeterConeHomeomorphStandardCircleCone
    (k : ℕ) :
    PerimeterCone k ≃ₜ StandardCircleCone where
  toEquiv := perimeterConeEquivStandardCircleCone k
  continuous_toFun :=
    perimeterConeToStandardCircleCone_continuous k
  continuous_invFun :=
    standardCircleConeToPerimeterCone_continuous k

@[simp] theorem perimeterConeHomeomorphStandardCircleCone_apply
    (k : ℕ) (z : PerimeterCone k) :
    perimeterConeHomeomorphStandardCircleCone k z =
      perimeterConeToStandardCircleCone k z := rfl

@[simp] theorem perimeterConeHomeomorphStandardCircleCone_symm_apply
    (k : ℕ) (z : StandardCircleCone) :
    (perimeterConeHomeomorphStandardCircleCone k).symm z =
      standardCircleConeToPerimeterCone k z := rfl

/-!
# One polygon vertex from the genuine oriented finite side pairings

The previously authored polygonBoundaryWord k in #745 is the ordered
list of signed edges for genus g=k+2. This module retains that exact
generator identity, but adds genuinely new FINITE combinatorial data:

* the two initial and k additional handles, in that order;
* the four corner POSITIONS in each ordered handle block;
* the paired sides of the same generator in opposite orientations;
* the b^{-1} endpoint seam to the NEXT handle, including the final
  cyclic seam from the last handle to the first;
* the ACTUAL equivalence relation generated by those pairings,
  implemented as Mathlib Relation.EqvGen, not a universal relation.

It proves that all corners of each handle are connected by gluing,
that the seam joins consecutive handles, and that all corners for
EVERY k are in one equivalence class. The actual Quotient of the
finite corner IDs is nonempty and a subsingleton: exactly ONE
combinatorial vertex is forced by the side-pairing relations.

The ordered block list flattens to the SAME polygonBoundaryWord k
already used by the source-indexed signed cellular differential and
categorical homology in #758/#823. Thus the formerly stipulated
one-vertex incidence convention receives an independent finite
polygon GLUING combinatorics supplier.

Crucial nonclaim: a Quotient of the FINITE set of corner IDs is NOT
the topological quotient of a closed disk by continuous boundary-side
gluing. It is not a CW complex, 2-cell attaching map, a topological
surface, manifold, fundamental-group or singular-homology theorem,
nor a hyperbolic surface, Selberg trace, GUE, Chowla, or RH result.
-/

/-- The first TWO handles and k additional handles, with distinct
source identities inherited from GenusLetter k. -/
abbrev Handle (k : ℕ) : Type :=
  Sum (Fin 2) (Fin k)

/-- The four starting corner positions of an ordered handle block.
The endpoint of the inverse b-edge is the NEXT handle's zero corner. -/
abbrev PolygonCorner (k : ℕ) : Type :=
  Handle k × Fin 4

/-- First handle in the fixed polygon order. -/
def firstHandle (k : ℕ) : Handle k :=
  Sum.inl 0

/-- Second handle in the fixed polygon order. -/
def secondHandle (k : ℕ) : Handle k :=
  Sum.inl 1

/-- The a-generator of each exact source-labelled handle. -/
def handleA (k : ℕ) : Handle k → GenusLetter k
  | Sum.inl i =>
      if i = 0 then baseLetter k 0 else baseLetter k 2
  | Sum.inr j => extraLetter k j false

/-- The b-generator of each exact source-labelled handle. -/
def handleB (k : ℕ) : Handle k → GenusLetter k
  | Sum.inl i =>
      if i = 0 then baseLetter k 1 else baseLetter k 3
  | Sum.inr j => extraLetter k j true

/-- A block of directed edges with the EXACT inherited generator IDs. -/
def handleEdges (k : ℕ) (h : Handle k) :
    List (SignedSide k) :=
  handleBlock k (handleA k h) (handleB k h)

@[simp] theorem firstHandleEdges (k : ℕ) :
    handleEdges k (firstHandle k) =
      handleBlock k (baseLetter k 0) (baseLetter k 1) := by
  simp [handleEdges, firstHandle, handleA, handleB]

@[simp] theorem secondHandleEdges (k : ℕ) :
    handleEdges k (secondHandle k) =
      handleBlock k (baseLetter k 2) (baseLetter k 3) := by
  simp [handleEdges, secondHandle, handleA, handleB]

@[simp] theorem extraHandleEdges (k : ℕ) (j : Fin k) :
    handleEdges k (Sum.inr j) =
      handleBlock k (extraLetter k j false) (extraLetter k j true) := rfl

/-- The complete ordered list of polygon handle blocks. Its first two
entries are the initial handles, followed by the extra handles in
their Fin k order; this is not a new, independently indexed word. -/
def orderedHandleBlocks (k : ℕ) :
    List (List (SignedSide k)) :=
  [handleEdges k (firstHandle k), handleEdges k (secondHandle k)] ++
    List.ofFn (fun j : Fin k => handleEdges k (Sum.inr j))

/-- SOURCE FIDELITY: flattening the ordered handle blocks is literally
the same signed polygon word from PR #751 / issue #745, including
all generator IDs, opposite edge orientations and concatenation order.
This statement is more precise than equal word lengths/cardinalities. -/
theorem orderedHandleBlocks_flatten_eq_polygonBoundaryWord (k : ℕ) :
    (orderedHandleBlocks k).flatten = polygonBoundaryWord k := by
  simp [orderedHandleBlocks, firstHandleEdges,
    secondHandleEdges, polygonBoundaryWord, firstTwoBoundary,
    extraBoundary, extraHandleBlocks]

/-- Each handle contributes exactly four actual signed polygon sides,
not an unchecked number of occurrences. -/
theorem handleEdges_length (k : ℕ) (h : Handle k) :
    (handleEdges k h).length = 4 := by
  simp [handleEdges]

/-- The exact signed source side beginning at one of a handle's four
corners. The original b^-1 side ends at the NEXT handle's zero corner. -/
def sideAt (k : ℕ) (h : Handle k) (i : Fin 4) : SignedSide k :=
  if i = 0 then (handleA k h, true)
  else if i = 1 then (handleB k h, true)
  else if i = 2 then (handleA k h, false)
  else (handleB k h, false)

@[simp] theorem sideAt_zero (k : ℕ) (h : Handle k) :
    sideAt k h 0 = (handleA k h, true) := by
  simp [sideAt]

@[simp] theorem sideAt_one (k : ℕ) (h : Handle k) :
    sideAt k h 1 = (handleB k h, true) := by
  simp [sideAt]

@[simp] theorem sideAt_two (k : ℕ) (h : Handle k) :
    sideAt k h 2 = (handleA k h, false) := by
  simp [sideAt]

@[simp] theorem sideAt_three (k : ℕ) (h : Handle k) :
    sideAt k h 3 = (handleB k h, false) := by
  simp [sideAt]

/-- The local four corners label EXACTLY the four ordered signed
sides of the inherited handleBlock and hence of #745's polygon word.
No extra edge occurrence or sign is introduced by sideAt. -/
theorem sideAt_list_eq_handleEdges (k : ℕ) (h : Handle k) :
    [sideAt k h 0, sideAt k h 1,
      sideAt k h 2, sideAt k h 3] = handleEdges k h := by
  simp [sideAt, handleEdges, handleBlock]

/-- Two base blocks plus one block for each extra source handle:
the block-count, not merely the one-vertex result, is source exact. -/
theorem orderedHandleBlocks_length (k : ℕ) :
    (orderedHandleBlocks k).length = k + 2 := by
  simp [orderedHandleBlocks]

/-- Opposite-oriented copy of the SAME source generator. -/
def oppositeSide (k : ℕ) (s : SignedSide k) : SignedSide k :=
  (s.1, !s.2)

/-- The two signed occurrences of each a-side and b-side are exactly
opposite under sideAt, not two arbitrarily identified edge labels. -/
theorem a_sides_opposite (k : ℕ) (h : Handle k) :
    sideAt k h 2 = oppositeSide k (sideAt k h 0) := by
  simp [oppositeSide]

theorem b_sides_opposite (k : ℕ) (h : Handle k) :
    sideAt k h 3 = oppositeSide k (sideAt k h 1) := by
  simp [oppositeSide]

/-- The oriented side-pairing involution changes sign and preserves
exactly the original source generator. -/
theorem oppositeSide_involutive (k : ℕ) (s : SignedSide k) :
    oppositeSide k (oppositeSide k s) = s := by
  cases s with
  | mk a direction =>
      cases direction <;> rfl

/-- No signed side is paired with itself. This is a hostile check
against folding equal-oriented sides rather than true inverses. -/
theorem oppositeSide_ne_self (k : ℕ) (s : SignedSide k) :
    oppositeSide k s ≠ s := by
  intro h
  cases s with
  | mk a direction =>
      cases direction <;> cases h

/-- Next handle in the actual ordered cyclic boundary. A completed
inverse b-edge at the last handle ends at the FIRST handle corner. -/
def nextHandle (k : ℕ) : Handle k → Handle k
  | Sum.inl i =>
      if i = 0 then secondHandle k
      else if hk : 0 < k then Sum.inr ⟨0, hk⟩ else firstHandle k
  | Sum.inr j =>
      if hj : j.val + 1 < k then Sum.inr ⟨j.val + 1, hj⟩
      else firstHandle k

@[simp] theorem nextHandle_first (k : ℕ) :
    nextHandle k (firstHandle k) = secondHandle k := by
  simp [nextHandle, firstHandle, secondHandle]

@[simp] theorem nextHandle_second_noExtras :
    nextHandle 0 (secondHandle 0) = firstHandle 0 := by
  simp [nextHandle, secondHandle, firstHandle]

/-- If an extra handle exists, the second base handle is immediately
followed by the extra handle with source index zero. -/
theorem nextHandle_second_withExtras (k : ℕ) (hk : 0 < k) :
    nextHandle k (secondHandle k) =
      Sum.inr (⟨0, hk⟩ : Fin k) := by
  simp [nextHandle, secondHandle, hk]

/-- Consecutive extra handle IDs are joined in their genuine Fin k
order, with no identification of an extra with a base source label. -/
theorem nextHandle_extra_step (k : ℕ) (j : Fin k)
    (hj : j.val + 1 < k) :
    nextHandle k (Sum.inr j) =
      Sum.inr (⟨j.val + 1, hj⟩ : Fin k) := by
  simp [nextHandle, hj]

/-- The last additional handle closes the cyclic boundary at the
first base handle; this is the final polygon boundary seam. -/
theorem nextHandle_extra_last (k : ℕ) (j : Fin k)
    (hj : ¬ j.val + 1 < k) :
    nextHandle k (Sum.inr j) = firstHandle k := by
  simp [nextHandle, hj]

/-- The ACTUAL direct corner identifications generated by pairing
the two occurrences of each named oriented generator.

For a forward a-edge (0→1) and inverse a-edge (2→3), orientation
reversal pairs 0 with 3, and 1 with 2.
For forward b (1→2) and inverse b (3→next 0), orientation reversal
pairs 2 with 3, and 1 with the next handle's 0 corner.

No other raw corner pairs are introduced. -/
inductive DirectCornerGluing (k : ℕ) :
    PolygonCorner k → PolygonCorner k → Prop where
  | a_start (h : Handle k) :
      DirectCornerGluing k (h, 0) (h, 3)
  | a_end (h : Handle k) :
      DirectCornerGluing k (h, 1) (h, 2)
  | b_end (h : Handle k) :
      DirectCornerGluing k (h, 2) (h, 3)
  | b_start (h : Handle k) :
      DirectCornerGluing k (h, 1) (nextHandle k h, 0)

/-- The smallest equivalence relation generated by the REAL declared
corner pairings, using Mathlib's equivalence closure. -/
def CornerConnected (k : ℕ) :
    PolygonCorner k → PolygonCorner k → Prop :=
  Relation.EqvGen (DirectCornerGluing k)

/-- The inverse of the first a-pairing joins local corner 3 to 0. -/
theorem local_three_to_zero (k : ℕ) (h : Handle k) :
    CornerConnected k (h, 3) (h, 0) :=
  (Relation.EqvGen.rel _ _ (DirectCornerGluing.a_start h)).symm

/-- The b-end pairing and previous a-start pairing join 2 to 0. -/
theorem local_two_to_zero (k : ℕ) (h : Handle k) :
    CornerConnected k (h, 2) (h, 0) :=
  Relation.EqvGen.trans _ _ _
    (Relation.EqvGen.rel _ _ (DirectCornerGluing.b_end h))
    (local_three_to_zero k h)

/-- The forward a-end gluing and b-end gluing join 1 to 0. -/
theorem local_one_to_zero (k : ℕ) (h : Handle k) :
    CornerConnected k (h, 1) (h, 0) :=
  Relation.EqvGen.trans _ _ _
    (Relation.EqvGen.rel _ _ (DirectCornerGluing.a_end h))
    (local_two_to_zero k h)

/-- Every one of the four local polygon corners is related to that
handle's corner zero by the actual generated relations. -/
theorem local_corner_to_zero (k : ℕ) (h : Handle k)
    (i : Fin 4) : CornerConnected k (h, i) (h, 0) := by
  fin_cases i
  · exact Relation.EqvGen.refl _
  · exact local_one_to_zero k h
  · exact local_two_to_zero k h
  · exact local_three_to_zero k h

/-- The b-side reverse-orientation seam, combined with local
pairings, connects each handle's corner zero to the NEXT handle's
corner zero, not an arbitrary unrelated vertex. -/
theorem handle_zero_to_next_zero (k : ℕ) (h : Handle k) :
    CornerConnected k (h, 0) (nextHandle k h, 0) :=
  Relation.EqvGen.trans _ _ _ (local_one_to_zero k h).symm
    (Relation.EqvGen.rel _ _ (DirectCornerGluing.b_start h))

/-- First and second handle zero corners are joined by the true
cyclic b-edge seam. -/
theorem first_to_second_zero (k : ℕ) :
    CornerConnected k (firstHandle k, 0) (secondHandle k, 0) := by
  simpa using handle_zero_to_next_zero k (firstHandle k)

/-- When k>0 the second base handle is joined to the first extra
handle, preserving the source-index identity of that extra handle. -/
theorem second_to_extra_zero (k : ℕ) (hk : 0 < k) :
    CornerConnected k (secondHandle k, 0)
      (Sum.inr (⟨0, hk⟩ : Fin k), 0) := by
  simpa [nextHandle_second_withExtras k hk] using
    (handle_zero_to_next_zero k (secondHandle k))

/-- Every extra handle zero corner is connected to the FIRST base
handle zero corner. This is a real induction over finite extra-handle
source indices, not an assumption that all corners are equivalent. -/
theorem first_to_extra_zero (k : ℕ) (j : Fin k) :
    CornerConnected k (firstHandle k, 0) (Sum.inr j, 0) := by
  have aux : ∀ n : ℕ, ∀ hn : n < k,
      CornerConnected k (firstHandle k, 0)
        (Sum.inr (⟨n, hn⟩ : Fin k), 0) := by
    intro n
    induction n with
    | zero =>
        intro hn
        exact Relation.EqvGen.trans _ _ _ (first_to_second_zero k)
          (second_to_extra_zero k hn)
    | succ n ih =>
        intro hn
        have hp : n < k := by omega
        have hstep : CornerConnected k
            (Sum.inr (⟨n, hp⟩ : Fin k), 0)
            (Sum.inr (⟨n + 1, hn⟩ : Fin k), 0) := by
          simpa [nextHandle_extra_step k (⟨n, hp⟩ : Fin k) hn] using
            (handle_zero_to_next_zero k
              (Sum.inr (⟨n, hp⟩ : Fin k)))
        exact Relation.EqvGen.trans _ _ _ (ih hp) hstep
  simpa using aux j.val j.isLt

/-- Every handle zero corner is connected to the first base handle.
The two base cases and the entire extra-handle induction are explicit. -/
theorem first_to_handle_zero (k : ℕ) (h : Handle k) :
    CornerConnected k (firstHandle k, 0) (h, 0) := by
  cases h with
  | inl i =>
      fin_cases i
      · exact Relation.EqvGen.refl _
      · exact first_to_second_zero k
  | inr j =>
      exact first_to_extra_zero k j

/-- Every actual finite polygon corner is related to the first corner
by the exact generated side-pairing relation, not a declared universal
relation. -/
theorem first_to_every_corner (k : ℕ) (v : PolygonCorner k) :
    CornerConnected k (firstHandle k, 0) v := by
  rcases v with ⟨h, i⟩
  exact Relation.EqvGen.trans _ _ _ (first_to_handle_zero k h)
    (local_corner_to_zero k h i).symm

/-- THE ONE-VERTEX COMBINATORIAL GLUING THEOREM: all pairs of polygon
corner IDs are equivalent under the genuine generated edge pairings. -/
theorem all_corners_connected (k : ℕ) (u v : PolygonCorner k) :
    CornerConnected k u v :=
  Relation.EqvGen.trans _ _ _ (first_to_every_corner k u).symm
    (first_to_every_corner k v)

/-- The actual setoid used for the polygon corner quotient. Its relation
is the EQUIVALENCE CLOSURE of the generated pairing, never an arbitrary
always-true relation independent of the polygon source. -/
def cornerSetoid (k : ℕ) : Setoid (PolygonCorner k) where
  r := CornerConnected k
  iseqv := Relation.EqvGen.is_equivalence _

/-- ACTUAL quotient of the FINITE polygon corner identifiers by the
smallest equivalence containing the opposite-side endpoint gluings.
This is not the topological quotient of a 2-disk. -/
def GluedPolygonCorners (k : ℕ) : Type :=
  Quotient (cornerSetoid k)

/-- The actual corner quotient has at most one equivalence class. -/
theorem gluedPolygonCorners_subsingleton (k : ℕ) :
    Subsingleton (GluedPolygonCorners k) := by
  constructor
  intro a b
  refine Quotient.inductionOn₂ a b ?_
  intro u v
  exact Quotient.sound (all_corners_connected k u v)

/-- The corner quotient is inhabited by the source-identified first
polygon corner, so it has EXACTLY ONE combinatorial vertex class. -/
theorem gluedPolygonCorners_nonempty (k : ℕ) :
    Nonempty (GluedPolygonCorners k) :=
  ⟨Quotient.mk (cornerSetoid k) (firstHandle k, (0 : Fin 4))⟩

/-- The finite oriented side pairings produce ONE (not zero) vertex
in the quotient of polygon corner IDS, for every genus g=k+2≥2. -/
@[instance_reducible] noncomputable def gluedPolygonCorners_unique (k : ℕ) :
    Unique (GluedPolygonCorners k) where
  default := (gluedPolygonCorners_nonempty k).some
  uniq _ := by
    let : Subsingleton (GluedPolygonCorners k) :=
      gluedPolygonCorners_subsingleton k
    exact Subsingleton.elim _ _

/-- The two initial handles are DIFFERENT generator-source blocks
before applying any gluing quotient, for every k. -/
theorem first_second_handles_distinct (k : ℕ) :
    firstHandle k ≠ secondHandle k := by
  simp [firstHandle, secondHandle]

/-- Essential seam control: the first and second handles are distinct
source IDs yet the b-pairing identifies their corner-zero vertices.
A fake gluing relation that only identifies corners WITHIN handles
cannot satisfy both facts. -/
theorem essential_b_seam_connects_distinct_handles (k : ℕ) :
    firstHandle k ≠ secondHandle k ∧
      CornerConnected k (firstHandle k, 0) (secondHandle k, 0) :=
  ⟨first_second_handles_distinct k, first_to_second_zero k⟩

/-- Concrete nonvacuous genus-two gluing: the first and second
handles have distinct source identifiers but their zero corner
classes become equal under the true side pairings. -/
theorem genusTwo_distinct_handles_glued :
    CornerConnected 0 (firstHandle 0, 0) (secondHandle 0, 0) :=
  first_to_second_zero 0

/-- Genus three truly has a first EXTRA handle, connected to the
second base handle by the b-edge seam rather than arbitrary equality. -/
theorem genusThree_extra_handle_glued :
    CornerConnected 1 (secondHandle 1, 0)
      (Sum.inr (⟨0, by decide⟩ : Fin 1), 0) :=
  second_to_extra_zero 1 (by decide)

/-- Retain the exact original count of 2g one-cell generator source IDs;
the one-vertex quotient does NOT identify or delete their edge labels. -/
theorem one_cell_source_count_unchanged (k : ℕ) :
    Fintype.card (GenusLetter k) = 2 * (k + 2) :=
  oneCellSourceCount k

    (hab : a ≤ b) (hbc : b ≤ c) (hcd : c ≤ d)
    (hint : IntervalIntegrable (fun x : ℝ => f (corner x y)) volume a d) :
    horizontal f a d y =
      horizontal f a b y + horizontal f b c y + horizontal f c d y := by

  have hac : a ≤ c := hab.trans hbc

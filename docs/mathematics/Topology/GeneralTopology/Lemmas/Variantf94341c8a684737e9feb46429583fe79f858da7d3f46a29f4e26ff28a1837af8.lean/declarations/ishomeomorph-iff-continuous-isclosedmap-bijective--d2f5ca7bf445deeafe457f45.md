lemma isHomeomorph_iff_continuous_isClosedMap_bijective  : IsHomeomorph f ↔

  ⟨fun hf => ⟨hf.continuous, hf.isClosedMap, hf.bijective⟩, fun ⟨hf, hf', hf''⟩ =>

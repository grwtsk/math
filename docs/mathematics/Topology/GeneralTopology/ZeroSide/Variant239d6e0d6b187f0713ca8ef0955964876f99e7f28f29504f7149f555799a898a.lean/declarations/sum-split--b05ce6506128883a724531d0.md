  have hdisj2 : Disjoint P.R (P.R.image D.σ) := by
    rw [disjoint_iff_ne]; rintro a ha _ hb rfl
    obtain ⟨w, hw, hwa⟩ := mem_image.mp hb
    exact P.σ_not_mem w hw (hwa ▸ ha)
  have hdisj1 : Disjoint D.onLine (P.R ∪ P.R.image D.σ) := by

theorem TendstoInDistribution.continuous_comp {F : Type*} [OpensMeasurableSpace E]

    (h : TendstoInDistribution X l Z μ μ') :

      congr
      rw [AEMeasurable.map_map_of_aemeasurable hg.aemeasurable h.aemeasurable_limit]

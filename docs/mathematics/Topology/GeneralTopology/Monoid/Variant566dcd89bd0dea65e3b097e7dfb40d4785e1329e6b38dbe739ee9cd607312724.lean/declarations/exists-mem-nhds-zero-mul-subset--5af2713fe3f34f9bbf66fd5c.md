theorem exists_mem_nhds_zero_mul_subset

  refine hK.induction_on ?_ ?_ ?_ ?_

  · rintro s t hst ⟨V, hV, hV'⟩

  · rintro s t ⟨V, V_in, hV'⟩ ⟨W, W_in, hW'⟩

/-
The graph realization in this file is an actual quotient.  It starts from the
disjoint union of vertices and edge-interval copies and generates precisely
the two endpoint identifications for every edge.  The topology is the quotient
topology of the coproduct topology in which vertices and edge labels are
discrete and each edge copy carries the supplied interval topology.

The abstract `ClosedInterval` contract intentionally does not assert
connectedness, density, or that every nonendpoint belongs to `interior`.
Consequently, the interior theorems below apply exactly to the declared
interior subset.  A future real-unit-interval adapter may discharge stronger
one-complex properties without changing this quotient construction.

/-- The disjoint union of vertices and interval-parametrized edge points. -/
inductive GraphPrePoint
    (Vertex : Type uVertex) (Edge : Type uEdge)
    (IntervalPoint : Type uInterval) where
  | vertex : Vertex → GraphPrePoint Vertex Edge IntervalPoint
  | edgePoint : Edge → IntervalPoint →
      GraphPrePoint Vertex Edge IntervalPoint

/--
The equivalence relation generated by the source and target endpoint
identifications.  Reflexivity, symmetry, and transitivity are constructors,
so this is the explicit equivalence closure of the two endpoint rules.
-/
inductive GraphEndpointEqv
    {Vertex : Type uVertex} {Edge : Type uEdge}
    (G : DirectedGraph Vertex Edge) (I : ClosedInterval.{uInterval}) :
    GraphPrePoint Vertex Edge I.Point →
      GraphPrePoint Vertex Edge I.Point → Prop where
  | refl (x) : GraphEndpointEqv G I x x
  | zero (e) : GraphEndpointEqv G I
      (.edgePoint e I.zero) (.vertex (G.source e))
  | one (e) : GraphEndpointEqv G I
      (.edgePoint e I.one) (.vertex (G.target e))
  | symm {x y} : GraphEndpointEqv G I x y → GraphEndpointEqv G I y x
  | trans {x y z} :
      GraphEndpointEqv G I x y → GraphEndpointEqv G I y z →
        GraphEndpointEqv G I x z

/--
`GraphEndpointEqv` is the least equivalence relation containing the two
endpoint rules: every other equivalence relation containing those rules also
contains every generated graph-endpoint equivalence.
-/
theorem graphEndpointEqv_minimal
    {Vertex : Type uVertex} {Edge : Type uEdge}
    {G : DirectedGraph Vertex Edge} {I : ClosedInterval.{uInterval}}
    {R : GraphPrePoint Vertex Edge I.Point →
      GraphPrePoint Vertex Edge I.Point → Prop}
    (hRefl : ∀ x, R x x)
    (hZero : ∀ e, R (.edgePoint e I.zero) (.vertex (G.source e)))
    (hOne : ∀ e, R (.edgePoint e I.one) (.vertex (G.target e)))
    (hSymm : ∀ {x y}, R x y → R y x)
    (hTrans : ∀ {x y z}, R x y → R y z → R x z)
    {x y : GraphPrePoint Vertex Edge I.Point}
    (hxy : GraphEndpointEqv G I x y) : R x y := by
  induction hxy with
  | refl x => exact hRefl x
  | zero e => exact hZero e
  | one e => exact hOne e
  | symm _ ih => exact hSymm ih
  | trans _ _ ihxy ihyz => exact hTrans ihxy ihyz

/-- The endpoint-generated setoid used by the graph quotient. -/
def graphEndpointSetoid
    {Vertex : Type uVertex} {Edge : Type uEdge}
    (G : DirectedGraph Vertex Edge) (I : ClosedInterval.{uInterval}) :
    Setoid (GraphPrePoint Vertex Edge I.Point) where
  r := GraphEndpointEqv G I
  iseqv := {
    refl := GraphEndpointEqv.refl
    symm := GraphEndpointEqv.symm
    trans := GraphEndpointEqv.trans
  }

/-- The geometric point type `V(G) ⊔ (E(G) × I)` modulo endpoints. -/
def GraphPoint
    {Vertex : Type uVertex} {Edge : Type uEdge}
    (G : DirectedGraph Vertex Edge) (I : ClosedInterval.{uInterval}) :
    Type (max uVertex uEdge uInterval) :=
  Quotient (graphEndpointSetoid G I)

/-- The quotient map from graph pre-points. -/
def graphQuotientMap
    {Vertex : Type uVertex} {Edge : Type uEdge}
    (G : DirectedGraph Vertex Edge) (I : ClosedInterval.{uInterval}) :
    GraphPrePoint Vertex Edge I.Point → GraphPoint G I :=
  fun x => Quotient.mk _ x

/-- Public concise name for the canonical graph quotient map. -/
def quotientMap
    {Vertex : Type uVertex} {Edge : Type uEdge}
    (G : DirectedGraph Vertex Edge) (I : ClosedInterval.{uInterval}) :
    GraphPrePoint Vertex Edge I.Point → GraphPoint G I :=
  graphQuotientMap G I

/-- The canonical inclusion of graph vertices into the quotient. -/
def vertexPoint
    {Vertex : Type uVertex} {Edge : Type uEdge}
    (G : DirectedGraph Vertex Edge) (I : ClosedInterval.{uInterval}) :
    Vertex → GraphPoint G I :=
  fun v => graphQuotientMap G I (.vertex v)

/-- The canonical interval-parametrized path of an edge in the quotient. -/
def edgePath
    {Vertex : Type uVertex} {Edge : Type uEdge}
    (G : DirectedGraph Vertex Edge) (I : ClosedInterval.{uInterval}) :
    Edge → I.Point → GraphPoint G I :=
  fun e t => graphQuotientMap G I (.edgePoint e t)

/-- The zero endpoint is identified with the source vertex. -/
theorem edgePath_zero
    {Vertex : Type uVertex} {Edge : Type uEdge}
    (G : DirectedGraph Vertex Edge) (I : ClosedInterval.{uInterval}) (e : Edge) :
    edgePath G I e I.zero = vertexPoint G I (G.source e) := by
  exact Quotient.sound (GraphEndpointEqv.zero e)

/-- The one endpoint is identified with the target vertex. -/
theorem edgePath_one
    {Vertex : Type uVertex} {Edge : Type uEdge}
    (G : DirectedGraph Vertex Edge) (I : ClosedInterval.{uInterval}) (e : Edge) :
    edgePath G I e I.one = vertexPoint G I (G.target e) := by
  exact Quotient.sound (GraphEndpointEqv.one e)

/--
The full endpoint-equivalence fiber of one vertex.  It contains the vertex
constructor itself, every zero endpoint sourced there, and every one endpoint
targeted there.  The endpoint generators preserve this predicate exactly.
-/
private def VertexFiberTag
    {Vertex : Type uVertex} {Edge : Type uEdge}
    (G : DirectedGraph Vertex Edge) (I : ClosedInterval.{uInterval})
    (v : Vertex) : GraphPrePoint Vertex Edge I.Point → Prop
  | .vertex w => w = v
  | .edgePoint e t =>
      (t = I.zero ∧ G.source e = v) ∨
      (t = I.one ∧ G.target e = v)

private theorem graphEndpointEqv_preserves_vertexFiberTag
    {Vertex : Type uVertex} {Edge : Type uEdge}
    {G : DirectedGraph Vertex Edge} {I : ClosedInterval.{uInterval}}
    {v : Vertex}
    {x y : GraphPrePoint Vertex Edge I.Point}
    (hxy : GraphEndpointEqv G I x y) :
    VertexFiberTag G I v x ↔ VertexFiberTag G I v y := by
  induction hxy with
  | refl _ => exact Iff.rfl
  | zero e =>
      constructor
      · intro h
        cases h with
        | inl hSource => exact hSource.2
        | inr hTarget => exact False.elim (I.zero_ne_one hTarget.1)
      · intro hSource
        exact Or.inl ⟨rfl, hSource⟩
  | one e =>
      constructor
      · intro h
        cases h with
        | inl hSource => exact False.elim (I.zero_ne_one hSource.1.symm)
        | inr hTarget => exact hTarget.2
      · intro hTarget
        exact Or.inr ⟨rfl, hTarget⟩
  | symm _ ih => exact ih.symm
  | trans _ _ ihxy ihyz => exact Iff.trans ihxy ihyz

/-- Distinct source vertices remain distinct in the graph quotient. -/
theorem vertexPoint_injective
    {Vertex : Type uVertex} {Edge : Type uEdge}
    {G : DirectedGraph Vertex Edge} {I : ClosedInterval.{uInterval}} :
    Function.Injective (vertexPoint G I) := by
  intro v w hEq
  have hRel : GraphEndpointEqv G I (.vertex v) (.vertex w) :=
    Quotient.exact hEq
  have hTag : VertexFiberTag G I v (.vertex w) :=
    (graphEndpointEqv_preserves_vertexFiberTag hRel).1 rfl
  exact hTag.symm

/--
An invariant tagging one fixed edge-interior pre-point.  Endpoint generators
cannot change this tag because interval interiors exclude zero and one.
-/
private def InteriorTag
    {Vertex : Type uVertex} {Edge : Type uEdge}
    {I : ClosedInterval.{uInterval}}
    (e : Edge) (t : I.Point) :
    GraphPrePoint Vertex Edge I.Point → Prop
  | .vertex _ => False
  | .edgePoint f u => f = e ∧ u = t

private theorem interiorTag_zero_false
    {Vertex : Type uVertex} {Edge : Type uEdge}
    {I : ClosedInterval.{uInterval}}
    {e : Edge} {t : I.Point}
    (ht : I.interior t) (f : Edge) :
    ¬ InteriorTag (Vertex := Vertex) e t (.edgePoint f I.zero) := by
  intro h
  have hzero : I.zero = t := h.2
  apply I.zero_not_mem_interior
  rw [hzero]
  exact ht

private theorem interiorTag_one_false
    {Vertex : Type uVertex} {Edge : Type uEdge}
    {I : ClosedInterval.{uInterval}}
    {e : Edge} {t : I.Point}
    (ht : I.interior t) (f : Edge) :
    ¬ InteriorTag (Vertex := Vertex) e t (.edgePoint f I.one) := by
  intro h
  have hone : I.one = t := h.2
  apply I.one_not_mem_interior
  rw [hone]
  exact ht

private theorem graphEndpointEqv_preserves_interiorTag
    {Vertex : Type uVertex} {Edge : Type uEdge}
    {G : DirectedGraph Vertex Edge} {I : ClosedInterval.{uInterval}}
    {e : Edge} {t : I.Point}
    (ht : I.interior t)
    {x y : GraphPrePoint Vertex Edge I.Point}
    (hxy : GraphEndpointEqv G I x y) :
    InteriorTag e t x ↔ InteriorTag e t y := by
  induction hxy with
  | refl _ => exact Iff.rfl
  | zero f =>
      constructor
      · intro h
        exact False.elim (interiorTag_zero_false ht f h)
      · intro h
        exact False.elim h
  | one f =>
      constructor
      · intro h
        exact False.elim (interiorTag_one_false ht f h)
      · intro h
        exact False.elim h
  | symm _ ih => exact ih.symm
  | trans _ _ ihxy ihyz => exact Iff.trans ihxy ihyz

/-- A quotient point lies in the declared interior of the displayed graph edge. -/
def EdgeInteriorPoint
    {Vertex : Type uVertex} {Edge : Type uEdge}
    (G : DirectedGraph Vertex Edge) (I : ClosedInterval.{uInterval})
    (e : Edge) (x : GraphPoint G I) : Prop :=
  ∃ t, I.interior t ∧ edgePath G I e t = x

/-- Each edge path is injective when restricted to interval-interior points. -/
theorem edgePath_injective_on_interior
    {Vertex : Type uVertex} {Edge : Type uEdge}
    {G : DirectedGraph Vertex Edge} {I : ClosedInterval.{uInterval}}
    {e : Edge} {t u : I.Point}
    (ht : I.interior t) (_hu : I.interior u)
    (hEq : edgePath G I e t = edgePath G I e u) :
    t = u := by
  have hRel : GraphEndpointEqv G I
      (.edgePoint e t) (.edgePoint e u) := Quotient.exact hEq
  have hTag : InteriorTag (Vertex := Vertex) e t (.edgePoint e u) :=
    (graphEndpointEqv_preserves_interiorTag ht hRel).1 ⟨rfl, rfl⟩
  exact hTag.2.symm

/-- Distinct graph edges have disjoint interiors in the quotient realization. -/
theorem distinct_edgeInteriors_disjoint
    {Vertex : Type uVertex} {Edge : Type uEdge}
    {G : DirectedGraph Vertex Edge} {I : ClosedInterval.{uInterval}}
    {e f : Edge} (hef : e ≠ f) {x : GraphPoint G I}
    (he : EdgeInteriorPoint G I e x)
    (hf : EdgeInteriorPoint G I f x) : False := by
  rcases he with ⟨t, ht, htx⟩
  rcases hf with ⟨u, _hu, hux⟩
  have hEq : edgePath G I e t = edgePath G I f u := htx.trans hux.symm
  have hRel : GraphEndpointEqv G I
      (.edgePoint e t) (.edgePoint f u) := Quotient.exact hEq
  have hTag : InteriorTag (Vertex := Vertex) e t (.edgePoint f u) :=
    (graphEndpointEqv_preserves_interiorTag ht hRel).1 ⟨rfl, rfl⟩
  exact hef hTag.1.symm

/-- No edge-interior point is a vertex point in the quotient realization. -/
theorem edgeInterior_avoids_vertices
    {Vertex : Type uVertex} {Edge : Type uEdge}
    {G : DirectedGraph Vertex Edge} {I : ClosedInterval.{uInterval}}
    {e : Edge} {x : GraphPoint G I}
    (he : EdgeInteriorPoint G I e x)
    {v : Vertex} (hv : vertexPoint G I v = x) : False := by
  rcases he with ⟨t, ht, htx⟩
  have hEq : edgePath G I e t = vertexPoint G I v := htx.trans hv.symm
  have hRel : GraphEndpointEqv G I
      (.edgePoint e t) (.vertex v) := Quotient.exact hEq
  have hTag : InteriorTag (Vertex := Vertex) e t (.vertex v) :=
    (graphEndpointEqv_preserves_interiorTag ht hRel).1 ⟨rfl, rfl⟩
  exact hTag

/-! ## Coproduct and quotient topologies -/

/--
The pre-realization topology.  The vertex summand is discrete (there is no
condition on vertex membership), while a set is open on the edge summand
exactly when its inverse image in every edge's interval copy is open.  This is
the coproduct of the discrete vertex space and the product of the discrete
edge-label space with the interval.
-/
def graphPreTopology
    {Vertex : Type uVertex} {Edge : Type uEdge}
    (_G : DirectedGraph Vertex Edge) (I : ClosedInterval.{uInterval}) :
    TopologicalSpaceLike (GraphPrePoint Vertex Edge I.Point) where
  IsOpen := fun U =>
    ∀ e, I.topology.IsOpen (fun t => U (.edgePoint e t))
  isOpen_univ := by
    intro _e
    exact I.topology.isOpen_univ
  isOpen_inter := by
    intro U V hU hV e
    exact I.topology.isOpen_inter _ _ (hU e) (hV e)
  isOpen_sUnion := by
    intro 𝒰 h𝒰 e
    let edgeFamily : Set (Set I.Point) :=
      fun A => ∃ U, U ∈ 𝒰 ∧
        A = fun t => U (GraphPrePoint.edgePoint e t)
    have hopen : ∀ A, A ∈ edgeFamily → I.topology.IsOpen A := by
      intro A hA
      obtain ⟨U, hU, rfl⟩ := hA
      exact h𝒰 U hU e
    have hOpenUnion := I.topology.isOpen_sUnion edgeFamily hopen
    have hEq : Set.sUnion edgeFamily =
        (fun t => (Set.sUnion 𝒰) (GraphPrePoint.edgePoint e t)) := by
      apply Set.ext
      intro t
      constructor
      · intro ht
        obtain ⟨A, ⟨U, hU, rfl⟩, htU⟩ := ht
        exact ⟨U, hU, htU⟩
      · intro ht
        obtain ⟨U, hU, htU⟩ := ht
        exact ⟨(fun s => U (GraphPrePoint.edgePoint e s)),
          ⟨U, hU, rfl⟩, htU⟩
    rw [hEq] at hOpenUnion
    exact hOpenUnion

/-- A subset supported only on vertices is open, exposing discreteness there. -/
theorem graphPreTopology_vertexSubset_open
    {Vertex : Type uVertex} {Edge : Type uEdge}
    (G : DirectedGraph Vertex Edge) (I : ClosedInterval.{uInterval})
    (U : Set Vertex) :
    (graphPreTopology G I).IsOpen
      (fun p => match p with
        | .vertex v => v ∈ U
        | .edgePoint _ _ => False) := by
  intro _e
  exact I.topology.isOpen_empty

/--
The quotient topology on the actual graph quotient.  By construction, a set
of graph points is open exactly when its inverse image under
`graphQuotientMap` is open in `graphPreTopology`.
-/
def graphRealizationTopology
    {Vertex : Type uVertex} {Edge : Type uEdge}
    (G : DirectedGraph Vertex Edge) (I : ClosedInterval.{uInterval}) :
    TopologicalSpaceLike (GraphPoint G I) :=
  quotientTopology (graphPreTopology G I) (graphQuotientMap G I)

/-- The canonical quotient map is continuous for the quotient topology. -/
theorem graphQuotientMap_continuous
    {Vertex : Type uVertex} {Edge : Type uEdge}
    (G : DirectedGraph Vertex Edge) (I : ClosedInterval.{uInterval}) :
    Continuous
      (graphPreTopology G I)
      (graphRealizationTopology G I)
      (graphQuotientMap G I) := by
  exact continuous_quotientMap
    (graphPreTopology G I) (graphQuotientMap G I)

/-- The concise `quotientMap` alias is continuous as well. -/
theorem quotientMap_continuous
    {Vertex : Type uVertex} {Edge : Type uEdge}
    (G : DirectedGraph Vertex Edge) (I : ClosedInterval.{uInterval}) :
    Continuous
      (graphPreTopology G I)
      (graphRealizationTopology G I)
      (quotientMap G I) :=
  graphQuotientMap_continuous G I

/-- Every canonical edge path is continuous. -/
theorem edgePath_continuous
    {Vertex : Type uVertex} {Edge : Type uEdge}
    (G : DirectedGraph Vertex Edge) (I : ClosedInterval.{uInterval}) (e : Edge) :
    Continuous I.topology (graphRealizationTopology G I) (edgePath G I e) := by
  intro U hU
  exact hU e

/--
The expected final-topology universal property: a map out of the graph
quotient is continuous exactly when its composite with the quotient map is
continuous on the pre-realization.
-/
theorem continuous_from_graphRealization_iff
    {Vertex : Type uVertex} {Edge : Type uEdge}
    {Y : Type uTarget}
    (G : DirectedGraph Vertex Edge) (I : ClosedInterval.{uInterval})
    (TY : TopologicalSpaceLike Y) (f : GraphPoint G I → Y) :
    Continuous (graphRealizationTopology G I) TY f ↔
      Continuous (graphPreTopology G I) TY
        (fun x => f (graphQuotientMap G I x)) := by
  constructor
  · intro hf V hV
    exact hf V hV
  · intro hf V hV
    exact hf V hV

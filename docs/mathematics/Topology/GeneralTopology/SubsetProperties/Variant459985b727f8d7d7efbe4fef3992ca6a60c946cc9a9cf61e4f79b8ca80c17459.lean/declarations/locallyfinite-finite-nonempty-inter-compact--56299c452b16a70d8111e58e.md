theorem LocallyFinite.finite_nonempty_inter_compact {ι : Type*} {f : ι → Set α}
    (hf : LocallyFinite f) {s : Set α} (hs : IsCompact s) : { i | (f i ∩ s).Nonempty }.Finite := by

  rintro i ⟨x, hx⟩

  exact mem_biUnion hct ⟨x, hx.1, hcx⟩

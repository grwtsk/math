    (hc₀ : 0 < c) (hc₁ : c < 1) :
    ∃ n : ℕ, a < c ^ n ∧ c ^ n < b := by
  have := exists_pow_lt_of_lt_one hb₀ hc₁

  by_contra! H

  exact Nat.find_min this (Nat.sub_one_lt hn) H

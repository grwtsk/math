/-!
# The repeated characteristic algebra at the two source pole exponents

The polynomial `(X^2 - 1/4)^2` has double roots at `1/2` and `-1/2`.
Its native `AdjoinRoot` algebra retains a nonzero square-zero class of
`X^2 - 1/4`. The Jacobson radical is exactly the ideal generated by that
class. Two-root evaluation removes the radical and gives `ℂ × ℂ`.

These are algebraic results for the source-selected polynomial. Identifying
polynomial action with differentiation on an analytic function space requires
an explicit representation, including its domain and boundary obligations.
The entire four-dimensional algebra is not its two-dimensional radical.
No arithmetic bound, zero simplicity, or terminal RH statement is assumed.

AI attribution: Codex authored this module and its proofs.
-/

/-- The unsquared characteristic factor at the two pole exponents. -/
def poleQuadratic : ℂ[X] := X ^ 2 - C (1 / 4)

def PolePolynomial : ℂ[X] := poleQuadratic ^ 2

/-- The native four-dimensional repeated-root polynomial operator algebra. -/
abbrev PoleOperatorAlgebra := AdjoinRoot PolePolynomial

/-- The class of the unsquared source characteristic factor. -/
def poleInfinitesimal : PoleOperatorAlgebra :=
  AdjoinRoot.root PolePolynomial ^ 2 - algebraMap ℂ PoleOperatorAlgebra (1 / 4)

theorem pole_quadratic_factorization :
    poleQuadratic = (X - C (1 / 2 : ℂ)) * (X - C (-1 / 2 : ℂ)) := by
  symm
  calc
    (X - C (1 / 2 : ℂ)) * (X - C (-1 / 2 : ℂ)) =
        X ^ 2 - C ((1 / 2 : ℂ) + -1 / 2) * X + C ((1 / 2 : ℂ) * (-1 / 2)) := by
      simp
      ring
    _ = poleQuadratic := by
      norm_num [poleQuadratic]
      ring

def polePositiveObservation : PoleOperatorAlgebra →ₐ[ℂ] ℂ :=
  AdjoinRoot.liftAlgHom _ (Algebra.ofId ℂ ℂ) (1 / 2) (by
    norm_num [PolePolynomial, poleQuadratic])

theorem pole_observation_infinitesimal : poleObservation poleInfinitesimal = 0 := by

  · intro x hx

    let : (RingHom.ker poleNegativeObservation.toRingHom).IsMaximal :=

    have hm := RingHom.mem_ker.mp (Ring.jacobson_le_of_isMaximal
      (RingHom.ker poleNegativeObservation.toRingHom) hx)

/-- One local double-root characteristic factor at a specified pole exponent. -/
abbrev ShiftedPoleFiber (r : ℂ) := AdjoinRoot ((X - C r) ^ 2)

/-- Its centered square-zero coordinate. -/
def shiftedPoleInfinitesimal (r : ℂ) : ShiftedPoleFiber r :=
  AdjoinRoot.mk ((X - C r) ^ 2) (X - C r)

theorem shifted_pole_infinitesimal_sq (r : ℂ) : shiftedPoleInfinitesimal r ^ 2 = 0 := by
  rw [shiftedPoleInfinitesimal, ← map_pow, AdjoinRoot.mk_self]

theorem shifted_pole_infinitesimal_eq_root_sub (r : ℂ) :
    shiftedPoleInfinitesimal r = AdjoinRoot.root ((X - C r) ^ 2) - algebraMap ℂ _ r := by
  simp [shiftedPoleInfinitesimal]

/-- The local characteristic coordinate becomes `r + ε` in native dual numbers. -/
def shiftedPoleToDual (r : ℂ) : ShiftedPoleFiber r →ₐ[ℂ] DualNumber ℂ :=
  AdjoinRoot.liftAlgHom _ (Algebra.ofId ℂ (DualNumber ℂ))
    (algebraMap ℂ (DualNumber ℂ) r + DualNumber.eps) (by simp)

/-- Native dual-number universality sends `ε` to the centered characteristic coordinate. -/
def dualToShiftedPole (r : ℂ) : DualNumber ℂ →ₐ[ℂ] ShiftedPoleFiber r :=
  DualNumber.lift ⟨(Algebra.ofId ℂ (ShiftedPoleFiber r), shiftedPoleInfinitesimal r),
    by rw [← pow_two, shifted_pole_infinitesimal_sq], fun _ ↦ Commute.all _ _⟩

theorem shifted_pole_to_dual_infinitesimal (r : ℂ) :
    shiftedPoleToDual r (shiftedPoleInfinitesimal r) = DualNumber.eps := by
  rw [shifted_pole_infinitesimal_eq_root_sub, map_sub]
  simp [shiftedPoleToDual]

theorem dual_to_shifted_pole_eps (r : ℂ) :
    dualToShiftedPole r DualNumber.eps = shiftedPoleInfinitesimal r := by
  simp [dualToShiftedPole]

/-- Each local repeated-root characteristic quotient is canonically a native dual-number algebra. -/
def shiftedPoleDualEquiv (r : ℂ) : ShiftedPoleFiber r ≃ₐ[ℂ] DualNumber ℂ :=
  AlgEquiv.ofAlgHom (shiftedPoleToDual r) (dualToShiftedPole r)
    (by
      apply DualNumber.algHom_ext
      simp only [AlgHom.comp_apply, dual_to_shifted_pole_eps,
        shifted_pole_to_dual_infinitesimal, AlgHom.id_apply])
    (by
      apply AdjoinRoot.algHom_ext
      simp only [AlgHom.comp_apply, shiftedPoleToDual, AdjoinRoot.liftAlgHom_root,
        map_add, AlgHom.commutes, dual_to_shifted_pole_eps, AlgHom.id_apply]
      rw [shifted_pole_infinitesimal_eq_root_sub]
      ring)

theorem pole_radical_mul_eq_zero {x y : PoleOperatorAlgebra}
    (hx : x ∈ Ring.jacobson PoleOperatorAlgebra)
    (hy : y ∈ Ring.jacobson PoleOperatorAlgebra) : x * y = 0 := by

/-- Every algebra representation of the quotient annihilates its defining polynomial.

    aeval (ρ (AdjoinRoot.root PolePolynomial)) PolePolynomial = 0 := by
  rw [aeval_algHom_apply, AdjoinRoot.aeval_eq, AdjoinRoot.mk_self, map_zero]

    ¬ ∃ ρ : PoleOperatorAlgebra →ₐ[ℂ] S, ρ (AdjoinRoot.root PolePolynomial) = T := by
  rintro ⟨ρ, hρ⟩
  exact hT (hρ ▸ pole_representation_annihilates_polynomial ρ)

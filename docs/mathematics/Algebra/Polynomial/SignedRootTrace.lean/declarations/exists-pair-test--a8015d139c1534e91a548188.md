theorem exists_pair_test {nodes : Finset ℂ}
    (hclosed : ∀ z ∈ nodes, conj z ∈ nodes) (weight : ℂ → ℝ)
    {z : ℂ} (hz : z ∈ nodes) (him : z.im ≠ 0) :
    ∃ p : Polynomial ℝ, p.degree < (nodes.card : WithBot ℕ) ∧

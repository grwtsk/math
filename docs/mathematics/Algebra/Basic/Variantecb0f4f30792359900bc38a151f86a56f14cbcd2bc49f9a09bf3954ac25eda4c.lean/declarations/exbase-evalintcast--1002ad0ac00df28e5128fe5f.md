def ExBase.evalIntCast {a : Q(ℤ)} (rα : Q(CommRing $α)) (va : ExBase sβ a) :
    AtomM (Result (ExBase sα) q($a)) :=
  match va with
  | .atom _ => do
    assumeInstancesCommute
    let (i, ⟨b', _⟩) ← addAtomQ q($a)
    pure ⟨b', .atom i, q(Eq.refl $b')⟩
  | .sum va => do
    let ⟨_, vc, p⟩ ← ExSum.evalIntCast rα va
    pure ⟨_, .sum vc, p⟩

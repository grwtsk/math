theorem isTorsion'_powers_iff (p : R) :
    IsTorsion' M (Submonoid.powers p) ↔ ∀ x : M, ∃ n : ℕ, p ^ n • x = 0 := by
  constructor
  · intro h x

multiplication by some power of `p`), the smallest `n` such that `p ^ n • x = 0`. -/

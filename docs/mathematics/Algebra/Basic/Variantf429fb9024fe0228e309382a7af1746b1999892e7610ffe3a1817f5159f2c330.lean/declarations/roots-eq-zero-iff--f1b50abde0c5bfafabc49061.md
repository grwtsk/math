    p.roots = 0 ↔ p = Polynomial.C (p.coeff 0) := by
  refine ⟨fun h => ?_, fun hp => by rw [hp, roots_C]⟩

  · obtain ⟨z, hz⟩ := IsAlgClosed.exists_root p hd.ne'

    simp at hz

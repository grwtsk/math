theorem conGen_eq (r : M → M → Prop) : conGen r = sInf { s : Con M | ∀ x y, r x y → s x y } :=
  le_antisymm
    (le_sInf (fun s hs x y (hxy : (conGen r).r x y) =>
      show s.r x y by
        apply ConGen.Rel.recOn (motive := fun x y _ => s.r x y) hxy
        · exact fun x y h => hs x y h
        · exact s.refl'
        · exact fun _ => s.symm'

        · exact fun _ _ => s.mul))

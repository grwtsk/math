theorem commutator_alternatingGroup_eq_top (h5 : 5 ≤ Fintype.card α) :
    commutator (alternatingGroup α) = ⊤ := by
  suffices closure {b : alternatingGroup α | (b : Perm α).IsThreeCycle} = ⊤ by

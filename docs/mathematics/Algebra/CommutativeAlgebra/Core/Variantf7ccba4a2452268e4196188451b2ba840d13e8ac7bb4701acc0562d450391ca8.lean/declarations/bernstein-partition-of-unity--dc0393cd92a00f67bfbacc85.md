theorem bernstein_partition_of_unity (n : ℕ) (t : ℝ) :
    (Finset.range (n + 1)).sum (fun k =>
      (Nat.choose n k : ℝ) * t ^ k * (1 - t) ^ (n - k)) = 1 := by
  have h := add_pow t (1 - t) n
  have h1 : t + (1 - t) = (1 : ℝ) := by ring
  rw [h1, one_pow] at h
  -- h : 1 = ∑ m ∈ range (n+1), t^m * (1-t)^(n-m) * ↑(n.choose m)

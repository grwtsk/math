import Mathlib.Algebra.GroupWithZero.Basic
import Mathematics.Polynomial.EquationSystem

/-!
# Explicit radical certificates

A `RadicalCertificate` is the proof-producing direction of the
Nullstellensatz used by the architecture.  It carries an exact identity

`claim ^ exponent = ∑ coefficientᵢ * generatorᵢ`.

Evaluating that identity at any common zero proves that the claim vanishes.
The converse Nullstellensatz is intentionally not asserted without the field
and closure hypotheses required by a particular geometric application.
-/

namespace Mathematics.Polynomial

open scoped BigOperators

universe u v

/-- The polynomial combination generated by a finite equation system. -/
noncomputable def EquationSystem.idealCombination
    {Variable : Type u} {K : Type v} [CommSemiring K]
    (system : EquationSystem Variable K)
    (coefficient :
      Fin system.generators.length → MvPolynomial Variable K) :
    MvPolynomial Variable K :=
  ∑ index, coefficient index * system.generators.get index

/--
An exact witness that a positive power of `claim` lies in the ideal generated
by the equations.
-/
structure RadicalCertificate
    {Variable : Type u} {K : Type v} [CommSemiring K]
    (system : EquationSystem Variable K)
    (claim : MvPolynomial Variable K) where
  exponent : Nat
  exponent_pos : 0 < exponent
  coefficient :
    Fin system.generators.length → MvPolynomial Variable K
  identity :
    claim ^ exponent = system.idealCombination coefficient

/-- Every generated polynomial combination vanishes at a certified solution. -/
theorem EquationSystem.eval_idealCombination_eq_zero
    {Variable : Type u} {K : Type v} [CommSemiring K]
    (system : EquationSystem Variable K)
    (solution : system.SolutionSpace)
    (coefficient :
      Fin system.generators.length → MvPolynomial Variable K) :
    MvPolynomial.eval solution.coordinates
        (system.idealCombination coefficient) = 0 := by
  classical
  simp only [EquationSystem.idealCombination, map_sum, map_mul]
  apply Finset.sum_eq_zero
  intro index _
  have generator_vanishes :
      MvPolynomial.eval solution.coordinates
          (system.generators.get index) = 0 := by
    simpa [EquationSystem.SolutionSpace.coordinates] using
      solution.property index
  rw [generator_vanishes, mul_zero]

/--
The exact radical identity certifies the geometric consequence: `claim`
vanishes on every common zero of the original equations.
-/
theorem RadicalCertificate.vanishesOn
    {Variable : Type u} {K : Type v} [Field K]
    {system : EquationSystem Variable K}
    {claim : MvPolynomial Variable K}
    (certificate : RadicalCertificate system claim) :
    system.VanishesOn claim := by
  intro solution
  apply eq_zero_of_pow_eq_zero
  rw [← MvPolynomial.eval_pow, certificate.identity]
  exact system.eval_idealCombination_eq_zero
    solution certificate.coefficient

/-- A named geometric-containment form of the same certified implication. -/
theorem RadicalCertificate.geometricConsequence
    {Variable : Type u} {K : Type v} [Field K]
    {system : EquationSystem Variable K}
    {claim : MvPolynomial Variable K}
    (certificate : RadicalCertificate system claim)
    (solution : system.SolutionSpace) :
    MvPolynomial.eval solution.coordinates claim = 0 :=
  certificate.vanishesOn solution

end Mathematics.Polynomial

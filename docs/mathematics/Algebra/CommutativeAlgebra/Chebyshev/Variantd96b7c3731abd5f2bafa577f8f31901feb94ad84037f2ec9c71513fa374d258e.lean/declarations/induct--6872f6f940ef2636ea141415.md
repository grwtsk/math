protected theorem induct (motive : ℤ → Prop)
    (zero : motive 0)
    (one : motive 1)
    (add_two : ∀ (n : ℕ), motive (↑n + 1) → motive ↑n → motive (↑n + 2))

    simpa only [Int.negSucc_eq, neg_add] using! neg_add_one n hn hnm

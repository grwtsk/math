/-!
# Finite polynomial equation systems

This module defines the affine solution space of a finite list of multivariate
polynomial equations.  It is deliberately independent of any scheduling or
heuristic layer: membership is the exact proposition that every listed
polynomial evaluates to zero.
-/

namespace Mathematics.Polynomial

universe u v w

/-- A finite, ordered system of polynomial equations over `K`. -/
structure EquationSystem
    (Variable : Type u) (K : Type v) [CommSemiring K] where
  generators : List (MvPolynomial Variable K)

/-- A point solves the system exactly when every indexed generator vanishes. -/
def EquationSystem.IsSolution
    {Variable : Type u} {K : Type v} [CommSemiring K]
    (system : EquationSystem Variable K)
    (point : Variable → K) : Prop :=
  ∀ index : Fin system.generators.length,
    MvPolynomial.eval point (system.generators.get index) = 0

/-- The affine solution space, retaining both coordinates and their proof. -/
abbrev EquationSystem.SolutionSpace
    {Variable : Type u} {K : Type v} [CommSemiring K]
    (system : EquationSystem Variable K) :=
  {point : Variable → K // system.IsSolution point}

/-- The underlying coordinate assignment of a certified solution. -/
def EquationSystem.SolutionSpace.coordinates
    {Variable : Type u} {K : Type v} [CommSemiring K]
    {system : EquationSystem Variable K}
    (solution : system.SolutionSpace) : Variable → K :=
  solution.1

/-- A polynomial vanishes on the complete certified solution space. -/
def EquationSystem.VanishesOn
    {Variable : Type u} {K : Type v} [CommSemiring K]
    (system : EquationSystem Variable K)
    (polynomial : MvPolynomial Variable K) : Prop :=
  ∀ solution : system.SolutionSpace,
    MvPolynomial.eval solution.coordinates polynomial = 0

/-- Pointwise equality of coordinates is equality in the solution space. -/
theorem EquationSystem.SolutionSpace.ext
    {Variable : Type u} {K : Type v} [CommSemiring K]
    {system : EquationSystem Variable K}
    {left right : system.SolutionSpace}
    (coordinates :
      ∀ coordinate,
        left.coordinates coordinate = right.coordinates coordinate) :
    left = right := by
  apply Subtype.ext
  funext coordinate
  exact coordinates coordinate

/--
A scale-indexed geometric family generated by polynomial systems, with an
explicit transport that preserves every equation.
-/
structure EquationFamily
    (Scale : Type u) (Variable : Type v) (K : Type w) [CommSemiring K] where
  system : Scale → EquationSystem Variable K
  transport :
    Scale → Scale → (Variable → K) → (Variable → K)
  transport_preserves :
    ∀ {source target} {point},
      (system source).IsSolution point →
        (system target).IsSolution
          (transport source target point)

/-- Transport a certified point through the declared polynomial family. -/
def EquationFamily.transportSolution
    {Scale : Type u} {Variable : Type v} {K : Type w} [CommSemiring K]
    (family : EquationFamily Scale Variable K)
    {source target : Scale}
    (solution : (family.system source).SolutionSpace) :
    (family.system target).SolutionSpace :=
  ⟨family.transport source target solution.coordinates,
    family.transport_preserves solution.property⟩

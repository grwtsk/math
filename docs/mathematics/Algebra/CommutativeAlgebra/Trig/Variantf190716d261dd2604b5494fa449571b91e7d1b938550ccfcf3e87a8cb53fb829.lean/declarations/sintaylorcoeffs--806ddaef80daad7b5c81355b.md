noncomputable def sinTaylorCoeffs (n : ℕ) : ℕ → ℚ := fun i =>
  if i ≤ n then
    if i % 2 = 1 then  -- odd terms only
      ((-1 : ℚ) ^ ((i - 1) / 2)) / (Nat.factorial i : ℚ)
    else 0
  else 0

/-- Taylor polynomial for sin of degree n -/

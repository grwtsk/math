noncomputable def cosTaylorCoeffs (n : ℕ) : ℕ → ℚ := fun i =>
  if i ≤ n then
    if i % 2 = 0 then  -- even terms only
      ((-1 : ℚ) ^ (i / 2)) / (Nat.factorial i : ℚ)
    else 0
  else 0

/-- Taylor polynomial for cos of degree n -/

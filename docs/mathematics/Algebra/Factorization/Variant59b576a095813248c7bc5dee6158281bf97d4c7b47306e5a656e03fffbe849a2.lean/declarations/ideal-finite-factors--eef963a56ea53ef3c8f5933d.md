theorem Ideal.finite_factors {I : Ideal R} (hI : I ≠ 0) :
    {v : HeightOneSpectrum R | v.asIdeal ∣ I}.Finite := by

  exact Subtype.coe_injective (HeightOneSpectrum.ext hvw)

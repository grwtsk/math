/-!
# Frozen-metric linear relaxation

The metric path is supplied first by `BoundedRicciReferenceFlow`.  Signal
updates are then linear maps in deviation coordinates for each frozen metric
sample.  Adding the fixed reference back is affine and is intentionally not
called linear.  Thresholds, clamping, learned transforms, and floating-point
execution are outside this interface.

The module also records the exact algebra generated by a linear involution:
`(I + C) / 2` is a projection and `(1-α)I + αC` is a linear relaxation.
These algebraic facts do not themselves supply a norm or self-adjointness
certificate.
-/

namespace DAF.Geometry

open DAF.Terrain

universe uV uSignal

/--
Linear signal evolution over a metric trajectory selected independently of
the signal.  Contractivity and cycle-energy descent are explicit fields,
rather than consequences of the word “diffusion”.
-/
structure FrozenMetricLinearRelaxation
    {V : Type uV}
    [Fintype V]
    [DecidableEq V]
    {T : Terrain V}
    (flow : BoundedRicciReferenceFlow T)
    (Signal : Type uSignal)
    [AddCommGroup Signal]
    [Module ℝ Signal] where
  signalNorm : Signal → ℝ
  signalNorm_nonnegative : ∀ y, 0 ≤ signalNorm y
  torsionEnergy : Signal → ℝ
  torsionEnergy_nonnegative : ∀ y, 0 ≤ torsionEnergy y
  step : ∀ _k : Fin flow.horizon, Signal →ₗ[ℝ] Signal
  contractive : ∀ k y, signalNorm (step k y) ≤ signalNorm y
  torsionNonincreasing :
    ∀ k y, torsionEnergy (step k y) ≤ torsionEnergy y

namespace FrozenMetricLinearRelaxation

variable {V : Type uV} [Fintype V] [DecidableEq V]
variable {T : Terrain V} {flow : BoundedRicciReferenceFlow T}
variable {Signal : Type uSignal} [AddCommGroup Signal] [Module ℝ Signal]

/-- Compose a chronological list so later steps act after earlier steps. -/
def composeChronological
    (steps : List (Signal →ₗ[ℝ] Signal)) : Signal →ₗ[ℝ] Signal :=
  steps.foldl (fun accumulated next => next.comp accumulated) LinearMap.id

/-- The first `k` frozen-metric steps, as one linear deviation map. -/
def prefixMap
    (R : FrozenMetricLinearRelaxation flow Signal)
    (k : Nat) : Signal →ₗ[ℝ] Signal :=
  composeChronological (List.ofFn R.step |>.take k)

/-- The complete map `A_(H-1) ∘ ⋯ ∘ A_0` is structurally linear. -/
def horizonMap
    (R : FrozenMetricLinearRelaxation flow Signal) : Signal →ₗ[ℝ] Signal :=
  composeChronological (List.ofFn R.step)

/-- The finite deviation trajectory obtained from the prefix maps. -/
def trajectory
    (R : FrozenMetricLinearRelaxation flow Signal)
    (initial : Signal) : Fin (flow.horizon + 1) → Signal :=
  fun k => R.prefixMap k.1 initial

@[simp]
theorem prefixMap_zero
    (R : FrozenMetricLinearRelaxation flow Signal)
    (k : Nat) : R.prefixMap k 0 = 0 :=
  (R.prefixMap k).map_zero

theorem horizonMap_add
    (R : FrozenMetricLinearRelaxation flow Signal)
    (x y : Signal) :
    R.horizonMap (x + y) = R.horizonMap x + R.horizonMap y :=
  R.horizonMap.map_add x y

theorem horizonMap_smul
    (R : FrozenMetricLinearRelaxation flow Signal)
    (a : ℝ)
    (x : Signal) :
    R.horizonMap (a • x) = a • R.horizonMap x :=
  R.horizonMap.map_smul a x

/-- Reconstructing an absolute state from a reference and deviation is affine. -/
def reconstructAbsolute (reference deviation : Signal) : Signal :=
  reference + deviation

end FrozenMetricLinearRelaxation

/--
A continuous interpolation is a separate proof-bearing capability.  Merely
having finite samples does not manufacture a continuous path; an instance must
supply both sample times and agreement at every sample.
-/
structure ContinuousRelaxationRaster
    {Signal : Type uSignal}
    [TopologicalSpace Signal]
    (H : Nat)
    (trajectory : Fin (H + 1) → Signal) where
  sampleTime : Fin (H + 1) → Set.Icc (0 : ℝ) (H : ℝ)
  path : Set.Icc (0 : ℝ) (H : ℝ) → Signal
  continuous_path : Continuous path
  agrees_at_samples : ∀ k, path (sampleTime k) = trajectory k

section InvolutionRelaxation

variable {Signal : Type uSignal} [AddCommGroup Signal] [Module ℝ Signal]

/--
A relaxation coefficient in the closed unit interval.  Keeping the bounds in
the type prevents a caller from silently applying the contractivity theorem
to an extrapolation parameter.
-/
structure RelaxationParameter where
  value : ℝ
  nonnegative : 0 ≤ value
  le_one : value ≤ 1

namespace RelaxationParameter

/-- The complementary coefficient is nonnegative. -/
theorem one_sub_nonnegative (parameter : RelaxationParameter) :
    0 ≤ 1 - parameter.value :=
  sub_nonneg.mpr parameter.le_one

end RelaxationParameter

/-- The averaging projection associated with a linear involution. -/
noncomputable def involutionProjection
    (C : Signal →ₗ[ℝ] Signal) : Signal →ₗ[ℝ] Signal :=
  ((1 : ℝ) / 2) • (LinearMap.id + C)

/-- A certified involution makes `(I + C) / 2` idempotent. -/
theorem involutionProjection_idempotent
    (C : Signal →ₗ[ℝ] Signal)
    (hC : C.comp C = LinearMap.id) :
    (involutionProjection C).comp (involutionProjection C) =
      involutionProjection C := by
  ext x
  have hCx : C (C x) = x := by
    have := LinearMap.congr_fun hC x
    simpa using this
  simp only [involutionProjection, LinearMap.comp_apply, LinearMap.smul_apply,
    LinearMap.add_apply, LinearMap.id_apply, map_smul, map_add]
  rw [hCx]
  module

/-- The explicit linear relaxation generated by an involution. -/
def involutionRelaxation
    (alpha : ℝ)
  (C : Signal →ₗ[ℝ] Signal) : Signal →ₗ[ℝ] Signal :=
  (1 - alpha) • LinearMap.id + alpha • C

/-- A proof-bearing admissible parameter selects the same linear operator. -/
def RelaxationParameter.operator
    (parameter : RelaxationParameter)
    (C : Signal →ₗ[ℝ] Signal) : Signal →ₗ[ℝ] Signal :=
  involutionRelaxation parameter.value C

/-- Invariant vectors are fixed by every relaxation parameter. -/
theorem involutionRelaxation_apply_of_invariant
    (alpha : ℝ)
    (C : Signal →ₗ[ℝ] Signal)
    {x : Signal}
    (hx : C x = x) :
    involutionRelaxation alpha C x = x := by
  simp only [involutionRelaxation, LinearMap.add_apply,
    LinearMap.smul_apply, LinearMap.id_apply, hx]
  module

/-- Anti-invariant vectors are scaled by `1 - 2α`. -/
theorem involutionRelaxation_apply_of_antiInvariant
    (alpha : ℝ)
    (C : Signal →ₗ[ℝ] Signal)
    {x : Signal}
    (hx : C x = -x) :
    involutionRelaxation alpha C x = (1 - 2 * alpha) • x := by
  simp only [involutionRelaxation, LinearMap.add_apply,
    LinearMap.smul_apply, LinearMap.id_apply, hx]
  module

end InvolutionRelaxation

section InvolutionRelaxationNormBounds

variable {Signal : Type uSignal}
variable [NormedAddCommGroup Signal] [NormedSpace ℝ Signal]

/--
Convex averaging with a norm isometry is nonexpansive.  Isometry is an
explicit hypothesis; involutivity or self-adjointness alone is not used to
manufacture a norm bound.
-/
theorem norm_involutionRelaxation_le
    (parameter : RelaxationParameter)
    (C : Signal →ₗ[ℝ] Signal)
    (isometry : ∀ x, ‖C x‖ = ‖x‖)
    (x : Signal) :
    ‖involutionRelaxation parameter.value C x‖ ≤ ‖x‖ := by
  have hcomplement : 0 ≤ 1 - parameter.value :=
    parameter.one_sub_nonnegative
  calc
    ‖involutionRelaxation parameter.value C x‖ =
        ‖(1 - parameter.value) • x + parameter.value • C x‖ := by
      rfl
    _ ≤ ‖(1 - parameter.value) • x‖ +
        ‖parameter.value • C x‖ := norm_add_le _ _
    _ = |1 - parameter.value| * ‖x‖ +
        |parameter.value| * ‖C x‖ := by
      simp only [norm_smul, Real.norm_eq_abs]
    _ = (1 - parameter.value) * ‖x‖ +
        parameter.value * ‖x‖ := by
      rw [isometry, abs_of_nonneg hcomplement,
        abs_of_nonneg parameter.nonnegative]
    _ = ‖x‖ := by ring

/--
For a strictly interior coefficient, every nonzero anti-invariant component
is contracted strictly.  The conclusion is deliberately restricted to the
anti-invariant component; it is not a global strict-contraction claim.
-/
theorem norm_involutionRelaxation_lt_of_antiInvariant
    (parameter : RelaxationParameter)
    (parameter_pos : 0 < parameter.value)
    (parameter_lt_one : parameter.value < 1)
    (C : Signal →ₗ[ℝ] Signal)
    {x : Signal}
    (antiInvariant : C x = -x)
    (x_ne_zero : x ≠ 0) :
    ‖involutionRelaxation parameter.value C x‖ < ‖x‖ := by
  have hscalar : |1 - 2 * parameter.value| < 1 := by
    rw [abs_lt]
    constructor <;> linarith
  have hnorm : 0 < ‖x‖ := norm_pos_iff.mpr x_ne_zero
  calc
    ‖involutionRelaxation parameter.value C x‖ =
        ‖(1 - 2 * parameter.value) • x‖ := by
      rw [involutionRelaxation_apply_of_antiInvariant
        parameter.value C antiInvariant]
    _ = |1 - 2 * parameter.value| * ‖x‖ := by
      rw [norm_smul, Real.norm_eq_abs]
    _ < 1 * ‖x‖ := mul_lt_mul_of_pos_right hscalar hnorm
    _ = ‖x‖ := one_mul _

end InvolutionRelaxationNormBounds

end DAF.Geometry

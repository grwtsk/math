theorem prod_ext_iff {f g : L₁ × L₂ →ₗ⁅R⁆ L} :
    f = g ↔ f.comp (inl _ _ _) = g.comp (inl _ _ _) ∧ f.comp (inr _ _ _) = g.comp (inr _ _ _) := by

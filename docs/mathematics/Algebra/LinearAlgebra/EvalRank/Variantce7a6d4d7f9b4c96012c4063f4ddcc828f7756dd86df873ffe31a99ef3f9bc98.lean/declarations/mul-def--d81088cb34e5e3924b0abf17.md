private theorem mul_def {d : ℤ} (x y x' y' : ℤ) :
    (⟨x, y⟩ * ⟨x', y'⟩ : Zsqrtd d) = ⟨x * x' + d * y * y', x * y' + y * x'⟩ := rfl

/- The first two columns are the powers `Φⁿ` and `Ψⁿ` of `Φ = 1 + √5 = 2φ` and

import DAF.Fold.ChainMap
import DAF.Hilbert.Heat

/-!
# Fold-compatible harmonic transport
-/

noncomputable section

namespace DAF.Fold

universe u

open DAF.Hilbert

variable {Index : Type u} [Fintype Index] [DecidableEq Index]
variable {terrain : GradedTerrainComplex Index}

/-- Coordinate-level harmonicity for the finite terrain Hodge--Dirac. -/
def IsHarmonic (vector : Index → ℂ) : Prop :=
  terrain.complex.hodgeDirac.mulVec vector = 0

namespace FoldUnitary

/-- A unitary chain-map fold preserves the harmonic kernel. -/
theorem fold_preserves_harmonicSubspace
    (fold : FoldUnitary terrain)
    {vector : Index → ℂ}
    (harmonic : IsHarmonic (terrain := terrain) vector) :
    IsHarmonic (terrain := terrain) (fold.operator.mulVec vector) := by
  unfold IsHarmonic at harmonic ⊢
  rw [Matrix.mulVec_mulVec,
    ← fold.fold_intertwines_hodgeDirac,
    ← Matrix.mulVec_mulVec, harmonic, Matrix.mulVec_zero]

/-- Fold transport commutes with the exact finite heat diffuser generated by
the terrain Laplacian. -/
theorem fold_intertwines_diffuser
    (fold : FoldUnitary terrain)
    (s : ℝ) :
    fold.operator * heatDiffuser terrain.complex.hodgeDirac s =
      heatDiffuser terrain.complex.hodgeDirac s * fold.operator := by
  have laplacianCommute :
      Commute fold.operator
        (terrain.complex.hodgeDirac * terrain.complex.hodgeDirac) := by
    rw [terrain.complex.hodgeDirac_square]
    exact fold.fold_intertwines_laplacian
  have scaledCommute :
      Commute fold.operator
        ((-(s : ℂ)) •
          (terrain.complex.hodgeDirac * terrain.complex.hodgeDirac)) :=
    laplacianCommute.smul_right (-(s : ℂ))
  exact scaledCommute.exp_right.eq

end FoldUnitary

end DAF.Fold

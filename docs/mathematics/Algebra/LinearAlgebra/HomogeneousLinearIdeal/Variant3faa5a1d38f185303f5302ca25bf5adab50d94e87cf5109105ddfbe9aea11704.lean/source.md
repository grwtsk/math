/-!
# Ideals of homogeneous linear equations

The ideal generated by a linear subspace in a symmetric algebra is exactly the
kernel of the algebra map to the symmetric algebra of the quotient vector space.
Over a field that target is a domain, so the linear equation ideal is prime and
radical, including singular and underdetermined systems. The finite polynomial
specialization uses the existing standard basis and symmetric-algebra APIs.

AI attribution: Codex authored this module and its proofs.
-/

variable {R : Type u} [CommRing R]
variable {V : Type v} [AddCommGroup V] [Module R V]
variable {A : Type w} [CommRing A] [Algebra R A]

/-- Linear relations in a symmetric algebra generate exactly the kernel of the
map induced by the quotient of the generating module. This is ideal equality. -/
theorem symmetric_linearIdeal_eq_ker (f : V →ₗ[R] A) (hf : IsSymmetricAlgebra f)
    (S : Submodule R V) :
    Ideal.span (f '' (S : Set V)) =
      RingHom.ker
        (hf.lift ((SymmetricAlgebra.ι R (V ⧸ S)).comp S.mkQ)).toRingHom := by
  let I : Ideal A := Ideal.span (f '' (S : Set V))
  let forward : A →ₐ[R] SymmetricAlgebra R (V ⧸ S) :=
    hf.lift ((SymmetricAlgebra.ι R (V ⧸ S)).comp S.mkQ)
  let g : V →ₗ[R] (A ⧸ I) := (Ideal.Quotient.mkₐ R I).toLinearMap.comp f
  have hg : S ≤ LinearMap.ker g := by
    intro v hv
    change Ideal.Quotient.mk I (f v) = 0
    exact Ideal.Quotient.eq_zero_iff_mem.mpr (Ideal.subset_span ⟨v, hv, rfl⟩)
  let inverse : SymmetricAlgebra R (V ⧸ S) →ₐ[R] (A ⧸ I) :=
    SymmetricAlgebra.lift (S.liftQ g hg)
  have hcomp : inverse.comp forward = Ideal.Quotient.mkₐ R I := by
    apply hf.algHom_ext
    ext v
    change inverse (forward (f v)) = Ideal.Quotient.mk I (f v)
    rw [show forward (f v) = SymmetricAlgebra.ι R (V ⧸ S) (S.mkQ v) by
      exact hf.lift_eq _ v]
    rw [SymmetricAlgebra.lift_ι_apply]
    rfl
  change I = RingHom.ker forward.toRingHom
  apply le_antisymm
  · apply Ideal.span_le.mpr
    rintro _ ⟨v, hv, rfl⟩
    change forward (f v) = 0
    rw [show forward (f v) = SymmetricAlgebra.ι R (V ⧸ S) (S.mkQ v) by
      exact hf.lift_eq _ v]
    have hs : S.mkQ v = 0 := (Submodule.Quotient.mk_eq_zero S).mpr hv
    rw [hs, map_zero]
  · intro a ha
    have hfa : forward a = 0 := RingHom.mem_ker.mp ha
    apply Ideal.Quotient.eq_zero_iff_mem.mp
    change (Ideal.Quotient.mkₐ R I) a = 0
    rw [← hcomp]
    change inverse (forward a) = 0
    rw [hfa, map_zero]

/-- Passing from a family of linear generators to its linear span does not
change the ideal generated by its image in an algebra. -/
theorem linearIdeal_span_eq {α : Type*} (f : V →ₗ[R] A) (v : α → V) :
    Ideal.span (f '' (Submodule.span R (Set.range v) : Set V)) =
      Ideal.span (Set.range fun i ↦ f (v i)) := by
  apply le_antisymm
  · apply Ideal.span_le.mpr
    rintro _ ⟨x, hx, rfl⟩
    change f x ∈ Ideal.span (Set.range fun i ↦ f (v i))
    induction hx using Submodule.span_induction with
    | mem x hx =>
        rcases hx with ⟨i, rfl⟩
        exact Ideal.subset_span ⟨i, rfl⟩
    | zero => simpa only [map_zero] using (Ideal.span (Set.range fun i ↦ f (v i))).zero_mem
    | add x y _ _ hx hy =>
        simpa only [map_add] using (Ideal.span (Set.range fun i ↦ f (v i))).add_mem hx hy
    | smul r x _ hx =>
        simpa only [map_smul, Algebra.smul_def] using
          (Ideal.span (Set.range fun i ↦ f (v i))).mul_mem_left (algebraMap R A r) hx
  · apply Ideal.span_le.mpr
    rintro _ ⟨i, rfl⟩
    exact Ideal.subset_span ⟨v i, Submodule.subset_span ⟨i, rfl⟩, rfl⟩

variable {K : Type*} [Field K]
variable {M : Type*} [AddCommGroup M] [Module K M]
variable {B : Type*} [CommRing B] [Algebra K B]

/-- A linear-relation ideal in a symmetric algebra over a field is prime,
because the symmetric algebra of the quotient vector space is a domain. -/
theorem symmetric_linearIdeal_isPrime (f : M →ₗ[K] B) (hf : IsSymmetricAlgebra f)
    (S : Submodule K M) : (Ideal.span (f '' (S : Set M))).IsPrime := by
  rw [symmetric_linearIdeal_eq_ker f hf S]
  exact RingHom.ker_isPrime _

/-- Arbitrary homogeneous linear generators of a symmetric algebra over a field
generate a prime ideal. No rank or invertibility hypothesis is needed. -/
theorem symmetric_linearGeneratorIdeal_isPrime {α : Type*} (f : M →ₗ[K] B)
    (hf : IsSymmetricAlgebra f) (v : α → M) :
    (Ideal.span (Set.range fun i ↦ f (v i))).IsPrime := by
  rw [← linearIdeal_span_eq f v]
  exact symmetric_linearIdeal_isPrime f hf _

/-- Arbitrarily many homogeneous linear equations in finitely many independent
variables over a field generate a prime ideal, regardless of the coefficient
matrix rank. -/
theorem homogeneous_linear_equationIdeal_isPrime {α σ : Type*} [Fintype σ]
    (L : Matrix α σ K) :
    (Ideal.span (Set.range fun i ↦
      ∑ j, MvPolynomial.C (L i j) * (MvPolynomial.X j : MvPolynomial σ K))).IsPrime := by
  classical
  let f : (σ → K) →ₗ[K] MvPolynomial σ K :=
    (Pi.basisFun K σ).constr K MvPolynomial.X
  have hf : IsSymmetricAlgebra f :=
    SymmetricAlgebra.IsSymmetricAlgebra.mvPolynomial σ (Pi.basisFun K σ)
  have h := symmetric_linearGeneratorIdeal_isPrime f hf L
  simpa only [f, Module.Basis.constr_apply_fintype, Pi.basisFun_equivFun,
    LinearEquiv.refl_apply, MvPolynomial.smul_eq_C_mul] using h

/-- The homogeneous linear equation ideal is radical over a field. -/
theorem homogeneous_linear_equationIdeal_isRadical {α σ : Type*} [Fintype σ]
    (L : Matrix α σ K) :
    (Ideal.span (Set.range fun i ↦
      ∑ j, MvPolynomial.C (L i j) * (MvPolynomial.X j : MvPolynomial σ K))).IsRadical :=
  (homogeneous_linear_equationIdeal_isPrime L).isRadical

/-- Matrix-coordinate form of the prime ideal theorem, with all scalar
coefficients lifted by `MvPolynomial.C`. -/
theorem homogeneous_mulVec_equationIdeal_isPrime {α σ : Type*} [Fintype σ]
    (L : Matrix α σ K) :
    (Ideal.span (Set.range ((L.map (MvPolynomial.C : K →+* MvPolynomial σ K)).mulVec
      MvPolynomial.X))).IsPrime := by
  change (Ideal.span (Set.range fun i ↦
    ∑ j, MvPolynomial.C (L i j) * (MvPolynomial.X j : MvPolynomial σ K))).IsPrime
  exact homogeneous_linear_equationIdeal_isPrime L

/-- Matrix-coordinate form of the radical ideal theorem. -/
theorem homogeneous_mulVec_equationIdeal_isRadical {α σ : Type*} [Fintype σ]
    (L : Matrix α σ K) :
    (Ideal.span (Set.range ((L.map (MvPolynomial.C : K →+* MvPolynomial σ K)).mulVec
      MvPolynomial.X))).IsRadical :=
  (homogeneous_mulVec_equationIdeal_isPrime L).isRadical

/-- At the finite algebraically closed field scope of Hilbert's Nullstellensatz,
the homogeneous linear equation ideal equals the ideal of its closed-point zero
locus. No algebraicity or existence target outside this finite scope is implied. -/
theorem homogeneous_mulVec_equationIdeal_eq_vanishingIdeal_zeroLocus
    [IsAlgClosed K] {α σ : Type*} [Fintype σ] (L : Matrix α σ K) :
    Ideal.span (Set.range ((L.map (MvPolynomial.C : K →+* MvPolynomial σ K)).mulVec
      MvPolynomial.X)) =
      MvPolynomial.vanishingIdeal K (MvPolynomial.zeroLocus K
        (Ideal.span (Set.range ((L.map (MvPolynomial.C : K →+* MvPolynomial σ K)).mulVec
          MvPolynomial.X)))) := by
  exact ((MvPolynomial.vanishingIdeal_zeroLocus_eq_radical _).trans
    (homogeneous_mulVec_equationIdeal_isRadical L).radical).symm

/-- A homogeneous linear system contains the origin in its zero locus. This
supplies a point only for the declared homogeneous linear equations. -/
theorem homogeneous_mulVec_origin_mem_zeroLocus {α σ : Type*} [Fintype σ]
    (L : Matrix α σ K) :
    (0 : σ → K) ∈ MvPolynomial.zeroLocus K
      (Ideal.span (Set.range ((L.map (MvPolynomial.C : K →+* MvPolynomial σ K)).mulVec
        MvPolynomial.X))) := by
  rw [MvPolynomial.zeroLocus_span]
  rintro _ ⟨i, rfl⟩
  simp [Matrix.mulVec, dotProduct]

end FormalResearch.PositiveAtlas

/-!
# Structured and diffuse coefficient projections

A dyadic Type II cutoff matrix can contain a large all-ones or otherwise
low-rank mode.  Consequently, a diffuse packet estimate must be applied only
after explicitly removing finite families of structured cluster modes.

This file records that separation as exact finite-dimensional data.  The
projections are idempotent and self-adjoint, their ranges are generated by
named cluster vectors, and every bilinear matrix pairing splits into four
structured/diffuse terms.  No estimate for any of those terms is stored in
the structure.
-/

/-- Finite structured-mode data on the two coefficient spaces.

The carrier finsets make the provenance of each generator inspectable.  The
range equalities, rather than a bare `FiniteDimensional` field, ensure that
the advertised rank bound is tied to those concrete generators.
-/
structure TypeIILocalStructure
    (Left Right : Type*) [Fintype Left] [Fintype Right]
    (rankBudget : ℕ) where
  leftProjection :
    EuclideanSpace ℂ Left →L[ℂ] EuclideanSpace ℂ Left
  rightProjection :
    EuclideanSpace ℂ Right →L[ℂ] EuclideanSpace ℂ Right

  leftProjection_idempotent :
    leftProjection.comp leftProjection = leftProjection
  rightProjection_idempotent :
    rightProjection.comp rightProjection = rightProjection
  leftProjection_selfAdjoint : leftProjection.adjoint = leftProjection
  rightProjection_selfAdjoint : rightProjection.adjoint = rightProjection

  LeftCluster : Type
  RightCluster : Type
  leftClusterFintype : Fintype LeftCluster
  rightClusterFintype : Fintype RightCluster

  leftClusterCarrier : LeftCluster → Finset Left
  rightClusterCarrier : RightCluster → Finset Right
  leftGenerator : LeftCluster → EuclideanSpace ℂ Left
  rightGenerator : RightCluster → EuclideanSpace ℂ Right
  leftGenerator_supported : ∀ cluster index,
    index ∉ leftClusterCarrier cluster → leftGenerator cluster index = 0
  rightGenerator_supported : ∀ cluster index,
    index ∉ rightClusterCarrier cluster → rightGenerator cluster index = 0

  leftRange_generated :
    LinearMap.range leftProjection.toLinearMap =
      Submodule.span ℂ (Set.range leftGenerator)
  rightRange_generated :
    LinearMap.range rightProjection.toLinearMap =
      Submodule.span ℂ (Set.range rightGenerator)
  rankBound :
    Fintype.card LeftCluster + Fintype.card RightCluster ≤ rankBudget

attribute [instance] TypeIILocalStructure.leftClusterFintype
attribute [instance] TypeIILocalStructure.rightClusterFintype

/-- Exact four-term expansion of a bilinear Type II pairing.

This is bookkeeping only.  In the intended analysis the diffuse/diffuse,
mixed, and structured/structured summands require different theorems.
-/
@[surface_novel, surface_published]
theorem TypeIILocalStructure.bilinearPairing_four_terms
    {Left Right : Type*} [Fintype Left] [Fintype Right] {rankBudget : ℕ}
    (localData : TypeIILocalStructure Left Right rankBudget)
    (left : EuclideanSpace ℂ Left) (matrix : Matrix Left Right ℂ)
    (right : EuclideanSpace ℂ Right) :
    bilinearMatrixPairing left matrix right =
      bilinearMatrixPairing (localData.leftDiffuse left) matrix
          (localData.rightDiffuse right) +
        bilinearMatrixPairing (localData.leftStructured left) matrix
          (localData.rightDiffuse right) +
        bilinearMatrixPairing (localData.leftDiffuse left) matrix
          (localData.rightStructured right) +
        bilinearMatrixPairing (localData.leftStructured left) matrix
          (localData.rightStructured right) := by
  classical
  have hleft := localData.leftStructured_add_leftDiffuse left
  have hright := localData.rightStructured_add_rightDiffuse right
  have hleftPoint (index : Left) :
      left index = localData.leftStructured left index +
        localData.leftDiffuse left index := by
    simpa using congrArg (fun vector : EuclideanSpace ℂ Left ↦ vector index)
      hleft.symm
  have hrightPoint (index : Right) :
      right index = localData.rightStructured right index +
        localData.rightDiffuse right index := by
    simpa using congrArg (fun vector : EuclideanSpace ℂ Right ↦ vector index)
      hright.symm
  simp only [bilinearMatrixPairing]
  simp_rw [hleftPoint, hrightPoint]
  simp only [add_mul, mul_add, Finset.sum_add_distrib]
  ring

end

/-!
# Block Schur provenance reduction

This module formalizes the algebraic block version of Phase-II provenance
reduction.

A block Schur solver is a linear map from autonomous coordinates to provenance
coordinates satisfying the normal equations for the ambient positive form. From
that datum the atlas constructs the canonical provenance projector on the product
space, proves orthogonality, and obtains the positive Schur residual.

This file proves the reduction once a normal-equation solver is supplied. A
later module must construct such a solver from an invertible finite-dimensional
provenance block, and a later extension may replace the inverse by a
Moore--Penrose pseudoinverse.
-/

universe u v

variable {P : Type u} {U : Type v}
variable [AddCommGroup P] [Module ℝ P]
variable [AddCommGroup U] [Module ℝ U]

/-- Include a provenance vector into the first block of the ambient product. -/
def provenanceInclusion : P →ₗ[ℝ] P × U where
  toFun p := (p, 0)
  map_add' x y := by
    ext <;> simp
  map_smul' r x := by
    ext <;> simp

/-- Include an autonomous vector into the second block of the ambient product. -/
def autonomousInclusion : U →ₗ[ℝ] P × U where
  toFun u := (0, u)
  map_add' x y := by
    ext <;> simp
  map_smul' r x := by
    ext <;> simp

@[simp]
theorem provenanceInclusion_apply (p : P) :
    provenanceInclusion (U := U) p = (p, 0) :=
  rfl

@[simp]
theorem autonomousInclusion_apply (u : U) :
    autonomousInclusion (P := P) u = (0, u) :=
  rfl

/-- A linear solver for the provenance normal equations of a positive block form.

For every autonomous coordinate u, solve u is the provenance correction whose
residual autonomous vector is orthogonal to every provenance direction. -/
structure BlockSchurSolver (G : PositiveForm (P × U)) where
  /-- Linear provenance correction satisfying the normal equation below. -/
  solve : U →ₗ[ℝ] P
  normal : ∀ p u,
    G.form (provenanceInclusion p)
      (autonomousInclusion u - provenanceInclusion (solve u)) = 0

namespace BlockSchurSolver

variable {G : PositiveForm (P × U)}

/-- The canonical block provenance projection generated by a normal-equation
solver. -/
def project (S : BlockSchurSolver G) : (P × U) →ₗ[ℝ] (P × U) where
  toFun x := (x.1 + S.solve x.2, 0)
  map_add' x y := by
    ext <;> simp [add_assoc, add_left_comm, add_comm]
  map_smul' r x := by
    ext <;> simp

@[simp]
theorem project_apply (S : BlockSchurSolver G) (x : P × U) :
    S.project x = (x.1 + S.solve x.2, 0) :=
  rfl

/-- The block provenance projection is idempotent. -/
theorem project_idempotent (S : BlockSchurSolver G) (x : P × U) :
    S.project (S.project x) = S.project x := by
  ext <;> simp [project]

/-- The block projection is orthogonal for the ambient positive form. -/
theorem project_orthogonal (S : BlockSchurSolver G) (x y : P × U) :
    G.form (S.project x) (y - S.project y) = 0 := by
  have hres : y - S.project y =
      autonomousInclusion y.2 - provenanceInclusion (S.solve y.2) := by
    ext <;> simp [project, provenanceInclusion, autonomousInclusion]
  rw [hres]
  exact S.normal (x.1 + S.solve x.2) y.2

/-- The normal-equation solver determines the Phase-II provenance projector. -/
def projector (S : BlockSchurSolver G) : ProvenanceProjector G where
  project := S.project
  idempotent := S.project_idempotent
  orthogonal := S.project_orthogonal

theorem project_autonomous (S : BlockSchurSolver G) (u : U) :
    S.project (autonomousInclusion u) = provenanceInclusion (S.solve u) := by
  ext <;> simp [project, provenanceInclusion, autonomousInclusion]

theorem residual_autonomous (S : BlockSchurSolver G) (u : U) :
    S.projector.residualMap (autonomousInclusion u) =
      autonomousInclusion u - provenanceInclusion (S.solve u) := by
  rw [ProvenanceProjector.residualMap_apply]
  change autonomousInclusion u - S.project (autonomousInclusion u) = _
  rw [S.project_autonomous]

/-- The positive Schur residual on autonomous coordinates. -/
def schurForm (S : BlockSchurSolver G) : PositiveForm U :=
  S.projector.residualForm.pullback autonomousInclusion

/-- The block Schur residual is positive semidefinite. -/
theorem schurForm_nonnegative (S : BlockSchurSolver G) (u : U) :
    0 ≤ S.schurForm.form u u :=
  S.schurForm.nonnegative u

/-- The residual form is the autonomous block minus the provenance energy of the
normal-equation correction. -/
theorem schurForm_eq_sub (S : BlockSchurSolver G) (u v : U) :
    S.schurForm.form u v =
      G.form (autonomousInclusion u) (autonomousInclusion v) -
        G.form (provenanceInclusion (S.solve u))
          (provenanceInclusion (S.solve v)) := by
  change
    G.form (S.projector.residualMap (autonomousInclusion u))
      (S.projector.residualMap (autonomousInclusion v)) = _
  rw [S.projector.residual_eq_sub_projected]
  simp [projector, project]

/-- The diagonal Schur subtraction is nonnegative. -/
theorem schur_diagonal_nonnegative (S : BlockSchurSolver G) (u : U) :
    0 ≤
      G.form (autonomousInclusion u) (autonomousInclusion u) -
        G.form (provenanceInclusion (S.solve u))
          (provenanceInclusion (S.solve u)) := by
  rw [← S.schurForm_eq_sub u u]
  exact S.schurForm_nonnegative u

/-- The ambient energy of a pure autonomous input splits into the explained
provenance correction and positive Schur residual. -/
theorem autonomous_energy_decomposition (S : BlockSchurSolver G) (u : U) :
    G.form (autonomousInclusion u) (autonomousInclusion u) =
      G.form (provenanceInclusion (S.solve u))
          (provenanceInclusion (S.solve u)) +
        S.schurForm.form u u := by
  linarith [S.schurForm_eq_sub u u]

end BlockSchurSolver

theorem orthogonal_of_eigen_plus_minus
    (G : PositiveForm V) {T : V →ₗ[ℝ] V}
    (hT : G.SelfAdjointFor T)
    {x y : V} (hx : T x = x) (hy : T y = -y) :
    G.form x y = 0 := by
  have h := hT x y
  rw [hx, hy] at h
  have hneg : G.form x (-y) = -G.form x y := by
    have hs := G.smul_right (-1 : ℝ) x y

  rw [hneg] at h
  linarith

/-!
# CA-03 qualified algebraic boundary presentation

This file computes a deliberately algebraic presentation.  It does **not**
identify the quotient below with singular, simplicial, or cellular homology of
any geometric space.

The degree-one symbols record a positively oriented outer curve and a
positively oriented inner curve.  The two-cell relation is therefore
`outer - inner`, represented by `(n, -n)`.
-/


namespace GRWTSK.Algebra.LinearAlgebra.C2.Variant06f6445a422506a181fe0e67145acd3ce26739faa07a3aae496561462e13e637

namespace Surface.CA03.BoundaryPresentation

noncomputable section

/-- Degree-two chains in the presentation. -/
abbrev C2 := ℤ

/-- Degree-one chains, ordered as `(outer, inner)`. -/
abbrev C1 := ℤ × ℤ

/-- Algebraic boundary of the sole two-cell generator. -/
def boundary2 : C2 →+ C1 where
  toFun n := (n, -n)
  map_zero' := by simp
  map_add' := by
    intro m n
    ext <;> simp [add_comm]

/-- All relations generated by algebraic two-cell boundaries. -/
def relations : AddSubgroup C1 :=
  boundary2.range

/-- The presented group `C₁ / im(∂₂)`.  This name carries no geometric
homology assertion. -/
abbrev H1 := C1 ⨸ relations

/-- Every explicit two-cell boundary is zero in the presentation. -/
theorem boundary2_class_eq_zero (n : C2) :
    (QuotientAddGroup.mk' relations) (boundary2 n) = 0 := by
  change (↑(boundary2 n) : H1) = 0
  exact
    (QuotientAddGroup.eq_zero_iff (N := relations) (boundary2 n)).2
      ⟨n, rfl⟩

/-- The positive outer-boundary symbol. -/
def outerChain : C1 :=
  (1, 0)

/-- The positive inner-boundary symbol.  It is positive counterclockwise,
not the induced clockwise orientation on the boundary of a punctured cell. -/
def innerChain : C1 :=
  (0, 1)

def outerClass : H1 :=
  (QuotientAddGroup.mk' relations) outerChain

def innerClass : H1 :=
  (QuotientAddGroup.mk' relations) innerChain

/-- The relation `(1, -1)` identifies the two positively oriented symbols. -/
theorem outerClass_eq_innerClass :
    outerClass = innerClass := by
  apply QuotientAddGroup.eq_iff_sub_mem.mpr
  refine ⟨1, ?_⟩
  norm_num [boundary2, outerChain, innerChain]

/-- Sum of outer and inner coefficients. -/
def total : C1 →+ ℤ where
  toFun x := x.1 + x.2
  map_zero' := by simp
  map_add' := by
    intro x y
    simp
    abel

/-- The relations are exactly the kernel of coefficient sum. -/
theorem relations_eq_total_ker :
    relations = total.ker := by
  apply le_antisymm
  · intro x hx
    rcases hx with ⟨n, rfl⟩
    simp [total, boundary2]
  · intro x hx
    have hsum : x.1 + x.2 = 0 := hx
    refine ⟨x.1, ?_⟩
    apply Prod.ext
    · rfl
    · dsimp [boundary2]
      omega

theorem total_surjective :
    Function.Surjective total := by
  intro n
  exact ⟨(n, 0), by simp [total]⟩

/-- Concrete additive classification of the presented quotient by coefficient
sum. -/
def H1EquivInt : H1 ≃+ ℤ := by
  change (C1 ⨸ relations) ≃+ ℤ
  rw [relations_eq_total_ker]
  refine (QuotientAddGroup.quotientKerEquivRange total).trans ?_
  rw [AddMonoidHom.range_eq_top.mpr total_surjective]
  exact AddSubgroup.topEquiv

/-- The presented group is a free integer module. -/
theorem h1_isTorsionFree : Module.Free ℤ H1 := by
  exact Module.Free.of_equiv H1EquivInt.toIntLinearEquiv.symm

end

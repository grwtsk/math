/-
Copyright (c) 2026 grwtsk contributors. MIT.
Human mathematical direction: grwtsk contributors. Formalization assistance: OpenAI Codex.

Pure declaration-level conceptual-seam contract. This module defines source,
coverage, dependency, reconstruction, subject-association and signature-preservation
relations only. It performs no parsing, model inference, filesystem mutation,
Lean execution, Git operation or attribution decision.
-/
import GRWTSK.FoundationsLogicAndTypeTheory.MathematicalOrganization.SubjectPlacement
import Mathlib.Data.List.Basic

set_option autoImplicit false

namespace GRWTSK.FoundationsLogicAndTypeTheory.MathematicalOrganization.ConceptualSeamPlan

open GRWTSK.FoundationsLogicAndTypeTheory.MathematicalOrganization.SubjectPlacement

/-- Half-open byte interval `[start, stop)` in one exact source occurrence. -/
structure ByteSpan where
  start : Nat
  stop : Nat
  ordered : start ≤ stop

/-- Exact adjacent coverage from `cursor` through `stop`. Empty regions are
permitted, but gaps and overlaps are not. -/
def CoversFrom : Nat → List ByteSpan → Nat → Prop
  | cursor, [], stop => cursor = stop
  | cursor, span :: rest, stop =>
      span.start = cursor ∧
      span.stop ≤ stop ∧
      CoversFrom span.stop rest stop

/-- Identity of one parsed source command and its complete owned byte region. -/
structure CommandIdentity where
  source : SourceFileIdentity
  ordinal : Nat
  span : ByteSpan
  rawDigest : String
  syntaxKind : String

/-- Whether a declaration is visible outside its source command/module context. -/
inductive DeclarationVisibility where
  | publicDecl
  | privateDecl
  | localDecl
  deriving DecidableEq, Repr

/-- An explicit source declaration versus one generated by elaborating a command. -/
inductive DeclarationOriginKind where
  | explicit
  | generated
  deriving DecidableEq, Repr

/-- Stable declaration surface used for before/after comparison. -/
structure DeclarationFingerprint where
  name : String
  kind : String
  elaboratedTypeDigest : String
  visibility : DeclarationVisibility
  deriving DecidableEq, Repr

/-- Attachment of every explicit or generated declaration to its generating
command. Generated declarations never become independent source spans. -/
structure DeclarationOrigin where
  declaration : DeclarationFingerprint
  commandOrdinal : Nat
  origin : DeclarationOriginKind
  deriving DecidableEq, Repr

/-- Context whose loss may change the meaning or elaboration of later commands. -/
structure ContextState where
  namespaceStack : List String
  sectionStack : List String
  variables : List String
  universes : List String
  options : List String
  scopedSyntax : List String
  attributes : List String
  localInstances : List String
  deriving DecidableEq, Repr

/-- One source command together with declarations and its incoming/outgoing
context observation. The command span owns its attached documentation and trivia. -/
structure CommandUnit where
  identity : CommandIdentity
  declarations : List DeclarationOrigin
  contextBefore : ContextState
  contextAfter : ContextState

/-- Typed reasons why two command units may not be separated freely. -/
inductive DependencyKind where
  | directReference
  | generatedFromCommand
  | mutualMembership
  | privateVisibility
  | localContext
  | noticeOwnership
  deriving DecidableEq, Repr

/-- Separation-forbidding dependency kinds. A direct public reference may cross a
module boundary through an explicit import; the other kinds require co-location
or a separately proved adapter. -/
def DependencyKind.hard : DependencyKind → Prop
  | .directReference => False
  | .generatedFromCommand => True
  | .mutualMembership => True
  | .privateVisibility => True
  | .localContext => True
  | .noticeOwnership => True

/-- Directed dependency between source command ordinals. -/
structure DependencyEdge where
  consumer : Nat
  provider : Nat
  kind : DependencyKind
  deriving DecidableEq, Repr

/-- A proposed conceptual component. Several subjects may be relevant, while the
selected target module remains singular and optional. -/
structure ConceptualSlice where
  ordinal : Nat
  commandOrdinals : List Nat
  primarySubject : Option SubjectIdentity
  secondarySubjects : List SubjectIdentity
  selectedTargetModule : Option String
  disposition : Disposition
  relation : RelationKind := .relevantTo

/-- A result-module dependency. `consumer` imports or otherwise depends on
`provider`. -/
structure ModuleDependency where
  consumer : String
  provider : String
  deriving DecidableEq, Repr

/-- `before` occurs strictly before `after` in one selected topological order. -/
def AppearsBefore (order : List String) (before after : String) : Prop :=
  ∃ prefix middle suffix,
    order = prefix ++ before :: middle ++ after :: suffix

/-- Command ordinals in exact source order. -/
def commandOrdinals (commands : List CommandUnit) : List Nat :=
  commands.map (fun command => command.identity.ordinal)

/-- Flattened slice assignment in slice order. -/
def sliceCommandOrdinals (slices : List ConceptualSlice) : List Nat :=
  slices.flatMap ConceptualSlice.commandOrdinals

/-- Every command belongs to exactly one slice, in exact source order. -/
def SlicesCoverExactly (commands : List CommandUnit)
    (slices : List ConceptualSlice) : Prop :=
  sliceCommandOrdinals slices = commandOrdinals commands

/-- All declaration fingerprints attached to the observed commands. -/
def declarationFingerprints (commands : List CommandUnit) :
    List DeclarationFingerprint :=
  commands.flatMap (fun command =>
    command.declarations.map DeclarationOrigin.declaration)

/-- Two commands are assigned to one conceptual slice. -/
def CommandsCoLocated (slices : List ConceptualSlice) (a b : Nat) : Prop :=
  ∃ slice ∈ slices,
    a ∈ slice.commandOrdinals ∧ b ∈ slice.commandOrdinals

/-- Every hard dependency remains within one physical/conceptual unit. -/
def HardDependenciesCoLocated (dependencies : List DependencyEdge)
    (slices : List ConceptualSlice) : Prop :=
  ∀ edge ∈ dependencies,
    edge.kind.hard → CommandsCoLocated slices edge.consumer edge.provider

/-- Machine-origin slice associations remain conceptual relevance only. -/
def MachineSlicesConservative (slices : List ConceptualSlice) : Prop :=
  ∀ slice ∈ slices, MachineAssignable slice.relation

/-- Pinned historical slices cannot receive a relocation target. -/
def PinnedSlicesStable (slices : List ConceptualSlice) : Prop :=
  ∀ slice ∈ slices,
    slice.disposition = .pinned → slice.selectedTargetModule = none

/-- Source-bound declaration-level decomposition plan. Bytes are represented as
natural numbers to keep the contract pure and independent of an operational byte
container. -/
structure SeamPlan where
  source : SourceFileIdentity
  sourceByteCount : Nat
  sourceBytes : List Nat
  reconstructedBytes : List Nat
  commands : List CommandUnit
  dependencies : List DependencyEdge
  slices : List ConceptualSlice
  observedDeclarations : List DeclarationFingerprint
  sourcePublicFingerprints : List DeclarationFingerprint
  resultPublicFingerprints : List DeclarationFingerprint
  sourceNotices : List String
  resultNotices : List String
  resultModuleOrder : List String
  resultModuleDependencies : List ModuleDependency
  attributionStatus : String
  sourceLength : sourceBytes.length = sourceByteCount
  commandSpansCover :
    CoversFrom 0 (commands.map (fun command => command.identity.span)) sourceByteCount
  uniqueCommandOrdinals : (commandOrdinals commands).Nodup
  slicesCoverCommands : SlicesCoverExactly commands slices
  declarationsCovered : declarationFingerprints commands = observedDeclarations
  uniqueDeclarationNames :
    (observedDeclarations.map DeclarationFingerprint.name).Nodup
  hardDependenciesStayTogether :
    HardDependenciesCoLocated dependencies slices
  conservativeRelations : MachineSlicesConservative slices
  pinnedSlicesStable : PinnedSlicesStable slices
  exactReconstruction : reconstructedBytes = sourceBytes
  publicSignaturesPreserved :
    resultPublicFingerprints = sourcePublicFingerprints
  noticesPreserved : resultNotices = sourceNotices
  uniqueResultModules : resultModuleOrder.Nodup
  resultModulesTopologicallyOrdered :
    ∀ edge ∈ resultModuleDependencies,
      AppearsBefore resultModuleOrder edge.provider edge.consumer

/-- Exact source reconstruction is a first-class plan field, not an inference
from equal hashes or successful parsing. -/
theorem SeamPlan.reconstructsExactBytes (plan : SeamPlan) :
    plan.reconstructedBytes = plan.sourceBytes :=
  plan.exactReconstruction

/-- The command partition covers the complete source byte range without an
unaccounted gap or overlap. -/
theorem SeamPlan.coversExactSourceRange (plan : SeamPlan) :
    CoversFrom 0
      (plan.commands.map (fun command => command.identity.span))
      plan.sourceByteCount :=
  plan.commandSpansCover

/-- Every hard generated/mutual/private/context/notice dependency remains in one
slice unless the plan has first changed that dependency through another theorem. -/
theorem SeamPlan.keepsHardDependenciesTogether (plan : SeamPlan) :
    HardDependenciesCoLocated plan.dependencies plan.slices :=
  plan.hardDependenciesStayTogether

/-- The selected public declaration surface is unchanged. -/
theorem SeamPlan.preservesPublicSignatures (plan : SeamPlan) :
    plan.resultPublicFingerprints = plan.sourcePublicFingerprints :=
  plan.publicSignaturesPreserved

/-- Source notices survive the decomposition exactly. -/
theorem SeamPlan.preservesNotices (plan : SeamPlan) :
    plan.resultNotices = plan.sourceNotices :=
  plan.noticesPreserved

/-- The result-module dependency graph has an explicit topological order. -/
theorem SeamPlan.ordersEveryResultDependency (plan : SeamPlan)
    (edge : ModuleDependency) (hedge : edge ∈ plan.resultModuleDependencies) :
    AppearsBefore plan.resultModuleOrder edge.provider edge.consumer :=
  plan.resultModulesTopologicallyOrdered edge hedge

/-- A machine-generated conceptual seam cannot be interpreted as mathematical
implication. -/
theorem SeamPlan.noMachineImplication (plan : SeamPlan)
    (slice : ConceptualSlice) (hslice : slice ∈ plan.slices) :
    slice.relation ≠ .implies := by
  have h := plan.conservativeRelations slice hslice
  intro himplies
  rw [himplies] at h
  exact machineAssignable_not_implies h

/-- Positive exact-coverage fixture. -/
def spanZeroTwo : ByteSpan :=
  { start := 0, stop := 2, ordered := by decide }

/-- Positive exact-coverage fixture. -/
def spanTwoThree : ByteSpan :=
  { start := 2, stop := 3, ordered := by decide }

/-- Hostile overlapping fixture. -/
def spanOneThree : ByteSpan :=
  { start := 1, stop := 3, ordered := by decide }

example : CoversFrom 0 [spanZeroTwo, spanTwoThree] 3 := by decide

example : ¬ CoversFrom 0 [spanZeroTwo, spanOneThree] 3 := by decide

example : ¬ ([0, 0] : List Nat).Nodup := by simp

example : ¬ MachineAssignable .implies :=
  machineAssignable_not_implies

end GRWTSK.FoundationsLogicAndTypeTheory.MathematicalOrganization.ConceptualSeamPlan

import Mathematics.Taxonomy.Status
import Mathlib.Logic.Relation

import Mathematics.Core.IdElements.StableId

/- @generated by mathematics-refactor: assembly -/

namespace Mathematics.Taxonomy

open Mathematics

structure DomainRecord where
  id : DomainId
  canonicalName : String
  definition : String
  parents : Finset DomainId
  provenance : Finset StableId
  deriving DecidableEq

structure DomainPath where
  root : DomainId
  segments : List DomainId
  deriving DecidableEq, Inhabited

namespace DomainPath

def terminal (path : DomainPath) : DomainId :=
  path.segments.getLastD path.root

def ids (path : DomainPath) : List DomainId :=
  path.root :: path.segments

end DomainPath

structure DomainRegistry where
  domains : List DomainRecord

namespace DomainRegistry

def UniqueIds (registry : DomainRegistry) : Prop :=
  (registry.domains.map (·.id)).Nodup

def IsParentOf (registry : DomainRegistry)
    (parent child : DomainId) : Prop :=
  ∃ record ∈ registry.domains,
    record.id = child ∧ parent ∈ record.parents

def ParentsResolve (registry : DomainRegistry) : Prop :=
  ∀ record ∈ registry.domains,
    ∀ parent ∈ record.parents,
      parent ∈ registry.domains.map (·.id)

def Acyclic (registry : DomainRegistry) : Prop :=
  ∀ id, ¬ Relation.TransGen registry.IsParentOf id id

/-- Strict parent reachability, with equality admitted for the root itself. -/
def ReachableFrom
    (registry : DomainRegistry) (root target : DomainId) : Prop :=
  root = target ∨ Relation.TransGen registry.IsParentOf root target

/-- The declared root occurs in the registry and reaches every domain. -/
def RootedAt (registry : DomainRegistry) (root : DomainId) : Prop :=
  root ∈ registry.domains.map (·.id) ∧
    ∀ record ∈ registry.domains, registry.ReachableFrom root record.id

/-- The distinguished root has no mathematical parent. -/
def RootHasNoParents
    (registry : DomainRegistry) (root : DomainId) : Prop :=
  ∀ record ∈ registry.domains,
    record.id = root → record.parents = ∅

def PathFrom (registry : DomainRegistry) :
    DomainId → List DomainId → Prop
  | _, [] => True
  | parent, child :: rest =>
      registry.IsParentOf parent child ∧ registry.PathFrom child rest

def ValidPath (registry : DomainRegistry) (path : DomainPath) : Prop :=
  path.root ∈ registry.domains.map (·.id) ∧
    registry.PathFrom path.root path.segments

def find? (registry : DomainRegistry) (id : DomainId) : Option DomainRecord :=
  registry.domains.find? fun record => record.id = id

end DomainRegistry

end Mathematics.Taxonomy

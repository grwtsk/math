def diff : Ordnode α → Ordnode α → Ordnode α
  | t₁, nil => t₁
  | t₁, t₂@(node _ l₂ x r₂) =>

      let (l₁, r₁) := split x t₁
      let l₁₂ := diff l₁ l₂

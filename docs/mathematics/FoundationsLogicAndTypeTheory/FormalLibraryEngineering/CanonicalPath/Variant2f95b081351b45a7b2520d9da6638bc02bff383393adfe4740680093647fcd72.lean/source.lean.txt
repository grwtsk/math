namespace Mathematics.Extractor

open Mathematics

structure CanonicalPath where
  value : String
  deriving DecidableEq, Repr, Inhabited

/-- Canonical artifact paths are nonempty, relative, and slash-normalized. -/
def CanonicalPath.Valid (path : CanonicalPath) : Prop :=
  path.value ≠ "" ∧
  path.value.startsWith "/" = false ∧
  path.value.contains '\\' = false ∧
  (path.value.splitOn "/").all (fun component ↦ component != "..") = true

end Mathematics.Extractor

  let initialState ← get
  let tasks ← jobs.mapM asTask'
  let mut results := []
  for task in tasks do
    try
      let result ← task.get
      let taskState ← saveState
      results := .ok (result, taskState) :: results
    catch e =>
      results := .error e :: results
  set initialState
  return results.reverse

/--

discarding state information.

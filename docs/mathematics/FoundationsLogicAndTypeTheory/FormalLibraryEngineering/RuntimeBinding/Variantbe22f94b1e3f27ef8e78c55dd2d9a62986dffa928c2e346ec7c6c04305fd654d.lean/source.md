import Mathematics.Concept.Representation

namespace Mathematics.Concept

/-- Runtime correspondence data, separated from its fixed refinement proof. -/
structure RuntimeBindingData
    (RuntimeInput FormalInput RuntimeOutput FormalOutput RuntimeFailure FormalFailure : Type)
    where
  conceptId : ConceptId
  operationId : StableId
  inputRelation : RuntimeInput → FormalInput → Prop
  outputRelation : RuntimeOutput → FormalOutput → Prop
  failureRelation : RuntimeFailure → FormalFailure → Prop
  executes : RuntimeInput → Except RuntimeFailure RuntimeOutput

/-- A runtime binding refines its formal boundary when every related input
produces either a related formal output or a related formal failure. -/
def RuntimeRefines
    {RuntimeInput FormalInput RuntimeOutput FormalOutput RuntimeFailure FormalFailure : Type}
    (binding : RuntimeBindingData RuntimeInput FormalInput RuntimeOutput FormalOutput
      RuntimeFailure FormalFailure) : Prop :=
  ∀ runtimeInput formalInput,
    binding.inputRelation runtimeInput formalInput →
      match binding.executes runtimeInput with
      | .ok runtimeOutput => ∃ formalOutput, binding.outputRelation runtimeOutput formalOutput
      | .error runtimeFailure => ∃ formalFailure, binding.failureRelation runtimeFailure formalFailure

/-- Proof-relevant generic runtime correspondence. -/
structure RuntimeBinding
    (RuntimeInput FormalInput RuntimeOutput FormalOutput RuntimeFailure FormalFailure : Type)
    where
  data : RuntimeBindingData RuntimeInput FormalInput RuntimeOutput FormalOutput
    RuntimeFailure FormalFailure
  refinementProof : RuntimeRefines data

end Mathematics.Concept

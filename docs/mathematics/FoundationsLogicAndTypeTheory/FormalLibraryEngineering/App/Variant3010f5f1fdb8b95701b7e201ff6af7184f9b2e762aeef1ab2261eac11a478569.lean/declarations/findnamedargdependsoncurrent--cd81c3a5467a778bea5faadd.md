private def findNamedArgDependsOnCurrent? : M (Option NamedArg) := do
  if (← get).namedArgs.isEmpty then
    return none
  else if (← get).fType.isArrow then
    return none
  else
    findNamedArgDependsOn? (← getFType) (← get).namedArgs

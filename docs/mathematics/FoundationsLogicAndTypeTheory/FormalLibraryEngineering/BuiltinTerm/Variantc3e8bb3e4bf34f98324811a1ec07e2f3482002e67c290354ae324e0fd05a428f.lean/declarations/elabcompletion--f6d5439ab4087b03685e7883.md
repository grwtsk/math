  if stx[0].isIdent then
    -- Add both an `id` and a `dot` `CompletionInfo` and have the language server figure out which

    try
      let e ← elabTerm stx[0] none
      addDotCompletionInfo stx e expectedType?
    catch _ =>
      s.restore

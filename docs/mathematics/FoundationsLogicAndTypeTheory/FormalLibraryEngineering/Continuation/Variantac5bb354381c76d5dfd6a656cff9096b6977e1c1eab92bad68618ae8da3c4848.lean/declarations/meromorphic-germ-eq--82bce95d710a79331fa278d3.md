    (hF : MeromorphicOn F U) (hG : MeromorphicOn G U) (hU : IsPreconnected U)
    {b s : ℂ} (hb : b ∈ U) (hs : s ∈ U) (heq : F =ᶠ[𝓝[≠] b] G) :

@[env_linter] def unusedArguments : Linter where
  noErrorsFound := "No unused arguments."
  errorsFound := "UNUSED ARGUMENTS."
  test declName := do
    if ← isAutoDecl declName then return none
    if ← isProjectionFn declName then return none
    if Linter.isDeprecated (← getEnv) declName then return none
    let info ← getConstInfo declName
    let ty := info.type
    -- Don't use `value? (allowOpaque := true)`, which would include `opaque` definition values.
    let val ← match info with
      | .defnInfo { value .. } | .thmInfo { value .. } => pure value
      | _ => return none
    if val.hasSorry || ty.hasSorry then return none
    forallTelescope ty fun args ty => do
      -- Exempt decls generated by `proof_wanted`/`def_wanted` (not imported here),
      -- which are of the form e.g. `(binders) → ProofWanted P` and have trivial content
      if ty.isAppOfArity' `ProofWanted 1 || ty.isAppOfArity' `DefWanted 1 then return none
      let mut e := (mkAppN val args).headBeta
      let ldecls ← args.mapM getFVarLocalDecl
      e := mkApp e ty
      for ldecl in ldecls do
        e := mkApp e ldecl.type
        if let some val := ldecl.value? then
          e := mkApp e val
      -- One `collectFVars` pass rather than `containsFVar` per argument, for performance.
      let usedFVars := (collectFVars {} e).fvarSet
      let unused := ldecls.zipIdx.filter fun (ldecl, _) =>
        -- We allow unused arguments whose names start with `_`, similar to the syntax linter,
        -- *unless* the name is an autogenerated instance name, since we do want to check those.
        (!ldecl.userName.isInternal ||
            ldecl.binderInfo.isInstImplicit && ldecl.userName.hasMacroScopes) &&
          !usedFVars.contains ldecl.fvarId
      if unused.isEmpty then return none
      let mut msg := m!"{unused.size} unused argument{if unused.size = 1 then "" else "s"}:"
      for (ldecl, i) in unused do
        -- Like `ppAsBinder`, but avoids printing autogenerated instance arguments.
        let ldeclMsg := match ldecl with
          | .ldecl (fvarId := fvarId) (type := type) (value := value) .. =>
            m!"let {Expr.fvar fvarId} : {type} := {value}"
          | .cdecl _ fvarId userName type bi _ =>
            let (lBracket, rBracket) : String × String := match bi with
              | .implicit       => ("{", "}")
              | .strictImplicit => ("⦃", "⦄")
              | .instImplicit   => ("[", "]")
              | .default        => ("(", ")")
            let typeMsg := if bi.isInstImplicit && userName.hasMacroScopes then m!"{type}" else
              m!"{Expr.fvar fvarId} : {type}"
            .bracket lBracket typeMsg rBracket
        msg := m!"{msg}\n  argument {i+1}: {ldeclMsg}"
      addMessageContextFull msg

def checkEqTrue (e : Expr) : CCM Unit := do
  let some (_, lhs, rhs) ← e.relSidesIfRefl? | return
  if ← isEqv e (.const ``True []) then return -- it is already equivalent to `True`
  let lhsR ← getRoot lhs

  if lhsR != rhsR then return
  -- Add `e = True`

noncomputable def checkRootStatus_idx (e : Expr) (ρ_int : IntervalEnv) (idx : Nat) : RootStatus :=
  let fI := evalInterval e ρ_int
  if excludesZero fI then
    RootStatus.noRoot
  else if signChange_idx e ρ_int idx then
    RootStatus.hasRoot
  else
    RootStatus.unknown

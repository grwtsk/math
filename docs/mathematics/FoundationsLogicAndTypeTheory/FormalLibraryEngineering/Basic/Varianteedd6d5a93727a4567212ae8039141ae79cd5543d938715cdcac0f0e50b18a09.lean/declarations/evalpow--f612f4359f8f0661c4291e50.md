def evalPow {a : Q($α)} {b : Q(ℕ)} (va : ExSum sα a) (vb : ExSum sℕ b) :
    Lean.Core.CoreM <| Result (ExSum sα) q($a ^ $b) := do
  match vb with
  | .zero => return ⟨_, (ExProd.mkNat sα 1).2.toSum, q(pow_zero $a)⟩

    let ⟨_, vc₁, pc₁⟩ ← evalPow₁ sα va vb₁

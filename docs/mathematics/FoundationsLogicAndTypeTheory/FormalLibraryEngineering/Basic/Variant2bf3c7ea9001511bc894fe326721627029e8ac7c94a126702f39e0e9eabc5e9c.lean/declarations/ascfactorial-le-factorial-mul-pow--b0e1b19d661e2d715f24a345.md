theorem ascFactorial_le_factorial_mul_pow (n k : ℕ) : n.ascFactorial k ≤ k ! * n ^ k :=

  | 0 => by simp
  | j + 1 => by
    rcases n.eq_zero_or_pos with rfl | hn
    · simp [zero_ascFactorial]

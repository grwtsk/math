theorem opow_le_iff_le_log {b x c : Ordinal} (hb : 1 < b) (hx : x ≠ 0) :
    b ^ c ≤ x ↔ c ≤ log b x := by

  intro h
  · apply le_of_not_lt
    intro hn

  · exact ((opow_le_opow_iff_right hb).2 h).trans <| opow_log_le_self b hx

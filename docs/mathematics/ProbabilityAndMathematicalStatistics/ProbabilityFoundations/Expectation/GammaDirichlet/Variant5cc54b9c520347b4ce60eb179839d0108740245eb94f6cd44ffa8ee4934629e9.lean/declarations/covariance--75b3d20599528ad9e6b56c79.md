noncomputable def covariance [Nonempty I]
    (parameters : DirichletParameters I) (i j : I) : ℝ := by
  classical
  exact if i = j then parameters.variance i
    else
      -(parameters.shape i * parameters.shape j) /
        (parameters.total ^ 2 * (parameters.total + 1))

@[simp]

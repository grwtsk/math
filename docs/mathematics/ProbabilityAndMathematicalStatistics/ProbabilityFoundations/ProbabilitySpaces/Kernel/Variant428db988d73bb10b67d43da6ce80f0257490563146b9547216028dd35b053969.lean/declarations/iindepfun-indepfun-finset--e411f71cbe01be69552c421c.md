    (hf_Indep : iIndepFun f κ μ) (hf_meas : ∀ i, Measurable (f i)) :

  · simp

    exists_ae_eq_isMarkovKernel hf_Indep.ae_isProbabilityMeasure hμ

  let πSβ := Set.pi (Set.univ : Set S) ''
    Set.pi (Set.univ : Set S) fun i => { s : Set (β i) | MeasurableSet[m i] s }

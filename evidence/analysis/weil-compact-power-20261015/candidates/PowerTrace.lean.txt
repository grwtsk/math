import Mathlib.Analysis.Complex.Norm
import Mathlib.Tactic

set_option autoImplicit false
open scoped BigOperators

namespace GRWTSK.Analysis.Complex.PowerTrace

/-- The denominator is the actual nonzero normalization, not a lower estimate. -/
def normalizedPower (p h v : ℂ) (n : ℕ) : ℂ := p * (h / v) ^ n

theorem selected_value (p v : ℂ) (n : ℕ) (hv : v ≠ 0) :
    normalizedPower p v v n = p := by
  simp [normalizedPower, hv]

theorem cancelled_value (h v : ℂ) (n : ℕ) :
    normalizedPower 0 h v n = 0 := by
  simp [normalizedPower]

theorem real_type (p h v : ℂ) (n : ℕ) (hv : star v = v) :
    normalizedPower (star p) (star h) v n = star (normalizedPower p h v n) := by
  simp [normalizedPower, hv, mul_comm]

/-- Source multiplicities are retained; this is a signed square identity. -/
theorem selected_pair_square (m₁ m₂ : ℝ) :
    (m₁ : ℂ) * Complex.I ^ 2 + (m₂ : ℂ) * (-Complex.I) ^ 2 =
      -((m₁ + m₂ : ℝ) : ℂ) := by
  norm_num [Complex.I_sq] <;> push_cast <;> ring

/-- A separate complete far-tail allowance, not a change to the pair trace. -/
theorem negative_with_tail (W M tail : ℝ) (hM : 0 < M)
    (hW : W ≤ -M + tail) (ht : tail < M / 2) : W < -M / 2 := by
  linarith

end GRWTSK.Analysis.Complex.PowerTrace

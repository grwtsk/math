import Mathlib

set_option autoImplicit false

namespace GRWTSK.Algebra.SignedNewton

variable {R : Type*} [CommRing R]

def inverseStep (a z : R) : R := a * (2 - z * a)

theorem inverseStep_eq_sub (a z : R) :
    inverseStep a z = a - a * (z * a - 1) := by
  unfold inverseStep
  ring

theorem residual_square (a z : R) :
    z * inverseStep a z - 1 = -(z * a - 1) ^ 2 := by
  unfold inverseStep
  ring

theorem source_error (mu a z : R) (h : mu * z = 1) :
    mu - inverseStep a z = mu * (z * a - 1) ^ 2 := by
  symm
  calc
    mu * (z * a - 1) ^ 2 = (mu * z) * (z * a ^ 2 - 2 * a) + mu := by ring
    _ = z * a ^ 2 - 2 * a + mu := by rw [h]; ring
    _ = mu - inverseStep a z := by unfold inverseStep; ring

theorem reconstruction (mu a z : R) (h : mu * z = 1) :
    mu = inverseStep a z + mu * (z * a - 1) ^ 2 := by
  rw [← source_error mu a z h]
  ring

theorem inverse_on_square_zero (a z : R) (h : (z * a - 1) ^ 2 = 0) :
    z * inverseStep a z = 1 := by
  have hr := residual_square a z
  rw [h, neg_zero] at hr
  exact sub_eq_zero.mp hr

end GRWTSK.Algebra.SignedNewton

namespace GRWTSK.Analysis.SignedResidual

noncomputable def scalarCorrection (a r : ℝ) : ℝ := -r / (2 * a)

theorem scalar_cross (a r : ℝ) (ha : a ≠ 0) :
    2 * a * scalarCorrection a r = -r := by
  unfold scalarCorrection
  field_simp [ha]
  <;> ring

theorem scalar_residual (a r : ℝ) (ha : a ≠ 0) :
    r + ((a + scalarCorrection a r) ^ 2 - a ^ 2) =
      (scalarCorrection a r) ^ 2 := by
  calc
    _ = (r + 2 * a * scalarCorrection a r) + (scalarCorrection a r) ^ 2 := by ring
    _ = _ := by rw [scalar_cross a r ha]; ring

theorem signed_difference_square (r l : ℝ) :
    (r - l) ^ 2 = r ^ 2 + l ^ 2 - 2 * r * l := by ring

theorem preserve_source (r l dr dl : ℝ) :
    r - l = (r + dr) - (l + dl) - (dr - dl) := by ring

end GRWTSK.Analysis.SignedResidual

/-- Complete three-term norm allowance, including all cross terms. -/
theorem three_square_bound (a b c : ℝ) :
    (a+b+c)^2 ≤ 3*(a^2+b^2+c^2) := by
  nlinarith [sq_nonneg (a-b), sq_nonneg (a-c), sq_nonneg (b-c)]

/-- Exact constants from the two cutoff derivatives at R=4L, L>=1. -/

    (hnegative : W ≤ -mass/2) (hlower : -bound*normSq ≤ W)
    (hbudget : bound*upper < mass/2) : False := by

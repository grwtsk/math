import Mathlib.Algebra.Order.BigOperators.Group.Finset
import Mathlib.Data.Real.Basic
import Mathlib.Tactic

set_option autoImplicit false
namespace GRWTSK.QuadraticForm.MomentPerturbation

/-- Finite Hankel pairing in one fixed, real coefficient basis. -/
def pairing {n : ℕ} (c : Fin n → ℝ) (m : ℕ → ℝ) : ℝ :=
  ∑ i, ∑ j, c i * c j * m (i.val + j.val)

def coefficientMass {n : ℕ} (c : Fin n → ℝ) : ℝ := ∑ i, |c i|

/-- The coefficient norm is retained; an entrywise moment tolerance is not
itself a tolerance for a potentially ill-conditioned polynomial test. -/
theorem pairing_error {n : ℕ} (c : Fin n → ℝ) (m mhat : ℕ → ℝ)
    (ε : ℝ) (herr : ∀ i j : Fin n, |m (i.val + j.val) - mhat (i.val + j.val)| ≤ ε) :
    |pairing c m - pairing c mhat| ≤ ε * coefficientMass c ^ 2 := by
  have hid : pairing c m - pairing c mhat =
      ∑ i, ∑ j, c i * c j * (m (i.val + j.val) - mhat (i.val + j.val)) := by
    simp only [pairing, mul_sub, Finset.sum_sub_distrib]
  rw [hid]
  calc
    |∑ i, ∑ j, c i * c j * (m (i.val + j.val) - mhat (i.val + j.val))|
        ≤ ∑ i, |∑ j, c i * c j * (m (i.val + j.val) - mhat (i.val + j.val))| :=
      Finset.abs_sum_le_sum_abs _ _
    _ ≤ ∑ i, ∑ j, |c i * c j * (m (i.val + j.val) - mhat (i.val + j.val))| := by
      apply Finset.sum_le_sum
      intro i _
      exact Finset.abs_sum_le_sum_abs _ _
    _ ≤ ∑ i, ∑ j, |c i| * |c j| * ε := by
      apply Finset.sum_le_sum
      intro i _
      apply Finset.sum_le_sum
      intro j _
      rw [abs_mul, abs_mul]
      exact mul_le_mul_of_nonneg_left (herr i j) (mul_nonneg (abs_nonneg _) (abs_nonneg _))
    _ = ε * coefficientMass c ^ 2 := by
      simp only [coefficientMass, pow_two, Finset.sum_mul, Finset.mul_sum]
      apply Finset.sum_congr rfl
      intro i _
      apply Finset.sum_congr rfl
      intro j _
      ring

/-- A finite strict margin can reject nonnegativity without a limiting theorem. -/
theorem negative_of_upper_error {n : ℕ} (c : Fin n → ℝ) (m mhat : ℕ → ℝ)
    (ε : ℝ) (herr : ∀ i j : Fin n, |m (i.val + j.val) - mhat (i.val + j.val)| ≤ ε)
    (hmargin : pairing c mhat + ε * coefficientMass c ^ 2 < 0) :
    pairing c m < 0 := by
  have h := (abs_le.mp (pairing_error c m mhat ε herr)).2
  linarith

/-- Equivalent supplier-facing form. No positivity assumption is discharged here. -/
theorem strict_error_contradiction {n : ℕ} (c : Fin n → ℝ) (m mhat : ℕ → ℝ)
    (ε : ℝ) (herr : ∀ i j : Fin n, |m (i.val + j.val) - mhat (i.val + j.val)| ≤ ε)
    (hsource : 0 ≤ pairing c m)
    (hmargin : pairing c mhat + ε * coefficientMass c ^ 2 < 0) : False := by
  exact (not_lt_of_ge hsource) (negative_of_upper_error c m mhat ε herr hmargin)

end GRWTSK.QuadraticForm.MomentPerturbation

import Mathlib.Data.Complex.Basic
import Mathlib.Tactic
import GRWTSK.Analysis.TriadicEnergy

set_option autoImplicit false

namespace GRWTSK.Analysis.ReflectionCollapse

open GRWTSK.Analysis.TriadicEnergy

/-- Even and odd reflection channels. -/
def evenPart (x y : ℂ) : ℂ := (x + y) / 2

def oddPart (x y : ℂ) : ℂ := (x - y) / 2

/-- Reflection collapse is lossless only when both parity channels are retained. -/
theorem reconstruct_left (x y : ℂ) : evenPart x y + oddPart x y = x := by
  simp [evenPart, oddPart]
  ring

theorem reconstruct_right (x y : ℂ) : evenPart x y - oddPart x y = y := by
  simp [evenPart, oddPart]
  ring

/-- Exact reflected energy split. -/
theorem energy_split (x y : ℂ) :
    energy x + energy y =
      2 * energy (evenPart x y) + 2 * energy (oddPart x y) := by
  simp only [energy, evenPart, oddPart, Complex.add_re, Complex.add_im,
    Complex.sub_re, Complex.sub_im, Complex.div_re, Complex.div_im,
    Complex.ofNat_re, Complex.ofNat_im]
  norm_num
  ring

/-- A fixed reflection slot has no odd component, but retains its indexed mass. -/
theorem oddPart_self (x : ℂ) : oddPart x x = 0 := by
  simp [oddPart]

end GRWTSK.Analysis.ReflectionCollapse

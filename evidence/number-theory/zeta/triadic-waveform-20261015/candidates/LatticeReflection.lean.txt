import Mathlib.Tactic

set_option autoImplicit false

namespace GRWTSK.Algebra.LatticeReflection

abbrev AdditivePair := ℤ × ℤ

def swap (p : AdditivePair) : AdditivePair := (p.2, p.1)

def differenceTranslate (k : ℤ) (p : AdditivePair) : AdditivePair :=
  (p.1 + k, p.2 - k)

@[simp] theorem swap_involution (p : AdditivePair) : swap (swap p) = p := by
  cases p
  rfl

/-- The fixed-total difference fiber carries the infinite-dihedral relation. -/
theorem swap_translate_swap (k : ℤ) (p : AdditivePair) :
    swap (differenceTranslate k (swap p)) = differenceTranslate (-k) p := by
  rcases p with ⟨a, b⟩
  simp [swap, differenceTranslate]
  constructor <;> ring

/-- The positive cone is not invariant under the difference translation. -/
theorem positive_cone_control : differenceTranslate 1 (1, 1) = (2, 0) := by
  norm_num [differenceTranslate]

end GRWTSK.Algebra.LatticeReflection

import Mathlib.Algebra.BigOperators.Group.Finset.Basic
import Mathlib.Tactic.Linarith
import Mathlib.Tactic.Ring

set_option autoImplicit false

namespace GRWTSK.Analysis.Sequences.HorizonWeightedAbelFlux

open scoped BigOperators

/-- Weighted endpoint energy on the finite prefix `0,...,N-1`. -/
def endpointEnergy (q v : ℕ → ℝ) (N : ℕ) : ℝ :=
  ∑ n ∈ Finset.range N, q n ^ 2 * (v n - v (n + 1))

/-- Complete signed off-diagonal work. -/
def signedFlux (q d v : ℕ → ℝ) (N : ℕ) : ℝ :=
  ∑ n ∈ Finset.range N, 2 * q n * d n * v (n + 1)

/-- Diagonal quadratic variation. -/
def correction (d v : ℕ → ℝ) (N : ℕ) : ℝ :=
  ∑ n ∈ Finset.range N, d n ^ 2 * v (n + 1)

/-- Exact finite Abel identity, retaining both endpoints. -/
theorem endpointEnergy_add_terminal_eq_initial_add_flux_add_correction
    (q d v : ℕ → ℝ)
    (hrec : ∀ n, q (n + 1) = q n + d n) :
    ∀ N : ℕ,
      endpointEnergy q v N + q N ^ 2 * v N =
        q 0 ^ 2 * v 0 + signedFlux q d v N + correction d v N := by
  intro N
  induction N with
  | zero => simp [endpointEnergy, signedFlux, correction]
  | succ N ih =>
      simp only [endpointEnergy, signedFlux, correction, Finset.sum_range_succ] at ih ⊢
      rw [hrec N]
      linarith [sq_nonneg (q N + d N)]

/-- A zero incoming seed and a terminally vanishing horizon weight leave no
boundary debt. -/
theorem endpointEnergy_eq_flux_add_correction
    (q d v : ℕ → ℝ)
    (hrec : ∀ n, q (n + 1) = q n + d n)
    (hq0 : q 0 = 0) {N : ℕ} (hvN : v N = 0) :
    endpointEnergy q v N = signedFlux q d v N + correction d v N := by
  have h := endpointEnergy_add_terminal_eq_initial_add_flux_add_correction
    q d v hrec N
  simp only [hq0, hvN, zero_pow, mul_zero, zero_mul, zero_add] at h
  exact h

/-- Nonnegative horizon weights make the signed flux no larger than the full
positive endpoint energy. -/
theorem signedFlux_le_endpointEnergy
    (q d v : ℕ → ℝ)
    (hrec : ∀ n, q (n + 1) = q n + d n)
    (hq0 : q 0 = 0) {N : ℕ} (hvN : v N = 0)
    (hv : ∀ n < N, 0 ≤ v (n + 1)) :
    signedFlux q d v N ≤ endpointEnergy q v N := by
  rw [endpointEnergy_eq_flux_add_correction q d v hrec hq0 hvN]
  have hc : 0 ≤ correction d v N := by
    unfold correction
    exact Finset.sum_nonneg (fun n hn =>
      mul_nonneg (sq_nonneg _) (hv n (Finset.mem_range.mp hn)))
  linarith

end GRWTSK.Analysis.Sequences.HorizonWeightedAbelFlux

open GRWTSK.Polynomial.QuarticTrace
open GRWTSK.RiemannHypothesis (NontrivialZetaZero)

/-- An explicitly equivalent supplier, not an established positivity theorem. -/
def QuarticTraceNonnegative : Prop :=
  ∀ s : ℂ, NontrivialZetaZero s → 0 ≤ witnessTrace 2 (s.re - 1 / 2) s.im

theorem quarticTraceNonnegative_iff :
    QuarticTraceNonnegative ↔ _root_.RiemannHypothesis := by
  constructor
  · intro h s hz htriv hpole
    have ha := (uniform_witness_nonneg_iff (s.re - 1 / 2) s.im).mp
      (h s ⟨hz, htriv, hpole⟩)
    linarith
  · intro h s hs
    apply (uniform_witness_nonneg_iff (s.re - 1 / 2) s.im).mpr
    have hre := h s hs.1 hs.2.1 hs.2.2
    linarith

/-- The analytic/arithmetic realization of these witnesses remains open. -/
theorem riemannHypothesis_of_relative_extraction
    (h : ∀ s : ℂ, NontrivialZetaZero s → ∀ ε : ℝ, 0 < ε →
      ∃ w scale residual : ℝ, 0 ≤ w ∧ 0 < scale ∧
        w = scale * witnessTrace 2 (s.re - 1 / 2) s.im + residual ∧
        residual ≤ scale * ε) : _root_.RiemannHypothesis := by
  apply quarticTraceNonnegative_iff.mp
  intro s hs
  exact GRWTSK.Order.PositiveScale.nonneg_of_residual_extraction _ (h s hs)

/-- Only one sufficiently sharp source test for each hypothetical off-line zero
is needed. This is still a conditional consumer with an unproved supplier. -/
theorem riemannHypothesis_of_finite_residual
    (h : ∀ s : ℂ, NontrivialZetaZero s → s.re ≠ 1 / 2 →
      ∃ w scale residual : ℝ, 0 ≤ w ∧ 0 < scale ∧
        w = scale * witnessTrace 2 (s.re - 1 / 2) s.im + residual ∧
        residual < 16 * scale * (s.re - 1 / 2) ^ 4) :
    _root_.RiemannHypothesis := by
  intro s hz htriv hpole
  by_contra hoff
  obtain ⟨w, scale, residual, hw, hs, heq, hr⟩ := h s ⟨hz, htriv, hpole⟩ hoff
  exact residual_contradiction (s.re - 1 / 2) s.im w scale residual hw hs heq hr

end GRWTSK.Zeta.Riemann

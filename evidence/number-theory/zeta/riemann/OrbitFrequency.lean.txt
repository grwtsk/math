import GRWTSK.NumberTheory.Zeta.Riemann.ZeroSymmetry
import GRWTSK.Algebra.Polynomial.Quartic.TraceWitness

set_option autoImplicit false

noncomputable section

namespace GRWTSK.Zeta.Riemann

open GRWTSK.RiemannHypothesis

/-- The actual affine rotation of the zeta variable, expressed in coordinates. -/
def frequency (s : ℂ) : ℂ := ⟨s.im, -(s.re - 1 / 2)⟩

theorem frequency_eq_rotation (s : ℂ) :
    frequency s = -Complex.I * (s - 1 / 2) := by
  apply Complex.ext <;> simp [frequency, Complex.mul_re, Complex.mul_im] <;> ring

/-- The Klein-four orbit gives precisely the listed algebraic roots in this order.
Repeated list entries are retained; no analytic multiplicity is asserted. -/
theorem frequency_orbit (s : ℂ) :
    [frequency (criticalReflection s), frequency s,
      frequency (zeroReflect s), frequency (zeroConjugate s)] =
    GRWTSK.Polynomial.QuarticTrace.roots (s.re - 1 / 2) s.im := by
  have hc : frequency (criticalReflection s) = ⟨s.im, s.re - 1 / 2⟩ := by
    rw [critical_reflection_eq_composite]
    apply Complex.ext <;> simp [frequency, zeroReflect, zeroConjugate] <;> ring
  have hr : frequency (zeroReflect s) = ⟨-s.im, s.re - 1 / 2⟩ := by
    apply Complex.ext <;> simp [frequency, zeroReflect] <;> ring
  have hconj : frequency (zeroConjugate s) = ⟨-s.im, -(s.re - 1 / 2)⟩ := by
    apply Complex.ext <;> simp [frequency, zeroConjugate]
  rw [hc, hr, hconj]
  rfl

/-- Reuse source zero transport instead of introducing an arbitrary model orbit. -/
theorem orbit_zero_members (s : ℂ) (hs : NontrivialZetaZero s) :
    NontrivialZetaZero (criticalReflection s) ∧ NontrivialZetaZero s ∧
      NontrivialZetaZero (zeroReflect s) ∧ NontrivialZetaZero (zeroConjugate s) :=
  ⟨hs.critical_reflection, hs, hs.reflect, hs.conjugate⟩

end GRWTSK.Zeta.Riemann
